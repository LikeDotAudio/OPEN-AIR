# oaComVisa/Tests/test_visa_manager.py

# Description: Tests for the VisaManager class, covering instrument connection,
#              communication, and disconnection.

import unittest
from unittest.mock import MagicMock, patch
import pyvisa as visa

from oaComVisa.Managers.visa_manager import VisaManager

class TestVisaManager(unittest.TestCase):
    """
    Tests for the VisaManager class, ensuring proper interaction with VISA instruments.
    """
    TEST_RESOURCE_NAME = "GPIB0::1::INSTR"
    TEST_WRITE_COMMAND = "*IDN?"
    TEST_READ_RESPONSE = "MY_VENDOR,MY_MODEL,MY_SN,MY_FW"

    def setUp(self):
        """
        Set up the test environment.
        Mocks pyvisa's ResourceManager and Instrument to simulate VISA communication.
        """
        self.mock_instrument = MagicMock(spec=visa.resources.MessageBasedResource)
        self.mock_instrument.query.return_value = self.TEST_READ_RESPONSE
        self.mock_instrument.read.return_value = self.TEST_READ_RESPONSE
        self.mock_instrument.write.return_value = None

        self.mock_resource_manager = MagicMock(spec=visa.ResourceManager)
        self.mock_resource_manager.open_resource.return_value = self.mock_instrument

        self.patcher = patch('pyvisa.ResourceManager', return_value=self.mock_resource_manager)
        self.mock_rm_class = self.patcher.start()

        self.visa_manager = VisaManager()

    def tearDown(self):
        """
        Clean up after tests.
        """
        self.patcher.stop()
        self.visa_manager.disconnect_all()

    def test_connect_to_instrument_success(self):
        """
        Goal: Verify that VisaManager can successfully connect to an instrument.
        """
        # BUILD
        resource_name = self.TEST_RESOURCE_NAME

        # OPERATE
        connected_instrument = self.visa_manager.connect_to_instrument(resource_name)

        # CHECK
        self.assertIsNotNone(connected_instrument)
        self.assertEqual(self.visa_manager.get_connected_resource(resource_name), connected_instrument)
        self.mock_resource_manager.open_resource.assert_called_once_with(resource_name)

    def test_connect_to_instrument_failure(self):
        """
        Goal: Verify that VisaManager handles connection failures gracefully.
        """
        # BUILD
        self.mock_resource_manager.open_resource.side_effect = visa.errors.VisaIOError(-1073807343)

        # OPERATE & CHECK
        with self.assertRaises(visa.errors.VisaIOError):
            self.visa_manager.connect_to_instrument("GPIB0::FAIL::INSTR")

    def test_write_and_read_command(self):
        """
        Goal: Verify that VisaManager can write commands and read responses from an instrument.
        """
        # BUILD
        self.visa_manager.connect_to_instrument(self.TEST_RESOURCE_NAME)
        
        # OPERATE
        response = self.visa_manager.write_and_read(self.TEST_RESOURCE_NAME, self.TEST_WRITE_COMMAND)

        # CHECK
        self.assertEqual(response, self.TEST_READ_RESPONSE)
        self.mock_instrument.query.assert_called_once_with(self.TEST_WRITE_COMMAND)

    def test_disconnect_instrument(self):
        """
        Goal: Verify that VisaManager can disconnect from a connected instrument.
        """
        # BUILD
        self.visa_manager.connect_to_instrument(self.TEST_RESOURCE_NAME)
        self.assertIsNotNone(self.visa_manager.get_connected_resource(self.TEST_RESOURCE_NAME))

        # OPERATE
        self.visa_manager.disconnect_instrument(self.TEST_RESOURCE_NAME)

        # CHECK
        self.assertIsNone(self.visa_manager.get_connected_resource(self.TEST_RESOURCE_NAME))
        self.mock_instrument.close.assert_called_once()

    def test_get_connected_resources(self):
        """
        Goal: Verify that get_connected_resources returns all currently connected instruments.
        """
        # BUILD
        self.visa_manager.connect_to_instrument("GPIB0::1::INSTR")
        self.visa_manager.connect_to_instrument("GPIB0::2::INSTR")

        # OPERATE
        connected_resources = self.visa_manager.get_connected_resources()

        # CHECK
        self.assertEqual(len(connected_resources), 2)
        self.assertIn("GPIB0::1::INSTR", connected_resources)
        self.assertIn("GPIB0::2::INSTR", connected_resources)
