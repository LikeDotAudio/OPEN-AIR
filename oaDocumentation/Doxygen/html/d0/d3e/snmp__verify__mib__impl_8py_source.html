121 def run_test(self):
122 if not self.snmp_manager: return
123 mib_path = self.selected_mib_path.get()
124 if not mib_path or mib_path == "No MIB selected":
125 self.text_area.insert(tk.END, "❌ ERROR: Please select a MIB file first.\n")
126 return
127
128 self.text_area.delete("1.0", tk.END)
129 self.text_area.insert(tk.END, f"Executing walk using MIB file: {mib_path}...\n")
130 self.text_area.insert(tk.END, "-"*40 + "\n")
131 self.update()
132
133 output = self.snmp_manager.run_verification(mib_path=mib_path)
134
135 # Apply Filter
136 filter_str = self.filter_var.get()
137 if filter_str:
138 lines = output.splitlines()
139 filtered_lines = [line.replace(filter_str, "") for line in lines]
140 output = "\n".join(filtered_lines)
141
142 # Update Counter
143 count = sum(1 for line in output.splitlines() if " = " in line)
144 self.counter_var.set(f"Objects: {count}")
145
146 self.text_area.insert(tk.END, output)
