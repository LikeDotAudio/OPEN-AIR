60 def publish_inventory(self, inventory_data):
61 if not self.mqtt_manager: return
62
63 matrix_log("comms", "visa", "publish_inventory", f"Fleet Bridge: Publishing inventory. Size: {len(str(inventory_data))} chars.", "DEBUG")
64
65 try:
66 self._publish_flattened_dict(inventory_data, self.topic_prefix)
67 except Exception as e:
68 matrix_log("comms", "visa", "publish_inventory", f"MQTT Bridge Error publishing flattened message: {e}", "ERROR")

70 def _publish_flattened_dict(self, data, base_topic):
71 if not self.mqtt_manager: return
72
73 if isinstance(data, dict):
74 if ("serial_number" in data and "device_type" in data
75 and "model" in data):
76 try:
77 payload = orjson.dumps(data, option=orjson.OPT_INDENT_2).decode()
78 self.mqtt_manager.publish(base_topic, payload)
79 except Exception as e:
80 matrix_log("comms", "visa", "_publish_flattened_dict", f"MQTT Bridge Error publishing device BLOB {base_topic}: {e}", "ERROR")
81 return
82
83 for key, value in data.items():
84 sanitized_key = key.replace("/", "_")
85 new_topic = f"{base_topic}/{sanitized_key}"
86 self._publish_flattened_dict(value, new_topic)
87 elif isinstance(data, list):
88 for index, item in enumerate(data):
89 new_topic = f"{base_topic}/{index}"
90 self._publish_flattened_dict(item, new_topic)
91 else:
92 if data is None: return
93 try:
94 self.mqtt_manager.publish(base_topic, str(data))
95 except Exception as e:
96 matrix_log("comms", "visa", "_publish_flattened_dict", f"MQTT Bridge Error publishing leaf node {base_topic}: {e}", "ERROR")
