99 def _on_bridge_status(self, message):
100 """Updates internal status from bridge broadcasts."""
101 try:
102 data = message.get_json_payload()
103 if "status" in data:
104 self.stats["status"] = data["status"]
105 # Trigger a broadcast to update GUI immediately
106 self._broadcast("STATUS_UPDATE", {"_message_type": "STATUS", "_stats": self.get_telemetry()})
107 except Exception: pass

109 def _heartbeat_loop(self):
110 """Periodically notifies GUI observers even if no messages are flowing."""
111 while self._is_running:
112 try:
113 # Update throughput even when idle
114 now = time.time()
115 elapsed = now - self.stats["start_time"]
116 if elapsed > 0:
117 self.stats["throughput_message_sec"] = round(self.stats["total_messages"] / elapsed, 2)
118
119 # Push telemetry to GUI
120 self._broadcast("HEARTBEAT", {"_message_type": "HEARTBEAT", "_stats": self.get_telemetry()})
121 except Exception: pass
122 time.sleep(1.0)

124 def _on_smpte2138_traffic(self, message):
125 """
126 Intercepts binary SMPTE2138 traffic and updates performance metrics.
127 """
128 topic = message.topic
129 payload = message.payload
130 now = time.time()
131
132 self.stats["total_messages"] += 1
133 self.stats["last_message_ts"] = now
134
135 decoded_data = None
136 message_type = "Unknown"
137
138 try:
139 if "/param/" in topic:
140 message_type = "SingleSetValue"
141 self.stats["params_processed"] += 1
142 decoded = param_pb2.SingleSetValuePayload()
143 decoded.ParseFromString(payload)
144 decoded_data = {
145 "slot": decoded.slot,
146 "oid": decoded.value.oid,
147 "value": self._extract_value(decoded.value.value)
148 }
149 elif "/cmd/" in topic:
150 message_type = "ExecuteCommand"
151 self.stats["cmds_processed"] += 1
152 decoded = param_pb2.ExecuteCommandPayload()
153 decoded.ParseFromString(payload)
154 decoded_data = {
155 "slot": decoded.slot,
156 "oid": decoded.oid,
157 "value": self._extract_value(decoded.value),
158 "respond": decoded.respond
159 }
160
161 if decoded_data:
162 decoded_data["_message_type"] = message_type
163 decoded_data["_topic"] = topic
164 decoded_data["_stats"] = self.get_telemetry()
165 self._broadcast(topic, decoded_data)
166
167 except Exception as e:
168 if LOCAL_DEBUG:
169 SMPTE2138_LOGGER.trace(f"⚠️ [MONITOR] Decode skipped for {topic}: {e}")

171 def _extract_value(self, value_obj):
172 """Helper to extract the active field from a SMPTE2138 Value object."""
173 kind = value_obj.WhichOneof("kind")
174 if not kind: return None
175 return getattr(value_obj, kind)

177 def get_telemetry(self):
178 """Returns the current performance and health snapshot."""
179 uptime = time.time() - self.stats["start_time"]
180 return {
181 "uptime_s": round(uptime, 1),
182 "message_count": self.stats["total_messages"],
183 "rate": self.stats["throughput_message_sec"],
184 "broker": f"{self.mqtt.broker_address}:{self.mqtt.broker_port}",
185 "connected": self.mqtt.is_connected(),
186 "status": self.stats["status"]
187 }
