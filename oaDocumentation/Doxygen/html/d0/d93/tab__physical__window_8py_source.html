58 """Loads last known window size and position from disk."""
59 try:
60 if os.path.exists(LAYOUT_CACHE_PATH):
61 with open(LAYOUT_CACHE_PATH) as f:
62 cache = json.load(f)
63 geom = cache.get("window_geometry")
64 if geom:
65 root.geometry(geom)
66 matrix_log("UI", "GUI_MANAGER", "restore_window_geometry", f"Restored window geometry: {geom}", level="INFO")
67 return
68
69 # Default center if no cache
70 sw, sh = root.winfo_screenwidth(), root.winfo_screenheight()
71 ww, wh = 1200, 800
72 x, y = (sw // 2) - (ww // 2), (sh // 2) - (wh // 2)
73 root.geometry(f"{ww}x{wh}+{x}+{y}")
74 except Exception as e:
