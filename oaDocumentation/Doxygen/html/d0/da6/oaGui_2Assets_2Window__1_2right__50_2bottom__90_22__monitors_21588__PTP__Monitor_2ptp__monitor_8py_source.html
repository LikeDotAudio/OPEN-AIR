114 def on_ptp_packet(self, data):
115 ordered, ts_str, tag = PTPDataProcessor.process_packet(data)
116 iid = self.packet_tree.insert("", 0, values=(ts_str, data["source_ip"], data["message_type"], data["domain"], data["sequence_id"], data["clock_identity"]), tags=(tag))
117 self._packet_data_cache[iid] = ordered
118
119 self.packet_tree.selection_set(iid); self.packet_tree.see(iid); self.on_packet_selecton_packet_select()
120 if len(self.packet_tree.get_children()) > 100:
121 last = self.packet_tree.get_children()[-1]; self.packet_tree.delete(last); self._packet_data_cache.pop(last, None)

123 def on_packet_select(self, event=None):
124 sel = self.packet_tree.selection()
125 if not sel: return
126 self.dissector_tree.delete(*self.dissector_tree.get_children())
127 data = self._packet_data_cache.get(sel[0])
128 if data:
129 PTPDissectorEngine.populate(self.dissector_tree, "", data)
130 if self.meter_panel: self.meter_panel.update(data.get("timestamp", 0))
