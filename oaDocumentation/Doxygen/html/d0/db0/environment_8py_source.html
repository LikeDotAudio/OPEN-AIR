14def verify_rust_pipeline(project_root, log):
15 """Build the centralised Rust core in-place (develop mode) if it exists."""
16 rust_core_dir = Path(project_root) / "oaRustCore"
17 if not rust_core_dir.exists():
18 return
19 log("🏗️ [NATIVE] Verifying high-performance Rust core...")
20 try:
21 env = os.environ.copy()
22 env["PIP_BREAK_SYSTEM_PACKAGES"] = "1"
23 subprocess.check_call(
24 [sys.executable, "-m", "maturin", "develop"],
25 cwd=str(rust_core_dir), env=env,
26 stdout=subprocess.DEVNULL, stderr=subprocess.STDOUT,
27 )
28 log("✨ [NATIVE] Rust pipeline verified and active.")
29 except Exception as e:
30 log(f"⚠️ [WARNING] Rust build check failed: {e}. "
31 f"System will attempt to run with graceful fallbacks.")

34def resolve_partition_scripts(project_root, log):
35 """Return (core_script, ui_script, web_script) as strings.
36
37 Core and UI are required — missing either is fatal. Web is optional; the
38 returned web_script may be None if frontEnd/Entry.py is absent.
39 """
40 project_root = Path(project_root)
41 core_script = project_root / "oaComBroker" / "Core" / "open_air_core.py"
42 ui_script = project_root / "oaGui" / "Managers" / "orchestration" / "loader_main_service.py"
43 web_script = project_root / "frontEnd" / "Entry.py"
44
45 if not core_script.exists():
46 log(f"🛑 CRITICAL FAILURE: Core script not found at {core_script}")
47 sys.exit(1)
48 if not ui_script.exists():
49 log(f"🛑 CRITICAL FAILURE: UI script not found at {ui_script}")
50 sys.exit(1)
51 if not web_script.exists():
52 log(f"⚠️ [WARNING] Web interface launcher not found at {web_script}. "
53 f"Skipping web UI.")
54 web_script = None
55
56 return (str(core_script), str(ui_script), str(web_script) if web_script else None)

64def build_partition_environments(project_root, log):
65 """Build (session_guid, child_env, core_env, ui_env) with the GUID injected."""
66 session_guid = _new_session_guid()
67 log(f"Session Identity established (Randomized): {session_guid}")
68
69 child_env = os.environ.copy()
70 child_env["OPEN_AIR_INSTANCE_GUID"] = session_guid
71 child_env["PYTHONPATH"] = os.pathsep.join(

73 )
74
75 core_env = child_env.copy()
76 core_env["OPEN_AIR_PARTITION_ID"] = "CORE"
