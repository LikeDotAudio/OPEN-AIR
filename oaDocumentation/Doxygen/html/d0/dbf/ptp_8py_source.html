76 def _on_external_data(self, message: MqttMessage):
77 """Bridge for PTP data received via MQTT (Core-to-UI)."""
78 payload = message.payload
79 if not payload: return
80
81 data = None
82 if isinstance(payload, (bytes, str)):
83 stripped = payload.strip() if isinstance(payload, str) else payload.strip()
84 if stripped and (stripped[0] in (ord('{'), ord('[')) if isinstance(stripped, bytes) else stripped[0] in ('{', '[')):
85 data = orjson.loads(payload)
86 else:
87 data = payload
88
89 if not data: return
90
91 if isinstance(data.get("message_type"), int):
92 data["message_type"] = PTPPacketParser.MSG_TYPES.get(data["message_type"], f"Unknown ({data['message_type']})")
93 PTPObserverRegistry.notify(data)

95 def _run_sniffer(self):
96 """Background loop for raw packet capture."""

102 def _process_packet(self, pkt):
103 """Dissects raw packets and distributes data."""
104 if not pkt or not pkt.haslayer(UDP): return
105
106 ptp_layer = None
107 if pkt.haslayer(PTP):
108 ptp_layer = pkt[PTP]
109 else:
110 payload = bytes(pkt[UDP].payload)
111 if len(payload) >= 34:
112 ptp_layer = PTP(payload)
113
114 if not ptp_layer: return
115
116 data = PTPPacketParser.tear_apart(pkt, ptp_layer)
117 self._handle_heartbeat(data)
118 PTPObserverRegistry.notify(data)

120 def _handle_heartbeat(self, data):
121 """Publishes activity status to MQTT at 1Hz."""
122 now = time.time()
123 if now - self.last_heartbeat >= self.heartbeat_interval:
124 self.last_heartbeat = now
125 if self.mqtt:
126 payload = {"status": "alive", "last_ptp_message": data["message_type"], "timestamp": data["timestamp"]}
127 self.mqtt.publish("OPEN-AIR/System/PTP/Heartbeat", orjson.dumps(payload).decode())
