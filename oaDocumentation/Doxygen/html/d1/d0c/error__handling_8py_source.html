31 """
32 Decorator for robust yet vocal error handling.
33
34 Lead with action: Wraps a function in a try-except block that logs
35 failures vocally and optionally displays them on a UI widget.
36
37 Args:
38 message (str): Contextual message to log on failure.
39 category (str): Subsystem category for the log.
40 fallback (Any): Value to return if the function fails (if re_raise is False).
41 re_raise (bool): Whether to re-propagate the exception after logging.
42 widget (Any): Optional Tkinter widget to display a 'Red Screen of Warning'.
43 """
44 def decorator(func: Callable):
45 @functools.wraps(func)
46 def wrapper(*args, **kwargs):
47 try:
48 return func(*args, **kwargs)
49 except Exception as e:
50 # 1. Capture forensic detail
51 tb = traceback.format_exc()
52 error_message = f"❌🔴 [FAILURE] {message} in '{func.__name__}': {e}"
53
54 # 2. Vocal Logging
55 FAILURE_LOGGER.bind(category=f"{get_emoji('FAILURE')} {category}").error(
56 f"{error_message}\n\n🕵️ FORENSIC TRACE:\n{tb}"
57 )
58
59 # 3. Visual 'Red Screen of Warning' (if widget provided)
60 if widget and hasattr(widget, 'config') and widget.winfo_exists():
61 try:
62 widget.config(
63 text=f"❌ CRITICAL FAILURE\n\n{message}\n{e}",
64 foreground="red",
65 font=("Consolas", 10, "bold"),
66 justify="center"
67 )
68 except:
69 pass # Don't let the warning trigger itself fail
70
71 if re_raise:
72 raise e
73 return fallback
74 return wrapper
75 return decorator
