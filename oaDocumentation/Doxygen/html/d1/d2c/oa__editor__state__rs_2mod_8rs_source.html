43 fn update_state(&self, path: Vec<String>, new_json_str: String) -> PyResult<()> {
44 let new_val: Value = serde_json::from_str(&new_json_str).unwrap_or_else(|_| serde_json::json!(null));
45 let mut data = self.data.write();
46
47 if path.is_empty() {
48 *data = new_val;
49 return Ok(());
50 }
51
52 let mut current = &mut *data;
53
54 for (i, key) in path.iter().enumerate() {
55 if i == path.len() - 1 {
56 // We are at the end, insert or replace
57 if let Some(obj) = current.as_object_mut() {
58 obj.insert(key.clone(), new_val.clone());
59 } else {
60 // Not an object, we need to convert it to one
61 *current = serde_json::json!({});
62 current.as_object_mut().unwrap().insert(key.clone(), new_val.clone());
63 }
64 } else {
65 // Navigate deeper, creating objects if necessary
66 if !current.is_object() {
67 *current = serde_json::json!({});
68 }
69
70 let obj = current.as_object_mut().unwrap();
71 if !obj.contains_key(key) {
72 obj.insert(key.clone(), serde_json::json!({}));
73 }
74
75 // Rust borrow checker requires this dance
76 current = current.get_mut(key).unwrap();
77 }
78 }
79
80 Ok(())
81 }
