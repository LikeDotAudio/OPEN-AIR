17# Description:
18# The 'Wall of Pity' is a detailed forensic report designed to provide
19# deep context on performance bottlenecks. Unlike the concise Wall of Shame,
20# it provides descriptive narratives for each category of offender, helping
21# developers understand *why* a specific architectural pattern is heavy.

24# - Forensic Narrator: Translates raw metrics into architectural insights.

28import html
29import os
30
31# --- Architectural Limits ---
32MAX_OFFENDERS_PER_CATEGORY = 50
33MAX_SUBSYSTEM_BREAKDOWN_ITEMS = 20

35def generate_wall_of_pity(performance_stats, ps):
36 """
37 Generates a highly detailed, descriptive performance report.
38 Designed to provide a 'blind coder' with full context on architectural bottlenecks.
39 The 'Wall of Pitty' provides deep empathy for the processor's struggles.
40 """
41 stats = ps.stats
42 vroot = ("<virtual>", 0, "total_execution")
43
44 # 1. Pre-process rich caller and subsystem attribution
45 func_metadata = {}
46 for func, (cc, nc, tt, ct, callers) in stats.items():
47 if func == vroot: continue
48
49 # Sort callers by who triggered this function the most
50 sorted_callers = sorted(callers.items(), key=lambda x: x[1][1], reverse=True)
51 top_callers = []
52 for c_func, c_stats in sorted_callers[:3]:
53 # Format caller name
54 c_name = c_func[2]
55 # Handle class methods or odd pstats names
56 if c_name.startswith("<") and c_name.endswith(">"):
57 # Clean up <method 'name' ...>
58 if "method" in c_name and "'" in c_name:
59 c_name = c_name.split("'")[1]
60
61 c_file = os.path.basename(c_func[0])
62 if c_func[0] == "~":
63 c_label = f"built-in:{c_name}"
64 else:
65 c_label = f"{c_name} (in {c_file}:{c_func[1]})"
66
67 # Calculate what percentage of total calls this caller is responsible for
68 pct = (c_stats[1] / nc * 100) if nc > 0 else 0
69 top_callers.append(f"{c_label} [{pct:.1f}%]")
70
71 # Escape the full path to avoid any weird chars
72 safe_path = html.escape(f"{func[0]}:{func[1]}")
73
74 func_metadata[func] = {
75 "caller_report": " | TRIGGERED BY: " + ", ".join(top_callers) if top_callers else " | TRIGGERED BY: <entry point>",
76 "full_path": safe_path
77 }
78
79 # 2. Define Descriptive Categories
80 categories = [
81 {
82 "id": "BLOCKING_PATHS",
83 "title": "CATEGORY 1: THE BLOCKERS (Cumulative Time Bottlenecks)",
84 "description": "These paths represent the heaviest execution branches. Even if the function itself is fast, its children are keeping the CPU occupied for significant durations. Fixing these usually requires architectural changes to how data flows through the system.",
85 "sort_key": lambda x: x['cumtime'],
86 "metric_label": "Total Time Held",
87 "metric_format": lambda x: f"{x['cumtime']:.4f}s"
88 },
89 {
90 "id": "CPU_INTENSIVE",
91 "title": "CATEGORY 2: THE GRINDERS (Self-Time / Hotspots)",
92 "description": "These functions are CPU 'hotspots'. The time is spent inside the function body itself, not waiting for others. This is the 'Wall of Shame' for inefficient loops, heavy regex, or unoptimized logic. Focus here for micro-optimizations.",
93 "sort_key": lambda x: x['tottime'],
94 "metric_label": "Internal Grind",
95 "metric_format": lambda x: f"{x['tottime']:.4f}s"
96 },
97 {
98 "id": "UPDATE_SPAM",
99 "title": "CATEGORY 3: THE SPAMMERS (High Frequency / Death by 1000 Cuts)",
100 "description": "Redundant logic and 'update spam'. These functions are called thousands of times. Even a 1ms execution becomes a disaster when triggered in a tight loop or every UI frame. Ask yourself: 'Do I really need to call this that often?'",
101 "sort_key": lambda x: x['ncalls'],
102 "metric_label": "Call Volume",
103 "metric_format": lambda x: f"{x['ncalls']} calls"
104 },
105 {
106 "id": "LATENCY_WEIGHT",
107 "title": "CATEGORY 4: THE SLUGGARDS (Slowest Single Execution)",
108 "description": "These functions represent latency spikes. A single call here is heavy enough to cause a visible stutter or a network timeout. This is often indicative of blocking I/O, heavy file reads, or synchronous network requests that should be async.",
109 "sort_key": lambda x: x['per_call'],
110 "metric_label": "Worst Latency",
111 "metric_format": lambda x: f"{x['per_call']:.6f}s/call"
112 }
113 ]
114
115 # 3. Build the descriptive report
116 report = [
117 "===================================================================================",
118 " OPEN-AIR PERFORMANCE WALL OF PITTY (INTELLIGENCE REPORT) ",
119 "===================================================================================",
120 "This report identifies the most significant architectural and logic-level offenders",
121 "discovered during the profiling session. Use the details below to trace exactly",
122 "who is responsible for the performance degradation.",
123 ""
124 ]
125
126 for cat in categories:
127 report.append(f"\n>>> {cat['title']}")
128 report.append("-" * len(cat['title']))
129 report.append(f"CONTEXT: {cat['description']}")
130 report.append("")
131
132 # USE CONSTANT
133 by_metric = sorted(performance_stats, key=cat['sort_key'], reverse=True)[:MAX_OFFENDERS_PER_CATEGORY]
134
135 for i, s in enumerate(by_metric, 1):
136 # Safe Fallback for Metadata
137 raw_key = s.get('raw_key')
138 if raw_key and raw_key in func_metadata:
139 meta = func_metadata[raw_key]
140 else:
141 meta = {
142 "full_path": html.escape(f"{s.get('filename', 'unknown')}:{s.get('lineno', '?')}"),
143 "caller_report": " | TRIGGERED BY: (Unknown Caller Context)"
144 }
145
146 subsystems = f" [Subsystems: {', '.join(s['roots'])}]" if s['roots'] else ""
147
148 # Sanitize Function Name for HTML Display
149 raw_name = s['funcname']
150 if raw_name.startswith("<") and raw_name.endswith(">"):
151 if "method" in raw_name and "'" in raw_name:
152 raw_name = raw_name.split("'")[1]
153
154 safe_funcname = html.escape(raw_name)
155 safe_caller_report = html.escape(meta['caller_report'])
156
157 line_1 = f" {i:2}. {cat['metric_label']}: {cat['metric_format'](s)} | FUNCTION: {safe_funcname}{subsystems}"
158 line_2 = f" LOCATION: {meta['full_path']}"
159 line_3 = f" DETAIL : {safe_caller_report}"
160
161 report.append(line_1)
162 report.append(line_2)
163 report.append(line_3)
164 report.append("")
165
166 # --- CATEGORY 5: SUBSYSTEM BREAKDOWN ---
167 report.append(f"\n>>> CATEGORY 5: SUBSYSTEM BREAKDOWN (Worst {MAX_SUBSYSTEM_BREAKDOWN_ITEMS} per Root Process)")
168 report.append("-" * 65)
169 report.append("CONTEXT: Performance offenders grouped by their origin process or root subsystem. This identifies which 'partition' of the app is causing the most cumulative lag.")
170 report.append("")
171
172 # Extract unique roots
173 all_roots = sorted(list(set(r for s in performance_stats for r in s['roots'])))
174 for root in all_roots:
175 report.append(f" [ ROOT SUBSYSTEM: {root} ]")
176 root_stats = [s for s in performance_stats if root in s['roots']]
177 root_offenders = sorted(root_stats, key=lambda x: x['cumtime'], reverse=True)[:MAX_SUBSYSTEM_BREAKDOWN_ITEMS]
178
179 for i, s in enumerate(root_offenders, 1):
180 raw_name = s['funcname']
181 if raw_name.startswith("<") and raw_name.endswith(">"):
182 if "method" in raw_name and "'" in raw_name:
183 raw_name = raw_name.split("'")[1]
184 safe_funcname = html.escape(raw_name)
185 report.append(f" {i:2}. {s['cumtime']:8.4f}s cum | {s['tottime']:8.4f}s self | {safe_funcname}")
186 report.append("")
187
188 # --- CATEGORY 6: THE FRAMEWORK OVERHEAD (Built-ins) ---
189 report.append(f"\n>>> CATEGORY 6: THE FRAMEWORK OVERHEAD (Top {MAX_SUBSYSTEM_BREAKDOWN_ITEMS} Built-in Spammers)")
190 report.append("-" * 65)
191 report.append("CONTEXT: These are internal Python built-ins or methods triggered by the app. High volume here often indicates overhead from libraries, debuggers, or inefficient use of language features (e.g., constant isinstance checks or lock contention).")
192 report.append("")
193
194 builtins = [s for s in performance_stats if s['filename'] == "~"]
195 builtin_spammers = sorted(builtins, key=lambda x: x['ncalls'], reverse=True)[:MAX_SUBSYSTEM_BREAKDOWN_ITEMS]
196 for i, s in enumerate(builtin_spammers, 1):
197 report.append(f" {i:2}. {s['ncalls']:10} calls | {s['tottime']:8.4f}s self | {html.escape(s['funcname'])}")
198
199 report.append("\n===================================================================================")
200 report.append("END OF PITTY REPORT - OPTIMIZE ACCORDINGLY")
201 report.append("===================================================================================")
202
203 return "\n".join(report)
