17# Description:
18# This module implements "Parameter Settling" and "Interaction Locking."
19# These mechanisms prevent feedback loops and jitter when a user is actively
20# modifying a parameter on a physical control surface or UI. It ensures that
21# the system reaches a stable terminal state after a burst of activity.
22#
23# Architectural Role:
24# - Prevents self-reflection loops at the router level.
25# - Implements a temporal "dead-zone" during active user interaction.
26# - Triggers final state synchronization (LINK_FEEDBACK) after silence.

35class SettleManager:
36 """
37 Manages interaction locks and terminal state 'settling' for topics.
38
39 The SettleManager tracks which parameters are currently being manipulated
40 by which instance. It uses threading.Timer objects to detect periods of
41 inactivity and broadcast final "settled" messages.
42 """
43 def __init__(self, ingest_callback):
44 """
45 Initializes the SettleManager with a reference to the router's ingest.
46
47 Args:
48 ingest_callback (fn): The router.ingest function for re-injection.
49 """
50 self._ingest_callback = ingest_callback
51 self._settle_timers = {}
52 self._settle_lock = threading.Lock()
53 self._locked_params_by_instance = {} # instance_id -> set(topics)
54
55 def is_parameter_locked(self, topic, instance_id=None):
56 """
57 Checks if a topic is currently locked, globally or for an instance.
58
59 Args:
60 topic (str): The logical address to check.
61 instance_id (str, optional): The GUID of a specific instance.
62
63 Returns:
64 bool: True if locked, False otherwise.
65 """
66 with self._settle_lock:
67 if instance_id:
68 locked_set = self._locked_params_by_instance.get(instance_id, set())
69 return topic in locked_set
70 else:
71 for locked_set in self._locked_params_by_instance.values():
72 if topic in locked_set:
73 return True
74 return False
75
76 def lock_parameter(self, topic, instance_id):
77 """
78 Engages an interaction lock for a specific topic/instance.
79
80 Args:
81 topic (str): The parameter to lock.
82 instance_id (str): The GUID of the instance claiming the lock.
83 """
84 with self._settle_lock:
85 if instance_id not in self._locked_params_by_instance:
86 self._locked_params_by_instance[instance_id] = set()
87 self._locked_params_by_instance[instance_id].add(topic)
88
89 def unlock_parameter(self, topic, instance_id):
90 """
91 Releases an interaction lock.
92
93 Args:
94 topic (str): The parameter to unlock.
95 instance_id (str): The GUID of the instance releasing the lock.
96 """
97 with self._settle_lock:
98 locked_set = self._locked_params_by_instance.get(instance_id)
99 if locked_set and topic in locked_set:
100 locked_set.remove(topic)
101 if not locked_set:
102 del self._locked_params_by_instance[instance_id]

104 def schedule_settling(self, topic, original_message):
105 """
106 Schedules a terminal LINK_FEEDBACK after a period of silence.
107
108 This method is called for every SPLICE_ACTION. It resets a 50ms
109 timer; if the timer expires without being reset, a terminal
110 message is broadcast to confirm the final parameter state.
111
112 Args:
113 topic (str): The parameter to settle.
114 original_message (dict): The last message received for this topic.
115 """
116 # Atomically cancel any existing timer for this specific topic.
117 with self._settle_lock:
118 if topic in self._settle_timers:
119 self._settle_timers[topic].cancel()
120
121 # Prepare the final settling callback task.
122 def _fire_settled():
123 settled_meta = original_message["meta"].copy()
124 settled_meta["message_type"] = "LINK_FEEDBACK"
125 settled_meta["is_settled"] = True
126
127 if app_constants.ROUTER_SETTLE_LOGS:
128 matrix_log("comms", "broker", "schedule_settling", f"⏳⏳🔄 [ROUTER] Settling: Firing final LINK_FEEDBACK for {topic}", "DEBUG")
129
130 # Unlock the parameter before re-ingesting to allow the feedback
131 # packet to pass through the router's filters.
132 self.unlock_parameter(topic, original_message.get("full_id"))
133
134 # Re-ingest the message as "settled" to inform all network spokes.
135 self._ingest_callback("SYSTEM", topic, original_message["value"], settled_meta)
136
137 # Cleanup the internal timer reference.
138 with self._settle_lock:
139 if self._settle_timers.get(topic) == t:
140 del self._settle_timers[topic]
141
142 # 50ms of silence is required to consider an interaction "settled."
143 t = threading.Timer(0.050, _fire_settled)
144
145 with self._settle_lock:
146 self._settle_timers[topic] = t
147
148 t.start()
