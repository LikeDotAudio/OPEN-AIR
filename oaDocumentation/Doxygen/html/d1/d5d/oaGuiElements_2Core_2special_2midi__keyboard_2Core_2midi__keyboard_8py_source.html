14# Resistor Color Code Mapping & MIDI Channel Logic
15RESISTOR_COLORS = {
16 0: "#000000", # Black (used for off/bg)
17 1: "#8B4513", # Brown
18 2: "#FF0000", # Red
19 3: "#FF8C00", # Orange
20 4: "#FFFF00", # Yellow
21 5: "#00FF00", # Green
22 6: "#0000FF", # Blue
23 7: "#EE82EE", # Violet
24 8: "#808080", # Gray
25 9: "#FFFFFF" # White
26}
27
28# ROYGBIVW sequence for wrapping
29WRAP_COLORS = [
30 "#FF0000", # Red
31 "#FF8C00", # Orange
32 "#FFFF00", # Yellow
33 "#00FF00", # Green
34 "#0000FF", # Blue
35 "#4B0082", # Indigo
36 "#EE82EE", # Violet
37 "#FFFFFF" # White
38]

40def get_midi_color(channel):
41 """Returns the color based on MIDI channel (0-15 typical in mido)."""
42 # Map to 1-indexed for the resistor color logic
43 ch = channel + 1
44
45 if 1 <= ch <= 9:
46 return RESISTOR_COLORS.get(ch, "#FFFFFF")
47 else:
48 # Wrap around starting from Red
49 idx = (ch - 10) % len(WRAP_COLORS)
50 return WRAP_COLORS[idx]

87 def _setup_keys(self):
88 self.delete("all")
89 self.key_map.clear()
90 self.id_to_note.clear()
91
92 # Determine dimensions
93 num_white_keys = (self.num_octaves * 7) + 1
94 kw = self.width / num_white_keys if num_white_keys > 0 else 10
95 kh = self.height
96
97 # 1. Draw White Keys first (Bottom Layer)
98 current_x = 0
99 for i in range(self.num_keys):
100 note = self.start_note + i
101 note_in_octave = note % 12
102 is_black = note_in_octave in [1, 3, 6, 8, 10]
103
104 if not is_black:
105 rect = self.create_rectangle(
106 current_x, 0, current_x + kw, kh,
107 fill="#ffffff", outline="#333333", tags=("white", f"note_{note}")
108 )
109 self.key_map[note] = rect
110 self.id_to_note[rect] = note
111 current_x += kw
112
113 # 2. Draw Black Keys (Top Layer)
114 current_x = 0
115 for i in range(self.num_keys):
116 note = self.start_note + i
117 note_in_octave = note % 12
118 is_black = note_in_octave in [1, 3, 6, 8, 10]
119
120 if is_black:
121 # Black keys are positioned between white keys
122 bx = current_x - (kw * 0.3)
123 rect = self.create_rectangle(
124 bx, 0, bx + (kw * 0.6), kh * 0.6,
125 fill="#000000", outline="#333333", tags=("black", f"note_{note}")
126 )
127 self.key_map[note] = rect
128 self.id_to_note[rect] = note
129 else:
130 current_x += kw

139 def _on_mouse_press(self, event):
140 items = self.find_closest(event.x, event.y)
141 if not items: return
142 item = items[0]
143 note = self.id_to_note.get(item)
144 if note and note not in self.active_mouse_notes:
145 self.active_mouse_notes.add(note)
146 if self.on_note_on:
147 self.on_note_on(note)
148 # Visual feedback is handled via handle_midi if looped back,
149 # or we can force it here for immediate response.
150 self.note_on(note, "#00ffff") # Cyan for mouse-over/press

152 def _on_mouse_release(self, event):
153 for note in list(self.active_mouse_notes):
154 if self.on_note_off:
155 self.on_note_off(note)
156 self.note_off(note)
157 self.active_mouse_notes.clear()

159 def _on_mouse_motion(self, event):
160 items = self.find_closest(event.x, event.y)
161 if not items: return
162 item = items[0]
163 note = self.id_to_note.get(item)
164
165 if note and note not in self.active_mouse_notes:
166 # Release old notes
167 for old_note in list(self.active_mouse_notes):
168 if self.on_note_off:
169 self.on_note_off(old_note)
170 self.note_off(old_note)
171 self.active_mouse_notes.clear()
172
173 # Press new note
174 self.active_mouse_notes.add(note)
175 if self.on_note_on:
176 self.on_note_on(note)
177 self.note_on(note, "#00ffff")

181 def handle_midi(self, message):
182 """Processes a mido message or a dict mirror."""
183 try:
184 m_type = ""
185 channel = 0
186 note = 0
187 velocity = 0
188
189 matrix_log("comms", "midi", "handle_midi", f"🎹 [KEYBOARD] Incoming MIDI: {message}", "DEBUG")
190
191 if isinstance(message, dict):
192 channel = message.get("channel", 0)
193 note = message.get("note", 0)
194 velocity = message.get("velocity", message.get("value", 0))
195 # Fallback to parsing 'raw' if fields are missing
196 if note == 0 and "raw" in message:
197 raw = message["raw"]
198 m_type = "note_on" if "note_on" in raw else "note_off"
199 # Simple regex-like extraction
200 import re
201 n_match = re.search(r"note=(\d+)", raw)
202 if n_match: note = int(n_match.group(1))
203 c_match = re.search(r"channel=(\d+)", raw)
204 if c_match: channel = int(c_match.group(1))
205 v_match = re.search(r"velocity=(\d+)", raw)
206 if v_match: velocity = int(v_match.group(1))
207 else:
208 m_type = message.get("type", "note_on" if velocity > 0 else "note_off")
209 else:
210 m_type = message.type
211 channel = message.channel if hasattr(message, "channel") else 0
212 note = message.note if hasattr(message, "note") else 0
213 velocity = message.velocity if hasattr(message, "velocity") else 0
214
215 if m_type in ["note_on", "note_off"]:
216 color = get_midi_color(channel)
217 matrix_log("comms", "midi", "handle_midi", f"🎹 [KEYBOARD] Dispatching: type={m_type}, note={note}, channel={channel}, color={color}", "DEBUG")
218 if m_type == "note_on" and velocity > 0:
219 self.note_on(note, color)
220 else:
221 self.note_off(note)
222 except Exception as e:
223 matrix_log("comms", "midi", "handle_midi", f"🎹 ❌ Keyboard Visualizer Error: {e}", "ERROR")
224 logger.error(f"🎹 Keyboard Visualizer Error: {e}")
