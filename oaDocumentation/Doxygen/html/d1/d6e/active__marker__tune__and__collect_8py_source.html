94 def _is_payload_true(self, payload) -> bool:
95 """Unifies JSON and raw-string truth evaluation for MQTT payloads."""
96 if not payload: return False
97
98 # 1. Attempt JSON parsing (primary)
99 try:
100 data = orjson.loads(payload)
101 if isinstance(data, dict):
102 return str(data.get("value", "")).lower() == "true"
103 except (orjson.JSONDecodeError, TypeError, ValueError):
104 pass
105
106 # 2. Fallback to raw string evaluation
107 raw_str = payload.decode("utf-8") if isinstance(payload, bytes) else str(payload)
108 return raw_str.lower() == "true"
