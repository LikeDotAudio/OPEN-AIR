20 def _create_zone_buttons(self):
21 """Generates the top-level zone filter buttons."""
22 if not self.zone_frame: return
23 for w in self.zone_frame.winfo_children(): w.destroy()
24
25 zones = sorted(self.grouped_markers.keys())
26 for zone in zones:
27 style = "Selected.TButton" if self.selected_zone == zone else "Custom.TButton"
28 btn = ttk.Button(self.zone_frame, text=zone, style=style,
29 command=lambda z=zone: self.on_zone_toggle(z))
30 btn.pack(side=tk.LEFT, padx=2)

51 if not self.selected_zone or not self.selected_group: return
