116 """Real-time update from EventBus."""
117 ts = meta.get('timestamp', time.time()) if isinstance(meta, dict) else time.time()
118 self.after(0, lambda: self._update_node(topic, value, ts))

120 def _update_node(self, topic, value, timestamp):
121 """Inserts or updates a node in the hierarchical tree."""
122 if not self.winfo_exists(): return
123
124 parts = topic.split('/')
125 path = ""
126 parent_id = ""
127
128 ts_str = time.strftime("%H:%M:%S", time.localtime(timestamp))
129 val_str = str(value)[:100] + ("..." if len(str(value)) > 100 else "")
130
131 for i, part in enumerate(parts):
132 path = f"{path}/{part}" if path else part
133 if path not in self._nodes:
134 # Create node
135 is_last = (i == len(parts) - 1)
136 v = val_str if is_last else ""
137 t = ts_str if is_last else ""
138 self._nodes[path] = self.tree.insert(parent_id, "end", text=part, values=(v, t), open=True)
139 elif i == len(parts) - 1:
140 # Update existing leaf
141 self.tree.item(self._nodes[path], values=(val_str, ts_str))
142
143 parent_id = self._nodes[path]
