4// WASM port of the procedural panel + screw background engine. Mirrors the
5// Python pipeline (oaGuiElements/Core/panels/panel_generator.py + layer_* +
6// panel_screw/screw_generator.py) and the Rust pattern engine
7// (oaRustCore_pkg/src/oa_pattern_engine_rs), composited entirely in Rust so the
8// browser only has to blit one finished RGBA buffer to a <canvas>.
9//
10// Two entry points are exported to JS:
11// generate_panel(width, height, config_json) -> RGBA bytes (w*h*4)
12// generate_screw(size, config_json) -> RGBA bytes (canvas_dim^2*4)
13// Everything is deterministic (seeded) — there is NO animation and NO I/O.

23const NORM: f64 = 127.5; // perlin (-1..1) -> (0..255)

25// ----------------------------------------------------------------------------
26// Small RGBA buffer with the blend ops the pipeline needs. Stored row-major as
27// [r,g,b,a, r,g,b,a, ...]; the canvas consumes it directly as ImageData.
28// ----------------------------------------------------------------------------
29struct Buf {
30 w: u32,
31 h: u32,
32 px: Vec<u8>,
33}
34
35impl Buf {
36 fn filled(w: u32, h: u32, rgba: [u8; 4]) -> Self {
37 let mut px = Vec::with_capacity((w * h * 4) as usize);
38 for _ in 0..(w * h) {
39 px.extend_from_slice(&rgba);
40 }
41 Buf { w, h, px }
42 }
43
44 fn transparent(w: u32, h: u32) -> Self {
45 Buf { w, h, px: vec![0u8; (w * h * 4) as usize] }
46 }
47
48 #[inline]
49 fn idx(&self, x: u32, y: u32) -> usize {
50 ((y * self.w + x) * 4) as usize
51 }
52
53 #[inline]
54 fn in_bounds(&self, x: i32, y: i32) -> bool {
55 x >= 0 && y >= 0 && (x as u32) < self.w && (y as u32) < self.h
56 }
57
58 // Multiply RGB by a single-channel (grayscale) plane / 255. Used for every
59 // "texture" / shading layer (brushed, hammered, vignette, gradient).
60 fn multiply_gray(&mut self, gray: &[u8]) {
61 for i in 0..(self.w * self.h) as usize {
62 let g = gray[i] as u16;
63 let p = i * 4;
64 self.px[p] = ((self.px[p] as u16 * g) / 255) as u8;
65 self.px[p + 1] = ((self.px[p + 1] as u16 * g) / 255) as u8;
66 self.px[p + 2] = ((self.px[p + 2] as u16 * g) / 255) as u8;
67 }
68 }
69
70 // Source-over alpha composite of a single RGBA pixel. Background stays opaque.
71 #[inline]
72 fn blend_over(&mut self, x: i32, y: i32, s: [u8; 4]) {
73 if !self.in_bounds(x, y) {
74 return;
75 }
76 let a = s[3] as f32 / 255.0;
77 if a <= 0.0 {
78 return;
79 }
80 let p = self.idx(x as u32, y as u32);
81 for c in 0..3 {
82 self.px[p + c] = (s[c] as f32 * a + self.px[p + c] as f32 * (1.0 - a)) as u8;
83 }
84 // Keep alpha at least as opaque as it was (panel base is opaque).
85 let na = a + (self.px[p + 3] as f32 / 255.0) * (1.0 - a);
86 self.px[p + 3] = (na * 255.0).min(255.0) as u8;
87 }
88
89 // Composite an entire RGBA sub-buffer with its top-left at (ox, oy).
90 fn composite(&mut self, src: &Buf, ox: i32, oy: i32) {
91 for sy in 0..src.h {
92 for sx in 0..src.w {
93 let sp = src.idx(sx, sy);
94 self.blend_over(
95 ox + sx as i32,
96 oy + sy as i32,
97 [src.px[sp], src.px[sp + 1], src.px[sp + 2], src.px[sp + 3]],
98 );
99 }
100 }
101 }
102
103 // Bresenham line of a flat RGBA color (alpha-composited).
104 fn line(&mut self, x1: f64, y1: f64, x2: f64, y2: f64, color: [u8; 4]) {
105 let mut x = x1.round() as i32;
106 let mut y = y1.round() as i32;
107 let x2i = x2.round() as i32;
108 let y2i = y2.round() as i32;
109 let dx = (x2i - x).abs();
110 let dy = (y2i - y).abs();
111 let sx = if x < x2i { 1 } else { -1 };
112 let sy = if y < y2i { 1 } else { -1 };
113 let mut err = dx - dy;
114 let mut guard = 0;
115 loop {
116 self.blend_over(x, y, color);
117 if x == x2i && y == y2i {
118 break;
119 }
120 let e2 = 2 * err;
121 if e2 > -dy {
122 err -= dy;
123 x += sx;
124 }
125 if e2 < dx {
126 err += dx;
127 y += sy;
128 }
129 guard += 1;
130 if guard > 20000 {
131 break;
132 }
133 }
134 }

136 // Gaussian blur the whole RGBA buffer in place (global blur).
137 fn blur_rgba(&mut self, sigma: f32) {
138 if sigma <= 0.0 {
139 return;
140 }
141 if let Some(img) = RgbaImage::from_raw(self.w, self.h, self.px.clone()) {
142 self.px = imageops::blur(&img, sigma).into_raw();
143 }
144 }
145
146 fn into_pixels(self) -> Vec<u8> {
147 self.px
148 }

171fn hex_to_rgb(hex: &str) -> [u8; 3] {
172 let h = hex.trim_start_matches('#');
173 if h.len() != 6 {
174 return [128, 128, 128];
175 }
176 [
177 u8::from_str_radix(&h[0..2], 16).unwrap_or(128),
178 u8::from_str_radix(&h[2..4], 16).unwrap_or(128),
179 u8::from_str_radix(&h[4..6], 16).unwrap_or(128),
180 ]
181}

183// ----------------------------------------------------------------------------
184// Grayscale texture generators (ported from the Rust pattern engine).
185// ----------------------------------------------------------------------------
186fn streaks_gray(w: u32, h: u32, vertical: bool, sigma: f64, seed: u32) -> Vec<u8> {
187 let perlin = Perlin::new(seed);
188 let (sw, sh) = if vertical {
189 (w, (h / 100).max(5))
190 } else {
191 ((w / 100).max(5), h)
192 };
193 let mut small: GrayImage = ImageBuffer::new(sw, sh);
194 for (x, y, p) in small.enumerate_pixels_mut() {
195 let v = perlin.get([x as f64 * sigma * 0.01, y as f64 * sigma * 0.01]);
196 *p = Luma([((v + 1.0) * NORM) as u8]);
197 }
198 imageops::resize(&small, w, h, imageops::FilterType::Lanczos3).into_raw()
199}

201fn hammered_gray(w: u32, h: u32, seed: u32) -> Vec<u8> {
202 let base = Perlin::new(seed);
203 let dimple = Perlin::new(seed.wrapping_add(1));
204 let mut out = vec![0u8; (w * h) as usize];
205 for y in 0..h {
206 for x in 0..w {
207 let b = base.get([x as f64 * 0.1, y as f64 * 0.1]);
208 let d = dimple.get([x as f64 * 0.02, y as f64 * 0.02]);
209 let c = (b * 0.7 + d * 0.3 + 1.0) * NORM;
210 out[(y * w + x) as usize] = c.clamp(0.0, 255.0) as u8;
211 }
212 }
213 out
214}

216// Multiply two grayscale planes (crosshatch weave).
217fn mul_planes(a: &[u8], b: &[u8]) -> Vec<u8> {
218 a.iter()
219 .zip(b.iter())
220 .map(|(&x, &y)| ((x as u16 * y as u16) / 255) as u8)
221 .collect()
222}

227#[wasm_bindgen]
228pub fn generate_panel(width: u32, height: u32, config_json: &str) -> Vec<u8> {
229 let w = width.max(1);
230 let h = height.max(1);
231
232 let cfg: Value = serde_json::from_str(config_json).unwrap_or(Value::Null);
233 // Prefer the "parameters" sub-block (matches DEFAULT_PANEL_CONFIG); else the
234 // top-level object.
235 let settings = cfg.get("parameters").cloned().unwrap_or(cfg.clone());
236 let settings = if settings.is_object() { settings } else { Value::Object(Default::default()) };
237
238 let seed = u32_of(&settings, "random_seed", 304);
239 let mut rng = StdRng::seed_from_u64(seed as u64);
240
241 // ---- Layer 1: substrate ----
242 let base = obj(&settings, "base_material").cloned().unwrap_or(Value::Null);
243 let sub_rgb = hex_to_rgb(&str_of(&base, "color", "#2a2a2a"));
244 let mut panel = Buf::filled(w, h, [sub_rgb[0], sub_rgb[1], sub_rgb[2], 255]);
245
246 let texture = str_of(&base, "texture_type", "flat");
247 let vertical = str_of(&base, "grain_direction", "horizontal") == "vertical";
248 match texture.as_str() {
249 "flat" => {}
250 "hammered" => {
251 let g = hammered_gray(w, h, seed);
252 panel.multiply_gray(&g);
253 }
254 "crosshatch" => {
255 let hori = streaks_gray(w, h, false, 10.0, seed);
256 let vert = streaks_gray(w, h, true, 10.0, seed.wrapping_add(7));
257 panel.multiply_gray(&mul_planes(&hori, &vert));
258 }
259 "brushed" => {
260 let g = streaks_gray(w, h, vertical, 20.0, seed);
261 panel.multiply_gray(&g);
262 }
263 "wrinkle" => {
264 // High-contrast two-octave crinkle (≈ PIL effect_noise multiply).
265 let p = Perlin::new(seed.wrapping_add(11));
266 let mut g = vec![0u8; (w * h) as usize];
267 for y in 0..h {
268 for x in 0..w {
269 let n = p.get([x as f64 * 0.08, y as f64 * 0.08]) * 0.6
270 + p.get([x as f64 * 0.21, y as f64 * 0.21]) * 0.4;
271 let v = (0.78 + 0.22 * n).clamp(0.0, 1.0);
272 g[(y * w + x) as usize] = (v * 255.0) as u8;
273 }
274 }
275 panel.multiply_gray(&g);
276 }
277 "enamel" => {
278 // Gentle low-frequency sheen (≈ PIL soft_light peel).
279 let p = Perlin::new(seed.wrapping_add(23));
280 let mut g = vec![0u8; (w * h) as usize];
281 for y in 0..h {
282 for x in 0..w {
283 let n = p.get([x as f64 * 0.03, y as f64 * 0.03]);
284 let v = (0.92 + 0.08 * n).clamp(0.0, 1.0);
285 g[(y * w + x) as usize] = (v * 255.0) as u8;
286 }
287 }
288 panel.multiply_gray(&g);
289 }
290 // any other named texture -> fine directional streaks.
291 _ => {
292 let g = streaks_gray(w, h, vertical, 5.0, seed);
293 panel.multiply_gray(&g);
294 }
295 }
296
297 // ---- Layer 2: paint + edge wear + lighting gradient ----
298 let paint = obj(&settings, "paint_layer").cloned().unwrap_or(Value::Null);
299 let edge = obj(&settings, "edge_wear").cloned().unwrap_or(Value::Null);
300 let scratches_cfg = obj(&settings, "panel_scratches")
301 .or_else(|| obj(&settings, "scratches"))
302 .cloned()
303 .unwrap_or(Value::Null);
304
305 // mask: 255 = paint shows, 0 = bare substrate.
306 let mut mask = vec![255u8; (w * h) as usize];
307
308 if bool_of(&edge, "enabled", false) {
309 let depth = f64_of(&edge, "scratch_depth", 30.0) as i32;
310 let count = (f64_of(&edge, "scratch_intensity", 0.5) * 50.0) as i32;
311 for _ in 0..count {
312 let edge_sel = rng.gen_range(0..4);
313 let (sx, sy, ex, ey) = match edge_sel {
314 0 => {
315 let x = rng.gen_range(0..w as i32);
316 (x, 0, x + rng.gen_range(-20..20), rng.gen_range(0..depth.max(1)))
317 }
318 1 => {
319 let x = rng.gen_range(0..w as i32);
320 (x, h as i32 - 1, x + rng.gen_range(-20..20), h as i32 - 1 - rng.gen_range(0..depth.max(1)))
321 }
322 2 => {
323 let y = rng.gen_range(0..h as i32);
324 (0, y, rng.gen_range(0..depth.max(1)), y + rng.gen_range(-20..20))
325 }
326 _ => {
327 let y = rng.gen_range(0..h as i32);
328 (w as i32 - 1, y, w as i32 - 1 - rng.gen_range(0..depth.max(1)), y + rng.gen_range(-20..20))
329 }
330 };
331 subtract_line(&mut mask, w, h, sx, sy, ex, ey);
332 }
333 }
334
335 // Deep substrate-revealing scratches.
336 let scount = u32_of(&scratches_cfg, "count", 0);
337 if scount > 0 && bool_of(&scratches_cfg, "reveals_substrate", false) {
338 let min_l = f64_of(&scratches_cfg, "min_length_px", 20.0);
339 let max_l = f64_of(&scratches_cfg, "max_length_px", 150.0).max(min_l + 1.0);
340 for _ in 0..scount {
341 let x1 = rng.gen_range(0.0..w as f64);
342 let y1 = rng.gen_range(0.0..h as f64);
343 let len = rng.gen_range(min_l..max_l);
344 let ang = rng.gen_range(0.0..(2.0 * PI));
345 subtract_line(&mut mask, w, h, x1 as i32, y1 as i32, (x1 + len * ang.cos()) as i32, (y1 + len * ang.sin()) as i32);
346 }
347 }
348
349 let opacity = f64_of(&paint, "opacity", 0.0).clamp(0.0, 1.0);
350 if opacity > 0.0 {
351 let pc = hex_to_rgb(&str_of(&paint, "color", "#4a5a6a"));
352 for i in 0..(w * h) as usize {
353 let m = mask[i] as f32 / 255.0; // how much paint applies here
354 let p = i * 4;
355 for c in 0..3 {
356 let substrate = panel.px[p + c] as f32;
357 let blended = substrate * (1.0 - opacity as f32) + pc[c] as f32 * opacity as f32;
358 panel.px[p + c] = (substrate * (1.0 - m) + blended * m) as u8;
359 }
360 }
361 }
362
363 // Lighting gradient (overhead): top brighter than bottom.
364 let grad = f64_of(&paint, "gradient_intensity", 0.2);
365 if grad > 0.0 {
366 let mut gray = vec![255u8; (w * h) as usize];
367 for y in 0..h {
368 let f = 1.0 - (y as f64 / h as f64) * (grad * 0.15);
369 let val = (255.0 * f) as u8;
370 for x in 0..w {
371 gray[(y * w + x) as usize] = val;
372 }
373 }
374 panel.multiply_gray(&gray);
375 }
376
377 // ---- Layer 3: studio haze (warm multiply) ----
378 let haze = obj(&settings, "studio_haze").cloned().unwrap_or(Value::Null);
379 if bool_of(&haze, "enabled", false) {
380 let inten = f64_of(&haze, "intensity", 0.15) as f32;
381 let warm = [180.0f32, 140.0, 50.0];
382 for i in 0..(w * h) as usize {
383 let p = i * 4;
384 for c in 0..3 {
385 let tint = 255.0 * (1.0 - inten) + warm[c] * inten;
386 panel.px[p + c] = ((panel.px[p + c] as f32 * tint) / 255.0) as u8;
387 }
388 }
389 }
390
391 // ---- Layer 4: rust oxidation ----
392 let rust = obj(&settings, "rust").cloned().unwrap_or(Value::Null);
393 if bool_of(&rust, "enabled", false) {
394 apply_rust(&mut panel, f64_of(&rust, "intensity", 0.5), &mut rng);
395 }
396
397 // ---- Layer 5: vignette edge fade ----
398 if bool_of(&edge, "enabled", false) {
399 let vig = f64_of(&edge, "vignette_intensity", 0.0);
400 if vig > 0.0 {
401 let depth = (f64_of(&edge, "fade_depth", 110.0) as u32).min(w.min(h) / 2).max(1);
402 panel.multiply_gray(&vignette_gray(w, h, vig, depth));
403 }
404 }
405
406 // ---- Layer 6: surface micro-scratches ----
407 if scount > 0 {
408 let inten = f64_of(&scratches_cfg, "intensity", 0.4);
409 let highlight = f64_of(&scratches_cfg, "depth_highlight", 0.5);
410 let sw = u32_of(&scratches_cfg, "width_px", 1);
411 let min_l = f64_of(&scratches_cfg, "min_length_px", 20.0);
412 let max_l = f64_of(&scratches_cfg, "max_length_px", 150.0).max(min_l + 1.0);
413 for _ in 0..scount {
414 let x1 = rng.gen_range(0.0..w as f64);
415 let y1 = rng.gen_range(0.0..h as f64);
416 let len = rng.gen_range(min_l..max_l);
417 let ang = rng.gen_range(0.0..(2.0 * PI));
418 let x2 = x1 + len * ang.cos();
419 let y2 = y1 + len * ang.sin();
420 let dark = [0, 0, 0, (255.0 * inten) as u8];
421 let light = [255, 255, 255, (255.0 * inten * highlight) as u8];
422 for o in 0..sw.max(1) {
423 let off = o as f64;
424 panel.line(x1 + off, y1 + off, x2 + off, y2 + off, dark);
425 }
426 panel.line(x1 + 1.0, y1 + 1.0, x2 + 1.0, y2 + 1.0, light);
427 }
428 }
429
430 // ---- Layer 7: grease / coffee stains ----
431 let grime = obj(&settings, "grime").cloned().unwrap_or(Value::Null);
432 let stain_count = u32_of(&grime, "stain_count", 0);
433 if stain_count > 0 {
434 let color = hex_to_rgb(&str_of(&grime, "color", "#0d0d0d"));
435 let opacity = f64_of(&grime, "opacity", 0.4);
436 let spread = u32_of(&grime, "stain_spread", 40).max(2);
437 let mut stains = Buf::transparent(w, h);
438 for _ in 0..stain_count {
439 let cx = rng.gen_range(0..w) as i32;
440 let cy = rng.gen_range(0..h) as i32;
441 let r = rng.gen_range((spread / 2) as i32..spread as i32).max(1);
442 fill_circle(&mut stains, cx, cy, r, [color[0], color[1], color[2], (255.0 * opacity) as u8]);
443 }
444 // Soften the blobs, then composite.
445 if let Some(img) = RgbaImage::from_raw(w, h, stains.px.clone()) {
446 stains.px = imageops::blur(&img, (spread as f32 / 2.0).max(1.0)).into_raw();
447 }
448 panel.composite(&stains, 0, 0);
449 }
450
451 // ---- Layer 8a: screws (auto-placed on the cover) ----
452 let screws = obj(&settings, "screws").cloned().unwrap_or(Value::Null);
453 let fold = obj(&settings, "metal_fold").cloned().unwrap_or(Value::Null);
454 if bool_of(&screws, "enabled", false) {
455 place_screws(&mut panel, &screws, &fold, &mut rng);
456 }
457
458 // ---- Layer 8b: metal fold creases ----
459 if bool_of(&fold, "enabled", false) {
460 apply_metal_fold(&mut panel, &fold);
461 }
462
463 // ---- Layer 8c: settled dust ----
464 let dust = obj(&settings, "dust").cloned().unwrap_or(Value::Null);
465 if bool_of(&dust, "enabled", false) {
466 let inten = f64_of(&dust, "intensity", 0.3);
467 let num = ((w * h) as f64 * 0.0005 * inten) as u32;
468 for _ in 0..num {
469 let x = rng.gen_range(0..w) as i32;
470 let y = rng.gen_range(0..h) as i32;
471 let c = rng.gen_range(180..=240) as u8;
472 let a = rng.gen_range(50..=150) as u8;
473 panel.blend_over(x, y, [c, c, c, a]);
474 }
475 }
476
477 // ---- Global blur ----
478 let gb = f64_of(&settings, "global_blur", 0.0) as f32;
479 if gb > 0.0 {
480 panel.blur_rgba(gb);
481 }
482
483 // Force opaque background (drop-shadow alphas may have lowered edges).
484 for i in 0..(w * h) as usize {
485 panel.px[i * 4 + 3] = 255;
486 }
487 panel.into_pixels()
488}

490// Subtract a drawn line from an L (grayscale) mask (carves bare substrate).
491fn subtract_line(mask: &mut [u8], w: u32, h: u32, x1: i32, y1: i32, x2: i32, y2: i32) {
492 let mut x = x1;
493 let mut y = y1;
494 let dx = (x2 - x).abs();
495 let dy = (y2 - y).abs();
496 let sx = if x < x2 { 1 } else { -1 };
497 let sy = if y < y2 { 1 } else { -1 };
498 let mut err = dx - dy;
499 let mut guard = 0;
500 loop {
501 if x >= 0 && y >= 0 && (x as u32) < w && (y as u32) < h {
502 mask[(y as u32 * w + x as u32) as usize] = 0;
503 }
504 if x == x2 && y == y2 {
505 break;
506 }
507 let e2 = 2 * err;
508 if e2 > -dy {
509 err -= dy;
510 x += sx;
511 }
512 if e2 < dx {
513 err += dx;
514 y += sy;
515 }
516 guard += 1;
517 if guard > 20000 {
518 break;
519 }
520 }
521}

523fn fill_circle(buf: &mut Buf, cx: i32, cy: i32, r: i32, color: [u8; 4]) {
524 for dy in -r..=r {
525 for dx in -r..=r {
526 if dx * dx + dy * dy <= r * r {
527 buf.blend_over(cx + dx, cy + dy, color);
528 }
529 }
530 }

533fn vignette_gray(w: u32, h: u32, intensity: f64, depth: u32) -> Vec<u8> {
534 let mut gray = vec![255u8; (w * h) as usize];
535 let depth_f = depth as f64;
536 for i in 0..depth {
537 let progress = i as f64 / depth_f;
538 let f = 1.0 - ((1.0 - progress).powf(1.2) * intensity * 0.8);
539 let v = (255.0 * f) as u8;
540 // rectangle outline at inset i
541 for x in i..(w - i) {
542 gray[(i * w + x) as usize] = v;
543 gray[((h - 1 - i) * w + x) as usize] = v;
544 }
545 for y in i..(h - i) {
546 gray[(y * w + i) as usize] = v;

548 }
549 }

551 if let Some(img) = GrayImage::from_raw(w, h, gray.clone()) {
552 gray = imageops::blur(&img, (depth as f32 / 2.0).max(2.0)).into_raw();
553 }

557fn apply_rust(panel: &mut Buf, intensity: f64, rng: &mut StdRng) {
558 let w = panel.w;
559 let h = panel.h;
560 let thresh = 255.0 - (intensity * 50.0);
561 // noise + threshold -> mask
562 let mut mask = vec![0u8; (w * h) as usize];
563 let mut noise = vec![0u8; (w * h) as usize];
564 for i in 0..(w * h) as usize {
565 let n = rng.gen_range(0..=255) as u8;
566 noise[i] = n;
567 if n as f64 > thresh {
568 mask[i] = 255;
569 }
570 }
571 // dilate 3x3 then blur (MaxFilter + GaussianBlur equivalent)
572 let dilated = dilate3(&mask, w, h);
573 let blurred = if let Some(img) = GrayImage::from_raw(w, h, dilated) {
574 imageops::blur(&img, 2.0).into_raw()
575 } else {
576 mask
577 };
578 for i in 0..(w * h) as usize {
579 let a = ((blurred[i] as u16 * noise[i] as u16) / 255) as u8;
580 let p = i * 4;
581 let af = a as f32 / 255.0;
582 panel.px[p] = (110.0 * af + panel.px[p] as f32 * (1.0 - af)) as u8;
583 panel.px[p + 1] = (50.0 * af + panel.px[p + 1] as f32 * (1.0 - af)) as u8;
584 panel.px[p + 2] = (20.0 * af + panel.px[p + 2] as f32 * (1.0 - af)) as u8;
585 }
586}

588fn dilate3(mask: &[u8], w: u32, h: u32) -> Vec<u8> {
589 let mut out = vec![0u8; (w * h) as usize];
590 for y in 0..h as i32 {
591 for x in 0..w as i32 {
592 let mut m = 0u8;
593 for dy in -1..=1 {
594 for dx in -1..=1 {
595 let nx = x + dx;
596 let ny = y + dy;
597 if nx >= 0 && ny >= 0 && (nx as u32) < w && (ny as u32) < h {
598 m = m.max(mask[(ny as u32 * w + nx as u32) as usize]);
599 }
600 }
601 }
602 out[(y as u32 * w + x as u32) as usize] = m;
603 }
604 }
605 out

608fn apply_metal_fold(panel: &mut Buf, cfg: &Value) {
609 let w = panel.w;
610 let h = panel.h;
611 let thickness = u32_of(cfg, "width_px", 20).max(1);
612 let creases = cfg.get("creases").and_then(|v| v.as_array()).cloned().unwrap_or_default();
613 let mut h_creases: Vec<f64> = vec![];
614 let mut v_creases: Vec<f64> = vec![];
615 for c in &creases {
616 let pos = f64_of(c, "position_pct", 0.5);
617 if str_of(c, "orientation", "horizontal") == "horizontal" {
618 h_creases.push(pos);
619 } else {
620 v_creases.push(pos);
621 }
622 }
623 h_creases.sort_by(|a, b| a.partial_cmp(b).unwrap_or(std::cmp::Ordering::Equal));
624 let mut bounds = vec![0.0];
625 bounds.extend(h_creases);
626 bounds.push(1.0);
627 for i in 0..(bounds.len() - 1) {
628 let y_start = (h as f64 * bounds[i]) as i32;
629 let y_end = (h as f64 * bounds[i + 1]) as i32;
630 if y_end <= y_start {
631 continue;
632 }
633 for j in 0..thickness as i32 {
634 let f = (1.0 - j as f64 / thickness as f64).powi(2);
635 let shadow = (160.0 * f) as u8;
636 let hi = (60.0 * f) as u8;
637 for x in 0..w as i32 {
638 panel.blend_over(x, y_end - 1 - j, [0, 0, 0, shadow]);
639 panel.blend_over(x, y_start + j, [255, 255, 255, hi]);
640 }
641 for y in y_start..y_end {
642 panel.blend_over(j, y, [255, 255, 255, hi]);
643 panel.blend_over(w as i32 - 1 - j, y, [0, 0, 0, shadow]);
644 }
645 }
646 }
647 for pos in v_creases {
648 let x = (w as f64 * pos) as i32;
649 if x > 0 && x < w as i32 - 1 {
650 for y in 0..h as i32 {
651 panel.blend_over(x - 1, y, [255, 255, 255, 60]);
652 panel.blend_over(x, y, [0, 0, 0, 100]);
653 panel.blend_over(x + 1, y, [0, 0, 0, 100]);
654 }
655 }
656 }

660fn place_screws(panel: &mut Buf, cfg: &Value, fold: &Value, rng: &mut StdRng) {
661 let w = panel.w as i32;
662 let h = panel.h as i32;
663 let size = u32_of(cfg, "size_px", 24);
664 let margin = 30i32;
665
666 let mut locs: Vec<String> = match cfg.get("locations") {
667 Some(Value::Array(a)) => a.iter().filter_map(|v| v.as_str().map(|s| s.to_lowercase())).collect(),
668 Some(Value::String(s)) => vec![s.to_lowercase()],
669 _ => vec!["top".into(), "bottom".into()],
670 };
671 if locs.is_empty() {
672 locs = vec!["top".into(), "bottom".into()];
673 }
674
675 let mut positions: Vec<(i32, i32)> = vec![];
676 if locs.iter().any(|l| l == "top") {
677 positions.push((margin, margin));
678 positions.push((w - margin, margin));
679 }
680 if locs.iter().any(|l| l == "bottom") {
681 positions.push((margin, h - margin));
682 positions.push((w - margin, h - margin));
683 }
684 if locs.iter().any(|l| l == "middle") {
685 positions.push((w / 2, margin));
686 positions.push((w / 2, h - margin));
687 }
688
689 // Repeat screws along folds, if requested.
690 if bool_of(fold, "enabled", false) && bool_of(fold, "repeat_screws", false) {
691 if let Some(creases) = fold.get("creases").and_then(|v| v.as_array()) {
692 for c in creases {
693 let pos = f64_of(c, "position_pct", 0.5);
694 if str_of(c, "orientation", "vertical") == "vertical" {
695 let x = (w as f64 * pos) as i32;
696 if locs.iter().any(|l| l == "top") {
697 positions.push((x, margin));
698 }
699 if locs.iter().any(|l| l == "bottom") {
700 positions.push((x, h - margin));
701 }
702 } else {
703 let y = (h as f64 * pos) as i32;
704 positions.push((margin, y - margin));
705 positions.push((w - margin, y - margin));
706 positions.push((margin, y + margin));
707 positions.push((w - margin, y + margin));
708 }
709 }
710 }
711 }
712
713 for (cx, cy) in positions {
714 let angle = rng.gen_range(0..90) as f64;
715 let screw = render_screw(size, cfg, angle);
716 panel.composite(&screw, cx - screw.w as i32 / 2, cy - screw.h as i32 / 2);
717 }
718}

728fn render_screw(size: u32, cfg: &Value, angle_deg: f64) -> Buf {
729 let canvas = screw_canvas_dim_internal(size);
730 let center = canvas as f64 / 2.0;
731 let radius = size as f64 / 2.0;
732 let mut img = Buf::transparent(canvas, canvas);
733
734 let head_type = str_of(cfg, "type", "fillister");
735 let rotation = angle_deg * PI / 180.0;
736 let damage = f64_of(cfg, "damage", 0.0);
737 let rust = f64_of(cfg, "rust", 0.0);
738 let finish = str_of(cfg, "finish", "chrome");
739 let base_hex = cfg
740 .get("color")
741 .and_then(|v| v.as_str())
742 .map(|s| s.to_string())
743 .unwrap_or_else(|| if finish == "chrome" { "#e0e0e0".into() } else { "#222222".into() });
744 let base = hex_to_rgb(&base_hex);
745
746 let mut rng = StdRng::seed_from_u64(size as u64 + (angle_deg as u64) * 123);
747 let perlin = Perlin::new(1234);
748
749 // 1. Drop shadow (fillister only).
750 if head_type == "fillister" {
751 let off = size as f64 * 0.15;
752 let scx = center + off;
753 let scy = center + off;
754 for y in 0..canvas {
755 for x in 0..canvas {
756 let dx = x as f64 - scx;
757 let dy = y as f64 - scy;
758 let dist = (dx * dx + dy * dy).sqrt();
759 if dist <= radius {
760 let falloff = 1.0 - dist / radius;
761 let a = (150.0 * falloff * falloff) as u8;
762 img.blend_over(x as i32, y as i32, [0, 0, 0, a]);
763 }
764 }
765 }
766 }
767
768 // 2. Head geometry + lighting + Robertson void + wear.
769 for y in 0..canvas {
770 for x in 0..canvas {
771 let dx = x as f64 - center;
772 let dy = y as f64 - center;
773 let dist = (dx * dx + dy * dy).sqrt();
774 if dist > radius {
775 continue;
776 }
777 let light = if head_type == "fillister" {
778 let so = radius * 0.3;
779 let lx = dx + so;
780 let ly = dy + so;
781 let ld = (lx * lx + ly * ly).sqrt();
782 0.5 + 0.5 * (1.0 - (ld / (radius * 1.5)).clamp(0.0, 1.0))
783 } else {
784 0.7
785 };
786 let mut r = (base[0] as f64 * light) as u8;
787 let mut g = (base[1] as f64 * light) as u8;
788 let mut b = (base[2] as f64 * light) as u8;
789
790 // Robertson square drive void (rotated).
791 let drive = radius * 0.55;
792 let rx = dx * rotation.cos() - dy * rotation.sin();
793 let ry = dx * rotation.sin() + dy * rotation.cos();
794 if rx.abs() < drive / 2.0 && ry.abs() < drive / 2.0 {
795 r = 20;
796 g = 20;
797 b = 20;
798 let e = 1.0;
799 if ry < -drive / 2.0 + e {
800 r = 100; g = 100; b = 100;
801 } else if rx < -drive / 2.0 + e {
802 r = 80; g = 80; b = 80;
803 } else if rx > drive / 2.0 - e {
804 r = 10; g = 10; b = 10;
805 } else if ry > drive / 2.0 - e {
806 r = 0; g = 0; b = 0;
807 }
808 } else if damage > 0.0 && rng.gen_bool((damage * 0.02).clamp(0.0, 1.0)) {
809 r = 200; g = 200; b = 200;
810 }
811
812 if rust > 0.0 {
813 let n = perlin.get([x as f64 * 0.2, y as f64 * 0.2]);
814 if n > 1.0 - rust {
815 r = 130; g = 60; b = 20;
816 }
817 }
818
819 let p = img.idx(x, y);
820 img.px[p] = r;
821 img.px[p + 1] = g;
822 img.px[p + 2] = b;
823 img.px[p + 3] = 255;
824 }
825 }
826
827 img
828}
