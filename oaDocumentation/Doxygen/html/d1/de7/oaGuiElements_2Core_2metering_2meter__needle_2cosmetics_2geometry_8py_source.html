81 def get_scaling_params(w, h, shape, line_width):
82 """Calculates the radius and global_y_shift for a given shape and container size."""
83 shape_key = shape.lower()
84 if shape_key not in SHAPE_MULTIPLIERS:
85 if shape_key in ["triangle", "pyramid"]: shape_key = "triangle"
86 elif shape_key in ["cylinder", "hotdog"]: shape_key = "hotdog" if shape_key == "hotdog" else "cylinder"
87 elif shape_key in ["trapezoid", "badge"]: shape_key = "trapezoid"
88 else: shape_key = "default"
89
90 m_w, m_h = SHAPE_MULTIPLIERS.get(shape_key, SHAPE_MULTIPLIERS["default"])
91 y_shift_factor = SHAPE_Y_SHIFTS.get(shape_key, 0.0)
92
93 # SCALING STRATEGY
94 avail_w = (w / 2.0) - (line_width / 2) - SAFE_MARGIN
95 avail_h = (h) - (line_width / 2) - SAFE_MARGIN
96
97 radius = min(avail_w / m_w, avail_h / m_h)
98 global_y_shift = y_shift_factor * radius
99
100 return radius, global_y_shift, shape_key
