35 def publish(self, topic: str, payload: Any, retain: bool = False, qos: int = 0) -> bool:
36 if not self.is_connected() or not self.client:
37 # Demote to TRACE to avoid log flooding during startup handshake
38 matrix_log("comms", "midi_mqtt", "publish", "📡 [MIDI-MQTT] Not connected. Cannot publish.", "TRACE")
39 return False
40 try:
41 # ⚡ EFFICIENT ENCODING: Use json for core transport compatibility
42 if isinstance(payload, dict) and "meta" not in payload:
43 # Inject local identity for echo prevention if not already there
44 payload["meta"] = payload.get("meta", {})
45 payload["meta"]["src"] = self.config.FULL_INSTANCE_ID
46 payload["meta"]["full_id"] = self.config.FULL_INSTANCE_ID
47
48 payload_str = json.dumps(payload)
49 matrix_log("comms", "midi_mqtt", "publish", f"📡📤 [MIDI-MQTT] Sending to {topic}: {payload_str[:100]}", "TRACE")
50 info = self.client.publish(topic, payload_str, qos=qos, retain=retain)
51 return info.rc == mqtt.MQTT_ERR_SUCCESS
52 except Exception as e:
53 matrix_log("comms", "midi_mqtt", "publish", f"📡❌ [MIDI-MQTT] Send Error: {e}", "ERROR")

79 def connect(self, connection_params: dict[str, Any]) -> bool:
80 host = connection_params.get("destination_host", self.config.MQTT_BROKER_ADDRESS)
81 port = connection_params.get("destination_port", self.config.MQTT_BROKER_PORT)
82 username = connection_params.get("username", self.config.MQTT_USERNAME)
83 password = connection_params.get("password", self.config.MQTT_PASSWORD)
84
85 # ⚡ UNIQUE IDENTITY: Prevent Client ID collisions in multi-partition environments
86 partition_id = os.environ.get("OPEN_AIR_PARTITION_ID", "SUP")
87 random_suffix = f"{random.getrandbits(16):04x}"
88 default_client_id = f"OPENAIR-{partition_id}-MIDI-{os.getpid()}-{random_suffix}"
89 client_id = connection_params.get("client_id", default_client_id)
90
91 matrix_log("comms", "midi_mqtt", "connect", f"📡📥 [MIDI-MQTT] Connecting to {host}:{port} as {client_id}.", "INFO")
92
93 self.client = mqtt.Client(client_id=client_id)
94 self.client.on_connect = self._on_connect
95 self.client.on_message = self._on_message
96 self.client.on_disconnect = self._on_disconnect
97
98 if connection_params.get("use_tls", False):
99 self.client.tls_set(tls_version=ssl.PROTOCOL_TLS)
100 if username and password:

103 try:
104 self.client.connect(host, port, 60)

108 import time
