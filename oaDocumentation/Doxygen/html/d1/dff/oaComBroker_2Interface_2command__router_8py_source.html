209 def _add_entry(self, message):
210 if not self.tree.winfo_exists(): return
211
212 # ⚡ ROBUSTNESS: Use safe .get() for all router message fields
213 utp = f"{message.get('timestamp', 0.0):.6f}"
214 source = message.get("source", "UNKNOWN") # Transport (e.g. MQTT)
215 logical_source = message.get("logical_source", source) # Identity (e.g. MIDI)
216 guid = message.get("logical_guid", message.get("guid", "LOCAL")) # Identity (e.g. 32_0/3)
217
218 strategy = message.get("strategy", "BROADCAST")
219 topic = message.get("topic", "N/A")
220 value = message.get("value", "N/A")
221
222 # Tags are now pre-calculated by the router (no brains in GUI)
223 tags = message.get("ui_tags", [])
224
225 # ⚡ DISPLAY ENRICHMENT: Add emoji for visual confirmation in the tree ONLY
226 display_source = logical_source
227 if "SPLINK" in tags:
228 display_source = f"🔗 {logical_source}"
229
230 # ⚡ STACK BEHAVIOR: Insert at TOP (index 0)
231 item_id = self.tree.insert("", 0, values=(utp, display_source, guid, strategy, topic, value), tags=tuple(tags))
232
233 if len(self.tree.get_children()) > 100:
234 self.tree.delete(self.tree.get_children()[-1])
235
236 # If autoscroll is on, make sure the latest (top) is visible
237 if self.autoscroll_var.get():
238 self.tree.see(item_id)

240 def on_select_packet(self, event):
241 """Populates the investigation pane and handles SPLINK selection logic."""
242 selected = self.tree.selection()
243 if not selected: return
244
245 item = self.tree.item(selected[0])
246 utp = item["values"][0]
247 topic = item["values"][4]
248 tags = item["tags"]
249
250 # ⚡ AUTO-DUAL: If this is a splinked message, try to find and show BOTH halves
251 if "SPLINK" in tags:
252 trigger_ts, result_ts = self.router.get_splink_relationship(utp)
253 if trigger_ts and result_ts:
254 self._src_utp = f"{trigger_ts:.6f}"
255 self._dest_utp = f"{result_ts:.6f}"
256 # Update Entries
257 item_src = self._find_item_by_ts(self._src_utp)
258 item_dest = self._find_item_by_ts(self._dest_utp)
259 if item_src: self.src_var.set(item_src["values"][4])
260 if item_dest: self.dest_var.set(item_dest["values"][4])
261 self.splink_btn.configure(fg="#00ff00")
262 self._refresh_investigation()
263 return
264
265 # ⚡ STANDARD ALTERNATING LOGIC (for non-splinked messages)
266 if not self.src_var.get():
267 self.src_var.set(topic)
268 self._src_utp = utp
269 self.splink_btn.configure(fg="#888888")
270 elif not self.dest_var.get():
271 if topic != self.src_var.get():
272 self.dest_var.set(topic)
273 self._dest_utp = utp
274 self.splink_btn.configure(fg="#00ff00")
275 else:
276 self.src_var.set("")
277 self._src_utp = None
278 else:
279 # Shift
280 self.src_var.set(self.dest_var.get())
281 self._src_utp = getattr(self, "_dest_utp", None)
282 self.dest_var.set(topic)
283 self._dest_utp = utp
