35

37 (width, 5.max(height / 100))

138 fn generate_metal_fold<'py>(&self, py: Python<'py>, width: u32, height: u32, config: &Bound<'_, PyDict>) -> PyResult<Bound<'py, PyBytes>> {
139 let mut img = RgbaImage::new(width, height);
140 let thickness: u32 = config.get_item("width_px")?.and_then(|v| v.extract().ok()).unwrap_or(20);
141
142 let creases = config.get_item("creases")?.and_then(|v| v.downcast_into::<PyList>().ok()).unwrap_or_else(|| PyList::empty_bound(py));
143
144 let mut h_creases: Vec<f64> = Vec::new();
145 let mut v_creases: Vec<f64> = Vec::new();
146
147 for item in creases.iter() {
148 let dict = item.downcast::<PyDict>()?;
149 let orientation: String = dict.get_item("orientation")?.and_then(|v| v.extract::<String>().ok()).unwrap_or_else(|| "horizontal".to_string());
150 let position: f64 = dict.get_item("position_pct")?.and_then(|v| v.extract::<f64>().ok()).unwrap_or(0.5);
151 if orientation == "horizontal" {
152 h_creases.push(position);
153 } else {
154 v_creases.push(position);
155 }
156 }
157
158 h_creases.sort_by(|a, b| a.partial_cmp(b).unwrap());
159 let mut h_bounds = vec![0.0];
160 h_bounds.extend(h_creases);
161 h_bounds.push(1.0);
162
163 for i in 0..(h_bounds.len() - 1) {
164 let y_start = (height as f64 * h_bounds[i]) as u32;
165 let y_end = (height as f64 * h_bounds[i+1]) as u32;
166
167 if y_end <= y_start { continue; }
168
169 for j in 0..thickness {
170 let j_f = j as f64;
171 let thick_f = thickness as f64;
172 let shadow_alpha = (160.0 * (1.0 - j_f / thick_f).powi(2)) as u8;
173 let highlight_alpha = (60.0 * (1.0 - j_f / thick_f).powi(2)) as u8;
174
175 // Bottom Shadow
176 if y_end > j {
177 for x in 0..width {
178 self.internal_blend_pixel(img.get_pixel_mut(x, y_end - 1 - j), [0, 0, 0, shadow_alpha]);
179 }
180 }
181
182 // Top Highlight
183 if y_start + j < height {
184 for x in 0..width {
185 self.internal_blend_pixel(img.get_pixel_mut(x, y_start + j), [255, 255, 255, highlight_alpha]);
186 }
187 }
188
189 // Left Highlight
190 if j < width {
191 for y in y_start..y_end {
192 self.internal_blend_pixel(img.get_pixel_mut(j, y), [255, 255, 255, highlight_alpha]);
193 }
194 }
195
196 // Right Shadow
197 if width > 1 + j {
198 for y in y_start..y_end {
199 self.internal_blend_pixel(img.get_pixel_mut(width - 1 - j, y), [0, 0, 0, shadow_alpha]);
200 }
201 }
202 }
203
204 // Crease gap
205 if i < h_bounds.len() - 2 {
206 let y = y_end;
207 if y < height {
208 for x in 0..width {
209 img.put_pixel(x, y, Rgba([0, 0, 0, 255]));
210 if y + 1 < height {
211 self.internal_blend_pixel(img.get_pixel_mut(x, y + 1), [255, 255, 255, 40]);
212 }
213 }
214 }
215 }
216 }
217
218 for position in v_creases {
219 let x = (width as f64 * position) as u32;
220 if x > 0 && x < width - 1 {
221 for y in 0..height {
222 self.internal_blend_pixel(img.get_pixel_mut(x - 1, y), [255, 255, 255, 60]);
223 img.put_pixel(x, y, Rgba([0, 0, 0, 100]));
224 if x + 1 < width {
225 img.put_pixel(x + 1, y, Rgba([0, 0, 0, 100]));
226 }
227 }
228 }
229 }
230
231 Ok(PyBytes::new_bound(py, img.as_raw()))

235 fn generate_screw<'py>(&self, py: Python<'py>, size: u32, config: &Bound<'_, PyDict>) -> PyResult<Bound<'py, PyBytes>> {
236 let padding = (size as f64 * 0.4) as u32;
237 let canvas_dim = size + padding * 2;
238 let center = canvas_dim as f64 / 2.0;
239 let radius = size as f64 / 2.0;
240
241 let mut img = RgbaImage::new(canvas_dim, canvas_dim);
242
243 let head_type: String = config.get_item("type")?.and_then(|v| v.extract().ok()).unwrap_or_else(|| "fillister".to_string());
244 let rotation_deg: f64 = config.get_item("angle")?.and_then(|v| v.extract().ok()).unwrap_or(0.0);
245 let rotation_rad = rotation_deg * PI / 180.0;
246
247 let damage_intensity: f64 = config.get_item("damage")?.and_then(|v| v.extract().ok()).unwrap_or(0.0);
248 let rust_intensity: f64 = config.get_item("rust")?.and_then(|v| v.extract().ok()).unwrap_or(0.0);
249
250 let material_finish: String = config.get_item("finish")?.and_then(|v| v.extract().ok()).unwrap_or_else(|| "chrome".to_string());
251 let base_color_hex: String = config.get_item("color")?.and_then(|v| v.extract().ok()).unwrap_or_else(|| {
252 match material_finish.as_str() {
253 "chrome" => "#e0e0e0".to_string(),
254 _ => "#222222".to_string(),
255 }
256 });
257
258 let base_rgb = self.convert_hex_to_rgb(&base_color_hex);
259
260 let mut rng = StdRng::seed_from_u64(size as u64 + (rotation_deg as u64 * 123));
261
262 // --- 1. Drop Shadow (Basic Circular) ---
263 if head_type == "fillister" {
264 let offset_x = (size as f64 * 0.15) as f64;
265 let offset_y = (size as f64 * 0.15) as f64;
266 let shadow_radius = radius;
267 let shadow_center_x = center + offset_x;
268 let shadow_center_y = center + offset_y;
269
270 for y in 0..canvas_dim {
271 for x in 0..canvas_dim {
272 let dx = x as f64 - shadow_center_x;
273 let dy = y as f64 - shadow_center_y;
274 let dist = (dx*dx + dy*dy).sqrt();
275 if dist <= shadow_radius {
276 let falloff = 1.0 - (dist / shadow_radius);
277 let alpha = (150.0 * falloff.powi(2)) as u8;
278 img.put_pixel(x, y, Rgba([0, 0, 0, alpha]));
279 }
280 }
281 }
282 }
283
284 // --- 2. Head Geometry and Lighting ---
285 let perlin = Perlin::new(1234);
286
287 for y in 0..canvas_dim {
288 for x in 0..canvas_dim {
289 let dx = x as f64 - center;
290 let dy = y as f64 - center;
291 let dist = (dx*dx + dy*dy).sqrt();
292
293 if dist <= radius {
294 let mut light_factor = 1.0;
295 if head_type == "fillister" {
296 let specular_offset = radius * 0.3;
297 let lx = dx + specular_offset;
298 let ly = dy + specular_offset;
299 let ldist = (lx*lx + ly*ly).sqrt();
300 light_factor = 0.5 + 0.5 * (1.0 - (ldist / (radius * 1.5)).clamp(0.0, 1.0));
301 } else {
302 light_factor = 0.7;
303 }
304
305 let mut r = (base_rgb[0] as f64 * light_factor) as u8;
306 let mut g = (base_rgb[1] as f64 * light_factor) as u8;
307 let mut b = (base_rgb[2] as f64 * light_factor) as u8;
308
309 // Robertson Void
310 let drive_size = radius * 0.55;
311 let rx = dx * rotation_rad.cos() - dy * rotation_rad.sin();
312 let ry = dx * rotation_rad.sin() + dy * rotation_rad.cos();
313
314 if rx.abs() < drive_size / 2.0 && ry.abs() < drive_size / 2.0 {
315 r = 20; g = 20; b = 20;
316 let edge_thresh = 1.0;
317 if ry < -drive_size/2.0 + edge_thresh { r = 100; g = 100; b = 100; }
318 else if rx < -drive_size/2.0 + edge_thresh { r = 80; g = 80; b = 80; }
319 else if rx > drive_size/2.0 - edge_thresh { r = 10; g = 10; b = 10; }
320 else if ry > drive_size/2.0 - edge_thresh { r = 0; g = 0; b = 0; }
321 } else if damage_intensity > 0.0 {
322 if rng.gen_bool(damage_intensity * 0.02) {
323 r = 200; g = 200; b = 200;
324 }
325 }
326
327 if rust_intensity > 0.0 {
328 let r_noise = perlin.get([x as f64 * 0.2, y as f64 * 0.2]);
329 if r_noise > 1.0 - rust_intensity {
330 r = 130; g = 60; b = 20;
331 }
332 }
333
334 // Compositionite
335 let pixel = img.get_pixel(x, y).0;
336 if pixel[3] == 0 {
337 img.put_pixel(x, y, Rgba([r, g, b, 255]));
338 } else {
339 let new_r = (r as u16 * 255 + pixel[0] as u16 * (255 - 255)) / 255;
340 let new_g = (g as u16 * 255 + pixel[1] as u16 * (255 - 255)) / 255;
341 let new_b = (b as u16 * 255 + pixel[2] as u16 * (255 - 255)) / 255;
342 img.put_pixel(x, y, Rgba([new_r as u8, new_g as u8, new_b as u8, 255]));
343 }
344 }
345 }
346 }
347
348 Ok(PyBytes::new_bound(py, img.as_raw()))
349 }

353 fn internal_draw_line(&self, img: &mut RgbaImage, x1: f64, y1: f64, x2: f64, y2: f64, color: [u8; 4]) {
354 let mut x = x1.round() as i32;
355 let mut y = y1.round() as i32;
356 let x2_i = x2.round() as i32;
357 let y2_i = y2.round() as i32;
358
359 let dx = (x2_i - x).abs();
360 let dy = (y2_i - y).abs();
361 let sx = if x < x2_i { 1 } else { -1 };
362 let sy = if y < y2_i { 1 } else { -1 };
363 let mut error_accumulator = dx - dy;
364
365 loop {
366 if x >= 0 && x < img.width() as i32 && y >= 0 && y < img.height() as i32 {
367 img.put_pixel(x as u32, y as u32, Rgba(color));
368 }
369
370 if x == x2_i && y == y2_i { break; }
371
372 let e2 = 2 * error_accumulator;
373 if e2 > -dy {
374 error_accumulator -= dy;
375 x += sx;
376 }
377 if e2 < dx {
378 error_accumulator += dx;
379 y += sy;
380 }
381
382 // ⚡ SAFETY: Absolute bail-out to prevent infinite loops if logic fails
383 if x.abs() > SAFETY_BAIL_OUT_LIMIT || y.abs() > SAFETY_BAIL_OUT_LIMIT { break; }
384 }
385 }

387 fn internal_blend_pixel(&self, pixel: &mut Rgba<u8>, source: [u8; 4]) {
388 let alpha_src = source[3] as f32 / 255.0;
389 let alpha_dst = pixel[3] as f32 / 255.0;
390
391 let out_alpha = alpha_src + alpha_dst * (1.0 - alpha_src);
392 if out_alpha > 0.0 {
393 pixel[0] = ((source[0] as f32 * alpha_src + pixel[0] as f32 * alpha_dst * (1.0 - alpha_src)) / out_alpha) as u8;
394 pixel[1] = ((source[1] as f32 * alpha_src + pixel[1] as f32 * alpha_dst * (1.0 - alpha_src)) / out_alpha) as u8;
395 pixel[2] = ((source[2] as f32 * alpha_src + pixel[2] as f32 * alpha_dst * (1.0 - alpha_src)) / out_alpha) as u8;
396 pixel[3] = (out_alpha * 255.0) as u8;
397 }

400 fn convert_hex_to_rgb(&self, hex_string: &str) -> [u8; 3] {
401 let hex = hex_string.trim_start_matches('#');
402 if hex.len() != 6 {
403 return [128, 128, 128];
404 }
405 let r = u8::from_str_radix(&hex[0..2], 16).unwrap_or(128);
406 let g = u8::from_str_radix(&hex[2..4], 16).unwrap_or(128);
407 let b = u8::from_str_radix(&hex[4..6], 16).unwrap_or(128);
408 [r, g, b]
409 }
