94 def _on_state_updated(self, json_data, source=None):
95 """Refreshes the tree when the master state changes."""
96 if source == self or source == self._ops_manager or source == self._drag_handler:
97 return # Avoid infinite loops

106 def _on_external_focus(self, path, source=None):
107 """Handles focus requests from other tabs."""

114 def _select_path_in_tree(self, path):
115 """Finds and selects the node corresponding to the given dot-notated path."""
116 if not path: return
117
118 # ⚡ EXPAND PARENTS
119 parts = path.split('.')
120 for i in range(1, len(parts)):
121 p = ".".join(parts[:i])
122 if self.tree.exists(p):
123 self.tree.item(p, open=True)
124
125 if self.tree.exists(path):
126 self.tree.selection_set(path)
127 self.tree.see(path)

129 def _on_tree_open(self, event):
130 """Lazy loads children when a node is expanded."""
131 node_id = self.tree.focus()
132 children = self.tree.get_children(node_id)
133 if len(children) == 1 and self.tree.item(children[0], "values")[0] == "dummy":
134 self.tree.delete(children[0])
135 path = self.tree.item(node_id, "values")[0]
136 data = state_manager.get_value_at_path(path)
137 if data is not None:
138 populate_tree(self.tree, node_id, data)
139
140 def _on_tree_select(self, event):
141 """Syncs selection with the global focus."""
142 selected = self.tree.selection()
143 if not selected: return
144
145 path = self.tree.item(selected[0], "values")[0]
146 event_bus.publish("FOCUS_REQUESTED", path=path, source=self)
