27def investigate_packet(message, mib_cache=None):
28 """
29 Executes Deep Packet Inspection to enrich message metadata.
30
31 This function analyzes the topic and transport source to apply protocol-
32 specific forensic tags. It modifies the 'meta' dictionary of the
33 provided message in-place.
34
35 Args:
36 message (dict): The normalized message packet to investigate.
37 mib_cache (dict, optional): A cache for SNMP OID-to-Name resolution.
38
39 Side Effects:
40 - Modifies message["meta"] by adding 'investigation', 'mib_resolution',
41 'blob', or 'mutation' keys.
42 """
43 topic = str(message["topic"])
44 source = message["source"]
45
46 # --- SNMP Investigation: Resolve numeric OIDs to human-readable names ---
47 if source == "SNMP" or topic.startswith(".1.3.6"):
48 mib = mib_cache.get(topic, "Unknown OID") if mib_cache else "Unknown OID"
49 message["meta"]["mib_resolution"] = mib
50 message["meta"]["investigation"] = f"SNMP MIB: {mib}"
51
52 # --- OSC Investigation: Map logical topics back to OSC addresses ---
53 if source == "OSC" or "osc_address" in message["meta"]:
54 osc_addr = message["meta"].get("osc_address",
55 "/" + topic.replace("OPEN-AIR/", ""))
56 message["meta"]["investigation"] = f"OSC Map: {osc_addr}"
57
58 # --- MIDI Investigation: Dissect raw bytes and identify signal type ---
59 if source == "MIDI" or "midi_raw" in message["meta"]:
60 raw = message["meta"].get("midi_raw", "Unknown Message")
61 port = message["meta"].get("midi_port", "Remote")
62 m_type = message["meta"].get("midi_type", "message")
63
64 # Flag real-time clock signals to differentiate from control changes.
65 if "clock" in raw.lower():
66 message["meta"]["investigation"] = f"🎹 MIDI [CLOCK] from {port}"
67 else:
68 message["meta"]["investigation"] = f"🎹 MIDI [{m_type.upper()}] " \
69 f"from {port}"
70
71 # --- YAK Hardware Safety: Flag potential hardware mutations ---
72 # Mutation events (SET, NAB) require elevated logging and audit trails.
73 if "YAK" in topic.upper():
74 if any(k in topic.upper() for k in ["SET", "NAB", "LEVEL", "FREQ"]):
75 message["meta"]["mutation"] = True
76 message["meta"]["investigation"] = "🚨 YAK Hardware Mutation"
77
78 # --- Splink Investigation: Trace brokered relationships ---
79 if message["meta"].get("splink_active"):
80 s_id = message["meta"].get("splink_id", "Link")
81 s_dest = message["meta"].get("splink_dest", "Destination")
82 message["meta"]["investigation"] = f"🔗 Splink Sync [{s_id}] ➜ {s_dest}"
83
84 # --- Data Volume Check: Detect oversized configuration blobs ---
85 val_str = str(message["value"])
86 if len(val_str) > 1000:
87 message["meta"]["blob"] = True
88 message["meta"]["investigation"] = f"📦 Large Configuration Blob " \
89 f"({len(val_str)} bytes)"
