55 def start(self):
56 """Starts the MQTT manager background threads."""

60 self._is_running = True

72 def _publish_status(self, status: str):

83 def _system_status_loop(self):
84 """Periodically prepares connection status and system paths."""
85 while self._is_running:
86 try:
87 if self.mqtt_client and self.mqtt_client.is_connected():
88 self._publish_status("ONLINE")
89
90 # Publish Paths
91 paths_payload = {
92 "root": str(app_paths.GLOBAL_PROJECT_ROOT),
93 "markers": str(app_paths.MARKERS_JSON_PATH),
94 "device_state": str(app_paths.DEVICE_STATE_CACHE_PATH),
95 "fleet": str(app_paths.STATE_VISA_FLEET_JSON_PATH),
96 "yak": str(app_paths.YAKETY_YAK_REPO_PATH),
97 "presets": str(app_paths.PRESET_REPO_PATH)
98 }
99 self.mqtt_client.publish("OPEN-AIR/System/Status/Paths", orjson.dumps(paths_payload), qos=1)
100 else:
101 self._publish_status("OFFLINE")
102 self._attempt_reconnect()
103
104 except Exception as e:
105 matrix_log("comms", "mqtt", "_system_status_loop", f"🚀 [MQTT] ERROR: Status loop failed: {e}", "ERROR")
106
107 time.sleep(5)

109 def _attempt_reconnect(self):
110 """Attempts to reconnect the MQTT client if disconnected."""
111 matrix_log("comms", "mqtt", "_attempt_reconnect", "📡 [MQTT] Attempting to reconnect...", "INFO")
112 try:
113 result = self.mqtt_client.connect(
114 app_constants.MQTT_BROKER_ADDRESS,
115 app_constants.MQTT_BROKER_PORT,
116 app_constants.MQTT_CLIENT_ID
117 )
118 if result == 0:
119 self._publish_status("ONLINE")
120 self.mqtt_client.loop_start()
121 self._sync_state_on_reconnect()
122 else:
123 self._publish_status("OFFLINE")
124 except Exception as e:
125 matrix_log("comms", "mqtt", "_attempt_reconnect", f"📡 [MQTT] Reconnect failed: {e}", "ERROR")
126 self._publish_status("OFFLINE")
127 self.mqtt_client.disconnect()

147 def _handle_service_command(self, message: MqttMessage):
148 try:
149 payload = message.payload
150 if not payload:
151 return
152 data = orjson.loads(payload) if isinstance(payload, (bytes, str)) else payload
153 action = data.get("action")
154 service_name = data.get("service")
155
156 matrix_log("comms", "mqtt", "_handle_service_command", f"🔧 [MQTT] MqttManager: Requested service {action}.", "DEBUG")
157
158 if action == "register" and service_name:
159 register_service(
160 self.mqtt_client, self.state_cache_manager,
161 app_constants.MQTT_BROKER_ADDRESS, app_constants.MQTT_BROKER_PORT,
162 app_constants.MQTT_CLIENT_ID, service_name, data.get("data", {})
163 )
164 elif action == "reregister_all":
165 re_register_all_services(self.mqtt_client, self.state_cache_manager)
166 elif action == "status" and service_name:
167 self._update_service_status(service_name, data.get("status", "UNKNOWN"))
168
169 except Exception as e:
170 matrix_log("comms", "mqtt", "_handle_service_command", f"MQTT: Service command parse error: {e}", "ERROR")
