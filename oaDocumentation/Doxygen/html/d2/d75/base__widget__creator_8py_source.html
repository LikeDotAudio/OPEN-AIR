32class BaseWidgetCreator(SyncBehavior):
33 """
34 Template Method orchestrator for constructing GUI widgets.
35 Decomposes monolithic build logic into specialized initialization phases.
36 """
37
38 # ⚡ COMPOSITE FLAG: Subclasses can set this to True to bypass automatic ghost box
39 # generation and handle ghosting recursively in their _assemble_ui.
40 is_composite = False
41
42 @classmethod
43 def build(cls, parent_widget, configuration, context=None, **kwargs):
44 """
45 Main entry point for widget construction.
46
47 Coordinates the entire assembly, synchronization, and system
48 registration lifecycle.
49
50 Inputs:
51 parent_widget (tk.Widget): The parent container in the UI tree.
52 configuration (dict): The homogenized schema defining the widget.
53 context (object): Shared service context (Builder, Mirror, etc.).
54 **kwargs: Additional overrides (e.g., render_tier, variable).
55
56 Returns:
57 tk.Widget: The constructed widget instance, or None on failure.
58
59 Side Effects:
60 - Instantiates a new widget in the UI tree.
61 - Registers the widget with the State Mirror and MQTT Router.
62 - Logs detailed exceptions if construction fails.
63 """
64 instance = cls()
65
66 # 1. Resolve Construction Context
67 ctx = cls._resolve_context(context, kwargs)
68
69 # ⚡ ROBUSTNESS: Handle case where ctx['builder'] might be None
70 builder = ctx.get('builder')
71 render_tier = kwargs.get('render_tier') or (getattr(builder, '_render_tier', 'high_res') if builder else 'high_res')
72
73 path = configuration.get("path")
74 label = configuration.get("label_active") or configuration.get("label", "Unknown")
75
76 try:
77 # 2. UI Assembly Phase
78 # ⚡ GHOST BYPASS: Standard widgets show a box; composite widgets descend.
79 if render_tier == 'ghost' and not instance.is_composite:
80 widget, canvas = instance._assemble_ghost_ui(parent_widget, configuration, context, **kwargs)
81 else:
82 result = instance._assemble_ui(parent_widget, configuration, context, **kwargs)
83 # Subclasses should return (widget, canvas) or just widget
84 if isinstance(result, tuple):
85 widget, canvas = result
86 else:
87 widget, canvas = result, None
88
89 if not widget:
90 return None
91
92 # 3. Post-Assembly Lifecycle
93 if render_tier != 'ghost':
94 # Industrial Background Sync
95 instance.register_for_bg_sync(widget, canvas, configuration, context)
96
97 # Global State & Communication Registration
98 if path and ctx['mirror']:
99 cls._register_with_system(widget, path, configuration, ctx, kwargs)
100
101 return widget
102
103 except Exception as e:
104 BUILDER_LOGGER.exception(f"{cls.__name__}: Error building widget '{label}' at {path}: {e}")
105 return None

107 @staticmethod
108 def _resolve_context(context, kwargs):
109 """
110 Extracts and standardizes engine references from the construction context.
111
112 Inputs:
113 context (object): The formal context object.
114 kwargs (dict): The caller's keyword arguments.
115
116 Returns:
117 dict: A mapping of canonical service names to their references.
118 """
119 return {
120 "builder": getattr(context, 'builder_instance', None) if context else kwargs.get('builder_instance'),
121 "mirror": getattr(context, 'state_mirror_engine', None) if context else kwargs.get('state_mirror_engine'),
122 "router": getattr(context, 'subscriber_router', None) if context else kwargs.get('subscriber_router'),
123 "topic": getattr(context, 'base_mqtt_topic_from_path', None) if context else kwargs.get('base_mqtt_topic_from_path')
124 }

126 @classmethod
127 def _register_with_system(cls, widget, path, config, ctx, kwargs):
128 """
129 Registers the widget variable with the State Mirror and MQTT router.
130
131 Inputs:
132 widget (tk.Widget): The widget being registered.
133 path (str): The unique widget path ID.
134 config (dict): Configuration data.
135 ctx (dict): Resolved services.
136 kwargs (dict): Optional variable override.
137
138 Side Effects:
139 - Subscribes the MQTT router to the widget's topic.
140 - Updates the State Mirror's widget registry.
141 """
142 variable = getattr(widget, 'variable', None) or kwargs.get('variable')
143 if not variable:
144 return
145
146 topic = ctx['mirror'].register_widget(
147 path, variable, ctx['topic'], config, instance=widget
148 )
149
150 if ctx['router'] and topic:
151 ctx['router'].subscribe_to_topic(
152 topic, ctx['mirror'].sync_incoming_mqtt_to_gui
153 )
154
155 ctx['mirror'].initialize_widget_state(path)

166 def _assemble_ghost_ui(self, parent_widget, configuration, context, **kwargs):
167 """
168 Industrial Placeholder Rendering (Ghost Mode).
169
170 Renders a lightweight high-contrast box for design-time interaction.
171 This provides instant visual feedback in the WYSIWYG editor without
172 the overhead of high-fidelity rendering.
173
174 Returns:
175 tuple: (ghost_frame, None)
176 """
177 geom = configuration.get("geometry", {})
178 width = geom.get("width", GHOST_WIDGET_DEFAULT_SIZE)
179 height = geom.get("height", GHOST_WIDGET_DEFAULT_SIZE)
180 label = configuration.get("label", configuration.get("_type", "Unknown"))
181
182 ghost = tk.Frame(parent_widget, width=width, height=height, bg="#333333",
183 highlightbackground="#00FF00", highlightthickness=1)
184 ghost.pack_propagate(False)
185
186 tk.Label(ghost, text=label, fg="#00FF00", bg="#333333", font=("Arial", 8, "bold")).pack(pady=(height//4, 0))
187 tk.Label(ghost, text=f"{width}x{height}", fg="#aaaaaa", bg="#333333", font=("Arial", 7)).pack()
188
189 return ghost, None
