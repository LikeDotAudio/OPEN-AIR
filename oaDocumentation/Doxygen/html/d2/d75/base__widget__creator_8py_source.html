15 """
16 Template Method orchestrator for constructing GUI widgets.
17 Decomposes monolithic build logic into specialized initialization phases.
18 """

20 # ⚡ COMPOSITE FLAG: Subclasses can set this to True to bypass automatic ghost box
21 # generation and handle ghosting recursively in their _assemble_ui.

24 @classmethod
25 def build(cls, parent_widget, config_data, context=None, **kwargs):
26 """
27 Main entry point for widget construction.
28 Coordinates assembly, synchronization, and system registration.
29 """
30 instance = cls()
31
32 # 1. Resolve Construction Context
33 ctx = cls._resolve_context(context, kwargs)
34
35 # ⚡ ROBUSTNESS: Handle case where ctx['builder'] might be None
36 builder = ctx.get('builder')
37 render_tier = kwargs.get('render_tier') or (getattr(builder, '_render_tier', 'high_res') if builder else 'high_res')
38
39 path = config_data.get("path")
40 label = config_data.get("label_active") or config_data.get("label", "Unknown")
41
42 try:
43 # 2. UI Assembly Phase
44 # ⚡ GHOST BYPASS: Standard widgets show a box; composite widgets descend.
45 if render_tier == 'ghost' and not instance.is_composite:
46 widget, canvas = instance._assemble_ghost_ui(parent_widget, config_data, context, **kwargs)
47 else:
48 result = instance._assemble_ui(parent_widget, config_data, context, **kwargs)
49 # Subclasses should return (widget, canvas) or just widget
50 if isinstance(result, tuple):
51 widget, canvas = result
52 else:
53 widget, canvas = result, None
54
55 if not widget:
56 return None
57
58 # 3. Post-Assembly Lifecycle
59 if render_tier != 'ghost':
60 # Industrial Background Sync
61 instance.register_for_bg_sync(widget, canvas, config_data, context)
62
63 # Global State & Communication Registration
64 if path and ctx['mirror']:
65 cls._register_with_system(widget, path, config_data, ctx, kwargs)
66
67 return widget
68
69 except Exception as e:
70 BUILDER_LOGGER.exception(f"{cls.__name__}: Error building widget '{label}' at {path}: {e}")
71 return None

75 """Extracts and standardizes various engine references from the construction context."""
76 return {

79 "router": getattr(context, 'subscriber_router', None) if context else kwargs.get('subscriber_router'),

83 @classmethod
84 def _register_with_system(cls, widget, path, config, ctx, kwargs):
85 """Registers the widget variable with the State Mirror and MQTT router."""
86 variable = getattr(widget, 'variable', None) or kwargs.get('variable')
87 if not variable:
88 return
89
90 topic = ctx['mirror'].register_widget(
91 path, variable, ctx['topic'], config, instance=widget
92 )
93
94 if ctx['router'] and topic:
95 ctx['router'].subscribe_to_topic(
96 topic, ctx['mirror'].sync_incoming_mqtt_to_gui
97 )
98
99 ctx['mirror'].initialize_widget_state(path)

105 def _assemble_ghost_ui(self, parent_widget, config_data, context, **kwargs):
106 """
107 Industrial Placeholder Rendering (Ghost Mode).
108 Renders a lightweight high-contrast box for design-time interaction.
109 """
110 geom = config_data.get("geometry", {})
111 w, h = geom.get("width", 100), geom.get("height", 100)
112 label = config_data.get("label", config_data.get("_type", "Unknown"))
113
114 ghost = tk.Frame(parent_widget, width=w, height=h, bg="#333333",
115 highlightbackground="#00FF00", highlightthickness=1)
116 ghost.pack_propagate(False)
117
118 tk.Label(ghost, text=label, fg="#00FF00", bg="#333333", font=("Arial", 8, "bold")).pack(pady=(h//4, 0))
119 tk.Label(ghost, text=f"{w}x{h}", fg="#aaaaaa", bg="#333333", font=("Arial", 7)).pack()
120
121 return ghost, None
