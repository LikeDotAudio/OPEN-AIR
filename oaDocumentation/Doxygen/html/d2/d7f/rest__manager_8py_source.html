115 def _start_health_monitor(self):
116 if self._health_thread and self._health_thread.is_alive(): return
117 self._health_thread = threading.Thread(target=self._health_loop, daemon=True, name="REST-HealthMonitor")
118 self._health_thread.start()
119
120 def _health_loop(self):
121 while self._should_run:
122 try:
123 # ⚡ MANDATORY: Always attempt to keep the service alive
124 is_local = self.is_running()
125 proc = get_process_on_port(REST_PORT)
126 is_sibling = proc and is_friendly_process(proc) and not is_local
127
128 if is_local:
129 self._sibling_active = False
130 elif is_sibling:
131 self._sibling_active = True
132 else:
133 self._sibling_active = False
134 if self._should_run:
135 self._launch_instance()
136
137 # Update status in cache
138 status_value = self.is_running() or self._sibling_active
139 if self.state_cache and hasattr(self.state_cache, 'handle_external_update'):
140 self.state_cache.handle_external_update(self.STATE_TOPICSTATE_TOPIC, status_value, source="REST-HB")
141 elif self.mqtt_transport and self.mqtt_transport.is_connected():
142 self.mqtt_transport.publish(self.STATE_TOPICSTATE_TOPIC, {"value": status_value, "source": "REST-HB"})
143
144 except Exception as e:
145 matrix_log("comms", "rest", "_health_loop", f"❌ [REST] Health loop error: {e}", "ERROR")
146
147 # Use short sleeps to be responsive to _should_run changes
148 for _ in range(20):
149 if not self._should_run: break
150 time.sleep(0.1)

152 def _launch_instance(self):
153 if not self._initialized or not self._should_run: return
154 proc = get_process_on_port(REST_PORT)
155 if proc:
156 if is_friendly_process(proc):
157 self._sibling_active = True
158 return
159 elif not zap_port(REST_PORT): return
160
161 try:
162 self.worker = UvicornWorker(self.app, host=REST_BIND_HOST, port=REST_PORT)
163 matrix_log("comms", "rest", "_launch_instance", f"🌐 [REST] Launching Mandatory API Service on {REST_BIND_HOST}:{REST_PORT} (URL: http://{REST_REPORT_HOST}:{REST_PORT})...", "INFO")
164 self.worker.start()
165 except Exception as e:
166 matrix_log("comms", "rest", "_launch_instance", f"❌ [REST] Launch failed: {e}", "ERROR")

168 def _shutdown_local_worker(self):
169 """Signals the local uvicorn worker to stop."""
170 if self.worker:
171 matrix_log("comms", "rest", "_shutdown_local_worker", "🛑 [REST] Stopping local worker service...", "INFO")
172 self.worker.stop()
173 self.worker.join(timeout=2.0)
174 self.worker = None

176 def start(self):
177 """Ensures the service is running. Always returns True as it is mandatory."""
178 self._should_run = True
179 if not self._initialized:
180 if not self._try_initialize(): return False
181
182 # ⚡ STANDALONE: Setup core MQTT transport for status reporting
183 if not self.mqtt_transport.is_connected():
184 connection_params = {
185 "destination_host": getattr(app_constants, "MQTT_BROKER_ADDRESS", "localhost"),
186 "destination_port": getattr(app_constants, "MQTT_BROKER_PORT", 1883)
187 }
188 self.mqtt_transport.connect(connection_params)
189
190 self._launch_instance()
191 return True

193 def stop(self):
194 """Signals the REST service and its monitor to shut down."""
195 matrix_log("comms", "rest", "stop", "🛑 [REST] Service shutdown initiated.", "INFO")
196 self._should_run = False
197 self._shutdown_local_worker()
198
199 # Disconnect core transport
200 if self.mqtt_transport:
201 self.mqtt_transport.disconnect()
202
203 if self._health_thread and self._health_thread.is_alive():
204 self._health_thread.join(timeout=1.0)
205 return True
