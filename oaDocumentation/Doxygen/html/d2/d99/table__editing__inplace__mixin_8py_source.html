187 # Increments any trailing digits in a string.
188 # This helper function is used for auto-incrementing cell values. If the string
189 # ends with numbers, it increments them; otherwise, it appends '1'.
190 # Inputs:
191 # text (str): The input string.
192 # Outputs:
193 # str: The incremented string.
194 def _increment_string_with_trailing_digits(self, text):
195 """
196 Increments any trailing digits in a string. If no trailing digits, appends '1'.
197 e.g., "Mic 1" -> "Mic 2", "Camera" -> "Camera 1", "Ch 09" -> "Ch 10"
198 """
199 match = re.search(r"(\d+)$", text)
200 if match:
201 num_str = match.group(1)
202 prefix = text[: -len(num_str)]
203 incremented_num = int(num_str) + 1
204 # Preserve leading zeros
205 new_num_str = str(incremented_num).zfill(len(num_str))
206 return f"{prefix}{new_num_str}"
207 else:
208 return f"{text} 1"
