146 def _perform_layout(self, w, h):
147 if w <= 1 or h <= 1: return
148 self.current_layout = self.layout_calc.calculate(w, h, self.configuration)
149 self.renderer.draw_static(self.current_layout, self.configuration)
150 self._refresh_frame()

152 # Reset resize flag after layout is complete and redraw is done
153 self.is_resizing = False

155 def _on_value_update(self, *args):
156 try:
157 # ⚡ SAFETY: Ensure we don't try to animate on a mock or dead canvas
158 if getattr(self, 'canvas', None) is None or not hasattr(self.canvas, 'winfo_exists'):
159 return
160 if not self.canvas.winfo_exists():
161 return
162
163 value = self.value_var.get()
164 self.physics.set_target(value)
165 if self._anim_timer_id is None:
166 self._last_anim_time = time.time() * 1000
167 self._animate()
168 except Exception as e:
169 logger.error(f"Error updating meter value: {e}")

171 def _animate(self):
172 # Prevent animation updates if a resize is in progress
173 if self.is_resizing:
174 self._anim_timer_id = None # Clear timer to avoid re-scheduling until resize is done
175 return
176
177 self._anim_timer_id = None
178 if not hasattr(self, 'current_layout'): return
179
180 now = time.time() * 1000
181 dt = now - self._last_anim_time
182 self._last_anim_time = now
183
184 dt = max(1.0, min(100.0, dt))
185 current_v, peak_v, overload_f, is_running, reached_min = self.physics.update(dt)
186
187 dyn_data = self.layout_calc.get_dynamic_coords(current_v, peak_v, overload_f, self.configuration, self.current_layout)
188 self.renderer.update_dynamic(dyn_data, overload_f, self.configuration)
189
190 if reached_min:
191 if self.state_mirror and self.configuration.path:
192 self.state_mirror.broadcast_gui_change_to_mqtt(self.configuration.path)
193
194 if is_running:
195 self._anim_timer_id = self.defer(20, self._animate)
