205 def _perform_background_sync(self, force=False):
206 """Internal execution logic for background sync."""
207 self._bg_sync_timer = None
208 if not self.winfo_exists() or not self.winfo_ismapped():
209 return

211 if not hasattr(self, 'scroll_frame'):
212 return
