115 # 3. Dynamic Handlers

117 if not canvas.winfo_exists(): return
118 cw, ch = float(canvas.winfo_width()), float(canvas.winfo_height())
119 frame._sync_fader_cap_position(cw if cw>1 else w, ch if ch>1 else h, val_var.get())

121 frame.sync_callback = _sync_pos

123 def _schedule_redraw(e=None):
124 if getattr(frame, "_resize_timer", None): canvas.after_cancel(frame._resize_timer)
125 cw, ch = float(canvas.winfo_width()), float(canvas.winfo_height())
126 frame._resize_timer = canvas.after(100, lambda: frame._draw_fader(cw if cw>1 else w, ch if ch>1 else h, val_var.get()))

128 frame._draw = _schedule_redraw
129 val_var.trace_add("write", _sync_pos)

139 canvas.bind("<B1-Motion>", frame._on_drag)

141 canvas.bind("<Button-2>", frame._jump_to_reff_point)
142 canvas.bind("<Alt-Button-1>", frame._open_manual_entry)
