28def purge_cache():
29 """
30 Deletes the local cache and running state files.
31 Excludes application logs but targets stateful data.
32 """
33 # Target non-log data directories and stateful logs
34 data_dirs = [
35 project_root / "oaDataCache",
36 project_root / "oaDataRunningFiles",
37 project_root / "oaDataSplinks",
38 project_root / "oaDataLogs" / "Reports",
39 project_root / ".pytest_cache" / "RUN",
40 project_root / "DATA" # Legacy compatibility
41 ]
42
43 # Specific files to remove
44 target_files = [
45 project_root / "oaDataCache" / "device_state_cache.json",
46 project_root / "oaDataCache" / "cache_layout_store.json",
47 project_root / "oaComProtocols" / "oaComSNMP" / "Assets" / "current.mib",
48 project_root / "oaComProtocols" / "oaComSNMP" / "Assets" / "OPEN-AIR.mib",
49 project_root / "oaComProtocols" / "oaComSNMP" / "Assets" / "openair_snmp_objects.txt",
50 project_root / "oaComProtocols" / "oaComSNMP" / "Assets" / "openair_snmp_set.log",
51 project_root / "oaComProtocols" / "oaComSNMP" / "Assets" / "bridge_debug.log"
52 ]
53
54 matrix_log("core", "system", "purge_cache", "📡📤📤 [CLEAR_CACHE] Starting local cache and state purge...", "INFO")
55
56 # 1. Clean individual files
57 for f in target_files:
58 if f.exists():
59 f.unlink()
60 matrix_log("core", "system", "purge_cache", f"🗑️ Deleted: {f.name}", "INFO")
61
62 # 2. Clean directories
63 for data_dir in data_dirs:
64 if data_dir.exists():
65 matrix_log("core", "system", "purge_cache", f"🗑️ Nuking contents of: {data_dir.name}", "INFO")
66 try:
67 # To delete CONTENTS and not the dir itself (to preserve permissions/structure)
68 items_purged = 0
69 for item in data_dir.iterdir():
70 try:
71 if item.is_file() or item.is_symlink():
72 item.unlink()
73 items_purged += 1
74 elif item.is_dir():
75 shutil.rmtree(item)
76 items_purged += 1
77 except Exception as e:
78 logger.error(f" └─ ❌ Failed to delete {item.name}: {e}")
79
80 if items_purged > 0:
81 matrix_log("core", "system", "purge_cache", f" └─ 💥 Purged {items_purged} items from {data_dir.name}.", "INFO")
82 except Exception as e:
83 logger.error(f" └─ ❌ Failed to access {data_dir.name}: {e}")
84 else:
85 matrix_log("core", "system", "purge_cache", f" └─ 🤷 {data_dir.name} directory not found.", "INFO")
86
87 # Recreate structure via path_initializer to ensure sanity
88 matrix_log("core", "system", "purge_cache", "🌱 Re-initializing directory structure integrity...", "INFO")
89 try:
90 from oaOchestration.Core.path_initializer import initialize_paths
91 initialize_paths()
92 matrix_log("core", "system", "purge_cache", "✨ Directory structure integrity verified.", "INFO")
93 except Exception as e:
94 logger.error(f"❌ Failed to initialize paths: {e}")
95
96 # 3. Clear in-memory state cache to prevent automatic recreation with old data
97 try:
98 import oaStateCache.Entry as StateCacheEntry
99 registry = StateCacheEntry.get_registry()
100 if registry and hasattr(registry, "clear_all_state"):
101 matrix_log("core", "system", "purge_cache", "🧠 Wiping in-memory State Registry...", "INFO")
102 registry.clear_all_state()
103 matrix_log("core", "system", "purge_cache", "🧠 In-memory State Registry wiped.", "INFO")
104 except Exception as e:
105 logger.error(f" └─ ❌ Failed to clear in-memory state cache: {e}")
106
107 # 4. Notify distributed processes via MQTT
108 try:
109 import paho.mqtt.publish as publish
110 matrix_log("core", "system", "purge_cache", "📡 Broadcasting global cache clear command...", "INFO")
111 publish.single("OPEN-AIR/System/Control/ClearCache", "true", hostname="localhost", port=1883)
112 except Exception as e:
113 logger.warning(f" └─ ⚠️ Failed to broadcast MQTT clear cache command: {e}")
114
115 matrix_log("core", "system", "purge_cache", "📡📤📤 [CLEAR_CACHE] Cache purge complete.", "INFO")
116
117if __name__ == "__main__":
118 purge_cache()
