85 def _attempt_connect(self) -> bool:
86 """Internal helper to perform a single connection attempt."""
87 uri = self._connection_params.get("connection_uri", "ws://localhost:8080")
88 try:
89 if self.ws_app:
90 try: self.ws_app.close()
91 except: pass
92
93 self.ws_app = websocket.WebSocketApp(uri,
94 on_open=self._on_open,
95 on_message=self._on_message,
96 on_error=self._on_error,
97 on_close=self._on_close)
98
99 self._ws_thread = threading.Thread(target=self.ws_app.run_forever)
100 self._ws_thread.daemon = True
101 self._ws_thread.start()
102
103 # Wait for either connection success or thread exit
104 start_time = time.time()
105 while time.time() - start_time < 2.0:
106 if self._is_connected:
107 return True
108 if not self._ws_thread.is_alive():
109 break
110 time.sleep(0.1)
111
112 return self._is_connected
113 except Exception as e:
114 matrix_log("comms", "websocket", "connect", f"📡❌ [WebSocketTransport] Connection exception: {e}", "ERROR")
115 self.ws_app = None
116 self._is_connected = False
117 return False
