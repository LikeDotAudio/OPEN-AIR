51def packet_callback(pkt, PTP, mqtt_client, MQTT_TOPIC):
52 """
53 Process captured network packets and extract PTP-specific information.
54 """
55 from scapy.all import IP, UDP
56 ptp_layer = None
57 source = "Unknown"
58 dport = 0
59
60 if pkt.haslayer(UDP):
61 source = pkt[IP].src if IP in pkt else "Unknown"
62 dport = pkt[UDP].dport
63
64 # Determine if the packet contains a recognized PTP layer.
65 if pkt.haslayer(PTP):
66 ptp_layer = pkt[PTP]
67 else:
68 # Attempt manual parsing if automatic layer identification failed.
69 payload = bytes(pkt[UDP].payload)
70 if len(payload) >= 34:
71 ptp_layer = PTP(payload)
72
73 if ptp_layer:
74 matrix_log("CORE", "PTP", inspect.currentframe().f_code.co_name, f"\n[+] Captured PTP Packet from {source} on port {dport}", level="INFO")
75 ptp_layer.show()
76
77 # Relay the captured metrics to the system-wide MQTT bus.
78 if mqtt_client:
79 # Prioritize direct field access to minimize processing latency.
80 m_type = int(ptp_layer.messageType)
81 domain = int(ptp_layer.domainNumber)
82 seq_id = int(ptp_layer.sequenceId)
83 port_id = ptp_layer.sourcePortIdentity
84
85 # Convert raw port identity to readable hex format for JSON.
86 clock_id = (port_id.hex(':') if isinstance(port_id, bytes)
87 else str(port_id))
88
89 data = {
90 "timestamp": time.time(),
91 "source_ip": source,
92 "udp_port": dport,
93 "message_type": m_type,
94 "domain": domain,
95 "sequence_id": seq_id,
96 "clock_identity": clock_id
97 }
98 mqtt_client.publish(MQTT_TOPIC, orjson.dumps(data).decode())
99 elif pkt.haslayer(UDP):
100 # Provide diagnostic feedback for unexpected traffic on PTP ports.
101 matrix_log("CORE", "PTP", inspect.currentframe().f_code.co_name, f"\n[+] Captured UDP Packet from {source} on port {dport}", level="INFO")
102 payload = bytes(pkt[UDP].payload)
103 matrix_log("CORE", "PTP", inspect.currentframe().f_code.co_name, f"Raw Payload (Hex): {payload.hex()}", level="INFO")
104 matrix_log("CORE", "PTP", inspect.currentframe().f_code.co_name, f"Raw Payload (Length): {len(payload)} bytes", level="INFO")
