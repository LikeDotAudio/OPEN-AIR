22 def generate(base_oid, master_script_path):
23 """
24 Generates a complete Bash Installer Script for the SNMP Manager.
25 Ensures path references are absolute and robust for snmpd access.
26 """
27 from oaConfigurationManager.FileReaders.config_reader import Config
28 from oaOchestration.Constants.project_paths import DATA_SNMP_DIR
29 configuration = Config.get_instance()
30
31 # We must use absolute paths for the snmpd.conf 'pass' command
32 # because snmpd runs as a system service and won't know the project root.
33 abs_master_path = os.path.abspath(master_script_path)
34 abs_data_dir = os.path.abspath(DATA_SNMP_DIR)
35
36 installer_lines = [
37 "#!/bin/bash",
38 "# OPEN-AIR SNMP Master Bridge: Automated Installer",
39 "",
40 "# 1. Install Prerequisites",
41 'echo "[SNMP] Installing snmpd and utilities..."',
42 "sudo apt-get update && sudo apt-get install snmpd snmp snmp-mibs-downloader -y",
43 "",
44 "# 2. Permission Fix (Traverse home to access project)",
45 'echo "[SNMP] Adjusting folder permissions for system access..."',
46 "sudo chmod o+x /home /home/anthony",
47 f"sudo chmod -R o+rwx {abs_data_dir}",
48
49 "# 3. Master Configuration",
50 "CONF_FILE='/etc/snmp/snmpd.conf'",
51 'echo "[SNMP] Configuring Master Bridge at $CONF_FILE..."',
52 "",
53 "# Clean up old configurations",
54 "sudo sed -i '/# --- BEGIN OPEN-AIR ---/,/# --- END OPEN-AIR ---/d' $CONF_FILE",
55 "sudo sed -i '/pass .1.3.6.1.4.1.65300/d' $CONF_FILE",
56 "sudo sed -i '/pass 1.3.6.1.4.1.65300/d' $CONF_FILE",
57 "",
58 "# Inject Fresh Master Bridge",
59 "sudo tee -a $CONF_FILE > /dev/null <<EOT",
60 "# --- BEGIN OPEN-AIR ---",
61 f"agentAddress udp:{configuration.SNMP_PORT}",
62 "view all included .1",
63 "rocommunity public default -V all",
64 "rwcommunity private default -V all",
65 f"pass {base_oid.lstrip('.')} {abs_master_path}",
66 "# --- END OPEN-AIR ---",
67 "EOT",
68 "",
69 "# 4. Service Management",
70 'echo "[SNMP] Restarting service..."',
71 "sudo systemctl restart snmpd",
72 "sudo systemctl enable snmpd",
73 "",
74 'echo "[SNMP] Setup Complete!"',
75 ]
76
77 return '\n'.join(installer_lines)
