63def initialize_logging(config, log_dir=None, partition="SYS"):
64 """
65 Configures the global Loguru infrastructure and sinks.
66 """
67 # Ensure partition has an emoji prefix if available
68 partition_with_emoji = f"{get_emoji(partition)} {partition}"
69
70 # Configure global defaults for the 'extra' dictionary.
71 logger.configure(
72 patcher=ptp_patcher,
73 extra={"category": "SYSTEM", "partition": partition_with_emoji, "protocol": None}
74 )
75
76 # Remove existing handlers to avoid duplicate output.
77 logger.remove()
78
79 debug_enabled = config.global_settings.get("debug_enabled", False)
80 console_level = "TRACE" if debug_enabled else "INFO"
81 file_level = "TRACE" if debug_enabled else "INFO"
82
83 # ⚡ PARTITION MUTING
84 if hasattr(config, "DEBUG_MATRIX"):
85 partition_toggle = config.DEBUG_MATRIX.get(f"SYS_{partition.upper()}")
86 if partition_toggle is False:
87 console_level = "WARNING"
88
89 # --- Log Formats ---
90 show_ts = config.global_settings.get("timestamp_logs", True)
91 ts_fmt = "<green>{extra[ptp_time]}</green>|" if show_ts else ""
92 log_format_console = f"{ts_fmt}{LOG_FORMAT_CONSOLE}"
93
94 ts_fmt_plain = "{extra[ptp_time]} | " if show_ts else ""
95 file_format_plain = f"{ts_fmt_plain}{FILE_FORMAT_PLAIN}"
96
97 jsonl_format = JSONL_FORMAT
98
99 # 1. --- Console Sink ---
100 logger.add(
101 sys.stderr,
102 format=log_format_console,
103 level=console_level,
104 enqueue=False,
105 filter=rust_gate_filter,
106 diagnose=False
107 )
108
109 if not log_dir:
110 return
111
112 # --- Ensure Log Directory Exists ---
113 try:
114 os.makedirs(log_dir, exist_ok=True)
115 run_log_dir = os.path.join(log_dir, "ApplicationRunLog")
116 error_log_dir = os.path.join(log_dir, "Errors")
117 comms_log_dir = os.path.join(log_dir, "Comms")
118
119 os.makedirs(run_log_dir, exist_ok=True)
120 os.makedirs(error_log_dir, exist_ok=True)
121 os.makedirs(comms_log_dir, exist_ok=True)
122
123 # 2. --- Application Log Sink ---
124 app_log_pattern = os.path.join(run_log_dir, "Application_{time}.log")
125 logger.add(
126 BatchLogSink(app_log_pattern, format_str=file_format_plain, batch_size=APP_LOG_BATCH_SIZE, interval=APP_LOG_INTERVAL),
127 format=file_format_plain, level=file_level,
128 filter=rust_gate_filter,
129 backtrace=True, diagnose=True
130 )
131
132 # 3. --- Isolated Error Log Sink ---
133 error_log_pattern = os.path.join(error_log_dir, "errors_{time}.log")
134 logger.add(
135 BatchLogSink(error_log_pattern, format_str=file_format_plain, batch_size=ERROR_LOG_BATCH_SIZE, interval=ERROR_LOG_INTERVAL),
136 format=file_format_plain, level="WARNING",
137 filter=rust_gate_filter,
138 backtrace=True, diagnose=True
139 )
140
141 # 4. --- Protocol and Broker Segregated Sinks ---
142 for proto in PROTOCOLS:
143 proto_dir = os.path.join(comms_log_dir, proto)
144 os.makedirs(proto_dir, exist_ok=True)
145 proto_pattern = os.path.join(proto_dir, f"{proto}_{{time}}.log")
146
147 # Use a closure for the filter to correctly capture 'proto'
148 def make_filter(p):
149 return lambda record: record["extra"].get("protocol") == p and rust_gate_filter(record)
150
151 logger.add(
152 BatchLogSink(proto_pattern, format_str=file_format_plain, batch_size=PROTOCOL_LOG_BATCH_SIZE, interval=PROTOCOL_LOG_INTERVAL),
153 format=file_format_plain, level=file_level,
154 filter=make_filter(proto),
155 backtrace=True, diagnose=True
156 )
157
158 except Exception as e:
159 print(f"CRITICAL: Logging filesystem initialization failed: {e}", file=sys.stderr)

166def initialize_test_logging(log_dir: str):
167 """Configures a dedicated sink for test run logs."""

188def get_logger(category: str, emoji_prefix: str = None):
189 """Returns a bound logger instance for a specific subsystem."""
190 if emoji_prefix is None and category.upper() in COMMS_ELEMENTS:
191 emoji = "📡"
192 cat_name = f"COMM: {category.upper()}"
193 else:
194 emoji = emoji_prefix if emoji_prefix else get_emoji(category)
195 cat_name = category.upper()
196
197 full_category = f"{emoji} {cat_name}"
198 padded_category = full_category.ljust(18)
199 return logger.bind(category=padded_category)
