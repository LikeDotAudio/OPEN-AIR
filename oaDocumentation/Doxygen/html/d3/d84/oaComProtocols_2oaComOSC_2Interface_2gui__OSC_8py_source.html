205 def _add_log_entry(self, direction, address, value, topic):
206 now = datetime.datetime.now()
207 timestamp = now.strftime("%H:%M:%S.%f")[:-3]
208
209 # ⚡ STACK BEHAVIOR: Insert at TOP
210 item_id = self.tree.insert("", 0, values=(timestamp, direction, address, value, topic or "-"), tags=(direction,))
211
212 # Cache for investigation
213 self._activity_cache[timestamp] = {
214 "timestamp": timestamp,
215 "direction": direction,
216 "address": address,
217 "value": value,
218 "topic": topic
219 }
220
221 if len(self.tree.get_children()) > 100:
222 last_item = self.tree.get_children()[-1]
223 last_ts = self.tree.item(last_item)["values"][0]
224 if last_ts in self._activity_cache: del self._activity_cache[last_ts]
225 self.tree.delete(last_item)
