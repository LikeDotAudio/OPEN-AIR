23 fn calculate_rotated_point(&self, px: f64, py: f64, cx: f64, cy: f64, cos_t: f64, sin_t: f64) -> (f64, f64) {
24 let delta_x = px - cx;
25 let delta_y = py - cy;
26 (
27 cos_t * delta_x - sin_t * delta_y + cx,
28 sin_t * delta_x + cos_t * delta_y + cy
29 )
30 }

32 fn calculate_position(&self, angle_deg: f64, distance: f64, center_x: f64, center_y: f64) -> (f64, f64) {
33 let rad = angle_deg * PI / 180.0;
34 (
35 center_x + distance * rad.cos(),
36 center_y + distance * rad.sin()
37 )

49 fn calculate_fader_geometry<'py>(&self, py: Python<'py>, config: &Bound<'py, PyDict>) -> PyResult<Bound<'py, PyDict>> {

51 let center_y: f64 = config.get_item("center_y")?.unwrap().extract()?;

53 let angle: f64 = config.get_item("angle")?.unwrap().extract()?;
54 let val_curr: f64 = config.get_item("val_curr")?.unwrap().extract()?;

56 let val_max: f64 = config.get_item("val_max")?.unwrap().extract()?;
57 let rot_curr: f64 = config.get_item("rot_curr")?.unwrap().extract()?;
58
59 let hitbox_width: f64 = config.get_item("hitbox_width")?.unwrap().extract()?;
60 let hitbox_padding: f64 = config.get_item("hitbox_padding")?.unwrap().extract()?;
61 let tick_count: usize = config.get_item("tick_count")?.unwrap().extract()?;
62 let tick_inner_offset: f64 = config.get_item("tick_inner_offset")?.unwrap().extract()?;
63 let tick_outer_offset: f64 = config.get_item("tick_outer_offset")?.unwrap().extract()?;
64 let cap_radius: f64 = config.get_item("cap_radius")?.unwrap().extract()?;
65
66 let t_ang_rad = (angle + 90.0) * PI / 180.0;
67 let cos_t = t_ang_rad.cos();
68 let sin_t = t_ang_rad.sin();
69
70 let result = PyDict::new_bound(py);
71
72 // 1. Hitbox Points
73 let mut hitbox_pts = Vec::with_capacity(8);
74 let hp1 = self.calculate_rotated_point(center_x - hitbox_width/2.0, center_y - track_length/2.0 - hitbox_padding, center_x, center_y, cos_t, sin_t);
75 let hp2 = self.calculate_rotated_point(center_x + hitbox_width/2.0, center_y - track_length/2.0 - hitbox_padding, center_x, center_y, cos_t, sin_t);
76 let hp3 = self.calculate_rotated_point(center_x + hitbox_width/2.0, center_y + track_length/2.0 + hitbox_padding, center_x, center_y, cos_t, sin_t);
77 let hp4 = self.calculate_rotated_point(center_x - hitbox_width/2.0, center_y + track_length/2.0 + hitbox_padding, center_x, center_y, cos_t, sin_t);
78 hitbox_pts.extend_from_slice(&[hp1.0, hp1.1, hp2.0, hp2.1, hp3.0, hp3.1, hp4.0, hp4.1]);
79 result.set_item("hitbox", hitbox_pts)?;
80
81 // 2. Track Points
82 let timestamp = self.calculate_rotated_point(center_x, center_y - track_length/2.0, center_x, center_y, cos_t, sin_t);
83 let te = self.calculate_rotated_point(center_x, center_y + track_length/2.0, center_x, center_y, cos_t, sin_t);
84 result.set_item("track", vec![timestamp.0, timestamp.1, te.0, te.1])?;
85
86 // 3. Tick Points
87 let mut tick_pts = Vec::with_capacity(tick_count * 4);
88 let tick_divisor = (tick_count - 1) as f64;
89 for i in 0..tick_count {
90 let local_y = (-track_length/2.0) + ((i as f64 / tick_divisor) * track_length);
91 let t_start = self.calculate_rotated_point(center_x - tick_inner_offset, center_y + local_y, center_x, center_y, cos_t, sin_t);
92 let t_end = self.calculate_rotated_point(center_x - tick_outer_offset, center_y + local_y, center_x, center_y, cos_t, sin_t);
93 tick_pts.extend_from_slice(&[t_start.0, t_start.1, t_end.0, t_end.1]);
94 }
95 result.set_item("ticks", tick_pts)?;
96
97 // 4. Cap Center
98 let denom = val_max - val_min;
99 let v_norm = if denom != 0.0 { (val_curr - val_min) / denom } else { 0.0 };
100 let cap_center = self.calculate_rotated_point(center_x, center_y + (-track_length/2.0) + (v_norm * track_length), center_x, center_y, cos_t, sin_t);
101 result.set_item("cap_center", vec![cap_center.0, cap_center.1])?;
102
103 // 5. Potentiometer Pointer
104 let pot_start_angle: f64 = 225.0;
105 let pot_extent_max: f64 = 270.0;
106 let percent_max: f64 = 100.0;
107 let pot_degree = pot_start_angle - (rot_curr / percent_max) * pot_extent_max;
108 let pot_rad = pot_degree * PI / 180.0;
109 let pointer_x = cap_center.0 + (cap_radius - 2.0) * pot_rad.cos();
110 let pointer_y = cap_center.1 - (cap_radius - 2.0) * pot_rad.sin();
111 result.set_item("pointer", vec![cap_center.0, cap_center.1, pointer_x, pointer_y])?;
112 result.set_item("pot_degree", pot_degree)?;
113
114 // 6. Label Position
115 let global_center_x: f64 = config.get_item("global_center_x")?.unwrap().extract()?;
116 let global_center_y: f64 = config.get_item("global_center_y")?.unwrap().extract()?;
117 let far_radius: f64 = config.get_item("far_radius")?.unwrap().extract()?;
118 let label_offset_base: f64 = config.get_item("label_offset_base")?.unwrap().extract()?;
119 let label_offset_step: f64 = config.get_item("label_offset_step")?.unwrap().extract()?;
120 let widget_id: usize = config.get_item("widget_id")?.unwrap().extract()?;
121
122 let label_dist = far_radius + label_offset_base + ((widget_id % 2) as f64) * label_offset_step;
123 let label_rad = angle * PI / 180.0;
124 let label_x = global_center_x + label_dist * label_rad.cos();
125 let label_y = global_center_y + label_dist * label_rad.sin();
126 result.set_item("label_pos", vec![label_x, label_y])?;
127
128 Ok(result)
129 }
