29# Global list of callbacks (observers) representing UI dashboard instances.
30_gui_observers = []

32def register_monitor_callback(callback_func):
33 """
34 Registers a GUI callback function to receive real-time YAK traffic.
35
36 Args:
37 callback_func (callable): Function accepting (topic, payload) strings.
38
39 Side Effects:
40 - Appends the function to the global '_gui_observers' list.
41 - Logs the registration event to the matrix gate.
42 """
43 if callback_func not in _gui_observers:
44 _gui_observers.append(callback_func)
45 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name,
46 "✅ [CONFIG] Yak Monitor GUI registered.", level="SUCCESS")
47
48def unregister_monitor_callback(callback_func):
49 """
50 Removes a GUI callback from the active notification list.
51
52 Args:
53 callback_func (callable): The previously registered callback.
54 """
55 if callback_func in _gui_observers:
56 _gui_observers.remove(callback_func)

58def handle_yak_monitor_traffic(message: MqttMessage):
59 """
60 Dispatches filtered MQTT traffic to all registered UI observers.
61
62 Typically invoked by the MQTT Router when traffic matching 'yak/monitor/#'
63 is detected. Ensures that the UI Partition's diagnostic dashboards are
64 kept in sync with background translation events.
65
66 Args:
67 message (MqttMessage): The incoming message containing topic and payload.
68
69 Warn:
70 - Callbacks are executed sequentially; long-running callbacks will
71 block the router thread.
72 """
73 topic = message.topic
74 payload = message.decode_payload()
75
76 # Notify all registered GUI dashboards.
77 for callback in _gui_observers:
78 try:
79 callback(topic, payload)
80 except Exception:
81 logger.exception("❌ [UI] Error updating Yak Monitor dashboard.")
