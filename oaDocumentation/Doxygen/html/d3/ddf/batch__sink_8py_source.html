18class BatchLogSink:

20 ⚡ HIGH PERFORMANCE SINK: Caches logs in memory and writes in batches.
21 Reduces I/O overhead and lock contention on the hot path.

54 def write(self, message):
55 """Standard Loguru sink write method."""
56 rust_sink = get_rust_sink()
57 if rust_sink:
58 # Direct handoff to Rust for non-blocking I/O
59 rust_sink.write(str(self._get_current_file()), str(message))

67 def _trigger_flush(self):
68 """Internal helper to write the buffer to disk."""
69 if not self.buffer:
70 return

72 target_file = self._get_current_file()
73
74 with self._lock:
75 lines_to_write = list(self.buffer)
76 self.buffer.clear()

78 try:
79 with open(target_file, "a", encoding="utf-8") as f:

85 def _flush_loop(self):
86 """Background thread to ensure logs are flushed periodically."""
87 while self._is_running:
88 try:
89 time.sleep(self.interval)

93 print(f"CRITICAL: Log flush loop error for {self.file_path_pattern}: {e}", file=sys.stderr)
