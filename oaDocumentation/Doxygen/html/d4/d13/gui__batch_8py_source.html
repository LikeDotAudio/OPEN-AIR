32 def _get_relative_coords(self, widget, ref_widget):
33 """
34 Calculates coordinates of widget relative to ref_widget.
35 OPTIMIZED: Caches results to prevent millions of redundant lookups.
36 """
37 if not hasattr(self, '_coord_cache'):
38 self._coord_cache = {}
39
40 wid = widget._w
41 if wid in self._coord_cache:
42 return self._coord_cache[wid]
43
44 relative_x, relative_y = 0, 0
45 current_widget = widget
46 ref_path = ref_widget._w if ref_widget else ""
47
48 while current_widget:
49 curr_path = current_widget._w
50 if curr_path == ref_path:
51 break
52 relative_x += current_widget.winfo_x()
53 relative_y += current_widget.winfo_y()
54
55 parent_path = current_widget.winfo_parent()
56 if not parent_path: break
57 current_widget = current_widget.nametowidget(parent_path)
58
59 if widget.winfo_ismapped():
60 self._coord_cache[wid] = (relative_x, relative_y)
61
62 return relative_x, relative_y
