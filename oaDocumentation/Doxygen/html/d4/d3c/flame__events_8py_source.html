32def process_stats_for_ui(ps):
33 """
34 Transforms raw pstats into a rich list of dicts with root attribution
35 and layer tagging (APP/LIB/CORE).
36 """
37 stats = ps.stats
38 vroot = ("<virtual>", 0, "total_execution")
39
40 # ⚡ VIRTUAL ROOT INJECTION (Mirroring make_graph.py logic)
41 # This ensures we can trace every function back to a top-level entry point.
42 roots = [f for f, data in stats.items() if not data[4]]
43 for func in list(stats.keys()):
44 if func == vroot: continue
45 cc, nc, tt, ct, callers = stats[func]
46 if not callers or func in roots:
47 # Inject link to virtual root
48 new_callers = dict(callers)
49 new_callers[vroot] = (cc, nc, tt, ct)
50 stats[func] = (cc, nc, tt, ct, new_callers)
51
52 # 1. Map hierarchy for root attribution
53 roots_to_analyze = [f for f, data in stats.items() if vroot in data[4]]
54
55 callees = defaultdict(list)
56 for func, data in stats.items():
57 for caller in data[4]:
58 callees[caller].append(func)
59
60 func_to_roots = defaultdict(set)
61 root_naming_map = {
62 'openair.py': 'MAIN', 'mqtt': 'MQTT', 'watchdog': 'WATCHDOG',
63 'gui': 'UI', 'worker': 'WORKER', 'visa': 'VISA',
64 'yak': 'YAK', 'manager': 'MANAGER', 'ptp': 'PTP'
65 }
66
67 MAX_LABEL_LENGTH = 10
68
69 def _format_root_label(file_path):
70 """Cleans and formats a file path into a standardized root label."""
71 base = os.path.basename(file_path)
72 clean = base.replace('.py', '').replace('gui_', '').replace('manager_', '')
73 return clean.upper()[:MAX_LABEL_LENGTH]
74
75 def get_root_label(f):
76 desc = f"{f[0]} {f[2]}".lower()
77 for pattern, label in root_naming_map.items():
78 if pattern in desc: return label
79 return _format_root_label(f[0])
80
81 for root_func in roots_to_analyze:
82 label = get_root_label(root_func)
83 queue = deque([root_func])
84 visited = {root_func}
85 while queue:
86 curr = queue.popleft()
87 func_to_roots[curr].add(label)
88 for child in callees[curr]:
89 if child not in visited:
90 visited.add(child)
91 queue.append(child)
92
93 # 2. Build final flattened performance stats
94 performance_stats = []
95 for func, (cc, nc, tt, ct, callers) in stats.items():
96 if func == vroot: continue
97 fname = func[2]
98
99 if func[0] == "~":
100 caller_names = [c[2] for c in callers if c[2] and c[2] != vroot[2]]
101 if caller_names:
102 fname = f"built-in: {fname} (via {', '.join(list(set(caller_names))[:2])})"
103 else:
104 fname = f"built-in: {fname}"
105
106 performance_stats.append({
107 'filename': func[0], 'lineno': func[1], 'funcname': fname or "<unknown>",
108 'ncalls': nc, 'tottime': tt, 'cumtime': ct,
109 'per_call': tt/nc if nc > 0 else 0,
110 'roots': sorted(list(func_to_roots[func])),
111 'raw_key': func
112 })
113
114 return performance_stats

116def generate_table_rows(performance_stats):
117 """Generates HTML table rows with data attributes for filtering/sorting."""
118 rows = []
119 performance_stats.sort(key=lambda x: x['cumtime'], reverse=True)
120 max_cumtime = max(s['cumtime'] for s in performance_stats) if performance_stats else 1
121
122 # ⚡ NO LIMIT: All events displayed
123 for stat in performance_stats:
124 contrib = (stat['cumtime'] / max_cumtime) * 100
125 filename = stat['filename']
126 layer = "LIB" if any(p in filename for p in ["site-packages", "lib/python"]) else "CORE" if (filename == "~" or "/usr/lib" in filename) else "APP"
127
128 tags_html = "".join([f'<span class="tag">{r}</span>' for r in stat["roots"]])
129
130 # Escape function name for HTML display
131 safe_funcname = html.escape(stat["funcname"])
132
133 row = f'<tr data-layer="{layer}" data-roots="{" ".join(stat["roots"])}">'
134 row += f'<td><div style="margin-bottom:4px"><span class="tag tag-layer-{layer}">{layer}</span>{tags_html}</div>'
135 row += f'<span style="color:#fff">{safe_funcname}</span><br><span style="color:#666;font-size:11px">{os.path.basename(filename)}:{stat["lineno"]}</span></td>'
136 row += f'<td class="stat-value">{stat["ncalls"]}</td><td class="stat-value">{stat["tottime"]:.6f}s</td><td class="stat-value">{stat["cumtime"]:.6f}s</td><td class="stat-value">{stat["per_call"]:.6f}s</td>'
137 row += f'<td><div style="width:{min(100, contrib):.1f}%;height:4px;background:#3498db;border-radius:2px"></div><span style="font-size:10px;color:#666">{contrib:.1f}%</span></td></tr>'
138 rows.append(row)
139
140 return "\n".join(rows)
