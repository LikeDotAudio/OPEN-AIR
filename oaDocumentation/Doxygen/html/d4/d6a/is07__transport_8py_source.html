108 def _attempt_connect(self) -> bool:
109 uri = self._connection_params.get("connection_uri", "ws://localhost:8085/is07")
110 try:
111 if self.ws_app:
112 try: self.ws_app.close()
113 except: pass
114
115 self.ws_app = websocket.WebSocketApp(uri,
116 on_open=self._on_open,
117 on_message=self._on_message,
118 on_error=self._on_error,

120
121 self._ws_thread = threading.Thread(target=self.ws_app.run_forever)
122 self._ws_thread.daemon = True
123 self._ws_thread.start()

125 # Wait for connection or timeout

248 self.client.on_connect = self._on_connect

252 if protocol == "secure-mqtt":

254 if username and password:
