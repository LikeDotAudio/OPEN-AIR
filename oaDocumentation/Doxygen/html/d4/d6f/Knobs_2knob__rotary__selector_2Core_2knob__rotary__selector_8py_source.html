86 def _prepare_canvas(self, canvas):
87 """Preserve industrial transparency slice while clearing other elements."""
88 for item in canvas.find_all():
89 if "panel_bg_slice" not in canvas.gettags(item):
90 canvas.delete(item)

92 if hasattr(canvas, 'panel_bg_image') and not canvas.find_withtag("panel_bg_slice"):

95 def _calc_layout(self, width, height, options):
96 """Calculate center points and usable radius."""
97 cx, cy = width / 2, height / 2
98 top_res = 20 if options.get("show_label") and options.get("main_label") else 0
99 bottom_res = 20
100 usable_h = height - top_res - bottom_res
101 adj_cy = cy + (top_res - bottom_res) / 2
102 radius = min(width, usable_h) / 2 - 25
103 return {'cx': cx, 'cy': cy, 'adj_cy': adj_cy, 'radius': radius}

105 def _calc_angles(self, num_pos, continuous):
106 """Determine angular start, step, and total span."""
107 start_angle, total_span = (90, 360) if continuous else (240, 300)
108 angle_step = total_span / (num_pos if continuous else max(1, num_pos - 1))
109 return {'start': start_angle, 'step': angle_step, 'span': total_span}

111 def _draw_track(self, canvas, layout, angles, color, continuous):
112 """Draw the circular or arc track."""
113 cx, adj_cy, radius = layout['cx'], layout['adj_cy'], layout['radius']
114 if continuous:
115 canvas.create_oval(cx - radius, adj_cy - radius, cx + radius, adj_cy + radius, outline=color, width=2)
116 else:
117 canvas.create_arc(cx - radius, adj_cy - radius, cx + radius, adj_cy + radius,
118 start=angles['start'], extent=-angles['span'], style=tk.ARC, outline=color, width=2)
119
120 def _draw_ticks_and_labels(self, canvas, layout, angles, current_idx, positions, colors):
121 """Draw position ticks and labels around the track."""
122 cx, adj_cy, radius = layout['cx'], layout['adj_cy'], layout['radius']
123 for i, pos_text in enumerate(positions):
124 angle_deg = angles['start'] - (i * angles['step'])
125 angle_rad = math.radians(angle_deg)
126 ts_x, ts_y = cx + (radius + 2) * math.cos(angle_rad), adj_cy - (radius + 2) * math.sin(angle_rad)
127 te_x, te_y = cx + (radius + 10) * math.cos(angle_rad), adj_cy - (radius + 10) * math.sin(angle_rad)
128 tl_x, tl_y = cx + (radius + 24) * math.cos(angle_rad), adj_cy - (radius + 24) * math.sin(angle_rad)
129
130 canvas.create_line(ts_x, ts_y, te_x, te_y, fill=colors['secondary'], width=1)
131 canvas.create_text(
132 tl_x, tl_y, text=str(pos_text),
133 fill=colors['indicator'] if i == current_idx else colors['fg'],
134 font=("Helvetica", 8, "bold" if i == current_idx else "normal"),
135 tags="industrial_text"
136 )
137
138 def _draw_knob_elements(self, canvas, layout, angles, current_idx, colors, options):
139 """Draw the physical knob body and pointer."""
140 cx, adj_cy, radius = layout['cx'], layout['adj_cy'], layout['radius']
141 p_angle = angles['start'] - (current_idx * angles['step'])
142 _draw_body(canvas, cx, adj_cy, radius - 5, options.get("shape", "circle"), colors['secondary'], 1,
143 rotation_angle=p_angle, outline_thickness=1, fill_color="", teeth=8)
144 _draw_pointer(canvas, cx, adj_cy, radius - 5, 4, p_angle, options.get("pointer_style", "line"),
145 colors['indicator'], length=radius+14, offset=0, no_center=options.get("no_center", False))
146
147 def _draw_text_overlays(self, canvas, width, height, colors, options):
148 """Draw the main title and selection value text."""
149 cx = width / 2
150 if options.get("show_label") and options.get("main_label"):
151 canvas.create_text(cx, 10, text=options.get("main_label"), fill=colors['fg'],
152 font=("Helvetica", 9, "bold"), anchor="n", tags="industrial_text")
153 if options.get("selection_text"):
154 canvas.create_text(cx, height - 10, text=options.get("selection_text"), fill=colors['indicator'],
155 font=("Helvetica", 9, "bold"), anchor="s", tags="industrial_text")
