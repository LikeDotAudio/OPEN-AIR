50 def handle_event(self, event_type: str, event_data: dict):
51 """
52 Simulates the overlay's event handling logic.
53 If active, it calls its own handler. If inactive, it passes to the underlying element.
54 """
55 print(f"DEBUG: Overlay handling event: {event_type} with data {event_data}") # Example debug log
56 if not self.is_active:
57 # Overlay is inactive, pass event to the underlying element if it has a handler for it.
58 if hasattr(self.underlying_element, event_type):
59 print(f"DEBUG: Overlay inactive, passing '{event_type}' to underlying element.") # Example debug log
60 getattr(self.underlying_element, event_type)(**event_data)
61 return
62
63 # Overlay is active, attempt to intercept and handle the event.
64 if event_type in self._event_handlers:
65 print(f"DEBUG: Overlay active, calling its '{event_type}' handler.") # Example debug log
66 self._event_handlers[event_type](**event_data)
67 else:
68 print(f"DEBUG: Overlay active, but no specific handler for '{event_type}'. Event not intercepted.") # Example debug log
69 # If no specific handler, it might still be considered intercepted if it doesn't reach the element below.
70 # For this test, we assume if it has a handler, it's intercepted.
71 # If no handler, it might still pass through, or be ignored by the overlay.
72 # For this test, focus is on 'intercepting' means not reaching below.
