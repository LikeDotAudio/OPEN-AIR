47 def build_services(self):
48 """
49 Instantiates concrete service implementations and maps their dependencies.
50 """
51 matrix_log("UI", "GUI_MANAGER", inspect.currentframe().f_code.co_name, "🏗️ [ROOT] Composing UI Services...", level="DEBUG")
52
53 # 1. Base Communication Layer
54 mqtt_conn = MqttConnectionManager()
55 sub_router = MqttSubscriberRouter()
56
57 # 2. State & Mirroring Layer
58 state_cache = StateRegistry(mqtt_conn)
59 state_cache.subscriber_router = sub_router
60
61 mirror_engine = StateMirrorEngine(
62 base_topic="OPEN-AIR",
63 subscriber_router=sub_router,
64 root=self.root,
65 state_cache_manager=state_cache
66 )
67 state_cache.state_mirror_engine = mirror_engine
68
69 # 3. Protocol Routing & Specialized Managers
70 protocol_router = ProtocolRouter.get_instance()
71 protocol_router.set_mqtt_manager(mqtt_conn)
72
73 splinker = ControlBroker.get_instance(state_cache, mqtt_conn)
74 protocol_router.set_splinker_manager(splinker)
75
76 # 4. Map to registry
77 self.services.update({
78 "mqtt_conn": mqtt_conn,
79 "sub_router": sub_router,
80 "state_cache": state_cache,
81 "mirror_engine": mirror_engine,
82 "protocol_router": protocol_router,
83 "splinker_manager": splinker
84 })
85
86 # Optional Managers (Conditional based on config)
87 if self.app_constants.SCAN_OSC:
88 self.services["osc_manager"] = OSCManager(state_cache_manager=state_cache, mqtt_connection_manager=mqtt_conn, run_bridge=True)
89 if hasattr(self.services["osc_manager"], "start"): self.services["osc_manager"].start()
90
91 if self.app_constants.SCAN_SNMP:
92 import oaComProtocols.oaComSNMP.Entry as snmp_entry
93 snmp_mgr = snmp_entry.get_manager(state_cache_manager=state_cache, mqtt_connection_manager=mqtt_conn, subscriber_router=sub_router, run_bridge=False)
94 self.services["snmp_manager"] = snmp_mgr
95 snmp_mgr.start()
96 protocol_router.set_snmp_manager(snmp_mgr)
97
98 if self.app_constants.SCAN_MIDI:
99 import oaComProtocols.oaComMidi.Entry as midi_entry
100 self.services["midi_manager"] = midi_entry.get_manager(state_cache_manager=state_cache, run_bridge=False)
101
102 # self.services["midi_manager"] = MidiManager(state_cache, run_bridge=False)
103 self.services["rest_manager"] = RESTManager(state_cache, protocol_router)
104
105 # ST 2138 SMPTE2138 Monitor Logic
106 import oaComProtocols.oaComSMPTE2138.Entry as smpte2138_monitor_entry
107 self.services["smpte2138_monitor_manager"] = smpte2138_monitor_entry.start_monitor(mqtt_conn, sub_router)
108
109 matrix_log("UI", "GUI_MANAGER", inspect.currentframe().f_code.co_name, "✅ [ROOT] UI Service Graph Composed.", level="SUCCESS")
