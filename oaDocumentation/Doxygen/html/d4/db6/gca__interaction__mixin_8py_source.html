13 def _on_press(self, event):
14 self.start_y = event.y
15 m_val = self._safe_get(self.master_value)
16 cap_y = self._get_y_from_val(m_val)
17 cap_h = 60
18 draw_w = self.req_width
19 offset_x = (self.width - draw_w) / 2 if self.width > draw_w else 0
20
21 if (cap_y - cap_h/2) <= event.y <= (cap_y + cap_h/2):
22 if getattr(self, 'mode', 'macro') == "micro":
23 if offset_x <= event.x <= (offset_x + draw_w):
24 strip_w = draw_w / self.num_channels
25 col_idx = int((event.x - offset_x) / strip_w)
26 if 0 <= col_idx < self.num_channels:
27 self.dragging_child = col_idx
28 self.start_val = self._safe_get(self.child_values[col_idx])
29 return
30 self.dragging_master = True
31 self.start_val = m_val
32 else:
33 self.dragging_master = True
34 self.start_val = self._get_val_from_y(event.y)
35 new_v = max(self.min_val, min(self.max_val, self.start_val))
36 self.master_value.set(new_v)
37 if hasattr(self, 'path') and self.path: self.state_mirror_engine.broadcast_gui_change_to_mqtt(self.path)

39 def _on_drag(self, event):
40 if getattr(self, 'dragging_master', False):
41 new_val = self._get_val_from_y(event.y)
42 new_val = max(self.min_val, min(self.max_val, new_val))
43 self.master_value.set(new_val)
44 if hasattr(self, 'path') and self.path: self.state_mirror_engine.broadcast_gui_change_to_mqtt(self.path)
45 elif getattr(self, 'dragging_child', -1) >= 0:
46 dy = self.start_y - event.y
47 val_range = self.max_val - self.min_val
48 pixel_range = self.height - 40
49 delta_val = (dy / pixel_range) * val_range
50 new_val = max(self.min_val, min(self.max_val, self.start_val + delta_val))
51 self.child_values[self.dragging_child].set(new_val)
52 if hasattr(self, 'path') and self.path:
53 self.state_mirror_engine.broadcast_gui_change_to_mqtt(f"{self.path}/ch_{self.dragging_child+1}")
54
55 def _on_release(self, event):

57 self.dragging_child = -1

59 def _on_mousewheel(self, event):
60 delta = 0
61 if sys.platform == "linux":
62 if event.num == 4: delta = 1
63 elif event.num == 5: delta = -1
64 else:
65 delta = 1 if event.delta > 0 else -1
66 if delta == 0: return
67
68 current_val = self._safe_get(self.master_value)
69 val_range = self.max_val - self.min_val
70 step = val_range * 0.05
71 new_val = max(self.min_val, min(self.max_val, current_val + (delta * step)))
72 self.master_value.set(new_val)
73 if hasattr(self, 'path') and self.path: self.state_mirror_engine.broadcast_gui_change_to_mqtt(self.path)

75 def _get_val_from_y(self, y):
76 draw_h = self.height - 40
77 norm = (draw_h - (y - 20)) / draw_h
78 return self.min_val + (norm * (self.max_val - self.min_val))
