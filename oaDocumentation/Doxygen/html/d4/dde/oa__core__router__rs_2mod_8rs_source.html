13struct CoreRouter {

37 // Inbound traffic (Protocol -> Global Bus)
38 fn push_inbound(&self, message: Py<PyAny>) {
39 // Send is non-blocking on unbounded channels. Failure to send implies
40 // a catastrophic channel closure, handled at the supervisor level.
41 let _ = self.inbound_transmitter.send(message);
42 }

44 fn pop_inbound(&self, py: Python<'_>) -> Option<Py<PyAny>> {
45 // try_recv ensures we don't hang the calling thread if no messages are pending.
46 self.inbound_receiver.try_recv().ok().map(|message| message.clone_ref(py))
47 }

49 // Outbound traffic (Global Bus -> Protocol)

56 }
