109 def _create_button_image(self, width, height, text, is_active, is_hovered, is_pressed):
110 width, height = max(1, int(width)), max(1, int(height))
111 image = Image.new("RGBA", (width, height), (0, 0, 0, 0))
112 draw = ImageDraw.Draw(image)
113
114 # Determine colors
115 if is_active:
116 bg = self.active_bg_color
117 border = None # ⚡ REMOVE BORDER WHEN ACTIVE
118 else:
119 bg = self.bg_color
120 border = "#333333"
121
122 if is_pressed:
123 bg = "#000000"
124 border = None # ⚡ REMOVE BORDER WHEN PRESSED
125
126 # Draw Base Rounded Rect
127 r = self.corner_radius
128 draw.rounded_rectangle([0, 0, width-1, height-1], r, fill=bg, outline=border, width=1)
129
130 # Glow Effect if active: Radiating light behind text
131 if is_active and self.glow_intensity > 0:
132 glow_color = self.active_color
133
134 # ⚡ RADIATING LIGHT: Create a soft radial glow centered behind the text
135 try:
136 # Parse hex color to RGB
137 r_c = int(glow_color[1:3], 16)
138 g_c = int(glow_color[3:5], 16)
139 b_c = int(glow_color[5:7], 16)
140
141 center_x, center_y = width / 2, height / 2
142 # ⚡ SCALED GLOW: Intensity scale 0-10. Radius and opacity now map to this scale.
143 # max_radius scales from 0% to ~80% of button size based on intensity (0-10)
144 max_radius = min(width, height) * 0.8 * (self.glow_intensity / 10.0)
145
146 # ⚡ SOFTER GRADIENT: steps scale with intensity for smoothness
147 num_steps = max(10, int(30 * (self.glow_intensity / 10.0)))
148 for i in range(num_steps, 0, -1):
149 radius = (max_radius / num_steps) * i
150 # Smooth quadratic falloff. Max alpha ~100 at intensity 10.
151 alpha_factor = (1 - (i / (num_steps + 1)))**2
152 alpha = int(100 * alpha_factor * (self.glow_intensity / 10.0))
153
154 if alpha > 0:
155 draw.ellipse(
156 [center_x - radius, center_y - radius, center_x + radius, center_y + radius],
157 fill=(r_c, g_c, b_c, alpha)
158 )
159 except Exception as e:
160 logger.error(f"Failed to render radiating glow: {e}")
161
162 # Draw Text
163 try:
164 display_text = get_text(text)
165 f_size = self.active_font_size if is_active else self.inactive_font_size
166 f_style = self.active_font_style if is_active else self.inactive_font_style
167 # This is a simplification; real app might use complex font loading
168 font = ImageFont.load_default()
169
170 # Simple text centering
171 text_x, text_y = width / 2, height / 2
172 draw.text((text_x, text_y), display_text, fill=self.active_text_color if is_active else self.text_color, anchor="mm")
173 except Exception as e:
174 logger.error(f"Error drawing button text: {e}")
175
176 return ImageTk.PhotoImage(image)
