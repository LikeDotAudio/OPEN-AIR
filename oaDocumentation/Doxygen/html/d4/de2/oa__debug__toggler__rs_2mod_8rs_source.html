15#[pyfunction]
16fn toggle_debug_flags_rs(project_root: String, target_state: bool) -> PyResult<bool> {
17 let target_state_str = if target_state { "True" } else { "False" };
18 // Pattern captures: LOCAL_DEBUG, BUILDER_DEBUG, or generic DEBUG assignments
19 let pattern = Regex::new(r"(?m)^(\s*(?:LOCAL_|BUILDER_)?[A-Z_]*DEBUG\s*=\s*)(True|False)(.*)$").unwrap();
20
21 let mut files_modified = 0;
22 let mut _flags_changed = 0;
23
24 let project_path = Path::new(&project_root);
25
26 for entry in WalkDir::new(project_path)
27 .into_iter()
28 .filter_entry(|e| {
29 let name = e.file_name().to_string_lossy();
30 // Respect project boundaries and ignore hidden/temporary directories
31 !name.starts_with('.') &&
32 name != "venv" &&
33 name != "node_modules" &&
34 name != "__pycache__" &&
35 name != "oaDataLogs"
36 })
37 .filter_map(|e| e.ok()) {
38
39 let path = entry.path();
40 if path.is_file() && path.extension().and_then(|s| s.to_str()) == Some("py") {
41 let content = match fs::read_to_string(path) {
42 Ok(c) => c,
43 Err(_) => continue,
44 };
45
46 if !pattern.is_match(&content) {
47 continue;
48 }
49
50 let mut replacements_in_file = 0;
51 let new_content = pattern.replace_all(&content, |caps: &regex::Captures| {
52 let current_value = &caps[2];
53 if current_value != target_state_str {
54 replacements_in_file += 1;
55 format!("{}{}{}", &caps[1], target_state_str, &caps[3])
56 } else {
57 caps[0].to_string()
58 }
59 });
60
61 if replacements_in_file > 0 {
62 if fs::write(path, new_content.to_string()).is_ok() {
63 files_modified += 1;
64 _flags_changed += replacements_in_file;
65 }
66 }
67 }
68 }
69
70 Ok(files_modified > 0)
71}
