54def maker_file_check_for_markers_file():
55 """
56 Checks for the MARKERS.csv file in the DATA directory and loads it if it exists.
57 """
58 current_function = inspect.currentframe().f_code.co_name
59 from oaOchestration.Core.path_initializer import DATA_RUNNING_DIR
60 target_path = DATA_RUNNING_DIR / "MARKERS.csv"
61
62 matrix_log("ui", "importer", "maker_file_check_for_markers_file",
63 f"📥📑🔍 [IMPORTER] {current_function}", "DEBUG")
64
65 if target_path.is_file():
66 matrix_log("ui", "importer", "maker_file_check_for_markers_file",
67 "📥📑✅ [SUCCESS] Found an existing MARKERS.csv file. Attempting to load.", "SUCCESS")
68 try:
69 with open(target_path, newline="") as csvfile:
70 reader = csv.DictReader(csvfile)
71 headers = reader.fieldnames if reader.fieldnames else CANONICAL_HEADERS
72 data = list(reader)
73
74 matrix_log("ui", "importer", "maker_file_check_for_markers_file",
75 "📥📑✅ [SUCCESS] Successfully loaded MARKERS.csv on startup.", "SUCCESS")
76 return headers, data
77 except Exception as e:
78 matrix_log("ui", "importer", "maker_file_check_for_markers_file",
79 f"📥📑❌ [ERROR] Error loading existing MARKERS.csv on startup: {e}", "ERROR")
80 else:
81 matrix_log("ui", "importer", "maker_file_check_for_markers_file",
82 "📥📑🟡 [IMPORTER] No existing MARKERS.csv found. Starting with a blank table.", "DEBUG")
83 return CANONICAL_HEADERS, []
