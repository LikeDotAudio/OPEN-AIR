43 def handle_mouse_press(self, event):
44 """Handles mouse button press events for zooming and panning."""
45 if event.inaxes != self.axes:
46 axes_bbox = self.axes.bbox
47 mouse_x, mouse_y = event.x, event.y
48
49 if event.dblclick:
50 if event.button == LEFT_CLICK:
51 self._handle_axis_double_click(mouse_x, mouse_y, axes_bbox)
52 else:
53 self.reset_view()
54 return
55
56 if event.button in [LEFT_CLICK, MIDDLE_CLICK]:
57 self._initialize_axis_mode(mouse_x, mouse_y, axes_bbox, event.button)
58 return
59
60 if event.button == MIDDLE_CLICK:
61 self.current_xlim = self.axes.get_xlim()
62 self.current_ylim = self.axes.get_ylim()
63 self.mouse_press_position = (event.x, event.y)
64 elif event.button == RIGHT_CLICK:
65 self.selection_rectangle_start = (event.xdata, event.ydata)
66 self.selection_rectangle = Rectangle(
67 (event.xdata, event.ydata), 0, 0,
68 linewidth=1, edgecolor='white', facecolor=(1, 1, 1, RECTANGLE_FACE_ALPHA)
69 )
70 self.axes.add_patch(self.selection_rectangle)
71 self.axes.figure.canvas.draw_idle()

73 def handle_mouse_release(self, event):
74 """Handles mouse button release events to finalize zooming and panning."""
75 if self.active_axis_zoom_mode or self.active_axis_pan_mode:
76 self.active_axis_zoom_mode = self.active_axis_pan_mode = self.mouse_press_position = None
77 self._notify_view_change()
78 return
79
80 if self.mouse_press_position and event.button == MIDDLE_CLICK:
81 self.mouse_press_position = None
82 self.axes.figure.canvas.draw()
83 self._notify_view_change()
84
85 if self.selection_rectangle and event.button == RIGHT_CLICK:
86 start_x, start_y = self.selection_rectangle_start
87 end_x, end_y = event.xdata, event.ydata
88
89 self.selection_rectangle.remove()
90 self.selection_rectangle = self.selection_rectangle_start = None
91
92 if start_x is not None and end_x is not None and start_x != end_x and start_y != end_y:
93 self.axes.set_xlim(min(start_x, end_x), max(start_x, end_x))
94 self.axes.set_ylim(min(start_y, end_y), max(start_y, end_y))
95 self.axes.figure.canvas.draw_idle()
96 self._notify_view_change()
97 elif self.on_context_menu_callback:
98 self.on_context_menu_callback(event)

100 def handle_mouse_motion(self, event):
101 """Handles mouse motion events for real-time zooming and panning."""
102 if (self.active_axis_zoom_mode or self.active_axis_pan_mode) and self.mouse_press_position:
103 press_x, press_y = self.mouse_press_position
104 axes_width = self.axes.bbox.width
105 axes_height = self.axes.bbox.height
106
107 if self.active_axis_zoom_mode:
108 if self.active_axis_zoom_mode == 'x':
109 zoom_scale = max(MINIMUM_ZOOM_SCALE, NORMAL_ZOOM_SCALE - ((event.x - press_x) / ZOOM_SENSITIVITY_DIVISOR))
110 min_val, max_val = self.current_xlim
111 center = (min_val + max_val) / 2
112 span = ((max_val - min_val) / 2) * zoom_scale
113 self.axes.set_xlim(center - span, center + span)
114 else:
115 zoom_scale = max(MINIMUM_ZOOM_SCALE, NORMAL_ZOOM_SCALE - ((event.y - press_y) / ZOOM_SENSITIVITY_DIVISOR))
116 min_val, max_val = self.current_ylim
117 center = (min_val + max_val) / 2
118 span = ((max_val - min_val) / 2) * zoom_scale
119 self.axes.set_ylim(center - span, center + span)
120 else:
121 if self.active_axis_pan_mode == 'x' and axes_width > 0:
122 delta_x = ((self.current_xlim[1] - self.current_xlim[0]) / axes_width) * (event.x - press_x)
123 self.axes.set_xlim(self.current_xlim[0] - delta_x, self.current_xlim[1] - delta_x)
124 elif axes_height > 0:
125 delta_y = ((self.current_ylim[1] - self.current_ylim[0]) / axes_height) * (event.y - press_y)
126 self.axes.set_ylim(self.current_ylim[0] - delta_y, self.current_ylim[1] - delta_y)
127
128 self.axes.figure.canvas.draw_idle()
129 return
130
131 if event.inaxes == self.axes:
132 if self.mouse_press_position and event.button == MIDDLE_CLICK:
133 delta_x = ((self.current_xlim[1] - self.current_xlim[0]) / self.axes.bbox.width) * (event.x - self.mouse_press_position[0])
134 delta_y = ((self.current_ylim[1] - self.current_ylim[0]) / self.axes.bbox.height) * (event.y - self.mouse_press_position[1])
135 self.axes.set_xlim(self.current_xlim[0] - delta_x, self.current_xlim[1] - delta_x)
136 self.axes.set_ylim(self.current_ylim[0] - delta_y, self.current_ylim[1] - delta_y)
137 self.axes.figure.canvas.draw_idle()
138 elif self.selection_rectangle:
139 start_x, start_y = self.selection_rectangle_start
140 end_x, end_y = event.xdata, event.ydata
141 self.selection_rectangle.set_width(abs(end_x - start_x))
142 self.selection_rectangle.set_height(abs(end_y - start_y))
143 self.selection_rectangle.set_xy((min(start_x, end_x), min(start_y, end_y)))
144 self.axes.figure.canvas.draw_idle()

146 def handle_scroll_wheel(self, event):
147 """Handles scroll wheel events for focal zooming."""
148 if event.inaxes != self.axes:
149 return
150
151 zoom_factor = (1.0 / DEFAULT_ZOOM_FACTOR) if event.button == "up" else DEFAULT_ZOOM_FACTOR
152 current_xlim = self.axes.get_xlim()
153 current_ylim = self.axes.get_ylim()
154 event_xdata, event_ydata = event.xdata, event.ydata
155
156 if self.axes.get_xscale() == 'log':
157 self.axes.set_xlim(event_xdata * (current_xlim[0] / event_xdata) ** zoom_factor, event_xdata * (current_xlim[1] / event_xdata) ** zoom_factor)
158 else:
159 self.axes.set_xlim((current_xlim[0] - event_xdata) * zoom_factor + event_xdata, (current_xlim[1] - event_xdata) * zoom_factor + event_xdata)
160
161 if self.axes.get_yscale() == 'log':
162 self.axes.set_ylim(event_ydata * (current_ylim[0] / event_ydata) ** zoom_factor, event_ydata * (current_ylim[1] / event_ydata) ** zoom_factor)
163 else:
164 self.axes.set_ylim((current_ylim[0] - event_ydata) * zoom_factor + event_ydata, (current_ylim[1] - event_ydata) * zoom_factor + event_ydata)
165
166 self.axes.figure.canvas.draw()
167 self._notify_view_change()

181 def _handle_axis_double_click(self, mouse_x, mouse_y, axes_bbox):
182 """Handles double-click events on axis labels to set custom limits."""
183 if axes_bbox.x0 <= mouse_x <= axes_bbox.x1:
184 if axes_bbox.y0 - AXIS_CLICK_THRESHOLD_PIXELS < mouse_y < axes_bbox.y0:
185 self._request_axis_limit('x', 'min')
186 elif mouse_y > axes_bbox.y1:
187 self._request_axis_limit('y', 'max')
188 elif axes_bbox.y0 <= mouse_y <= axes_bbox.y1:
189 if axes_bbox.x0 - AXIS_CLICK_THRESHOLD_PIXELS < mouse_x < axes_bbox.x0:
190 self._request_axis_limit('y', 'min')
191 elif mouse_x > axes_bbox.x1:
192 self._request_axis_limit('x', 'max')

194 def _request_axis_limit(self, axis_name, limit_side):
195 """Opens a dialog to manually set an axis limit."""
196 current_limits = self.axes.get_xlim() if axis_name == 'x' else self.axes.get_ylim()
197 limit_index = 0 if limit_side == 'min' else 1
198
199 new_value = simpledialog.askfloat("Limit", f"{axis_name.upper()} {limit_side}:", initialvalue=current_limits[limit_index])
200
201 if new_value is not None:
202 if axis_name == 'x':
203 self.axes.set_xlim(left=new_value) if limit_side == 'min' else self.axes.set_xlim(right=new_value)
204 else:
205 self.axes.set_ylim(bottom=new_value) if limit_side == 'min' else self.axes.set_ylim(top=new_value)
206 self.axes.figure.canvas.draw_idle()
207 self._notify_view_change()

209 def _initialize_axis_mode(self, mouse_x, mouse_y, axes_bbox, mouse_button):
210 """Sets the active zooming or panning mode based on click location."""
211 mode_attribute = 'active_axis_zoom_mode' if mouse_button == LEFT_CLICK else 'active_axis_pan_mode'
212
213 if (axes_bbox.x0 <= mouse_x <= axes_bbox.x1) and (axes_bbox.y0 - AXIS_CLICK_THRESHOLD_PIXELS < mouse_y < axes_bbox.y0 or mouse_y > axes_bbox.y1):
214 setattr(self, mode_attribute, 'x')
215 self.mouse_press_position = (mouse_x, mouse_y)
216 self.current_xlim = self.axes.get_xlim()
217 elif (axes_bbox.y0 <= mouse_y <= axes_bbox.y1) and (axes_bbox.x0 - AXIS_CLICK_THRESHOLD_PIXELS < mouse_x < axes_bbox.x0 or mouse_x > axes_bbox.x1):
218 setattr(self, mode_attribute, 'y')
219 self.mouse_press_position = (mouse_x, mouse_y)
220 self.current_ylim = self.axes.get_ylim()
