19def validate_and_sanitize_json(data: dict) -> dict:
20 """
21 Ensures the data is a valid JSON structure before publishing.
22 Uses Rust for strict validation and Python fallback for serializability check.
23 """
24 if not isinstance(data, dict):
25 logger.error(f"❌ JSON validation error: Expected dict, got {type(data)}")
26 return data
27
28 if RUST_ENABLED:

30 # Rust performs deep validation and ensures strict compliance
31 oasafetycore_rs.validate_json(data)

34 logger.error(f"❌ [RUST] JSON validation failed: {e}")
35 return data
36
37 # --- Python Fallback Logic ---

39 # Fallback to orjson serialization check
