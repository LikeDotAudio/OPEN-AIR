121 def _refresh_tree(self):
122 """Fetches the tree keys from the server and rebuilds the view."""
123 def worker():
124 try:
125 # We use the new tree endpoint to get keys
126 response = requests.get(f"{self.base_url}/api/v1/system/tree", timeout=2)
127 if response.status_code == 200:
128 data = response.json()
129 self.after(0, lambda: self._rebuild_tree_ui(data))
130 except Exception as e:
131 err_msg = str(e)
132 self.after(0, lambda: self.result_text.insert(tk.END, f"\n❌ Failed to sync tree: {err_msg}"))
133
134 threading.Thread(target=worker, daemon=True).start()

136 def _rebuild_tree_ui(self, data):
137 """Reconstructs the hierarchical Treeview from a flat dict."""
138 self.tree.delete(*self.tree.get_children())
139
140 # Build node structure
141 nodes = {} # path -> id
142
143 # Sort topics to ensure parents are created before children
144 topics = sorted(data.keys())
145
146 for topic in topics:
147 parts = topic.split('/')
148 path = ""
149 parent_id = ""
150
151 for i, part in enumerate(parts):
152 path = f"{path}/{part}" if path else part
153 if path not in nodes:
154 # If it's the last part, show the value
155 val_str = str(data[topic]) if i == len(parts)-1 else ""
156 nodes[path] = self.tree.insert(parent_id, "end", text=part, values=(val_str,), open=True)
157 parent_id = nodes[path]

159 def _test_selected(self, event=None):
160 selected = self.tree.selection()
161 if not selected: return
162
163 # Reconstruct path from tree hierarchy
164 path_parts = []
165 curr = selected[0]
166 while curr:
167 path_parts.insert(0, self.tree.item(curr)["text"])
168 curr = self.tree.parent(curr)
169
170 endpoint = "/".join(path_parts)
171 self._run_test(endpoint)

177 def _run_test(self, endpoint):
178 """Executes a GET request against an endpoint and displays the raw JSON result."""
179 self.result_text.delete("1.0", tk.END)
180 self.result_text.insert(tk.END, f"📡 REQUEST: GET {self.base_url}/{endpoint}\n")

183 def worker():
184 try:
185 import time
186 t0 = time.time()
187 response = requests.get(f"{self.base_url}/{endpoint}", timeout=5)
188 t1 = time.time()
189
190 content = json.dumps(response.json(), indent=4)
191
192 self.after(0, lambda: self._display_result(response.status_code, content, t1-t0))
193 except Exception as e:
194 err_msg = str(e)
195 self.after(0, lambda: self.result_text.insert(tk.END, f"❌ TEST FAILED: {err_msg}"))
196
197 threading.Thread(target=worker, daemon=True).start()
