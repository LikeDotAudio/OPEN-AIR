22# Creates the base Matplotlib figure, axes, and canvas for embedding into a Tkinter frame.
23# This function initializes a new Matplotlib figure and adds a subplot to it,
24# then embeds this figure into a Tkinter canvas widget.

26# parent_frame (tk.Frame): The Tkinter frame to embed the plot into.

30# tuple: A tuple containing the Matplotlib figure, axes, and FigureCanvasTkAgg instance.
31def create_base_plot(parent_frame: tk.Frame, config: dict[str, Any]) -> tuple:
32 """
33 Creates the FigureCanvasTkAgg and basic Axis.
34 Returns (figure, axis, canvas).
35 """
36 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"🔬🏗️📊 [BUILDER] graph_builder: Creating base Matplotlib plot for '{config.get('path', 'Unknown')}'.", level="DEBUG")
37
38 layout_config = config.get("layout", {})
39 # ⚡ DIMENSION ENFORCEMENT: Use explicit geometry or layout, fallback to 500x400
40 geom = config.get("geometry", {})
41 width = config.get("width") or geom.get("width") or layout_config.get("width") or 500
42 height = config.get("height") or geom.get("height") or layout_config.get("height") or 400
43
44 # Ensure we don't start with 0 or 1 which triggers the "pixel wide" bug
45 width = max(1, int(float(width)))
46 height = max(1, int(float(height)))
47
48 # ⚡ Enable transparency at the Figure level
49 fig = Figure(
50 figsize=(width / 100, height / 100),
51 dpi=100,
52 facecolor='none'
53 )
54 # ⚡ Enable transparency at the Axis level
55 ax = fig.add_subplot(111, facecolor='none')
56
57 canvas = FigureCanvasTkAgg(fig, master=parent_frame)
58 canvas_widget = canvas.get_tk_widget()
59
60 # ⚡ Ensure the Tkinter widget itself is configured for transparency
61 # (Though it still needs the Industrial Transparency slice to look perfect)
62 try:
63 canvas_widget.configure(highlightthickness=0, bd=0)
64 canvas_widget.pack(side=tk.TOP, fill=tk.BOTH, expand=True)
65 except tk.TclError as e:
66 matrix_log("UI", "GUI_ELEMENTS", "create_base_plot", f"⚠️ Canvas widget configuration skipped: {e}", "TRACE")
67
68 return fig, ax, canvas
