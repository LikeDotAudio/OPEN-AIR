52 def start(self):
53 if self._running: return
54 self._running = True
55
56 existing_callback = self.mqtt.on_message_callback
57 def failover_message_handler(client, userdata, message):
58 if existing_callback:
59 try: existing_callback(client, userdata, message)
60 except Exception: pass
61 self._on_heartbeat(message)
62
63 self.mqtt.on_message_callback = failover_message_handler
64 self.mqtt.subscribe(self.discovery_topic)
65
66 threading.Thread(target=self._heartbeat_loop, daemon=True).start()
67 threading.Thread(target=self._monitor_loop, daemon=True).start()

69 if app_constants.ROUTER_FAILOVER_LOGS:
70 matrix_log("comms", "broker", "start",
71 f"🛡️ [FAILOVER] Manager active. Instance: {self.guid}", "INFO")

73 def _heartbeat_loop(self):
74 while self._running:
75 try:
76 payload = {
77 "guid": self.guid,
78 "active": self.is_active,
79 "start_ts": self._start_ts,
80 "timestamp": time.time()
81 }
82 self.mqtt.publish(self.heartbeat_topic,
83 orjson.dumps(payload).decode(), retain=False)
84 except Exception: pass
85 time.sleep(self.HEARTBEAT_INTERVAL)
86
87 def _on_heartbeat(self, message):
88 if "System/Failover/Heartbeat/" in message.topic:
89 try:
90 data = orjson.loads(message.payload)
91 peer_guid = data.get("guid")
92 if peer_guid and peer_guid != self.guid:
93 with self._lock:
94 self._peers[peer_guid] = {
95 "timestamp": time.time(),
96 "start_ts": data.get("start_ts", time.time())
97 }
98 except Exception: pass
99
100 def _monitor_loop(self):
101 while self._running:
102 now = time.time()
103 with self._lock:
104 dead_peers = [g for g, p in self._peers.items()
105 if (now - p["timestamp"]) > self.FAILOVER_TIMEOUT]
106 for g in dead_peers:
107 del self._peers[g]
108 if app_constants.ROUTER_FAILOVER_LOGS:
109 matrix_log("comms", "broker", "_monitor_loop",
110 f"💀 [FAILOVER] Peer {g} lost.", "WARNING")
111
112 candidates = [{"guid": self.guid, "start_ts": self._start_ts}]
113 for g, p in self._peers.items():
114 candidates.append({"guid": g, "start_ts": p["start_ts"]})
115
116 candidates.sort(key=lambda x: (x["start_ts"], x["guid"]))
117
118 winner = candidates[0]["guid"]
119 should_be_active = (winner == self.guid)
120
121 if should_be_active != self.is_active:
122 self._transition_state(should_be_active)
123
124 time.sleep(0.5)
125
126 def _transition_state(self, become_active):
127 self.is_active = become_active
128 role = "PRIMARY" if become_active else "SHADOW"
129
130 try:
131 status_payload = {"guid": self.guid, "role": role,
132 "active": become_active}
133 self.mqtt.publish(self.status_topic,
134 orjson.dumps(status_payload).decode(), retain=True)
135 except Exception: pass
136
137 if become_active:
138 if app_constants.ROUTER_FAILOVER_LOGS:
139 matrix_log("comms", "broker", "_transition_state",
140 "👑 [FAILOVER] Promoted to MASTER.", "SUCCESS")
141 self._set_router_state(True)
142 else:
143 if app_constants.ROUTER_FAILOVER_LOGS:
144 matrix_log("comms", "broker", "_transition_state",
145 "💤 [FAILOVER] Entering passive SHADOW mode.", "INFO")
146 self._set_router_state(False)

148 def _set_router_state(self, active):
149 if hasattr(self.router, "set_active_state"):
150 self.router.set_active_state(active)
