39 """
40 Manages the firehose buffer and provides forensic investigation tools.

42 The Monitor acts as a passive observer within the router, capturing

45 def __init__(self, local_guid):
46 """
47 Initializes the monitor with a rolling buffer.
48
49 Args:
50 local_guid (str): The unique ID of the local router instance.

52 self.local_guid = local_guid
53 # ⚡ OPTIMIZATION: Use deque with maxlen for O(1) appends and auto-eviction.
54 self.firehose = deque(maxlen=2000)
55 self._firehose_lock = threading.Lock()

60 self._byte_count = 0
61 self._last_telemetry_ts = time.time()

64 "bps": 0,
65 "latency_ms": 0,

85 def append_to_firehose(self, message):
86 """
87 Maintains the rolling "firehose" buffer (last 2000 packets).
88
89 Args:
90 message (dict): The normalized message packet to store.
91 """
92 with self._firehose_lock:
93 # ⚡ O(1) append to the left
94 self.firehose.appendleft(message)
95
96 # Update Telemetry Counters
97 self._message_count += 1
98 # ⚡ OPTIMIZATION: Use a cheap estimation instead of orjson.dumps on every message.
99 # This reduces CPU usage in the critical ingest path.
100 self._byte_count += 256 # Assume 256 bytes per packet on average
101
102 # Calculate Latency (Ingest TS vs Current TS)
103 if "timestamp" in message:
104 latency = (time.time() - message["timestamp"]) * 1000
105 self._telemetry["latency_ms"] = (self._telemetry["latency_ms"] * 0.9) + (latency * 0.1)
106
107 # Publish telemetry once per second
108 now = time.time()
109 if now - self._last_telemetry_ts >= 1.0:
110 self._publish_telemetry(now)
111
112 def _publish_telemetry(self, now):
113 elapsed = now - self._last_telemetry_ts
114 if elapsed <= 0: return
115
116 self._telemetry["pps"] = int(self._message_count / elapsed)
117 self._telemetry["bps"] = int((self._byte_count * 8) / elapsed)
118 self._telemetry["total_messages"] += self._message_count
119
120 # Reset interval counters
121 self._message_count = 0
122 self._byte_count = 0
123 self._last_telemetry_ts = now
124
125 # Broadcast to observers (UI)
126 telemetry_message = {
127 "timestamp": now,
128 "source": "SYSTEM",
129 "topic": "OPEN-AIR/System/Monitor/Telemetry",
130 "value": self._telemetry.copy(),
131 "meta": {"message_type": "TELEMETRY", "is_settled": True}
132 }
133 self.broadcast_to_observers(telemetry_message)

135 def get_splink_relationship(self, message_ts):
136 """
137 Correlates a message with its connected "Splink" partner.
138
139 Splinks represent logical patches between topics (e.g., Fader A
140 controls EQ Gain B). This method identifies the partner message
141 in the firehose based on the metadata links.
142
143 Args:
144 message_ts (float/str): The timestamp of the message to investigate.
145
146 Returns:
147 tuple: (source_ts, dest_ts) or (None, None).
148 """
149 with self._firehose_lock:
150 match = next((m for m in self.firehose if
151 f"{m['timestamp']:.6f}" == message_ts or m['timestamp'] == message_ts), None)
152
153 if not match:
154 return None, None
155
156 s_id = (match["meta"].get("splink_id") or
157 match["meta"].get("splinker_source"))
158
159 if not s_id:
160 return None, None
161
162 is_active = match["meta"].get("splink_active")
163
164 if is_active:
165 candidates = [m for m in self.firehose if
166 m["meta"].get("splinker_source") == s_id]
167 else:
168 candidates = [m for m in self.firehose if
169 m["meta"].get("splink_id") == s_id]
170
171 if not candidates:
172 return (match["timestamp"] if is_active else None), (None if is_active else match["timestamp"])
173
174 partner = min(candidates, key=lambda m: abs(m["timestamp"] - match["timestamp"]))
175
176 if is_active:
177 return match["timestamp"], partner["timestamp"]
178 else:
179 return partner["timestamp"], match["timestamp"]

181 def get_dpi_report(self, message_ts):
182 """
183 Generates a human-readable forensic report for a specific packet.
184
185 Constructs a visual ASCII report detailing the packet's origin,
186 session ID, routing strategy, and DPI-enriched metadata.
187
188 Args:
189 message_ts (float/str): The timestamp of the packet to report on.
190
191 Returns:
192 str: The formatted report.
193 """
194 match = None
195 with self._firehose_lock:
196 match = next((m for m in self.firehose if
197 f"{m['timestamp']:.6f}" == message_ts or m['timestamp'] == message_ts), None)
198
199 if not match:
200 return "Packet not found in firehose buffer."
201
202 report = []
203 source = match['source']
204 src_desc = SOURCE_DESCRIPTIONS.get(source, f"❓ [{source}] - Unknown origin.")
205
206 report.append("╔════════════ PACKET INVESTIGATION REPORT ════════════╗")
207 report.append(f" TIME (UTP) : {match['timestamp']}")
208
209 p_id = match.get("partition", "UNKNOWN")
210 is_local_session = (match['guid'] == self.local_guid)
211 is_local_partition = (p_id == app_constants.PARTITION_ID)
212
213 p_desc = f"{p_id} ({'HERE' if is_local_partition else 'REMOTE'})"
214 s_desc = f"{match['guid']} ({'THIS SESSION' if is_local_session else 'REMOTE'})"
215
216 report.append(f" SESSION : {s_desc}")
217 report.append(f" PARTITION : {p_desc}")
218 report.append("╟──────────────────────────────────────────────────────╢")
219 report.append(f" SOURCE : {src_desc}")
220 report.append(f" TOPIC/PATH : {match['topic']}")
221 report.append(f" RAW VALUE : {match['value']}")
222 report.append("╟──────────────────────────────────────────────────────╢")
223
224 strat = match.get("strategy", "BROADCAST")
225 report.append(f" STRATEGY : {strat}")
226
227 lifecycle_parts = []
228 for char in strat:
229 if char in EMOJI_TO_WORD:
230 lifecycle_parts.append(f"[{EMOJI_TO_WORD[char]}]")
231
232 if lifecycle_parts:
233 if "[PUSH]" in lifecycle_parts:
234 p_idx = lifecycle_parts.index("[PUSH]")
235 source_part = " ".join(lifecycle_parts[:p_idx])
236 dest_part = " ".join(lifecycle_parts[p_idx+1:])
237 lifecycle_str = f"{source_part} -> PUSH -> {dest_part}"
238 else:
239 lifecycle_str = " -> ".join(lifecycle_parts)
240 report.append(f" LIFECYCLE : {lifecycle_str}")
241
242 if match["meta"]:
243 report.append("╟── DEEP PACKET INSPECTION (DPI) ──────────────────────╢")
244 for k, v in match["meta"].items():
245 if k in ["ui_tags"]: continue
246 k_disp = k.replace("_", " ").upper()
247 prefix = "🔬 "
248 if "mib" in k.lower(): prefix = "Ⓢ 🔍 "
249 if "osc" in k.lower(): prefix = "🅾️ 🔍 "
250 if "midi" in k.lower(): prefix = "🎹 🔍 "
251 if "mutation" in k.lower(): prefix = "🚨 ⚡ "
252 report.append(f" {prefix}{k_disp}: {v}")
253
254 try:
255 if isinstance(match['value'], dict):
256 report.append("╟── PAYLOAD DISSECTION ────────────────────────────────╢")
257 pretty_json = orjson.dumps(match['value'],
258 option=orjson.OPT_INDENT_2).decode()
259 report.append(f"{pretty_json}")
260 except:
261 pass
262
263 report.append("╚════════════════════════ END ════════════════════════╝")
264 return "\n".join(report)
