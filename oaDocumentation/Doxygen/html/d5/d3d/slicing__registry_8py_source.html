83 def _detect_visual_folds(self):
84 """Scans children for widgets defining a physical layout fold."""
85 folds = []
86 if not hasattr(self, 'scroll_frame'): return folds
87
88 s_ry = self.scroll_frame.winfo_rooty()
89 wh = self.scroll_frame.winfo_height()
90 if wh <= 0: return folds
91
92 for child in self.scroll_frame.winfo_children():
93 if self._is_fold_widget(child):
94 wy = child.winfo_rooty() + (child.winfo_height() / 2) - s_ry
95 folds.append({"position_pct": wy / wh, "orientation": "horizontal"})
96
97 folds.sort(key=lambda x: x["position_pct"])
98 return folds

105 def _folds_require_update(self, new_folds, bg_config):
106 """Determines if the detected folds differ from the current background configuration."""
107 fold_params = bg_config.get("parameters", bg_config).get("metal_fold", {})
108 existing = [c for c in fold_params.get("creases", []) if c.get('orientation') == 'horizontal']
109
110 if len(new_folds) != len(existing): return True
111 for f, e in zip(new_folds, existing):
112 if abs(f["position_pct"] - float(e["position_pct"])) > SLICING_POSITION_EPSILON:
113 return True
114 return False
