76def _execute_pip_command(action, package_name, console_print_func,
77 debug_log_func):
78 """
79 Invokes the 'pip' module to perform a specific package action.
80 """
81 # Use current interpreter to avoid 'pip' version or path mismatches.
82 command = [sys.executable, "-m", "pip", action, package_name,
83 FLAG_BREAK_SYSTEM_PACKAGES]
84 if action == ACTION_UNINSTALL:
85 command.append(FLAG_ASSUME_YES)
86
87 try:
88 # Check=False allows us to handle non-zero exits via status code.
89 result = subprocess.run(command, capture_output=True, text=True,
90 check=False)
91 if result.returncode == 0:
92 if LOCAL_DEBUG:
93 logger.success(f"✅✅✅ [SUCCESS] Pip {action} successful "
94 f"for {package_name}.")
95 return True
96 else:
97 if action == ACTION_UNINSTALL:
98 stderr_lower = result.stderr.lower()
99 # Treat 'already uninstalled' as a success to avoid false errors.
100 if (ERROR_NOT_INSTALLED in stderr_lower or
101 "permission denied" in stderr_lower):
102 return True
103
104 error_message = f"❌ [ERROR] Pip {action} failed for {package_name}: {result.stderr.strip()}"
105 console_print_func(error_message)
106 logger.error(f"🔍📦🔗 [DEPENDENCY] {error_message}")
107 return False
108 except Exception as e:
109 error_message = f"💥 [CRITICAL] Pip execution error: {e}"
110 console_print_func(error_message)
111 logger.error(f"🔍📦🔗 [DEPENDENCY] {error_message}")
112 return False

114def action_check_dependancies(console_print_func, debug_log_func,

117 Validates all listed packages and optionally repairs them via pip.

131 elif friendly_name == "python-usbtmc": package_name_for_pip = "python-usbtmc"
132 elif friendly_name == "python-vxi11": package_name_for_pip = "python-vxi11"
133 elif friendly_name == "pyserial": package_name_for_pip = "pyserial"
134 elif friendly_name == "beautifulsoup4 (bs4)": package_name_for_pip = "beautifulsoup4"
135 elif friendly_name == "Pillow (for support)": package_name_for_pip = "Pillow"
136 elif friendly_name == "python-osc": package_name_for_pip = "python-osc"
137 elif friendly_name == "python-rtmidi": package_name_for_pip = "python-rtmidi"
138 elif friendly_name == "protobuf": package_name_for_pip = "protobuf"
