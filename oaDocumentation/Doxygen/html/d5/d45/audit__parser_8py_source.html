13def parse_audit_log(file_path):
14 """
15 Parses an audit log file and returns a list of result dictionaries.
16 Captures full 1:1 content for reporting.
17 """
18 if not os.path.exists(file_path):
19 return []
20
21 with open(file_path, encoding="utf-8") as f:
22 # ⚡ OPTIMIZATION: Limit read to 2MB.
23 content = f.read(2000000)
24
25 # Split by "## File: "
26 sections = re.split(r'## File: ', content)
27 results = []
28
29 for section in sections[1:]: # Skip the session header
30 lines = section.split('\n')
31 if not lines:
32 continue
33
34 header_line = lines[0].strip()
35
36 # Extract filename and status
37 # Example: AuditArchitecture.toml (PASSED)
38 match = re.search(r'^(.*?)\s*\‍((PASSED|FAILED|SKIPPED|ERROR|UNEXPECTED ERROR)\‍)', header_line)
39 if match:
40 name = match.group(1).strip()
41 status_raw = match.group(2).upper()
42
43 # Map statuses to reporting categories
44 status_map = {
45 "PASSED": "passed",
46 "FAILED": "failed",
47 "SKIPPED": "skipped",
48 "ERROR": "error",
49 "UNEXPECTED ERROR": "error"
50 }
51 status = status_map.get(status_raw, "error")
52
53 # Capture EVERYTHING after the header line as the body
54 # We strip the trailing separator '---' if it exists at the end
55 body = "\n".join(lines[1:]).strip()
56 if body.endswith('---'):
57 body = body[:-3].strip()
58
59 # For passed audits, the body goes to description
60 # For failed/error, the body goes to cause
61 description = ""
62 cause = ""
63
64 if status == "passed":
65 description = body
66 else:
67 cause = body
68 # Extract a short summary for the description if it's a failure
69 desc_match = re.search(r'\*\*Partial Output:\*\*\s*(.*?)\s*(?:---|##|$)', section, re.DOTALL)
70 if desc_match:
71 description = desc_match.group(1).strip().split('\n')[0]
72 else:
73 description = f"Architectural audit failure in {name}"
74
75 results.append({
76 "name": f"Audit: {name}",
77 "status": status,
78 "description": description,
79 "cause": cause, # Preserve raw newlines, HTML generator will handle formatting
80 "duration": "N/A"
81 })
82
83 return results
