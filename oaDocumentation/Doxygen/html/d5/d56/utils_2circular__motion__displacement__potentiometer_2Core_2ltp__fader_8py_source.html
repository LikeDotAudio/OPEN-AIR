85 def update_fader_position(self):
86 """Calculates the fader's center position in the circular array."""
87 distance = NEAR_RADIUS + (self.track_length / RADIUS_DIVISOR)

90 def render_fader(self):
91 """Draws all fader components (track, ticks, cap, pointer) on the canvas."""
92 self.canvas.delete(self.tag_root)
93 if not self.is_visible:
94 return
95
96 # Prepare config for batch calculation (RUST OPTIMIZED)
97 # Some values are hardcoded in the original script or based on constants
98 TICK_INNER_OFFSET = 15
99 TICK_OUTER_OFFSET_MAJOR = TICK_INNER_OFFSET + 10
100 TICK_OUTER_OFFSET_MINOR = TICK_INNER_OFFSET + 5
101
102 config = {
103 "center_x": float(self.center_x),
104 "center_y": float(self.center_y),
105 "track_length": float(self.track_length),
106 "angle": float(self.angle),
107 "val_curr": float(self.value_current),
108 "val_min": float(self.value_min),
109 "val_max": float(self.value_max),
110 "rot_curr": float(self.rotation_current),
111 "hitbox_width": 40.0, # Approximate
112 "hitbox_padding": 10.0,
113 "tick_count": TICK_COUNT,
114 "tick_inner_offset": float(TICK_INNER_OFFSET),
115 "tick_outer_offset": float(TICK_OUTER_OFFSET_MAJOR), # We handle minor ticks below
116 "cap_radius": float(CAP_RADIUS),
117 "global_center_x": 600.0, # Default from CMDP
118 "global_center_y": 450.0,
119 "far_radius": float(FAR_RADIUS),
120 "label_offset_base": float(LABEL_FAR_RADIUS_OFFSET),
121 "label_offset_step": float(LABEL_STAGGER_OFFSET),
122 "widget_id": self.widget_id
123 }

125 geo = CircularMath.calculate_fader_geometry(config)

127 if geo:
128 # 1. Track
129 self.canvas.create_line(*geo["track"], fill=TRACK_BACKGROUND_COLOR, width=6, capstyle=tk.ROUND, tags=self.tag_root)
130 self.canvas.create_line(*geo["track"], fill=TRACK_FOREGROUND_COLOR, width=2, capstyle=tk.ROUND, tags=self.tag_root)
131
132 # 2. Ticks
133 tick_pts = geo["ticks"]
134 for i in range(TICK_COUNT):
135 # Major/Minor logic
136 if i % MAJOR_TICK_INTERVAL != 0:
137 # Recalculate end point for minor tick if Rust gave us major for all
138 # (In a real scenario, we'd adjust Rust to handle both)
139 p1x, p1y = tick_pts[i*4], tick_pts[i*4+1]
140 p2x, p2y = tick_pts[i*4+2], tick_pts[i*4+3]
141 # Simple interpolation to shorten minor ticks
142 p2x = p1x + (p2x - p1x) * (MINOR_TICK_LENGTH / MAJOR_TICK_LENGTH)
143 p2y = p1y + (p2y - p1y) * (MINOR_TICK_LENGTH / MAJOR_TICK_LENGTH)
144 self.canvas.create_line(p1x, p1y, p2x, p2y, fill=TICK_COLOR, width=2, tags=self.tag_root)
145 else:
146 self.canvas.create_line(tick_pts[i*4], tick_pts[i*4+1], tick_pts[i*4+2], tick_pts[i*4+3], fill=TICK_COLOR, width=2, tags=self.tag_root)
147
148 # 3. Cap
149 cx, cy = geo["cap_center"]
150 cap_fill = CAP_HOVER_COLOR if self.is_hovered else CAP_NORMAL_COLOR
151 cap_outline = CAP_HOVER_OUTLINE if self.is_hovered else CAP_NORMAL_OUTLINE
152 cap_outline_width = CAP_HOVER_WIDTH if self.is_hovered else CAP_NORMAL_WIDTH
153 self.canvas.create_oval(cx - CAP_RADIUS, cy - CAP_RADIUS, cx + CAP_RADIUS, cy + CAP_RADIUS,
154 fill=cap_fill, outline=cap_outline, width=cap_outline_width, tags=(self.tag_root, "cap"))
155
156 # 4. Sweep & Pointer
157 pot_deg = geo["pot_degree"]
158 self.canvas.create_arc(cx - CAP_RADIUS + CAP_INNER_PADDING, cy - CAP_RADIUS + CAP_INNER_PADDING,
159 cx + CAP_RADIUS - CAP_INNER_PADDING, cy + CAP_RADIUS - CAP_INNER_PADDING,
160 start=SWEEP_START_ANGLE, extent=-(SWEEP_START_ANGLE - pot_deg),
161 style=tk.ARC, outline=self.color_highlight, width=4, tags=self.tag_root)
162
163 self.canvas.create_line(*geo["pointer"], fill=self.color_highlight, width=POINTER_WIDTH, tags=self.tag_root)
164
165 # 5. Values & Label
166 self.canvas.create_text(cx, cy + ROTATION_VALUE_Y_OFFSET, text=str(int(self.rotation_current)), fill=self.color_highlight, font=("Arial", 9, "bold"), tags=self.tag_root)
167 self.canvas.create_text(cx, cy - CURRENT_VALUE_Y_OFFSET, text=str(int(self.value_current)), fill=VALUE_TEXT_COLOR, font=("Arial", 8), tags=self.tag_root)
168
169 is_interaction_active = self.is_dragging or self.is_hovered
170 label_x, label_y = (ACTIVE_LABEL_X, ACTIVE_LABEL_Y) if is_interaction_active else geo["label_pos"]
171 label_font = ("Arial", 12 if is_interaction_active else 10, "bold" if is_interaction_active else "normal")
172 self.canvas.create_text(label_x, label_y, text=self.label, fill=self.color_highlight, font=label_font, tags=self.tag_root)

175 # Fallback to old rendering if Rust is missing
176 self._render_fallback()

178 def _render_fallback(self):
179 base_x, base_y = self.center_x, self.center_y
180 track_angle = self.angle + ANGLE_90_DEGREES
181 half_length = self.track_length / RADIUS_DIVISOR
182
183 # Track
184 start_point = CircularMath.rotate_point(base_x, base_y - half_length, base_x, base_y, track_angle)
185 end_point = CircularMath.rotate_point(base_x, base_y + half_length, base_x, base_y, track_angle)
186 self.canvas.create_line(start_point, end_point, fill=TRACK_BACKGROUND_COLOR, width=6, capstyle=tk.ROUND, tags=self.tag_root)
187 self.canvas.create_line(start_point, end_point, fill=TRACK_FOREGROUND_COLOR, width=2, capstyle=tk.ROUND, tags=self.tag_root)
188
189 # Ticks
190 for i in range(TICK_COUNT):
191 normalization_factor = i / TICK_INTERVAL_NORMALIZER
192 local_y = (-half_length) + (normalization_factor * self.track_length)
193 tick_length = MAJOR_TICK_LENGTH if i % MAJOR_TICK_INTERVAL == 0 else MINOR_TICK_LENGTH
194
195 left_tick_p1 = CircularMath.rotate_point(base_x - TICK_HORIZONTAL_OFFSET, base_y + local_y, base_x, base_y, track_angle)
196 left_tick_p2 = CircularMath.rotate_point(base_x - TICK_HORIZONTAL_OFFSET - tick_length, base_y + local_y, base_x, base_y, track_angle)
197 self.canvas.create_line(left_tick_p1, left_tick_p2, fill=TICK_COLOR, width=2, tags=self.tag_root)
198
199 right_tick_p1 = CircularMath.rotate_point(base_x + TICK_HORIZONTAL_OFFSET, base_y + local_y, base_x, base_y, track_angle)
200 right_tick_p2 = CircularMath.rotate_point(base_x + TICK_HORIZONTAL_OFFSET + tick_length, base_y + local_y, base_x, base_y, track_angle)
201 self.canvas.create_line(right_tick_p1, right_tick_p2, fill=TICK_COLOR, width=2, tags=self.tag_root)
202
203 # Cap
204 value_normalization = (self.value_current - self.value_min) / (self.value_max - self.value_min)
205 cap_center_x, cap_center_y = CircularMath.rotate_point(base_x, base_y + (-half_length + value_normalization * self.track_length), base_x, base_y, track_angle)
206
207 cap_fill = CAP_HOVER_COLOR if self.is_hovered else CAP_NORMAL_COLOR
208 cap_outline = CAP_HOVER_OUTLINE if self.is_hovered else CAP_NORMAL_OUTLINE
209 cap_outline_width = CAP_HOVER_WIDTH if self.is_hovered else CAP_NORMAL_WIDTH
210 self.canvas.create_oval(cap_center_x - CAP_RADIUS, cap_center_y - CAP_RADIUS, cap_center_x + CAP_RADIUS, cap_center_y + CAP_RADIUS,
211 fill=cap_fill, outline=cap_outline, width=cap_outline_width, tags=(self.tag_root, "cap"))
212
213 # Sweep & Pointer
214 pointer_angle_degrees = SWEEP_START_ANGLE - (self.rotation_current / 100.0) * SWEEP_EXTENT_DEGREES
215 self.canvas.create_arc(cap_center_x - CAP_RADIUS + CAP_INNER_PADDING, cap_center_y - CAP_RADIUS + CAP_INNER_PADDING,
216 cap_center_x + CAP_RADIUS - CAP_INNER_PADDING, cap_center_y + CAP_RADIUS - CAP_INNER_PADDING,
217 start=SWEEP_START_ANGLE, extent=-(SWEEP_START_ANGLE - pointer_angle_degrees),
218 style=tk.ARC, outline=self.color_highlight, width=4, tags=self.tag_root)
219
220 pointer_radians = math.radians(pointer_angle_degrees)
221 pointer_end_x = cap_center_x + (CAP_RADIUS - POINTER_LENGTH_ADJUSTMENT) * math.cos(pointer_radians)
222 pointer_end_y = cap_center_y - (CAP_RADIUS - POINTER_LENGTH_ADJUSTMENT) * math.sin(pointer_radians)
223 self.canvas.create_line(cap_center_x, cap_center_y, pointer_end_x, pointer_end_y, fill=self.color_highlight, width=POINTER_WIDTH, tags=self.tag_root)
224
225 # Values & Label
226 self.canvas.create_text(cap_center_x, cap_center_y + ROTATION_VALUE_Y_OFFSET, text=str(int(self.rotation_current)), fill=self.color_highlight, font=("Arial", 9, "bold"), tags=self.tag_root)
227 self.canvas.create_text(cap_center_x, cap_center_y - CURRENT_VALUE_Y_OFFSET, text=str(int(self.value_current)), fill=VALUE_TEXT_COLOR, font=("Arial", 8), tags=self.tag_root)
228
229 is_interaction_active = self.is_dragging or self.is_hovered
230 if is_interaction_active:
231 label_x, label_y = ACTIVE_LABEL_X, ACTIVE_LABEL_Y
232 else:
233 stagger_distance = FAR_RADIUS + LABEL_FAR_RADIUS_OFFSET + (self.widget_id % 2) * LABEL_STAGGER_OFFSET
234 label_x, label_y = CircularMath.get_position(self.angle, stagger_distance)
235
236 label_font = ("Arial", 12 if is_interaction_active else 10, "bold" if is_interaction_active else "normal")
237 self.canvas.create_text(label_x, label_y, text=self.label, fill=self.color_highlight, font=label_font, tags=self.tag_root)
