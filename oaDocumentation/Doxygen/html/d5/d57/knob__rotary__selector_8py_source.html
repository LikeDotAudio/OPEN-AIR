66 def _calc_layout(self, width, height, options):
67 """Calculate center points and usable radius."""
68 cx, cy = width / 2, height / 2
69 top_res = 20 if options.get("show_label") and options.get("main_label") else 0
70 bottom_res = 20
71 usable_h = height - top_res - bottom_res
72 adj_cy = cy + (top_res - bottom_res) / 2
73 radius = min(width, usable_h) / 2 - 25
74 return {'cx': cx, 'cy': cy, 'adj_cy': adj_cy, 'radius': radius}

76 def _calc_angles(self, num_pos, continuous):
77 """Determine angular start, step, and total span."""
78 start_angle, total_span = (90, 360) if continuous else (240, 300)
79 angle_step = total_span / (num_pos if continuous else max(1, num_pos - 1))
80 return {'start': start_angle, 'step': angle_step, 'span': total_span}

82 def _draw_track(self, canvas, layout, angles, color, continuous):
83 """Draw the circular or arc track."""
84 cx, adj_cy, radius = layout['cx'], layout['adj_cy'], layout['radius']
85 if continuous:
86 canvas.create_oval(cx - radius, adj_cy - radius, cx + radius, adj_cy + radius, outline=color, width=2)
87 else:
88 canvas.create_arc(cx - radius, adj_cy - radius, cx + radius, adj_cy + radius,
89 start=angles['start'], extent=-angles['span'], style=tk.ARC, outline=color, width=2)

91 def _draw_ticks_and_labels(self, canvas, layout, angles, current_idx, positions, colors):
92 """Draw position ticks and labels around the track."""
93 cx, adj_cy, radius = layout['cx'], layout['adj_cy'], layout['radius']
94 for i, pos_text in enumerate(positions):
95 angle_deg = angles['start'] - (i * angles['step'])
96 angle_rad = math.radians(angle_deg)
97 ts_x, ts_y = cx + (radius + 2) * math.cos(angle_rad), adj_cy - (radius + 2) * math.sin(angle_rad)
98 te_x, te_y = cx + (radius + 10) * math.cos(angle_rad), adj_cy - (radius + 10) * math.sin(angle_rad)
99 tl_x, tl_y = cx + (radius + 24) * math.cos(angle_rad), adj_cy - (radius + 24) * math.sin(angle_rad)
100
101 canvas.create_line(ts_x, ts_y, te_x, te_y, fill=colors['secondary'], width=1)
102 canvas.create_text(
103 tl_x, tl_y, text=str(pos_text),
104 fill=colors['indicator'] if i == current_idx else colors['fg'],
105 font=("Helvetica", 8, "bold" if i == current_idx else "normal"),
106 tags="industrial_text"
107 )

109 def _draw_knob_elements(self, canvas, layout, angles, current_idx, colors, options):
110 """Draw the physical knob body and pointer."""
111 cx, adj_cy, radius = layout['cx'], layout['adj_cy'], layout['radius']
112 p_angle = angles['start'] - (current_idx * angles['step'])
113 _draw_body(canvas, cx, adj_cy, radius - 5, options.get("shape", "circle"), colors['secondary'], 1,
114 rotation_angle=p_angle, outline_thickness=1, fill_color="", teeth=8)
115 _draw_pointer(canvas, cx, adj_cy, radius - 5, 4, p_angle, options.get("pointer_style", "line"),
116 colors['indicator'], length=radius+14, offset=0, no_center=options.get("no_center", False))
