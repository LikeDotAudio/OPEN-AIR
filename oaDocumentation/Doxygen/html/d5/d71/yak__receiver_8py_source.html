29class YakReceiverManager:
30 """
31 Handles instrument responses and dispatches data to the MQTT broker.
32
33 This manager correlates incoming instrument responses with previously
34 issued commands, parses the payload based on YAK definitions, and
35 updates the system state via MQTT and the state cache.
36
37 Responsibilities:
38 - Response Correlation: Matches 'Rx_Outbox' messages to 'Tx_Inbox' IDs.
39 - Payload De-serialization: Parses semi-colon delimited SCPI replies.
40 - State Propagation: Updates the StateCache with hardware-fresh values.
41 - Legacy Correction: Handles hardware quirks for specific trigger paths.
42
43 Constraints:
44 - Operates within the UI Partition.
45 - Depends on 'YakTranslator' for context retrieval.
46 - Assumes 'OPEN-AIR/Proxy/Rx_Outbox' topic availability.
47 """
48

80 def _on_rx_outbox_message(self, message: MqttMessage):
81 """
82 Callback for processing messages from the instrument proxy.
83
84 Extracts the correlation ID to retrieve the original YAK command
85 context, then hands off the raw response for parsing.
86
87 Args:
88 message: The MQTT message containing response, command, and ID.
89
90 Side Effects:
91 - Purges correlation context from the YakTranslator on match.
92 """
93 try:
94 payload_data = message.get_json_payload()
95 except Exception as e:
96 logger.error(f"❌ [RX] Payload parse failure: {e}")
97 return
98
99 response_value = payload_data.get("response")
100 correlation_id = payload_data.get("correlation_id")
101
102 if correlation_id and response_value:
103 # Retrieve the 'why' behind this response from the translator.
104 command_context = self.yak_translator.retrieve_command_context(
105 correlation_id)
106 if command_context:
107 path_parts = command_context.get("path_parts")
108 command_details = command_context.get("command_details")
109
110 if path_parts and command_details:
111 self.process_response(path_parts,
112 {"Outputs": command_details},
113 response_value)
114 else:
115 logger.error(f"❌ [RX] Incomplete context for {correlation_id}")
116 else:
117 logger.error(f"❌ [RX] Unknown correlation ID: {correlation_id}")
118 else:
119 logger.error("❌ [RX] Missing response or correlation ID.")

121 def process_response(self, path_parts: list, command_details: dict,
122 response: str):
123 """
124 Parses an instrument response and publishes values to the MQTT broker.
125
126 Args:
127 path_parts: The hierarchical path to the command in the YAK repo.
128 command_details: The expected output structure (dictionary).
129 response: The raw string response from the instrument.
130
131 Side Effects:
132 - Publishes one or more MQTT messages to 'OPEN-AIR/yak/.../value'.
133 - Updates the StateCache if available.
134 """
135 outputs = command_details.get("Outputs", {})
136 # Split hardware response; SCPI standard usually implies ';' delimiter.
137 response_parts = [p.strip() for p in response.split(";")]
138 output_keys = list(outputs.keys())
139
140 # --- Hardware Quirk Correction ---
141 # Correct for legacy bandwidth setting key-order discrepancies
142 # observed in specific firmware versions.
143 if path_parts == self.NAB_BANDWIDTH_TRIGGER_PATH and len(output_keys) >= 5:
144 if (output_keys[3].startswith("Sweep_Time_s") and
145 output_keys[4].startswith("Continuous_Mode_On")):
146 output_keys[3], output_keys[4] = output_keys[4], output_keys[3]
147 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name,
148 "🔄 [CORRECT] Key swap applied for bandwidth path.",
149 level="DEBUG")
150
151 if len(response_parts) != len(output_keys):
152 logger.error(f"❌ [RX] Length mismatch! Expected {len(output_keys)}, "
153 f"got {len(response_parts)}.")
154 return
155
156 # Construct the canonical YAK output topic hierarchy.
157 base_output_topic = "/".join(["OPEN-AIR", "yak"] + path_parts[:4] + ["Outputs"])
158
159 for i, key in enumerate(output_keys):
160 raw_value = response_parts[i]
161 output_topic = f"{base_output_topic}/{key}/value"
162
163 # Route to Cache Manager for atomic persistence.
164 if self.state_cache_manager:
165 self.state_cache_manager.handle_external_update(
166 output_topic, raw_value, source="VISA")
167 else:
168 self.mqtt_util.get_client_instance().publish(
169 topic=output_topic, payload=raw_value, qos=0, retain=True)
170
171 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name,
172 f"📡📤📤 [STATE] {output_topic} -> {raw_value}",
173 level="DEBUG")
