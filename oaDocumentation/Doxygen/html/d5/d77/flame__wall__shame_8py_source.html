31MAX_SUBSYSTEM_BREAKDOWN_ITEMS = 20

33def generate_wall_of_shame(performance_stats, ps):
34 """
35 Generates a concise, metrics-focused performance report.
36 Focuses on raw numbers for quick identification of top offenders.
37 """
38 # 1. Define Categories
39 categories = [
40 ("CATEGORY 1: BLOCKING / HEAVY PATHS (Cumulative Time)", lambda x: x['cumtime']),
41 ("CATEGORY 2: CPU INTENSIVE LOOPS (Self Time)", lambda x: x['tottime']),
42 ("CATEGORY 3: UPDATE SPAM / REDUNDANCY (Call Frequency)", lambda x: x['ncalls']),
43 ("CATEGORY 4: LATENCY / PAYLOAD WEIGHT (Slowest Single Calls)", lambda x: x['per_call'])
44 ]
45
46 # 2. Build lines
47 shame_lines = ["--- PERFORMANCE WALL OF SHAME (TOP OFFENDERS) ---", ""]
48
49 for title, sort_key in categories:
50 shame_lines.append(f"\n>>> {title}")
51 # USE CONSTANT
52 by_metric = sorted(performance_stats, key=sort_key, reverse=True)[:MAX_OFFENDERS_PER_CATEGORY]
53
54 for i, s in enumerate(by_metric, 1):
55 # Sanitize Function Name for HTML Display (remove <method ...>)
56 raw_name = s['funcname']
57 if raw_name.startswith("<") and raw_name.endswith(">"):
58 if "method" in raw_name and "'" in raw_name:
59 raw_name = raw_name.split("'")[1]
60
61 safe_funcname = html.escape(raw_name)
62
63 if "Frequency" in title:
64 val_str = f"{s['ncalls']:10} calls | {s['tottime']:8.4f}s self"
65 elif "Latency" in title:
66 val_str = f"{s['per_call']:10.6f}s/call"
67 elif "CPU" in title:
68 val_str = f"{s['tottime']:10.4f}s | {s['ncalls']:8} calls"
69 else: # Cumulative
70 val_str = f"{s['cumtime']:10.4f}s | {s['ncalls']:8} calls"
71
72 shame_lines.append(f" {i:2}. {val_str} | {safe_funcname}")
73
74 # 3. Add Subsystem Breakdown
75 shame_lines.append(f"\n>>> CATEGORY 5: SUBSYSTEM BREAKDOWN (Worst {MAX_SUBSYSTEM_BREAKDOWN_ITEMS} per Root Process)")
76 all_roots = sorted(list(set(r for s in performance_stats for r in s['roots'])))
77 for root in all_roots:
78 shame_lines.append(f" [ ROOT SUBSYSTEM: {root} ]")
79 root_stats = [s for s in performance_stats if root in s['roots']]
80 root_offenders = sorted(root_stats, key=lambda x: x['cumtime'], reverse=True)[:MAX_SUBSYSTEM_BREAKDOWN_ITEMS]
81 for i, s in enumerate(root_offenders, 1):
82 raw_name = s['funcname']
83 if raw_name.startswith("<") and raw_name.endswith(">"):
84 if "method" in raw_name and "'" in raw_name:
85 raw_name = raw_name.split("'")[1]
86 safe_funcname = html.escape(raw_name)
87 shame_lines.append(f" {i:2}. {s['cumtime']:8.4f}s cum | {safe_funcname}")
88
89 # 4. Add Framework Overhead
90 shame_lines.append(f"\n>>> CATEGORY 6: THE FRAMEWORK OVERHEAD (Top {MAX_SUBSYSTEM_BREAKDOWN_ITEMS} Built-in Spammers)")
91 builtins = [s for s in performance_stats if s['filename'] == "~"]
92 builtin_spammers = sorted(builtins, key=lambda x: x['ncalls'], reverse=True)[:MAX_SUBSYSTEM_BREAKDOWN_ITEMS]
93 for i, s in enumerate(builtin_spammers, 1):
94 shame_lines.append(f" {i:2}. {s['ncalls']:10} calls | {s['tottime']:8.4f}s self | {html.escape(s['funcname'])}")
95
96 return "\n".join(shame_lines)
