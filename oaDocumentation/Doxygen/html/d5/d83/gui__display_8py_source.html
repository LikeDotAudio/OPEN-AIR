139 def _on_initial_build_complete(self):

142 # ⚡ FINAL SETTLE: After all widgets are created and the layout has had a moment
143 # to calculate, trigger a final, forced reslice and background sync on all builders.
144 # DELAY INCREASED: To 500ms to allow X11 window manager to assign geometry.
145 def _final_settle():
146 for loader_instance in self.module_loader.get_all_builders():
147 if loader_instance and hasattr(loader_instance, 'dynamic_gui') and loader_instance.dynamic_gui.winfo_exists():
148 builder = loader_instance.dynamic_gui
149 builder._trigger_reslice_all(force=True)
150 builder._trigger_background_sync(force=True)

152 self.after(500, _final_settle)
