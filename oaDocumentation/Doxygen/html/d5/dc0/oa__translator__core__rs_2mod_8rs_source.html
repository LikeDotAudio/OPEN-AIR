162fn internal_diff_values(path: &str, old: &Value, new: &Value, diffs: &mut HashMap<String, Value>) {
163 match (old, new) {
164 (Value::Object(old_obj), Value::Object(new_obj)) => {
165 for (key, new_v) in new_obj {
166 let new_path = if path.is_empty() { key.clone() } else { format!("{}/{}", path, key) };
167 if let Some(old_v) = old_obj.get(key) {
168 internal_diff_values(&new_path, old_v, new_v, diffs);
169 } else {
170 diffs.insert(new_path, new_v.clone());
171 }
172 }
173 }
174 (old_v, new_v) if old_v != new_v => {
175 diffs.insert(path.to_string(), new_v.clone());
176 }
177 _ => {}
178 }
179}

193 let old_val: Value = depythonize_bound(old.clone().into_any()).map_err(|e| PyErr::new::<pyo3::exceptions::PyValueError, _>(e.to_string()))?;
194 let new_val: Value = depythonize_bound(new.clone().into_any()).map_err(|e| PyErr::new::<pyo3::exceptions::PyValueError, _>(e.to_string()))?;
195
196 let mut diffs = HashMap::new();
197 internal_diff_values("", &old_val, &new_val, &mut diffs);
198
199 let result = PyDict::new_bound(py);
200 for (key, value) in diffs {
201 result.set_item(key, pythonize(py, &value).map_err(|e| PyErr::new::<pyo3::exceptions::PyRuntimeError, _>(e.to_string()))?)?;
202 }
203 Ok(result)
