29 The 'Traffic Light' controller for the Fleet Manager.
30 - Defaults to RED (Scanning/Uninitialized).
31 - Turns GREEN only when a valid Fleet JSON is published.

34 # Initializes the FleetStatusMonitor.
35 # This constructor sets up the monitor to subscribe to scan start and complete events
36 # and publishing this initial state to the GUI status light.
37 # Inputs:
38 # state_mirror_engine: The state mirror engine for MQTT synchronization.
39 # subscriber_router (MqttSubscriberRouter): The MQTT subscriber router.
40 # Outputs:
41 # None.
42 def __init__(self, state_mirror_engine, subscriber_router: MqttSubscriberRouter):
43 self.state_mirror_engine = state_mirror_engine
44 self.subscriber_router = subscriber_router # Store subscriber_router
45 self.base_topic = "OPEN-AIR/System/Status/Fleet"
46 self.current_state = "RED" # Default state

48 # 1. Listen for the signals
49 # 'Start' comes from the Visa Manager beginning its work

61 # Callback for when a device fleet scan starts.
62 # This method updates the internal state to "RED" and publishes this color
63 # to the GUI status light, indicating that a scan is in progress.
64 # Inputs:
65 # message (MqttMessage): The MQTT message object.
66 # Outputs:
67 # None.
68 def _on_scan_start(self, message: MqttMessage):
69 self.current_state = "RED"
70 self._publish_color("red")
71 if LOCAL_DEBUG: sys_logger.debug("Fleet Scan Started - Status Red")
72
73 # Callback for when a device fleet scan completes.
74 # This method processes the scan completion payload, checks the number of devices found,
75 # and updates the GUI status light to GREEN if devices were found, or RED otherwise.
76 # Inputs:
77 # message (MqttMessage): The MQTT message object.
78 # Outputs:
79 # None.
80 def _on_scan_complete(self, message: MqttMessage):
81 payload = message.payload
82 try:
83 if isinstance(payload, (bytes, str)):
84 data = orjson.loads(payload)
85 else:
86 data = payload
87
88 num_devices = data.get("num_devices", 0)
89 if num_devices > 0:
90 self.current_state = "GREEN"
91 self._publish_color("green")
92 if LOCAL_DEBUG: sys_logger.debug(f"Fleet Scan Complete - {num_devices} devices found. Status Green")
93 else:
94 self.current_state = "RED"
95 self._publish_color("red")
96 if LOCAL_DEBUG: sys_logger.debug("Fleet Scan Complete - No devices found. Status Red")
97 except orjson.JSONDecodeError as e:
98 sys_logger.warning(f"Malformed scan complete payload received: {e}")
99 self.current_state = "RED"
100 self._publish_color("red")
101 except Exception:
102 sys_logger.exception("Error processing scan complete payload")
103 self.current_state = "RED"
104 self._publish_color("red")
105
106 # Publishes the specified color to the GUI Status Light via MQTT.
107 # This method constructs a JSON payload containing the color and a timestamp,
108 # then publishes it to a dedicated MQTT topic that controls the GUI status indicator.
109 # Inputs:
110 # color (str): The color to set the status light to (e.g., "red", "green").
111 # Outputs:
112 # None.
113 def _publish_color(self, color):
114 """Tells the GUI Status Light what color to be."""
115 target_topic = "OPEN-AIR/GUI/Global/Header/StatusLight"
116 payload = {"color": color, "timestamp": time.time()}
117 publish_payload(target_topic, orjson.dumps(payload).decode())
