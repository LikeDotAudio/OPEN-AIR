32 @classmethod
33 def resolve(cls, data):
34 """
35 Recursively resolves shorthands in a dictionary.
36 """
37 if not isinstance(data, dict):
38 return data
39
40 new_data = {}
41 for k, expanded_value in data.items():
42 if isinstance(expanded_value, dict):
43 expanded_value = cls.resolve(expanded_value)
44
45 target_key = cls.MAPPING.get(k, k)
46
47 # ⚡ SPECIAL HANDLING: x/y should ONLY be row/column in a layout context
48 if k == "x":
49 if any(key in data for key in ["sticky", "weight", "col_span"]):
50 target_key = "row"
51 else:
52 target_key = "width" # Fallback for cap.x etc.
53 elif k == "y":
54 if any(key in data for key in ["sticky", "weight", "row_span"]):
55 target_key = "column"
56 else:
57 target_key = "height" # Fallback for cap.y etc.
58
59 if target_key not in new_data or target_key == k:
60 new_data[target_key] = expanded_value
61
62 return new_data
