148 fn calculate_circular_ticks(&self,
149 py: Python<'_>,
150 center_x: f64, center_y: f64,
151 tick_values: &Bound<'_, PyList>,
152 min_val: f64, max_val: f64,
153 start_angle_deg: f64, end_angle_deg: f64, extent_deg: f64,
154 tick_start_radius: f64, tick_length: f64,
155 counter_clockwise: bool) -> PyResult<Py<PyAny>> {
156
157 let range_val = max_val - min_val;
158 let mut results = Vec::new();
159
160 for item in tick_values.iter() {
161 let tick_val: f64 = item.extract()?;
162 let percentage = if range_val != 0.0 { (tick_val - min_val) / range_val } else { 0.0 };
163
164 let current_angle_deg = if counter_clockwise {
165 end_angle_deg + (percentage * extent_deg)
166 } else {
167 start_angle_deg - (percentage * extent_deg)
168 };
169
170 let current_angle_rad = current_angle_deg * PI / 180.0;
171 let cos_a = current_angle_rad.cos();
172 let sin_a = current_angle_rad.sin();
173
174 let x_start = center_x + tick_start_radius * cos_a;
175 let y_start = center_y - tick_start_radius * sin_a;
176 let x_end = center_x + (tick_start_radius - tick_length) * cos_a;
177 let y_end = center_y - (tick_start_radius - tick_length) * sin_a;
178
179 results.push((x_start, y_start, x_end, y_end));
180 }
181
182 Ok(results.into_py(py))
183 }

191 [r, g, b]
192 }
