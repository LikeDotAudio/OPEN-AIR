125 indicator_width, indicator_height = button_width * INDICATOR_WIDTH_RATIO, button_height * INDICATOR_HEIGHT_RATIO

137 canvas.create_text(center_x, y_offset_coordinate + button_height * 0.6, text=button_text, fill="white", font=BUTTON_TEXT_FONT, tags="button_elements")

144 def adjust_color_lightness(self, hex_color, factor):
145 """Helper to lighten/darken a hex color."""
146 hex_color = hex_color.lstrip('#')
147 if len(hex_color) == 3:
148 hex_color = ''.join([char * 2 for char in hex_color])
149 try:
150 red, green, blue = tuple(int(hex_color[i:i+2], 16) for i in (0, 2, 4))
151 red = max(0, min(255, int(red * factor)))
152 green = max(0, min(255, int(green * factor)))
153 blue = max(0, min(255, int(blue * factor)))
154 return f"#{red:02x}{green:02x}{blue:02x}"
155 except Exception:
156 return "#8B0000"
