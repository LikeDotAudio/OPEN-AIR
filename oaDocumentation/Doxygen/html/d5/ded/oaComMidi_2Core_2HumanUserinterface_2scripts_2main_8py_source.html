92def update_table_lcd(table: list[list[int]], message, direction: Direction) -> None:
93 """Updates the lcd portion of the table based on the midi message
94 :parameter table: The table to update
95 :parameter message: The midi message to parse
96 :parameter direction: The direction of the midi message (incoming from or outgoing to rtp)
97 """
98
99 length = (len(message.data) - 1) - LCD_POSITION_INDEX
100 starting_position = message.data[LCD_POSITION_INDEX] # This is the "id" of the starting lcd segment
101
102 position = starting_position
103
104 # Parses through the message.data and upates each affected character in the table induvidually
105 for char_index in range(LCD_POSITION_INDEX + 1, LCD_POSITION_INDEX + length + 1): # for each ascii character in the message
106
107 cell = lcd_seg_to_cell(table, position)
108
109 new_char = chr(message.data[char_index])
110
111 seg = position % SEGMENTS_PER_LCD
112
113 # replace the character at the position seg in the string
114 str_val = table[cell.row][cell.column]
115 str_val = str_val[:seg] + new_char + str_val[seg + 1:]
116 table[cell.row][cell.column] = str_val
117
118 position += 1

121def lcd_seg_to_cell(table: list[list[int]], position: int) -> Cell:
122 """Returns the cell corresponding to the lcd segment
123 :parameter table: The table containing the lcd text data
124 :parameter position: The position of the lcd segment
125 :return: The cell corresponding to the lcd segment
126 """
127
128 lcd_section = position // SEGMENTS_PER_LCD # the lcd section (1-8 is first row, 9-16 is second row)
129
130 lcd_row_index = midiParser.get_row_num('lcd_line_' + str(lcd_section // LCDS_PER_ROW), table)
131 lcd_column_index = lcd_section % LCDS_PER_ROW
132
133 if lcd_row_index is None:
134 print("lcd_line_" + str(lcd_section // LCDS_PER_ROW) + " not found")
135 return None
136
137 return Cell(lcd_row_index, lcd_column_index)
