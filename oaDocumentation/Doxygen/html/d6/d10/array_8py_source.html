34 def _initialize_group(self, group_name: str):

36 var = tk.BooleanVar(value=True)
37 self.vars[group_name] = var

44 def _toggle_group(self, group_name: str):
45 is_visible = self.vars[group_name].get()
46 state = "expanded" if is_visible else "collapsed"
47 for widget in self.groups.get(group_name, []):
48 if hasattr(widget, "set_view_state"):
49 widget.set_view_state(state)

51 def show_menu(self, event):

76class BlueprintDataInjector:
77 """Handles recursive injection of data and view managers into JSON blueprints."""
78 @classmethod
79 def inject(cls, config: Any, data: dict, view_manager: ViewManager | None = None):
80 """Recursively injects data context and view manager into the configuration."""
81 if isinstance(config, dict):
82 cls._inject_into_dict(config, data, view_manager)
83 elif isinstance(config, list):
84 cls._inject_into_list(config, data, view_manager)

86 @classmethod
87 def _inject_into_dict(cls, config: dict, data: dict, vm: ViewManager | None):
88 # Specific injection for collapsible blocks
89 if config.get("type") == "OcaCollapsibleBlock" and vm:
90 config["_view_manager"] = vm
91
92 for key, value in config.items():
93 if isinstance(value, (dict, list)):
94 cls.inject(value, data, vm)
95 elif isinstance(value, str) and "{{" in value:
96 config[key] = cls._resolve_string_placeholders(value, data)
97
98 @classmethod
99 def _inject_into_list(cls, config: list, data: dict, vm: ViewManager | None):
100 for i, value in enumerate(config):
101 if isinstance(value, (dict, list)):
102 cls.inject(value, data, vm)
103 elif isinstance(value, str) and "{{" in value:
104 config[i] = cls._resolve_string_placeholders(value, data)
105
106 @staticmethod
107 def _resolve_string_placeholders(text: str, data: dict) -> Any:
108 """Replaces {{key}} placeholders with values from the data context."""
109 for key, val in data.items():
110 placeholder = f"{{{{{key}}}}}"
111 if text == placeholder:
112 return val
113 if placeholder in text:
114 text = text.replace(placeholder, str(val))
115 return text

120 def expand_blueprint(blueprint: dict, data_array: list[dict], view_manager: ViewManager) -> dict[str, Any]:
121 """Creates a collection of item configurations by mapping data to a blueprint."""
122 synthetic_fields = {}
123 # Use orjson for optimized string template generation
124 blueprint_template = orjson.dumps(blueprint).decode()
125
126 for idx, item in enumerate(data_array):
127 item_id = str(item.get("id", f"item_{idx}"))

129 # Materialize item configuration from template

133 except Exception as e:
134 logger.error(f"ArrayExpander: Failed to materialize element {idx}: {e}")
