30 def _on_knob_drag(self, event):
31 if self.state.get("start_y") is None: return
32 dy = self.state["start_y"] - event.y
33
34 min_val, max_val = self.widget_config["min"], self.widget_config["max"]
35 base_sensitivity = (max_val - min_val) / 200.0
36 if self.widget_config["fine_pitch"]:
37 base_sensitivity /= 10.0
38
39 if (event.state & 0x000C) == 0x000C:
40 base_sensitivity /= 2.0
41
42 delta = dy * base_sensitivity
43 raw_new_val = self.state["start_value"] + delta
44
45 if self.widget_config["infinity"]:
46 range_span = max_val - min_val
47 new_val = min_val + ((raw_new_val - min_val) % range_span)
48 else:
49 new_val = max(min_val, min(max_val, raw_new_val))
50
51 if self.variable.get() != new_val:
52 self.variable.set(new_val)
53 self._broadcast_cb()

61 def _on_mousewheel(self, event):
62 current_val = self.variable.get()
63 val_range = self.widget_config["max"] - self.widget_config["min"]
64 step = val_range * 0.05
65 delta = 0
66 if sys.platform == "linux":
67 delta = 1 if event.num == 4 else -1
68 else:
69 delta = 1 if event.delta > 0 else -1
70 new_val = max(self.widget_config["min"], min(self.widget_config["max"], current_val + (delta * step)))
71 self.variable.set(new_val)
