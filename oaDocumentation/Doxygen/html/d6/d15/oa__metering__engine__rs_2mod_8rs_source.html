89 fn update(&mut self, dt_ms: f64) -> PyResult<(f64, f64, f64, bool, bool)> {
90 let now_ms = SystemTime::now()
91 .duration_since(UNIX_EPOCH)
92 .unwrap()
93 .as_secs_f64() * 1000.0;
94
95 let mut full_range = self.max_val - self.min_val;
96 if full_range <= 0.0 { full_range = 1.0; }
97
98 // 1. State Transitions & Target Selection
99 if self.state == MeterState::Holding {
100 let hold_dur = self.hold_time.max(self.dwell_time);
101 if now_ms - self.hold_start_time >= hold_dur {
102 self.state = MeterState::Decaying;
103 }
104 }
105
106 let effective_target = match self.state {
107 MeterState::Tracking => self.target_value,
108 MeterState::Decaying => self.min_val,
109 _ => self.current_value,
110 };
111
112 // 2. Main Bar Movement
113 let diff = effective_target - self.current_value;
114 let epsilon = full_range * 0.001;
115 let mut reached_min = false;
116
117 if diff.abs() < epsilon {
118 self.current_value = effective_target;
119 match self.state {
120 MeterState::Tracking => {
121 self.state = MeterState::Holding;
122 self.hold_start_time = now_ms;
123 }
124 MeterState::Decaying => {
125 self.state = MeterState::Idle;
126 reached_min = true;
127 }
128 _ => {}
129 }
130 } else {
131 let time_param = if diff > 0.0 {
132 self.attack_ms
133 } else if self.state == MeterState::Tracking {
134 self.release_ms
135 } else {
136 self.fall_time
137 };
138
139 if time_param <= 0.0 {
140 self.current_value = effective_target;
141 } else {
142 let step = (full_range / time_param) * dt_ms;
143 if diff > 0.0 {
144 self.current_value = (self.current_value + step).min(effective_target);
145 } else {
146 self.current_value = (self.current_value - step).max(effective_target);
147 }
148 }
149 }
150
151 // 3. Floating Peak Line
152 if self.peak_display {
153 if self.current_value > self.peak_value {
154 self.peak_value = self.current_value;
155 self.peak_hold_start_time = now_ms;
156 } else {
157 if now_ms - self.peak_hold_start_time >= self.peak_hold_time {
158 if self.peak_display_fall_time > 0.0 {
159 let p_step = (full_range / self.peak_display_fall_time) * dt_ms;
160 self.peak_value = (self.peak_value - p_step).max(self.min_val);
161 } else {
162 self.peak_value = self.current_value;
163 }
164 }
165 }
166 }
167
168 // 4. Overload LED
169 if self.show_peak_hold {
170 if self.current_value >= self.upper_range {
171 self.overload_expiry = now_ms + self.peak_hold_time;
172 self.overload_fade_factor = 1.0;
173 } else if now_ms < self.overload_expiry {
174 self.overload_fade_factor = 1.0;
175 } else if now_ms < self.overload_expiry + self.overload_fade_time {
176 if self.overload_fade_time > 0.0 {
177 let elapsed_fade = now_ms - self.overload_expiry;
178 self.overload_fade_factor = (1.0 - (elapsed_fade / self.overload_fade_time)).max(0.0);
179 }
180 } else {
181 self.overload_fade_factor = 0.0;
182 }
183 }
184
185 // 5. Activity Check
186 let mut is_active = self.state != MeterState::Idle;
187 if self.peak_display && (self.peak_value > self.current_value + epsilon) {
188 is_active = true;
189 }
190 if self.overload_fade_factor > 0.0 {
191 is_active = true;
192 }
193
194 self.is_running = is_active;
195 Ok((self.current_value, self.peak_value, self.overload_fade_factor, self.is_running, reached_min))
196 }
