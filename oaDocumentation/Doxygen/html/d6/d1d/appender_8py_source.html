33# Appends data from an unknown CSV file to the marker table.
34# This function prompts the user to select a CSV file, converts its content
35# to the standardized marker format, and then imports this new data into the
36# active table editor.
37# Inputs:
38# importer_tab_instance: The instance of the importer tab.
39# editor_instance: The instance of the table editor.
40# Outputs:
41# None.
42def append_markers_file_action(importer_tab_instance, editor_instance):
43 current_function = inspect.currentframe().f_code.co_name
44 file_path = filedialog.askopenfilename(
45 defaultextension=".csv",
46 filetypes=[("CSV files", "*.csv"), ("All files", "*.*")],
47 )
48 if not file_path:
49 logger.debug("🟢️️️🟡 'Append CSV Marker Set' action cancelled by user.", file=importer_tab_instance.current_file,
50 version=importer_tab_instance.current_version,
51 function=f"{current_function}",
52 )
53 return
54 headers, new_data = Marker_convert_csv_unknow_report_to_csv(file_path=file_path)
55 if headers and new_data:
56 # Pass the new data to the table editor for handling

58 save_markers_file_internally(importer_tab_instance)
