26 # Initializes the table data from a CSV file.
27 # This function checks for the existence of a CSV file at the given path.
28 # If the file exists, it reads its contents and publishes each row to MQTT
29 # to seed the application's state cache. If the file does not exist,
30 # it creates a new blank CSV file with the specified headers.
31 # Inputs:
32 # csv_path (str): The full path to the CSV file.
33 # headers (list): A list of column headers for the CSV file.
34 # data_topic (str): The base MQTT topic for publishing data from the CSV.
35 # Outputs:
36 # None.
37 def initialize_from_csv(self, csv_path, headers, data_topic):
38 """
39 Checks for a CSV file. If it exists, reads it and publishes data to MQTT
40 to seed the state cache. If not, creates a blank CSV with headers.
41 """
42 reader = TableCsvReader()
43 writer = TableCsvWriter()
44
45 if os.path.exists(csv_path):
46 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"Found existing CSV at {csv_path}. Publishing contents to seed state cache.", level="DEBUG")
47 _headers, data_list = reader.read_from_csv(csv_path)
48
49 if not data_list:
50 return # File exists but is empty
51
52 key_preference = ["gpib_address", "serial_number", "resource_string"]
53
54 for i, row in enumerate(data_list):
55 item_key = None
56 for key_name in key_preference:
57 if key_name in row and row[key_name]:
58 item_key = row[key_name]
59 break
60 if not item_key:
61 item_key = f"row_{i}"
62
63 # Publish to MQTT to seed the cache
64 field_topic = get_topic(data_topic, "data", item_key)
65 mqtt_publisher_service.publish_payload(field_topic, orjson.dumps(row).decode())
66 else:
67 if headers: # Only create file if headers are known
68 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"No CSV found at {csv_path}. Creating blank file with headers.", level="DEBUG")
69 # Create a blank file with just the headers
70 writer.write_to_csv(csv_path, headers, [])
