27 """
28 Runs snmpwalk locally.
29 If no MIB is provided, it returns RAW numerical OIDs (-On).
30 If a MIB is provided, it returns translated symbolic names (-Os) with diagnostics.
31 """
32 temp_mib = str(SNMP_TEMP_MIB)
33 active_mib_path = mib_path
34
35 # ⚡ TARGET PLUMPING: If walking the absolute root, append .1 to reach the populated v1 tree
36 if base_oid == ".1.3.6.1.4.1.65300":
37 base_oid = f"{base_oid}.1"
38
39 # --- RAW MODE: No MIB provided ---
40 if not active_mib_path and not mib_content:
41 # -Cc: Continue on error (sometimes helpful for walk)
42 cmd = ["snmpwalk", "-v2c", "-c", "public", "-On", "localhost", base_oid]
43 try:
44 result = subprocess.run(cmd, capture_output=True, text=True, timeout=10)
45 output = result.stdout
46 if result.stderr: output = "ERRORS/WARNINGS:\n" + result.stderr + "\n" + "-"*40 + "\n" + output
47 return output if output.strip() else "No OID data returned. Is snmpd running?"
48 except Exception as e:
49 snmp_logger.error(f"Raw Walk Exception: {e}")
50 return f"Raw Walk Exception: {e}"
51
52 # --- MIB MODE: MIB provided or requested ---
53 diagnostics = ["SNMP DIAGNOSTIC REPORT", "="*40]
54
55 # 1. Check for system MIBs
56 std_mib_path = "/usr/share/snmp/mibs"
57 fallback_path = "/usr/share/apps/snmpb/mibs"
58
59 has_std_mibs = os.path.exists(os.path.join(std_mib_path, "SNMPv2-SMI.txt")) or \
60 os.path.exists(os.path.join(std_mib_path, "SNMPv2-SMI"))
61
62 if not has_std_mibs:
63 diagnostics.append("! SYSTEM MIBS MISSING: Found no 'SNMPv2-SMI' in /usr/share/snmp/mibs.")
64 if os.path.exists(fallback_path):
65 diagnostics.append(f"* Using found fallback MIB path: {fallback_path}")
66 else:
67 diagnostics.append(" Fix: sudo apt-get install snmp-mibs-downloader")
68 else:
69 diagnostics.append("- System MIBs found in /usr/share/snmp/mibs.")
70
71 # 2. Process custom MIB
72 module_name = "OPENAIR-MIB"
73 if not active_mib_path and mib_content:
74 try:
75 with open(temp_mib, "w") as f:
76 f.write(mib_content)
77 active_mib_path = temp_mib
78 except Exception as e:
79 snmp_logger.error(f"FAILED to write temporary MIB: {e}")
80 diagnostics.append(f"! FAILED to write temporary MIB: {e}")
81
82 if active_mib_path and os.path.exists(active_mib_path):
83 diagnostics.append(f"- Custom MIB found: {os.path.basename(active_mib_path)}")
84 # Detect module name
85 try:
86 with open(active_mib_path) as f:
87 first_lines = f.read(2000)
88 match = re.search(r"^([\w-]+)\s+DEFINITIONS\s+::=\s+BEGIN", first_lines, re.MULTILINE)
89 if match:
90 module_name = match.group(1)
91 diagnostics.append(f"- Module name detected: {module_name}")
92 except Exception as e:
93 snmp_logger.debug(f"Failed to detect module name from MIB: {e}")
94 else:
95 diagnostics.append("! No custom MIB provided or found.")
96
97 # Phase 2: Execution
98 env = os.environ.copy()
99
100 if active_mib_path:
101 mib_dir = os.path.dirname(os.path.abspath(active_mib_path))
102 search_paths = [mib_dir]
103 if os.path.exists(std_mib_path): search_paths.append(std_mib_path)
104 if os.path.exists(fallback_path): search_paths.append(fallback_path)
105
106 search_path_str = ":".join(search_paths)
107 env["MIBDIRS"] = f"{search_path_str}" # ⚡ ABSOLUTE search path
108
109 # Build Command
110 # -OS: Full symbolic names
111 # -m ALL: Load all MIBs in search path
112 # -Cc: Continue on error (critical for identity nodes)
113 cmd = ["snmpwalk", "-v2c", "-c", "public",
114 "-Cc",
115 "-M", search_path_str,
116 "-m", "ALL",
117 "-OS",
118 "localhost", base_oid]
119 else:
120 cmd = ["snmpwalk", "-v2c", "-c", "public", "localhost", base_oid]
121
122 diagnostics.append(f"> Executing walk on {base_oid}...")
123 diagnostics.append("="*40 + "\n")
124
125 try:
126 result = subprocess.run(cmd, capture_output=True, text=True, timeout=10, env=env)
127 output = result.stdout
128 if result.stderr:
129 output = "SNMPWALK ERRORS/WARNINGS:\n" + result.stderr + "\n" + "-"*40 + "\n" + output
130
131 if not output.strip() and result.returncode == 0:
132 output = "No data returned. Bridge is active but tree is empty."
133 elif not output.strip():
134 output = "No data returned and walk failed. Check snmpd logs."
135 except Exception as e:
136 snmp_logger.error(f"Process Exception during SNMP walk: {e}")
137 output = f"Process Exception: {e}"
138
139 return "\n".join(diagnostics) + output
