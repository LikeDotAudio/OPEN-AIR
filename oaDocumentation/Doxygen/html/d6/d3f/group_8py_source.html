16def group_markers(raw_marker_data):
17 """
18 ⚡ DATA TRANSFORMATION: Organizes raw list into nested Zone/Group dictionary.
19 Inputs:
20 raw_marker_data (list): Raw list of marker dictionaries.
21 Returns:
22 defaultdict: Nested structure {zone: {group: [markers]}}
23 """
24 matrix_log("UI", "SHOWTIME", inspect.currentframe().f_code.co_name, "🟢 [DATA] Grouping marker data by Zone and Group.", level="DEBUG")
25 grouped = defaultdict(lambda: defaultdict(list))
26 for row in raw_marker_data:
27 zone = row.get("ZONE", "N/A")
28 group = row.get("GROUP", "N/A")
29 grouped[zone][group].append(row)
30 return grouped

32def sort_markers(grouped_markers):
33 """
34 ⚡ DATA ORDERING: Alphabetically sorts devices within each group.
35 Inputs:
36 grouped_markers (dict): The nested structure to sort.
37 """
38 matrix_log("UI", "SHOWTIME", inspect.currentframe().f_code.co_name, "🔵 [DATA] Sorting markers by Device Name.", level="DEBUG")
39 for zone, groups in grouped_markers.items():
40 for group, devices in groups.items():
41 devices.sort(key=lambda x: x.get("NAME", ""))
