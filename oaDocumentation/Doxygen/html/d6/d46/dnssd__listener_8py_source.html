32class DNSSDListener:
33 """
34 Listens for Bonjour/DNSSD announcements on the local network
35 and bridges them to the MQTT Hub using the StandaloneMqttPublisher.
36 """
37 def __init__(self, mqtt_publisher, rx_callback=None):
38 self.mqtt_publisher = mqtt_publisher
39 self.rx_callback = rx_callback
40 self.zeroconf = None
41 self.known_services = {}
42 self.browsers = [] # Holds all our service browsers
43 self.services_being_browsed = set() # Tracks types we are already browsing
44
45 def add_service_type_to_browse(self, type_):
46 """Adds a new service type to the list of browsed services."""
47 if type_ in self.services_being_browsed:
48 return
49
50 print(f"🕵️ [DNSSD] New service type found: {type_}. Adding to browser.")
51 self.services_being_browsed.add(type_)
52 browser = ServiceBrowser(self.zeroconf, type_, self)
53 self.browsers.append(browser)

55 def _create_topic(self, action, type, name):
56 """Creates a hierarchical MQTT topic from the service type and name."""
57 # type: _googlecast._tcp.local. -> googlecast/tcp/local
58 type_path = type.strip('._').replace('._', '/').replace('.', '/')
59 # name: My Device._googlecast._tcp.local. -> My_Device
60 name_sanitized = name.replace(f".{type}", "").replace(" ", "_")
61 return f"OPEN-AIR/DNSSD/{action}/{type_path}/{name_sanitized}"

63 def remove_service(self, zeroconf, type, name):
64 topic = self._create_topic("Removed", type, name)
65 payload = {
66 "action": "removed",
67 "type": type,
68 "name": name,
69 "origin_source": "oaComDNSSD"
70 }
71 self.known_services.pop(name, None)
72 print(f"📉 [DNSSD] Removed: {name} ({type})")
73 if self.rx_callback:
74 # Pass name as source, and type as summary for clarity
75 self.rx_callback(name, f"Removed {type}", payload)
76 self.mqtt_publisher.publish(topic, payload)

78 def add_service(self, zeroconf, type, name):
79 # This is now only called for service *instances*, so this call is correct.
80 info = zeroconf.get_service_info(type, name)
81 if info:
82 # Prefer parsed_addresses if available (handles IPv4/IPv6 correctly)
83 addresses = info.parsed_addresses() if hasattr(info, 'parsed_addresses') else []
84 # Fallback for older zeroconf versions or specific cases
85 if not addresses and info.addresses:
86 try:
87 addresses = [socket.inet_ntoa(addr) for addr in info.addresses]
88 except Exception as e:
89 print(f"⚠️ [DNSSD] Could not convert addresses for {name}: {e}")
90 addresses = [] # Ensure addresses is a list
91
92 properties = {}
93 if info.properties:
94 for k, v in info.properties.items():
95 # Attempt decoding with UTF-16 as it's sometimes used for properties, fall back to UTF-8
96 try:
97 k_str = k.decode('utf-16') if isinstance(k, bytes) else str(k)
98 v_str = v.decode('utf-16', errors='ignore') if isinstance(v, bytes) else str(v)
99 except UnicodeDecodeError:
100 k_str = k.decode('utf-8') if isinstance(k, bytes) else str(k)
101 v_str = v.decode('utf-8', errors='ignore') if isinstance(v, bytes) else str(v)
102 properties[k_str] = v_str
103
104 source_ip = addresses[0] if addresses else "Unknown"
105
106 payload = {
107 "action": "added",
108 "type": type, # e.g. _http._tcp.local.
109 "name": name, # e.g. MyRavennaDevice
110 "server": info.server,
111 "port": info.port,
112 "addresses": addresses,
113 "source_ip": source_ip,
114 "properties": properties, # This is the TXT record payload ("package")
115 "origin_source": "oaComDNSSD"
116 }
117
118 payload_hash = hash(json.dumps(payload, sort_keys=True))
119 if self.known_services.get(name) == payload_hash:
120 return # Skip republishing unchanged records
121 self.known_services[name] = payload_hash
122
123 topic = self._create_topic("Discovered", type, name)
124 print(f"📈 [DNSSD] Discovered/Updated: {name} ({type}) at {addresses}:{info.port}") # Enhanced print statement
125 if self.rx_callback:
126 # Pass 'name' as source and 'type' as summary for better differentiation in the GUI
127 self.rx_callback(name, type, payload)
128 self.mqtt_publisher.publish(topic, payload)
