55 def _launch_wysiwyg_editor(self):
56 """
57 Orchestrates the spawning of the external WYSIWYG editor process.
58 Ensures a singleton process per file.
59 """
60 if not hasattr(self, 'json_filepath') or not self.json_filepath:
61 builder_logger.error("🏗️🚫 Editor launch failed: No target JSON file found.")
62 return
63
64 if self._is_editor_active():
65 if str(self._editor_file) == str(self.json_filepath):
66 return # Already editing this file
67 self._terminate_active_editor()
68
69 self._spawn_editor_process()

75 def _is_editor_active(self):
76 """Checks if a previously spawned editor is still running."""
77 return self._editor_process and self._editor_process.poll() is None

79 def _terminate_active_editor(self):
80 """Gracefully shuts down the current editor process."""
81 try:
82 self._editor_process.terminate()
83 self._editor_process.wait(timeout=1.0)
84 except Exception:
85 self._editor_process.kill()
86 self._editor_process = None

88 def _spawn_editor_process(self):
89 """Physical execution of the subprocess call for the editor."""
90 runner_path = Path(__file__).resolve().parent.parent.parent.parent / "oaGuiEditorWYSIWYG" / "Managers" / "run_builder.py"
91
92 if not runner_path.exists():
93 builder_logger.error(f"🏗️🚫 Orchestrator runner missing: {runner_path}")
94 return
95
96 try:
97 # ⚡ FORENSIC LOGGING: Capture stderr to identify why the process might fail to start
98 BuilderContextMenuMixin._editor_process = subprocess.Popen(
99 [sys.executable, str(runner_path), str(self.json_filepath)],
100 stderr=subprocess.PIPE,
101 text=True
102 )
103 BuilderContextMenuMixin._editor_file = self.json_filepath
104
105 # Check for immediate failure
106 try:
107 _, stderr = BuilderContextMenuMixin._editor_process.communicate(timeout=0.5)
108 if stderr:
109 builder_logger.error(f"🏗️🚫 Editor Process Immediate Error: {stderr.strip()}")
110 except subprocess.TimeoutExpired:
111 # Process is still running, which is good
112 pass
113
114 except Exception as e:
115 builder_logger.exception(f"🏗️🚫 Process spawn failure: {e}")
