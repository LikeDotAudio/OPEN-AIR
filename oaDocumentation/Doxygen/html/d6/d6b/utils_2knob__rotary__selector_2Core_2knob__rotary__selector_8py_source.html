67 def _calc_layout(self, width, height, options):
68 """Calculate center points and usable radius."""
69 cx, cy = width / 2, height / 2
70 top_res = 20 if options.get("show_label") and options.get("main_label") else 0
71 bottom_res = 20
72 usable_h = height - top_res - bottom_res
73 adj_cy = cy + (top_res - bottom_res) / 2
74 radius = min(width, usable_h) / 2 - 25
75 return {'cx': cx, 'cy': cy, 'adj_cy': adj_cy, 'radius': radius}

77 def _calc_angles(self, num_pos, continuous):
78 """Determine angular start, step, and total span."""
79 start_angle, total_span = (90, 360) if continuous else (240, 300)
80 angle_step = total_span / (num_pos if continuous else max(1, num_pos - 1))
81 return {'start': start_angle, 'step': angle_step, 'span': total_span}

83 def _draw_track(self, canvas, layout, angles, color, continuous):
84 """Draw the circular or arc track."""
85 cx, adj_cy, radius = layout['cx'], layout['adj_cy'], layout['radius']
86 if continuous:
87 canvas.create_oval(cx - radius, adj_cy - radius, cx + radius, adj_cy + radius, outline=color, width=2)
88 else:
89 canvas.create_arc(cx - radius, adj_cy - radius, cx + radius, adj_cy + radius,
90 start=angles['start'], extent=-angles['span'], style=tk.ARC, outline=color, width=2)

92 def _draw_ticks_and_labels(self, canvas, layout, angles, current_idx, positions, colors):
93 """Draw position ticks and labels around the track."""
94 cx, adj_cy, radius = layout['cx'], layout['adj_cy'], layout['radius']
95 for i, pos_text in enumerate(positions):
96 angle_deg = angles['start'] - (i * angles['step'])
97 angle_rad = math.radians(angle_deg)
98 ts_x, ts_y = cx + (radius + 2) * math.cos(angle_rad), adj_cy - (radius + 2) * math.sin(angle_rad)
99 te_x, te_y = cx + (radius + 10) * math.cos(angle_rad), adj_cy - (radius + 10) * math.sin(angle_rad)
100 tl_x, tl_y = cx + (radius + 24) * math.cos(angle_rad), adj_cy - (radius + 24) * math.sin(angle_rad)
101
102 canvas.create_line(ts_x, ts_y, te_x, te_y, fill=colors['secondary'], width=1)
103 canvas.create_text(
104 tl_x, tl_y, text=str(pos_text),
105 fill=colors['indicator'] if i == current_idx else colors['fg'],
106 font=("Helvetica", 8, "bold" if i == current_idx else "normal"),
107 tags="industrial_text"
108 )

110 def _draw_knob_elements(self, canvas, layout, angles, current_idx, colors, options):
111 """Draw the physical knob body and pointer."""
112 cx, adj_cy, radius = layout['cx'], layout['adj_cy'], layout['radius']
113 p_angle = angles['start'] - (current_idx * angles['step'])
114 _draw_body(canvas, cx, adj_cy, radius - 5, options.get("shape", "circle"), colors['secondary'], 1,
115 rotation_angle=p_angle, outline_thickness=1, fill_color="", teeth=8)
116 _draw_pointer(canvas, cx, adj_cy, radius - 5, 4, p_angle, options.get("pointer_style", "line"),
117 colors['indicator'], length=radius+14, offset=0, no_center=options.get("no_center", False))

119 def _draw_text_overlays(self, canvas, width, height, colors, options):
120 """Draw the main title and selection value text."""
121 cx = width / 2
122 if options.get("show_label") and options.get("main_label"):
123 canvas.create_text(cx, 10, text=options.get("main_label"), fill=colors['fg'],
124 font=("Helvetica", 9, "bold"), anchor="n", tags="industrial_text")
125 if options.get("selection_text"):
126 canvas.create_text(cx, height - 10, text=options.get("selection_text"), fill=colors['indicator'],
127 font=("Helvetica", 9, "bold"), anchor="s", tags="industrial_text")
