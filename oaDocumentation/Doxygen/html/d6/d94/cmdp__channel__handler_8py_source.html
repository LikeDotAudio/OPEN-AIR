64 def update_position(self):
65 """
66 ⚡ MATH ONLY: Calculates new physical coordinates based on state variables.
67 """
68 try: angle = float(self.angle_var.get())
69 except Exception as e:
70 logger.error(f"Failed to get float angle from {self.angle_var}: {e}")
71 angle = 0.0
72 rad = math.radians(angle)
73 near, far = self.widget_ref.near_radius, self.widget_ref.far_radius
74 self.track_len = far - near
75 dist = near + (self.track_len / 2)
76 center_x, center_y = self.widget_ref.center_x, self.widget_ref.center_y
77 self.x = center_x + dist * math.cos(rad)
78 self.y = center_y + dist * math.sin(rad)

91 def calculate_rotated_point(self, px, py, cx, cy, cos_a, sin_a):
92 """Vectorized point rotation using pre-calculated trig values."""
93 delta_x, delta_y = px - cx, py - cy
94 return cos_a * delta_x - sin_a * delta_y + cx, sin_a * delta_x + cos_a * delta_y + cy
