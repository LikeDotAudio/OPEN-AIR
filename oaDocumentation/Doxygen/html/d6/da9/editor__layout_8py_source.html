109class SashManager:
110 """Handles the positioning of PanedWindow sashes."""

112 @staticmethod
113 def setup(editor):
114 """Sets up the initial sash positions after the window is mapped."""
115 def _on_map(e):
116 # ⚡ [UI] Unbind after first map to prevent "snapping back" on minimize/restore
117 if hasattr(editor, 'window') and editor.window.winfo_exists():
118 editor.window.unbind("<Map>", map_id)
119 editor.window.after(100, lambda: SashManager.set_initial(editor))
120
121 map_id = editor.window.bind("<Map>", _on_map, add="+")
122
123 @staticmethod
124 def set_initial(editor):
125 """Calculates and applies the initial sash positions using percentage-based logic."""
126 try:
127 editor.main_pane.update_idletasks()
128 width = editor.main_pane.winfo_width()
129 if width <= 1:
130 editor.window.after(200, lambda: SashManager.set_initial(editor))
131 return
132
133 matrix_log("ui", "gui_builder", "layout", f"📐📐📐 [RENDER] Setting initial sashes for width: {width}", "DEBUG")
134
135 # 1. Left Sidebar: 30% of width
136 left_pos = int(width * 0.30)
137
138 # 2. Right Sidebar: 30% of width (measured from the right)
139 right_pos = int(width * (1.0 - 0.30))
140
141 editor.main_pane.sash_place(0, left_pos, 0)
142 editor.main_pane.sash_place(1, right_pos, 0)
143 except Exception as e:
144 matrix_log("ui", "gui_builder", "layout", f"⚠️🎨🤦‍♂️ [RENDER] Sash placement failed: {e}", "TRACE")
