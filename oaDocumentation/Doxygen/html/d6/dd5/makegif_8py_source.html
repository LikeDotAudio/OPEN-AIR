63# Creates a set of bar and line objects for a single layer of the wave animation.
64# This helper function sets up the visual components for a layer, including a gradient
65# of bars and a line, with specified alpha, color, and width properties.
66# Inputs:
67# alpha_bar (float): The transparency (alpha) for the bar graph.
68# color_line (str): The color for the line plot.
69# width_line (float): The line width for the line plot.
70# alpha_line (float): The transparency (alpha) for the line plot.
71# Outputs:
72# tuple: A tuple containing a list of bar objects and a line object.
73def create_wave_layer(alpha_bar, color_line, width_line, alpha_line):
74 # Always use the standard 'bar_colors' gradient
75 bars = axes.bar(
76 x_vals, np.zeros(num_bars), width=BAR_WIDTH, color=bar_colors, alpha=alpha_bar
77 )
78 (line) = axes.plot([], [], color=color_line, linewidth=width_line, alpha=alpha_line)
79 return bars, line

101# Calculates the height of the wave pattern at a given time `time_phase`.
102# This function generates a dynamic wave based on sine functions, x-coordinates,
103# and time-dependent offsets, contributing to the animation's movement.
104# Inputs:
105# time_phase (float): The current time in the animation cycle.
106# offset_x (float): An x-axis offset for the wave.
107# offset_t (float): A time-based offset for the wave.
108# Outputs:
109# numpy.ndarray: An array of wave heights.
110def calculate_wave_height(time_phase, offset_x, offset_t):
111 return np.abs(np.sin(x_vals + offset_x - time_phase) * np.sin(0.5 * x_vals + time_phase + offset_t))
112

114# Updates the animation for each frame.
115# This function is the core of the animation logic. It calculates the wave heights
116# heights of the bars and the data of the lines for each layer.
117# Inputs:
118# frame (int): The current frame number in the animation sequence.
119# Outputs:
120# list: A list of all updated Matplotlib artists (bars and lines) for the frame.
121def update_animation_frame(frame):
122 progress = frame / FRAMES
123 time_phase = 2 * np.pi * progress # Perfect Loop
124
125 # --- MATH FOR LAYERS ---
126
127 # 1. Deep Background
128 height_layer1 = calculate_wave_height(time_phase, 1.0, 0) * envelope * BASE_HEIGHT_LAYER1 * (1.0 + 0.1 * np.sin(time_phase))
129
130 # 2. Ghost
131 height_layer2 = calculate_wave_height(time_phase, 2.0, 1.5) * envelope * BASE_HEIGHT_LAYER2 * (1.0 + 0.15 * np.sin(2 * time_phase))
132
133 # 3. Main
134 raw_main = calculate_wave_height(time_phase, 0, 0) + (np.abs(np.sin(2 * x_vals - time_phase * 2) * 0.3))
135 height_layer3 = raw_main * envelope * BASE_HEIGHT_LAYER3 * (1.0 + 0.1 * np.sin(time_phase))
136
137 # 4. Electric
138 raw_elec = calculate_wave_height(time_phase * 2, 0.5, 3.0)
139 height_layer4 = raw_elec * envelope * BASE_HEIGHT_LAYER4 * (1.0 + 0.2 * np.sin(3 * time_phase))
140
141 # 5. THE HARMONIC SPIKE
142 # Trigger Logic: Active around t=3.5
143 spike_trigger = np.exp(SPIKE_TRIGGER_STEEPNESS * (time_phase - SPIKE_TRIGGER_CENTER) ** 2)
144
145 # Harmonics: Sharp peaks
146 harmonics = np.abs(np.sin(12 * x_vals - time_phase * 5) * np.sin(18 * x_vals + time_phase))
147 height_layer5 = harmonics * envelope * BASE_HEIGHT_LAYER5 * spike_trigger
148
149 # --- UPDATE PLOTS ---
150 def update_layer_set(bars, line, heights, line_offset):
151 for bar, height in zip(bars, heights):
152 bar.set_height(height)
153 line.set_data(x_vals, heights + line_offset)
154
155 # All lines now sit tight to the bars (offset 0.2)
156 # This keeps them on the "Same X Axis" visually
157 update_layer_set(bars1, line1, height_layer1, LINE_OFFSET_STANDARD)
158 update_layer_set(bars2, line2, height_layer2, LINE_OFFSET_STANDARD)
159 update_layer_set(bars3, line3, height_layer3, LINE_OFFSET_STANDARD)
160 update_layer_set(bars4, line4, height_layer4, LINE_OFFSET_ELECTRIC)
161
162 # Spike layer also sits tight
163 update_layer_set(bars5, line5, height_layer5, LINE_OFFSET_STANDARD)
164
165 return (
166 list(bars1)
167 + list(bars2)
168 + list(bars3)
169 + list(bars4)
170 + list(bars5)
171 + [line1, line2, line3, line4, line5]
172 )
