47def generate_flamegraph_with_flameprof(ps, output_svg):
48 """Renders the pstats data into an interactive SVG flame graph."""
49 if not flameprof:
50 print("? 'flameprof' library not found. Install it with: pip install flameprof")
51 return False
52
53 # ? PRECONDITION VALIDATION
54 if not ps or not hasattr(ps, 'stats') or not ps.stats:
55 print("?? No profile data collected.")
56 return False
57
58 ps.strip_dirs()
59 old_limit = sys.getrecursionlimit()
60 sys.setrecursionlimit(DEFAULT_RECURSION_LIMIT)
61
62 stats = ps.stats
63 total_tottime = 0
64 for func, data in stats.items():
65 cc, nc, tt, ct, callers = data
66 tt = max(0, tt) if tt == tt else 0
67 ct = max(tt, ct) if ct == ct else tt
68 stats[func] = (cc, nc, tt, ct, callers)
69 total_tottime += tt
70
71 roots = [f for f, data in stats.items() if not data[4]]
72 vroot = ("<virtual>", 0, "total_execution")
73
74 for func in list(stats.keys()):
75 if func == vroot: continue
76 cc, nc, tt, ct, callers = stats[func]
77
78 new_callers = {}
79 for c_func, c_data in callers.items():
80 c_cc, c_nc, c_tt, c_ct = c_data
81 new_callers[c_func] = (c_cc, c_nc, c_tt, max(c_ct, MIN_EXECUTION_TIME))
82
83 if not new_callers or func in roots:
84 new_callers[vroot] = (cc, nc, tt, max(ct, MIN_EXECUTION_TIME))
85 stats[func] = (cc, nc, tt, ct, new_callers)
86
87 stats[vroot] = (1, 1, 0, max(total_tottime, MIN_EXECUTION_TIME), {})
88
89 out = io.StringIO()
90 # Increased width to 6000 (5x standard) for better resolution
91 flameprof.render(ps.stats, out, width=FLAMEGRAPH_STANDARD_WIDTH * FLAMEGRAPH_WIDTH_MULTIPLIER, threshold=DEFAULT_THRESHOLD)
92 svg_content = out.getvalue()
