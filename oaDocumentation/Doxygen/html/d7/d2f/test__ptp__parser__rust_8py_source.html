42 # Byte 0: messageType 0 (Sync)
43 # Byte 20-29: Port ID

47 payload[4] = 0x01 # Domain 1

51 # Port Identity: 01:02:03:04:05:06:07:08 + Port 123
52 payload[20:28] = b"\x01\x02\x03\x04\x05\x06\x07\x08"

54 payload[29] = 0x7B # Port 123 (0x7B)
55
56 pkt = MockPkt(bytes(payload))
57 data = PTPPacketParser.tear_apart(pkt)
58
59 self.assertIsNotNone(data)
