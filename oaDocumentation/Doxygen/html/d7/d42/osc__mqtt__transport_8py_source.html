69 def connect(self, connection_params: dict[str, Any]) -> bool:

71 port = connection_params.get("destination_port", 1883)

75 # ⚡ UNIQUE IDENTITY: Prevent Client ID collisions in multi-partition environments

77 random_suffix = f"{random.getrandbits(16):04x}"
78 default_client_id = f"OPENAIR-{partition_id}-OSC-{os.getpid()}-{random_suffix}"
79 client_id = connection_params.get("client_id", default_client_id)

86 else:

89 self.client.on_connect = self._on_connect

98 try:
