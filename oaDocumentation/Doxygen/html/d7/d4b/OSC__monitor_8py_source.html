126 def _start_server(self):
127 if self._running: return
128
129 port = self.listen_port.get()
130 dispatcher = Dispatcher()
131 dispatcher.set_default_handler(self._on_osc_message_on_osc_message)
132
133 try:
134 self.server = BlockingOSCUDPServer(("0.0.0.0", port), dispatcher)
135 self.server_thread = threading.Thread(target=self.server.serve_forever, daemon=True)
136 self.server_thread.start()
137 self._running = True
138
139 self.status_var.set(f"Listening on port: {port}")
140 self.status_label.config(fg="#00ff00")
141 self.toggle_btn.config(text="Stop Server", bg="#ff4444")
142 self.port_entry.config(state="disabled")
143 except Exception as e:
144 self.status_var.set(f"Port {port} in use / Error")
145 self.status_label.config(fg="#ff4444")
146 messagebox.showerror("Server Error", f"Could not start server on port {port}:\n\n{e}")
147 self._running = False
148 self.toggle_btn.config(text="Start Server", bg="#00aa00")
149 self.port_entry.config(state="normal")

165 def _on_osc_message(self, address, *args):
166 if not self._running: return
167
168 # Flatten value
169 if len(args) == 1:
170 value = args[0]
171 if isinstance(value, float): value = round(value, 4)
172 else:
173 value = str(args)
174
175 timestamp = datetime.now().strftime("%H:%M:%S.%f")[:-3]
176
177 # Schedule UI update on main thread
178 self.root.after(0, lambda: self._sync_ui(address, value, timestamp))

180 def _sync_ui(self, address, value, timestamp):
181 # 1. Update Recent Activity Log (Top)
182 self.log_tree.insert("", 0, values=(timestamp, address, value))
183 # Keep only last 50 events
184 if len(self.log_tree.get_children()) > 50:
185 self.log_tree.delete(self.log_tree.get_children()[-1])
186
187 # 2. Update Hierarchy Tree (Bottom)
188 parts = [p for p in address.split("/") if p]
189 parent = ""
190 full_node_path = address
191
192 for i, part in enumerate(parts):
193 node_path = "/" + "/".join(parts[:i+1])
194
195 if node_path not in self.nodes:
196 is_leaf = (i == len(parts) - 1)
197 display_val = value if is_leaf else ""
198 display_ts = timestamp if is_leaf else ""
199
200 node_id = self.tree.insert(parent, "end", text=part, values=(display_val, display_ts), open=True)
201 self.nodes[node_path] = node_id
202 else:
203 if i == len(parts) - 1:
204 node_id = self.nodes[node_path]
205 self.tree.item(node_id, values=(value, timestamp))
206
207 parent = self.nodes[node_path]
208
209 # 3. Visual Fading Logic
210 # Update change history
211 if full_node_path in self.change_history:
212 self.change_history.remove(full_node_path)
213 self.change_history.insert(0, full_node_path)
214 if len(self.change_history) > 10:
215 self.change_history.pop()
216
217 # Re-apply tags to all nodes in history
218 for i, path in enumerate(self.change_history):
219 if path in self.nodes:
220 node_id = self.nodes[path]
221 tag = f"fade_{min(i, 4)}"
222 self.tree.item(node_id, tags=(tag,))
