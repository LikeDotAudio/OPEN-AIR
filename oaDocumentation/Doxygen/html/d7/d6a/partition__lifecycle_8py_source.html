3# Spawn / monitor / shutdown of the Core, UI and (optional) Web partitions.
4# The supervisor's monitoring loop lives here too. Each partition's restart
5# policy depends on `is_mission_critical`:
6# - Core / UI exit → restart if mission-critical, else shut the system down.
7# - Web exit → restart if mission-critical, else leave it down
8# (Web is auxiliary and never forces a full shutdown).

16def interpret_exit_code(code):
17 """Human-readable description of common subprocess exit reasons."""
18 if code == 0: return "The service closed gracefully."
19 if code == -11: return "The service crashed due to a critical memory error (Segmentation Fault)."
20 if code == -15: return "The service was stopped by a termination request."
21 if code == -9: return "The service was forcefully killed by the system."
22 if code == 1: return "The service failed to start or encountered a generic error."
23 return f"The service exited with an unhandled status code: {code}"

40def spawn_partitions(python_exe, scripts, envs, log):
41 """Spawn UI → Core → (optional) Web. Returns dict of {p_core, p_ui, p_web, processes}."""
42 core_script, ui_script, web_script = scripts
43 child_env, core_env, ui_env = envs
44
45 log("Spawning Partition B (UI)...")
46 p_ui = subprocess.Popen([python_exe, ui_script], env=ui_env)
47
48 log("Spawning Partition A (Core)...")
49 p_core = subprocess.Popen([python_exe, core_script], env=core_env)
50
51 # processes[0]=core, processes[1]=ui (used by monitoring loop to restart in place).
52 processes = [p_core, p_ui]
53
54 p_web = None
55 if web_script:
56 log("Spawning Web Interface (http://localhost:8000)...")
57 p_web = subprocess.Popen([python_exe, web_script], env=child_env)
58 processes.append(p_web)
59
60 return {"p_core": p_core, "p_ui": p_ui, "p_web": p_web, "processes": processes}

63def _restart_or_shutdown(name, exited, restart_args, restart_env, *,
64 is_mission_critical, shutdown_requested, log, processes, index):
65 """Common restart-or-shutdown decision for Core/UI. Returns the live Popen (new or restarted)."""
66 code = exited.returncode
67 desc = interpret_exit_code(code)
68 if is_mission_critical and not shutdown_requested[0]:
69 log(f"⚠️ {name} has stopped ({desc}). Restarting automatically...")
70 time.sleep(1.0)
71 new_p = subprocess.Popen(restart_args, env=restart_env)
72 processes[index] = new_p
73 return new_p
74 if name.startswith("The User Interface") and code == 0:
75 log("👋 The User Interface was closed normally. System complete.")
76 else:
77 log(f"🛑 {name} has exited ({desc}). Shutting down the entire system.")
78 shutdown_requested[0] = True
79 return exited

82def monitor_processes(state, scripts, envs, *, python_exe,
83 is_mission_critical, shutdown_requested, log):
84 """Monitor Core/UI/Web liveness until a shutdown is requested."""
85 core_script, ui_script, web_script = scripts
86 child_env, core_env, ui_env = envs
87 p_core, p_ui, p_web = state["p_core"], state["p_ui"], state["p_web"]

91 time.sleep(0.5) # Throttle to minimise CPU.

93 if p_core.poll() is not None:

98 shutdown_requested=shutdown_requested,

100 if shutdown_requested[0]: break
101
102 if p_ui.poll() is not None:
103 p_ui = _restart_or_shutdown(
104 "The User Interface", p_ui,
105 [python_exe, ui_script], ui_env,
106 is_mission_critical=is_mission_critical,
107 shutdown_requested=shutdown_requested,
108 log=log, processes=processes, index=1)
109 if shutdown_requested[0]: break

111 # Web is auxiliary — its exit NEVER forces a system shutdown.
112 if p_web is not None and p_web.poll() is not None:
113 code = p_web.returncode
114 desc = interpret_exit_code(code)
115 if is_mission_critical and not shutdown_requested[0]:
116 log(f"⚠️ The Web Interface has stopped ({desc}). Restarting automatically...")
117 time.sleep(1.0)
118 p_web = subprocess.Popen([python_exe, web_script], env=child_env)
119 processes[2] = p_web
120 else:
121 log(f"ℹ️ The Web Interface has exited ({desc}). Core system continues running.")
122 p_web = None # Stop polling.

125def shutdown_processes(processes, log):
126 """Terminate (then kill, if necessary) all child processes."""
127 log("Terminating child processes...")
128 for p in processes:
129 if p and p.poll() is None:
130 p.terminate()
131 start = time.time()
132 while p.poll() is None and (time.time() - start) < 2:
133 time.sleep(0.1)
134 if p.poll() is None:
135 p.kill()
136 p.wait()
