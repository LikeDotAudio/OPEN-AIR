43class StateMirrorEngine(RegistryMixin, SyncQueueMixin):
44 """
45 Manages the synchronization of GUI widget states with an MQTT broker.
46
47 Responsibilities:
48 - Bidirectional State Sync: Mirrors local UI variables to MQTT topics.
49 - Throttling: Prevents flooding the broker with high-frequency UI events.
50 - Change Filtering: Only broadcasts when the value meaningfully changes.
51 - Registry Integration: Maintains a map of topics to UI widgets.
52
53 Constraints:
54 - Operates within the UI Partition.
55 - Requires a valid 'root' Tkinter instance for thread-safe UI updates.
56 - Depends on 'oaComProtocols.oaComMQTT' for network egress.
57 """

115 def register_widget(self, widget_id, tk_variable, tab_name, config,
116 update_callback=None, instance=None):
117 """
118 Links a UI widget's variable to an MQTT topic.
119
120 Args:
121 widget_id (str): Unique identifier for the widget.
122 tk_variable (tk.Variable): The Tkinter variable to track.
123 tab_name (str): The UI tab context for topic calculation.
124 config (dict): The widget's JSON definition/manifest.
125 update_callback (callable, optional): Invoked on external changes.
126 instance (object, optional): Reference to the widget class instance.
127
128 Returns:
129 str: The fully qualified MQTT topic mapped to this widget.
130 """
131 if self.is_inert: return None
132
133 dynamics = config.get("dynamics", {})
134 topic_override = dynamics.get("path") or dynamics.get("sub")
135 full_topic = self.topic_calculator.calculate(topic_override or widget_id, tab_name)
136
137 info = {
138 "var": tk_variable, "tab": tab_name, "id": widget_id,
139 "config": config, "update_callback": update_callback,
140 "topic": full_topic, "last_broadcast_ts": 0,
141 "last_sent_val": None, "instance": instance
142 }
143 self._register_to_internal_dicts(widget_id, info)
144
145 def _auto_broadcast(*args):
146 # Automated hook into Tkinter's trace system.
147 if not self._silent_update and not self._binding_suspended:
148 self.broadcast_gui_change_to_mqtt(widget_id)
149
150 if tk_variable:
151 tk_variable.trace_add("write", _auto_broadcast)
152 self._registration_queue.put(widget_id)
153 return full_topic

155 def initialize_widget_state(self, widget_id):
156 """
157 Polls the local cache to set the widget's initial visual state.
158
159 Args:
160 widget_id (str): The ID of the widget to initialize.
161
162 Returns:
163 bool: True if initial state was successfully applied from cache.
164 """
165 widget_info = self._get_widget_info(widget_id)
166 if not widget_info or not self.state_cache_manager: return False
167
168 full_topic = widget_info["topic"]
169 cached = self.state_cache_manager.rust_cache.get(full_topic)
170 if cached is None: return False
171
172 try:
173 data = cached if isinstance(cached, (dict, list)) else orjson.loads(cached)
174 raw_val = ValueProcessor.extract_value(data, widget_info["config"])
175 final_val = ValueProcessor.normalize(raw_val, widget_info["var"],
176 widget_info["config"])
177
178 if final_val is not None:
179 widget_info["last_sent_val"] = final_val
180 self.update_queue.put((widget_info["var"], final_val, widget_id))
181 self._schedule_queue_processing()
182 if widget_info["update_callback"]:
183 self.root.after(0, lambda: self._safe_execute_callback(
184 widget_info["update_callback"], data, widget_id))
185 return True
186 except Exception as e:
187 matrix_log("ui", "state_mirror", "initialize_widget_state",
188 f"❌ [CACHE] Init failure for {widget_id}: {e}", "ERROR")
189 return False

211 def broadcast_gui_change_to_mqtt(self, widget_id, extra_payload=None):
212 """
213 Calculates and publishes a local UI change to the MQTT broker.
214
215 Implements throttling and settled-state logic to minimize network
216 chatter during active user interaction (e.g., sliding a fader).
217
218 Args:
219 widget_id (str): The identifier of the widget that changed.
220 extra_payload (dict, optional): Metadata to merge into the manifest.
221 """
222 if not widget_id or self.is_inert or self._suppress_broadcast or \
223 self._silent_update or self._binding_suspended:
224 return
225
226 # Ignore high-frequency internal metadata topics.
227 if "/visibility/" in widget_id or "/left_meter" in widget_id or \
228 "/right_meter" in widget_id:
229 return
230
231 widget_info = self._get_widget_info(widget_id)
232 if not widget_info: return
233
234 instance = widget_info.get("instance")
235 is_locked = getattr(instance, "is_locked", False)
236 is_settled = not getattr(instance, "is_sliding", False)
237
238 metadata = extra_payload or {}
239 if "LOCKED" not in metadata: metadata["LOCKED"] = is_locked
240 if "SETTLED" not in metadata: metadata["SETTLED"] = is_settled
241
242 # --- Structural Identity Injection ---
243 configuration = widget_info.get("config", {})
244 if "bin_id" in configuration: metadata["bin_id"] = configuration["bin_id"]
245 if "block_name" in configuration: metadata["block_name"] = configuration["block_name"]
246 if "field_name" in configuration: metadata["field_name"] = configuration["field_name"]
247
248 now = time.time()
249 # Global and per-widget throttling gates.
250 if (now - self._last_global_broadcast_ts) < self._GLOBAL_THROTTLE_MS: return
251 if (now - widget_info.get("last_broadcast_ts", 0)) < 0.050: return
252
253 try:
254 current_val = widget_info["var"].get()
255 except Exception as e:
256 matrix_log("ui", "state_mirror", "broadcast_gui_change_to_mqtt",
257 f"❌ [GUI] Value retrieval failure for {widget_id}: {e}",
258 "ERROR")
259 return
260
261 # Change-only filter.
262 if current_val == widget_info.get("last_sent_val"): return
263
264 widget_info["last_sent_val"] = current_val
265 widget_info["last_broadcast_ts"] = now
266 self._last_global_broadcast_ts = now
267
268 # Route through cache manager if available, otherwise publish direct.
269 if self.state_cache_manager:
270 self.state_cache_manager.handle_external_update(
271 widget_info["topic"], current_val, source="GUI", metadata=metadata)
272 else:
273 from oaStateCache.Core.manifest.builder import create_manifest
274 payload = create_manifest(current_val, widget_info["topic"],
275 "GUI", metadata)
276 mqtt_publisher_service.publish_payload(
277 widget_info["topic"], orjson.dumps(payload).decode())

279 def sync_incoming_mqtt_to_gui(self, message: MqttMessage):
280 """
281 Parses an incoming MQTT message and updates the corresponding UI widget.
282
283 Args:
284 message (MqttMessage): The raw message from the broker.
285 """
286 if self.is_inert or not message.payload: return
287 try:
288 data = message.payload if isinstance(message.payload, (dict, list)) else orjson.loads(message.payload)
289 if not isinstance(data, dict): return
290
291 # Prevent feedback loops from our own broadcasts.
292 from oaStateCache.Core.manifest.echo_canceller import is_echo
293 if is_echo(data): return
294
295 widget_id = self.topic_to_widget_id.get(message.topic)
296 widget_info = self._get_widget_info(widget_id) if widget_id else None
297 if not widget_info: return
298
299 # Prevent remote updates from overriding a widget currently
300 # under local user control (ghost-touch protection).
301 from oaStateCache.Core.manifest.ghost_lock import is_ghost_touch_locked
302 if is_ghost_touch_locked(data, widget_info.get("instance")): return
303
304 raw_val = ValueProcessor.extract_value(data, widget_info["config"])
305 final_val = ValueProcessor.normalize(raw_val, widget_info["var"],
306 widget_info["config"])
307
308 if final_val is not None:
309 widget_info["last_sent_val"] = final_val
310 # If we are the origin (rare), set immediately.
311 # Otherwise, queue for the main thread.
312 if data.get("origin_source") == widget_id:
313 self._silent_update = True
314 try:
315 widget_info["var"].set(final_val)
316 finally:
317 self._silent_update = False
318 else:
319 self.update_queue.put((widget_info["var"], final_val, widget_id))
320 self._schedule_queue_processing()
321
322 if widget_info["update_callback"]:
323 self.root.after(0, lambda: self._safe_execute_callback(
324 widget_info["update_callback"], data, widget_id))
325
326 except Exception as e:
327 matrix_log("ui", "state_mirror", "sync_incoming_mqtt_to_gui",
328 f"❌ [SYNC] Sync failure for {message.topic}: {e}", "ERROR")
