16def get_directory_tree(path):
17 """Recursively build a dictionary representing the folder structure and JSON files."""
18 tree = {"name": os.path.basename(path), "type": "directory", "children": []}
19 try:
20 entries = sorted(os.listdir(path))
21 for entry in entries:
22 # Ignore hidden files or __init__.py
23 if entry.startswith('.') or entry.startswith('__'):
24 continue
25 full_path = os.path.join(path, entry)
26 if os.path.isdir(full_path):
27 tree["children"].append(get_directory_tree(full_path))
28 elif entry.endswith('.json'):
29 try:
30 with open(full_path, encoding='utf-8') as f:
31 content = json.load(f)
32 except Exception as e:
33 content = {"error": f"Could not parse JSON: {e}"}
34 tree["children"].append({
35 "name": entry,
36 "type": "file",
37 "content": content,
38 "path": full_path.replace(GUI_FRAMES_DIR, "")
39 })
40 except PermissionError:
41 pass
42 return tree

44def strip_volatile(obj):
45 """Remove runtime-only keys (e.g. value.current_value) so they never persist."""
46 if isinstance(obj, dict):
47 obj.pop("current_value", None)
48 for v in obj.values():
49 strip_volatile(v)
50 elif isinstance(obj, list):
51 for v in obj:
52 strip_volatile(v)
53 return obj

56def _extract_readme_json(md_text):
57 """Return the first ```json fenced block in a markdown file, parsed; or None."""
58 m = re.search(r"```json\s*\n(.*?)\n```", md_text, re.DOTALL)
59 if not m:
60 return None
61 try:
62 return json.loads(m.group(1))
63 except Exception:
64 return None

67def _add_components(content, category, relpath, components, legends, seen):
68 """Append widget entries from a parsed sample dict (deduped by name); merge
69 every _LEGEND enum block into the shared legends lookup."""
70 for key, schema in content.items():
71 # _LEGEND blocks enumerate allowed enum values (visualization_types,
72 # knob_styles, …). Merge them (union) into the global legends lookup.
73 if key == "_LEGEND" and isinstance(schema, dict):
74 for lk, lv in schema.items():
75 if isinstance(lv, list):
76 bucket = legends.setdefault(lk, [])
77 for item in lv:
78 if item not in bucket:
79 bucket.append(item)
80 continue
81 if key == "_README":
82 continue
83 # Skip non-widget entries (no "type") and names already supplied.
84 if not isinstance(schema, dict) or "type" not in schema:
85 continue
86 if key in seen:
87 continue
88 seen.add(key)
89 components.append({
90 "name": key,
91 "category": category,
92 "type": schema.get("type", "unknown"),
93 "schema": schema,
94 "path": relpath,
95 })

98def get_grab_bag():
99 """Build the WYSIWYG palette. The libControl component READMEs are the source
100 of truth: each <Component>/Readme.md embeds one ```json sample block. We scan
101 those first (authoritative), then fall back to oaGuiElements/*/sample.json for
102 any widget a README has not already provided (deduped by sample name). Returns
103 a flat, categorized component list + merged enum legends for the editor."""
104 components = []
105 legends = {}
106 seen = set()
107
108 # 1. libControl component READMEs — authoritative.
109 lib_root = os.path.join(FRONTEND_DIR, "libControl")
110 for dirpath, _dirnames, filenames in os.walk(lib_root):
111 readme = next((f for f in filenames if f.lower() == "readme.md"), None)
112 if not readme:
113 continue
114 full = os.path.join(dirpath, readme)
115 try:
116 content = _extract_readme_json(open(full, encoding="utf-8").read())
117 except Exception:
118 content = None
119 if not content:
120 continue
121 rel = os.path.relpath(dirpath, lib_root).split(os.sep)
122 # Category = the libControl group folder (buttons, faders, …); a component
123 # sitting at the libControl root (BreakLine) is grouped as "structure".
124 category = rel[0] if len(rel) > 1 else "structure"
125 _add_components(content, category, os.path.relpath(full, FRONTEND_DIR),
126 components, legends, seen)
127
128 # 2. oaGuiElements/*/sample.json — fallback for widgets with no libControl README.
129 oag_root = os.path.abspath(os.path.join(FRONTEND_DIR, "..", "oaGuiElements"))
130 for dirpath, _dirnames, filenames in os.walk(oag_root):
131 if "sample.json" not in filenames:
132 continue
133 full = os.path.join(dirpath, "sample.json")
134 try:
135 with open(full, encoding="utf-8") as f:
136 content = json.load(f)
137 except Exception:
138 continue
139 rel_parts = os.path.relpath(dirpath, oag_root).split(os.sep)
140 # Category = first meaningful folder segment (skip Core/Assets wrappers).
141 meaningful = [p for p in rel_parts if p not in ("Core", "Assets", ".")]
142 category = meaningful[0] if meaningful else "General"
143 _add_components(content, category, os.path.relpath(full, oag_root),
144 components, legends, seen)
145
146 components.sort(key=lambda c: (c["category"].lower(), c["name"].lower()))
147 return {"components": components, "legends": legends}

256def open_in_chrome(url):
257 """Open `url` in Chrome, falling back to Chromium then the default browser."""
258 # Browsers webbrowser may already know about.
259 for name in ("google-chrome", "chrome", "chromium", "chromium-browser"):
260 try:
261 webbrowser.get(name).open(url)
262 return
263 except webbrowser.Error:
264 pass
265 # Otherwise locate a Chrome/Chromium binary directly.
266 candidates = [
267 "google-chrome", "google-chrome-stable", "chromium", "chromium-browser",
268 "/Applications/Google Chrome.app/Contents/MacOS/Google Chrome",
269 r"C:\Program Files\Google\Chrome\Application\chrome.exe",
270 r"C:\Program Files (x86)\Google\Chrome\Application\chrome.exe",
271 ]
272 for cand in candidates:
273 path = cand if os.path.exists(cand) else shutil.which(cand)
274 if path:
275 try:
276 webbrowser.get(f'"{path}" %s').open(url)
277 return
278 except webbrowser.Error:
279 pass
280 # Last resort: the system default browser.
281 webbrowser.open(url)
