14fn scrape_tables(py: Python<'_>, html_content: String) -> PyResult<Bound<'_, PyList>> {
15 let document = Html::parse_document(&html_content);
16 let table_selector = Selector::parse("table").unwrap();
17 let row_selector = Selector::parse("tr").unwrap();
18 let cell_selector = Selector::parse("th, td").unwrap();
19
20 let tables = PyList::empty_bound(py);
21
22 for table in document.select(&table_selector) {
23 let mut headers = Vec::new();
24 let table_rows = PyList::empty_bound(py);
25
26 for (i, row) in table.select(&row_selector).enumerate() {
27 let cells: Vec<_> = row.select(&cell_selector).collect();
28 if cells.is_empty() { continue; }
29
30 if i == 0 {
31 // Assume first row is headers
32 for cell in cells {
33 headers.push(cell.text().collect::<Vec<_>>().join(" ").trim().to_string());
34 }
35 } else {
36 let row_dict = PyDict::new_bound(py);
37 for (j, cell) in cells.iter().enumerate() {
38 let text = cell.text().collect::<Vec<_>>().join(" ").trim().to_string();
39 let header = headers.get(j).cloned().unwrap_or_else(|| format!("col_{}", j));
40 let _ = row_dict.set_item(header, text);
41 }
42 let _ = table_rows.append(row_dict);
43 }
44 }
45 let _ = tables.append(table_rows);
46 }
47
48 Ok(tables)
49}
