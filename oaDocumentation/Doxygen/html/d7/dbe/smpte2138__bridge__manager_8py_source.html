133 def handle_router_event(self, topic, value, meta=None):
134 """
135 Direct entry point from ProtocolRouter dispatch.
136 Bypasses raw MQTT subscription for higher reliability.
137 """
138 # ⚡ DEBUG: Log arrival of event
139 if _is_debug():
140 matrix_log("comms", "smpte2138", "handle_router_event", f"📥 [ST2138-IN] topic={topic} enabled={self.bridge_enabled}", "TRACE")
141
142 if not self.bridge_enabled: return
143
144 # ⚡ V3.1.13 NAMESPACE EXCLUSION:
145 # System and Monitor topics should never be mirrored to the ST2138 bus.
146 if "/System/" in topic or "/Monitor/" in topic:
147 return
148
149 meta = meta or {}
150 # The Asynchronous "Listen-and-Filter" Loop
151 # Filter out traffic generated by SMPTE2138 itself
152 if meta.get("origin_source") == "SMPTE2138":
153 return
154
155 # ⚡ ENHANCEMENT: Handle enriched payloads (e.g., from MIDI or REST)
156 # If value is a dict, extract the primary numeric or boolean value.
157 real_val = value
158 if isinstance(value, dict) and "value" in value:
159 real_val = value["value"]
160 # Merge dictionary into metadata if it's not already there
161 meta = meta or {}
162 for k, v in value.items():
163 if k != "value": meta[k] = v
164
165 meta = meta or {}
166 bin_id = meta.get("bin_id")
167 block_name = meta.get("block_name")
168 field_name = meta.get("field_name")
169
170 # 1. Resolve Slot (Default to self.slot if not in meta)
171 slot = self.slot
172 if bin_id:
173 slot = self._derive_slot(bin_id)
174
175 # 2. Resolve OID
176 oid = self.topic_to_oid.get(topic)
177
178 # ⚡ ARCHITECT'S CHOICE: Mirror the builder hierarchy if structural metadata is present
179 if block_name and field_name:
180 oid = f"{block_name}/{field_name}"
181
182 # ⚡ GENERIC FALLBACK: If no OID is mapped, use a portion of the topic path.
183 if not oid:
184 # We strip the root namespace to create a relative OID
185 oid = topic.replace("OPEN-AIR/", "")
186
187 if not oid:
188 # Silent skip for unmapped or empty topics
189 return
190
191 # ⚡ DEBUG: Log resolved OID
192 if _is_debug():
193 matrix_log("comms", "smpte2138", "handle_router_event", f"📝 [ST2138-OID] oid={oid} value={real_val} type={type(real_val)}", "DEBUG")
194
195 try:
196 if isinstance(real_val, (int, float, bool)):
197 # Convert bool to float for ST2138 standard
198 f_val = float(real_val) if not isinstance(real_val, bool) else (1.0 if real_val else 0.0)
199 self._publish_parameter(oid, f_val, slot_override=slot)
200 else:
201 self._publish_command(oid, str(real_val), slot_override=slot)
202 except Exception as e:
203 SMPTE2138_LOGGER.error(f"❌ [BRIDGE] Router Event Translation failure: {e}")

207 def _derive_slot(self, bin_id: str) -> int:
208 """
209 Converts a dot-separated Bin ID (e.g. '50.100.5.1.1') into a uint32 slot.
210 Strips dots and converts to integer. Handles up to ~4 billion.
211 """
212 try:
213 # Flatten 50.100.0.3.1 -> 50100031
214 numeric_str = str(bin_id).replace(".", "")
215 slot = int(numeric_str)
216 # Clamp to uint32 max
217 return slot & 0xFFFFFFFF
218 except (ValueError, TypeError):
219 return self.slot

258 def _on_remote_control(self, message):
259 """Processes remote start/stop commands."""
260 try:
261 if not message.payload: return
262 data = message.get_json_payload()
263
264 if "active" in data:
265 new_state = bool(data["active"])
266
267 # 🛡️ RESILIENCE: If it's a retained message and it's trying to disable us,
268 # we ignore it to ensure we start in an ACTIVE state.
269 if message.retain and not new_state:
270 matrix_log("comms", "smpte2138", "_on_remote_control", "🛡️ [BRIDGE] Ignoring retained DISABLE command on startup.", "DEBUG")
271 return
272
273 if new_state != self.bridge_enabled:
274 self.bridge_enabled = new_state
275 status_message = "ENABLED" if self.bridge_enabled else "DISABLED"
276 matrix_log("comms", "smpte2138", "_on_remote_control", f"🔄 [BRIDGE] Bridge translation is now {status_message}.", "INFO")
277 self._publish_bridge_status()
278 except Exception as e:
279 SMPTE2138_LOGGER.error(f"❌ [BRIDGE] Remote control failure: {e}")
