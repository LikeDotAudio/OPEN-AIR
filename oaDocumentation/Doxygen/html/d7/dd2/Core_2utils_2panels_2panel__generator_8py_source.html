77 def generate_procedural_panel(width, height, configuration_data={}):
78 """
79 Generates a PIL Image based on detailed physical parameters.
80 Includes disk caching to prevent redundant generation.
81 """
82 #prefer the 'parameters' key if it exists, otherwise use the top-level dict
83 settings = configuration_data.get("parameters", configuration_data)

85 # ⚡ RESOLUTION CONTROL: Extract scale_factor (default 1.0)
86 scale_factor = float(settings.get("scale_factor", 1.0))
87
88 # --- 0. Check Cache First ---
89 matrix_log("ui", "gui_builder", "generate_procedural_panel", f"Checking for procedural panel in cache: {width}x{height} (Scale: {scale_factor})", "TRACE")
90 cached_image = AssetCacheManager.load_from_cache("panel", width, height, configuration_data)
91 if cached_image:
92 matrix_log("ui", "gui_builder", "generate_procedural_panel", "Retaining procedural panel from disk cache.", "DEBUG")
93 return cached_image

95 # --- 1. Extract Parameters ---
96 random_seed = settings.get("random_seed")
97 if not random_seed:
98 random_seed = random.randint(1, MAX_RANDOM_SEED)
99 random.seed(random_seed)
100
101 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, f"Generating NEW Procedural Panel ({width}x{height}) Scale: {scale_factor} Seed: {random_seed}", "info".upper())
102
103 # --- 2. Layer Configs ---
104 base_material_settings = settings.get("base_material", {})
105 paint_layer_settings = settings.get("paint_layer", {})
106 edge_wear_settings = settings.get("edge_wear", {"enabled": False})
107 panel_scratch_settings = settings.get("panel_scratches", settings.get("scratches", {"count": 0}))
108 grime_settings = settings.get("grime", {"stain_count": 0})
109 rust_settings = settings.get("rust", {"enabled": False})
110 dust_settings = settings.get("dust", {"enabled": False, "intensity": 0.3})
111 screws_settings = settings.get("screws", {"enabled": False})
112 fold_settings = settings.get("metal_fold", {"enabled": False})
113 haze_settings = settings.get("studio_haze", {"enabled": False})
114
115 # --- Layer 1: The Substrate (The Metal Panel) ---
116 substrate_color_hex = base_material_settings.get("color", "#2a2a2a")
117 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, f"Layer 1. Substrate: {substrate_color_hex}", "trace".upper())
118 substrate_color_rgba = PanelUtils.hex_to_rgba(substrate_color_hex)
119 panel_image = Image.new('RGBA', (width, height), substrate_color_rgba)
120
121 texture_type = base_material_settings.get("texture_type", "flat")
122 grain_intensity = float(base_material_settings.get("grain_intensity", DEFAULT_GRAIN_INTENSITY))
123
124 if texture_type == "hammered":
125 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Applying hammered texture.", "trace".upper())
126 hammered_texture = SubstrateFactory.generate_hammered(width, height, grain_intensity, scale_factor=scale_factor).convert("RGBA")
127 panel_image = ImageChops.multiply(panel_image, hammered_texture)
128 elif texture_type == "wrinkle":
129 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Applying wrinkle noise texture.", "trace".upper())
130 # Wrinkle noise at target res
131 noise_w, noise_h = int(width * scale_factor), int(height * scale_factor)
132 wrinkle_texture = Image.effect_noise((noise_w, noise_h), sigma=WRINKLE_SIGMA).convert("RGBA")
133 if scale_factor > 1.0:
134 wrinkle_texture = wrinkle_texture.resize((width, height), resample=Image.LANCZOS)
135 panel_image = ImageChops.multiply(panel_image, wrinkle_texture)
136 elif texture_type == "crosshatch":
137 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Weaving crosshatch streaks.", "trace".upper())
138 horizontal_streaks = SubstrateFactory.generate_streaks(width, height, vertical=False, sigma=STREAK_SIGMA_CROSSHATCH, scale_factor=scale_factor)
139 vertical_streaks = SubstrateFactory.generate_streaks(width, height, vertical=True, sigma=STREAK_SIGMA_CROSSHATCH, scale_factor=scale_factor)
140 crosshatch_weave = ImageChops.multiply(horizontal_streaks, vertical_streaks).convert("RGBA")
141 panel_image = ImageChops.multiply(panel_image, crosshatch_weave)
142 elif texture_type == "enamel":
143 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Applying enamel peel texture.", "trace".upper())
144 # Adjusted resize factor based on scale
145 adj_enamel_factor = max(1, ENAMEL_RESIZE_FACTOR / scale_factor)
146 peel_texture = Image.effect_noise((int(width // adj_enamel_factor), int(height // adj_enamel_factor)), sigma=STREAK_SIGMA_CROSSHATCH).resize((width, height), Image.BICUBIC).convert("RGBA")
147 panel_image = ImageChops.soft_light(panel_image, peel_texture)
148 else:
149 streak_sigma = STREAK_SIGMA_BRUSHED if texture_type == "brushed" else STREAK_SIGMA_DEFAULT
150 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, f"Applying {texture_type} streak texture.", "trace".upper())
151 is_vertical = (base_material_settings.get("grain_direction") == "vertical")
152 directional_streaks = SubstrateFactory.generate_streaks(width, height, vertical=is_vertical, sigma=streak_sigma, scale_factor=scale_factor).convert("RGBA")
153 panel_image = ImageChops.multiply(panel_image, directional_streaks)
154
155 # --- Layer 2: The Paint Layer ---
156 paint_color_rgb = PanelUtils.hex_to_rgb(paint_layer_settings.get("color", "#4a5a6a"))
157 paint_opacity = float(paint_layer_settings.get("opacity", DEFAULT_PAINT_OPACITY))
158 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, f"Layer 2. Paint: {paint_layer_settings.get('color')} (Opacity: {paint_opacity})", "trace".upper())
159 paint_mask_image = Image.new('L', (width, height), MAX_OPACITY_VALUE)
160
161 if edge_wear_settings.get("enabled", False):
162 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Etching edge wear scratches.", "trace".upper())
163 scratch_depth_limit = int(edge_wear_settings.get("scratch_depth", DEFAULT_EDGE_SCRATCH_DEPTH))
164 edge_scratch_mask = Image.new('L', (width, height), 0)
165 edge_scratch_draw = ImageDraw.Draw(edge_scratch_mask)
166 edge_scratch_count = int(edge_wear_settings.get("scratch_intensity", DEFAULT_EDGE_SCRATCH_INTENSITY) * EDGE_SCRATCH_COUNT_MULTIPLIER)
167 for _ in range(edge_scratch_count):
168 selected_edge = random.choice(["top", "bottom", "left", "right"])
169 if selected_edge == "top":
170 start_x, start_y = random.randint(0, width - 1), 0
171 end_x, end_y = start_x + random.randint(-EDGE_SCRATCH_OFFSET_RANGE, EDGE_SCRATCH_OFFSET_RANGE), random.randint(0, scratch_depth_limit)
172 elif selected_edge == "bottom":
173 start_x, start_y = random.randint(0, width - 1), height - 1
174 end_x, end_y = start_x + random.randint(-EDGE_SCRATCH_OFFSET_RANGE, EDGE_SCRATCH_OFFSET_RANGE), height - 1 - random.randint(0, scratch_depth_limit)
175 elif selected_edge == "left":
176 start_x, start_y = 0, random.randint(0, height - 1)
177 end_x, end_y = random.randint(0, scratch_depth_limit), start_y + random.randint(-EDGE_SCRATCH_OFFSET_RANGE, EDGE_SCRATCH_OFFSET_RANGE)
178 else:
179 start_x, start_y = width - 1, random.randint(0, height - 1)
180 end_x, end_y = width - 1 - random.randint(0, scratch_depth_limit), start_y + random.randint(-EDGE_SCRATCH_OFFSET_RANGE, EDGE_SCRATCH_OFFSET_RANGE)
181 edge_scratch_draw.line((start_x, start_y, end_x, end_y), fill=MAX_OPACITY_VALUE, width=random.randint(1, 2))
182 paint_mask_image = ImageChops.subtract(paint_mask_image, edge_scratch_mask)
183
184 surface_scratch_count = int(panel_scratch_settings.get("count", 0))
185 if surface_scratch_count > 0 and panel_scratch_settings.get("reveals_substrate", False):
186 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Carving deep substrate-revealing scratches.", "trace".upper())
187 substrate_reveal_mask = Image.new('L', (width, height), 0)
188 substrate_reveal_draw = ImageDraw.Draw(substrate_reveal_mask)
189 for _ in range(surface_scratch_count):
190 start_x, start_y = random.randint(0, width - 1), random.randint(0, height - 1)
191 scratch_length = random.randint(int(panel_scratch_settings.get("min_length_px", DEFAULT_MIN_SCRATCH_LENGTH)), int(panel_scratch_settings.get("max_length_px", DEFAULT_MAX_SCRATCH_LENGTH)))
192 scratch_angle = random.uniform(0, 2 * math.pi)
193 end_x, end_y = start_x + scratch_length * math.cos(scratch_angle), start_y + scratch_length * math.sin(scratch_angle)
194 substrate_reveal_draw.line((start_x, start_y, end_x, end_y), fill=MAX_OPACITY_VALUE, width=int(panel_scratch_settings.get("width_px", DEFAULT_SCRATCH_WIDTH)))
195 paint_mask_image = ImageChops.subtract(paint_mask_image, substrate_reveal_mask.filter(ImageFilter.GaussianBlur(SCRATCH_GAUSSIAN_BLUR_RADIUS)))
196
197 if paint_opacity > 0:
198 solid_paint_layer = Image.new('RGBA', (width, height), paint_color_rgb + (MAX_OPACITY_VALUE,))
199 blended_paint_layer = Image.blend(panel_image, solid_paint_layer, max(0.0, min(1.0, paint_opacity)))
200 panel_image = Image.composite(blended_paint_layer, panel_image, paint_mask_image)
201
202 gradient_intensity = float(paint_layer_settings.get("gradient_intensity", DEFAULT_GRADIENT_INTENSITY))
203 if gradient_intensity > 0:
204 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Applying linear lighting gradient.", "trace".upper())
205 lighting_gradient = VignetteLayer.generate_linear_gradient(width, height, gradient_intensity)
206 panel_image = ImageChops.multiply(panel_image, lighting_gradient)
207
208 # --- Layer 3: Studio Haze ---4
209 if haze_settings.get("enabled", False):
210 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Layer 3. Infusing warm studio haze.", "trace".upper())
211 haze_alpha = int(MAX_OPACITY_VALUE * float(haze_settings.get("intensity", DEFAULT_HAZE_INTENSITY)))
212 studio_haze_layer = Image.new('RGBA', (width, height), STUDIO_HAZE_COLOR + (haze_alpha,))
213 panel_image = ImageChops.multiply(panel_image, studio_haze_layer)
214
215 # --- Layer 4: Rust ---
216 if rust_settings.get("enabled", False):
217 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Layer 4. Spawning rust oxidation spots.", "trace".upper())
218 rust_intensity = float(rust_settings.get("intensity", 0.5))
219 rust_layer_image = RustLayer.generate_rust_spots(width, height, rust_intensity)
220 panel_image = Image.alpha_composite(panel_image, rust_layer_image)
221
222 # --- Layer 5: Edge Fade ---
223 vignette_intensity_value = float(edge_wear_settings.get("vignette_intensity", 0))
224 if edge_wear_settings.get("enabled", False) and vignette_intensity_value > 0:
225 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Layer 5. Applying vignette edge fade.", "trace".upper())
226 fade_depth_limit = min(int(edge_wear_settings.get("fade_depth", DEFAULT_FADE_DEPTH)), min(width, height) // 2)
227 vignette_layer_image = VignetteLayer.generate_vignette(width, height, vignette_intensity_value, fade_depth_limit)
228 panel_image = ImageChops.multiply(panel_image, vignette_layer_image)
229
230 # --- Layer 6: Scratches ---
231 if int(panel_scratch_settings.get("count", 0)) > 0:
232 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Layer 6. Adding surface micro-scratches.", "trace".upper())
233 scratch_layer_image = ScratchLayer.generate_scratches(width, height, panel_scratch_settings)
234 panel_image = Image.alpha_composite(panel_image, scratch_layer_image)
235
236 # --- Layer 7: Stains ---
237 if int(grime_settings.get("stain_count", 0)) > 0:
238 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Layer 7. Adding grease and coffee stains.", "trace".upper())
239 stains_layer_image = StainsLayer.generate_stains(width, height, grime_settings)
240 panel_image = Image.alpha_composite(panel_image, stains_layer_image)
241
242 # --- Layer 8: Details ---
243 if screws_settings.get("enabled", False):
244 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Layer 8. Drilling screws into the panel.", "trace".upper())
245 screws_layer_image = ScrewLayer.generate_screws(width, height, screws_settings, fold_settings)
246 panel_image = Image.alpha_composite(panel_image, screws_layer_image)
247 if fold_settings.get("enabled", False):
248 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Layer 8. Folding metal creases.", "trace".upper())
249 fold_layer_image = MetalFoldLayer.generate_metal_fold(width, height, fold_settings)
250 panel_image = Image.alpha_composite(panel_image, fold_layer_image)
251 if dust_settings.get("enabled", False):
252 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Layer 8. Settling fine dust particles.", "trace".upper())
253 dust_intensity = float(dust_settings.get("intensity", 0.3))
254 dust_layer_image = DustLayer.generate_dust(width, height, dust_intensity)
255 panel_image = Image.alpha_composite(panel_image, dust_layer_image)
256
257 blur_radius_value = float(settings.get("global_blur", DEFAULT_GLOBAL_BLUR))
258 if blur_radius_value > 0:
259 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, f"Applying global Gaussian blur: {blur_radius_value}", "trace".upper())
260 panel_image = panel_image.filter(ImageFilter.GaussianBlur(radius=blur_radius_value))
261
262 matrix_log("ui", "gui_builder", inspect.currentframe().f_code.co_name, "Procedural panel generation complete. Saving to disk cache.", "success".upper())
263 AssetCacheManager.save_to_cache("panel", width, height, configuration_data, panel_image)
264 return panel_image
