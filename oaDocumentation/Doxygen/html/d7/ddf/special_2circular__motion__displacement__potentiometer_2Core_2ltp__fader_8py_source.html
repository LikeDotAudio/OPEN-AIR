85 def update_fader_position(self):
86 """Calculates the fader's center position in the circular array."""
87 distance = NEAR_RADIUS + (self.track_length / RADIUS_DIVISOR)
88 self.center_x, self.center_y = CircularMath.get_position(self.angle, distance)
89
90 def render_fader(self):
91 """Draws all fader components (track, ticks, cap, pointer) on the canvas."""
92 self.canvas.delete(self.tag_root)
93 if not self.is_visible:
94 return
95
96 base_x, base_y = self.center_x, self.center_y
97 track_angle = self.angle + ANGLE_90_DEGREES
98 half_length = self.track_length / RADIUS_DIVISOR
99
100 # Track
101 start_point = CircularMath.rotate_point(base_x, base_y - half_length, base_x, base_y, track_angle)
102 end_point = CircularMath.rotate_point(base_x, base_y + half_length, base_x, base_y, track_angle)
103 self.canvas.create_line(start_point, end_point, fill=TRACK_BACKGROUND_COLOR, width=6, capstyle=tk.ROUND, tags=self.tag_root)
104 self.canvas.create_line(start_point, end_point, fill=TRACK_FOREGROUND_COLOR, width=2, capstyle=tk.ROUND, tags=self.tag_root)
105
106 # Ticks
107 for i in range(TICK_COUNT):
108 normalization_factor = i / TICK_INTERVAL_NORMALIZER
109 local_y = (-half_length) + (normalization_factor * self.track_length)
110 tick_length = MAJOR_TICK_LENGTH if i % MAJOR_TICK_INTERVAL == 0 else MINOR_TICK_LENGTH
111
112 left_tick_p1 = CircularMath.rotate_point(base_x - TICK_HORIZONTAL_OFFSET, base_y + local_y, base_x, base_y, track_angle)
113 left_tick_p2 = CircularMath.rotate_point(base_x - TICK_HORIZONTAL_OFFSET - tick_length, base_y + local_y, base_x, base_y, track_angle)
114 self.canvas.create_line(left_tick_p1, left_tick_p2, fill=TICK_COLOR, width=2, tags=self.tag_root)
115
116 right_tick_p1 = CircularMath.rotate_point(base_x + TICK_HORIZONTAL_OFFSET, base_y + local_y, base_x, base_y, track_angle)
117 right_tick_p2 = CircularMath.rotate_point(base_x + TICK_HORIZONTAL_OFFSET + tick_length, base_y + local_y, base_x, base_y, track_angle)
118 self.canvas.create_line(right_tick_p1, right_tick_p2, fill=TICK_COLOR, width=2, tags=self.tag_root)
119
120 # Cap
121 value_normalization = (self.value_current - self.value_min) / (self.value_max - self.value_min)
122 cap_center_x, cap_center_y = CircularMath.rotate_point(base_x, base_y + (-half_length + value_normalization * self.track_length), base_x, base_y, track_angle)
123
124 cap_fill = CAP_HOVER_COLOR if self.is_hovered else CAP_NORMAL_COLOR
125 cap_outline = CAP_HOVER_OUTLINE if self.is_hovered else CAP_NORMAL_OUTLINE
126 cap_outline_width = CAP_HOVER_WIDTH if self.is_hovered else CAP_NORMAL_WIDTH
127 self.canvas.create_oval(cap_center_x - CAP_RADIUS, cap_center_y - CAP_RADIUS, cap_center_x + CAP_RADIUS, cap_center_y + CAP_RADIUS,
128 fill=cap_fill, outline=cap_outline, width=cap_outline_width, tags=(self.tag_root, "cap"))
129
130 # Sweep & Pointer
131 pointer_angle_degrees = SWEEP_START_ANGLE - (self.rotation_current / 100.0) * SWEEP_EXTENT_DEGREES
132 self.canvas.create_arc(cap_center_x - CAP_RADIUS + CAP_INNER_PADDING, cap_center_y - CAP_RADIUS + CAP_INNER_PADDING,
133 cap_center_x + CAP_RADIUS - CAP_INNER_PADDING, cap_center_y + CAP_RADIUS - CAP_INNER_PADDING,
134 start=SWEEP_START_ANGLE, extent=-(SWEEP_START_ANGLE - pointer_angle_degrees),
135 style=tk.ARC, outline=self.color_highlight, width=4, tags=self.tag_root)
136
137 pointer_radians = math.radians(pointer_angle_degrees)
138 pointer_end_x = cap_center_x + (CAP_RADIUS - POINTER_LENGTH_ADJUSTMENT) * math.cos(pointer_radians)
139 pointer_end_y = cap_center_y - (CAP_RADIUS - POINTER_LENGTH_ADJUSTMENT) * math.sin(pointer_radians)
140 self.canvas.create_line(cap_center_x, cap_center_y, pointer_end_x, pointer_end_y, fill=self.color_highlight, width=POINTER_WIDTH, tags=self.tag_root)
141
142 # Values & Label
143 self.canvas.create_text(cap_center_x, cap_center_y + ROTATION_VALUE_Y_OFFSET, text=str(int(self.rotation_current)), fill=self.color_highlight, font=("Arial", 9, "bold"), tags=self.tag_root)
144 self.canvas.create_text(cap_center_x, cap_center_y - CURRENT_VALUE_Y_OFFSET, text=str(int(self.value_current)), fill=VALUE_TEXT_COLOR, font=("Arial", 8), tags=self.tag_root)
145
146 is_interaction_active = self.is_dragging or self.is_hovered
147 if is_interaction_active:
148 label_x, label_y = ACTIVE_LABEL_X, ACTIVE_LABEL_Y
149 else:
150 stagger_distance = FAR_RADIUS + LABEL_FAR_RADIUS_OFFSET + (self.widget_id % 2) * LABEL_STAGGER_OFFSET
151 label_x, label_y = CircularMath.get_position(self.angle, stagger_distance)
152
153 label_font = ("Arial", 12 if is_interaction_active else 10, "bold" if is_interaction_active else "normal")
154 self.canvas.create_text(label_x, label_y, text=self.label, fill=self.color_highlight, font=label_font, tags=self.tag_root)
