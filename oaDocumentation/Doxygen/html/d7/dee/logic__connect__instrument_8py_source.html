44 def setup_visa_instrument(self, resource_name):
45 """Establishes a connection to a VISA instrument.
46
47 Parameters:
48 - resource_name: A string representing the VISA resource address
49 (e.g., 'TCPIP::192.168.1.1::INSTR', 'GPIB0::7::INSTR').
50
51 Returns:
52 - A pyvisa.Resource object on success.
53 - None if the connection fails or the resource is unavailable.

55 Side effects and thread-safety:
56 - Performs blocking I/O to initialize the hardware interface.
57 - Configures instrument timeout (5s) and termination characters.
58 """
59 if LOCAL_DEBUG:
60 matrix_log("comms", "visa", inspect.currentframe().f_code.co_name if "inspect" in globals() else "unknown", f"Connecting to instrument: {resource_name}.", "DEBUG")

62 rm = pyvisa.ResourceManager()

64 # Set default communication parameters required for reliable

69 inst.query_delay = 0.1

79 def connect_instrument_logic(self, resource_name):
80 """Handles the full connection sequence to a VISA instrument.
81
82 This method coordinates the setup, session storage, and metadata
83 extraction for a target instrument.
84
85 Parameters:
86 - resource_name: The VISA resource address string.
87
88 Returns:
89 - The pyvisa.Resource instance on success.
90 - False if the connection or identification query fails.
91
92 Side effects and thread-safety:
93 - Updates the global visa_proxy with the new instrument session.
94 - Dispatches multiple MQTT/GUI status messages for device metadata.
95 - Performs synchronous SCPI queries; may block the calling thread.
96 """
97 try:
98 self.inst = self.setup_visa_instrument(resource_name)
99 if not self.inst:
100 # Reset proxy and update UI to reflect failed connection state.
101 self.visa_proxy.set_instrument_instance(inst=None)
102 self.gui_publisher._publish_status("connected", False)
103 self.gui_publisher._publish_status("disconnected", True)
104 return False
105
106 # Update the central proxy so other managers can access the
107 # active instrument session.
108 self.visa_proxy.set_instrument_instance(inst=self.inst)
109
110 # Query instrument identity. Standard SCPI response format is:
111 # <Manufacturer>,<Model>,<Serial>,<Firmware>
112 idn_response = self.inst.query("*IDN?")
113 idn_parts = idn_response.strip().split(",")
114 manufacturer = idn_parts[0].strip() if len(idn_parts) >= 1 else "N/A"
115 model = idn_parts[1].strip() if len(idn_parts) >= 2 else "N/A"
116 serial_number = idn_parts[2].strip() if len(idn_parts) >= 3 else "N/A"
117 firmware = idn_parts[3].strip() if len(idn_parts) >= 4 else "N/A"
118
119 # Broadcast hardware metadata to the GUI and monitoring layers.
120 self.gui_publisher._publish_status("brand", manufacturer)
121 self.gui_publisher._publish_status("device_model", model)
122 self.gui_publisher._publish_status("device_series", model)
123 self.gui_publisher._publish_status("device_serial_number", serial_number)
124 self.gui_publisher._publish_status("device_firmware", firmware)
125 self.gui_publisher._publish_status("visa_resource", resource_name)
126 self.gui_publisher._publish_status(
127 "Time_connected", datetime.datetime.now().strftime("%H:%M:%S")
128 )
129 self.gui_publisher._publish_status("connected", True)
130 self.gui_publisher._publish_status("disconnected", False)
131 self.gui_publisher._publish_status("CONNECTED", "CONNECTED")
132
133 return self.inst
134 except Exception as e:
135 # ⚡ OPTIMIZATION: Downgrade from exception() to error() to prevent massive tracebacks during expected query failures.
136 visa_logger.error(f"Error during connection logic for {resource_name}: {e}")
137 return False
