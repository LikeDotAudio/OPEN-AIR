39 def remove_service(self, zeroconf, type, name):
40 topic = f"OPEN-AIR/MDNS/Removed/{name.split('.')[0]}"
41 payload = {
42 "action": "removed",
43 "type": type,
44 "name": name,
45 "origin_source": "oaComMDNS"

54 def add_service(self, zeroconf, type, name):
55 info = zeroconf.get_service_info(type, name)
56 if info:
57 # Prefer parsed_addresses if available (handles IPv4/IPv6 correctly)
58 addresses = info.parsed_addresses() if hasattr(info, 'parsed_addresses') else []
59 # Fallback for older zeroconf versions or specific cases
60 if not addresses and info.addresses:
61 try:
62 addresses = [socket.inet_ntoa(addr) for addr in info.addresses]
63 except Exception as e:
64 print(f"⚠️ [MDNS] Could not convert addresses for {name}: {e}")
65 addresses = [] # Ensure addresses is a list
66
67 properties = {}
68 if info.properties:
69 for k, v in info.properties.items():
70 # Attempt decoding with UTF-16 as it's sometimes used for properties, fall back to UTF-8
71 try:
72 k_str = k.decode('utf-16') if isinstance(k, bytes) else str(k)
73 v_str = v.decode('utf-16', errors='ignore') if isinstance(v, bytes) else str(v)
74 except UnicodeDecodeError:
75 k_str = k.decode('utf-8') if isinstance(k, bytes) else str(k)
76 v_str = v.decode('utf-8', errors='ignore') if isinstance(v, bytes) else str(v)
77 properties[k_str] = v_str
78
79 source_ip = addresses[0] if addresses else "Unknown"
80
81 payload = {
82 "action": "added",
83 "type": type, # e.g. _http._tcp.local.
84 "name": name, # e.g. MyRavennaDevice
85 "server": info.server,
86 "port": info.port,
87 "addresses": addresses,
88 "source_ip": source_ip,
89 "properties": properties, # This is the TXT record payload ("package")
90 "origin_source": "oaComMDNS"
91 }
92
93 payload_hash = hash(json.dumps(payload, sort_keys=True))
94 if self.known_services.get(name) == payload_hash:
95 return # Skip republishing unchanged records
96 self.known_services[name] = payload_hash
97
98 # Topic differentiates by the first part of the service name (e.g., "MyRavennaDevice")
99 topic = f"OPEN-AIR/MDNS/Discovered/{name.split('.')[0]}"
100 print(f"📈 [MDNS] Discovered/Updated: {name} ({type}) at {addresses}:{info.port}") # Enhanced print statement
101 if self.rx_callback:
102 # Pass 'name' as source and 'type' as summary for better differentiation in the GUI
103 self.rx_callback(name, type, payload)
104 self.mqtt_publisher.publish(topic, payload)
