45 def synchronize_to_viewport(self, viewport_w: int, viewport_h: int):
46 """
47 Calculates and applies the target dimensions for the content frame to ensure it fills
48 the viewport or scrolls correctly.
49 """
50 if not self.canvas.winfo_exists():
51 return None
52
53 # Determine requested content size
54 req_w = self.scroll_frame.winfo_reqwidth()
55 req_h = self.scroll_frame.winfo_reqheight()
56
57 # logic: fill viewport width if not scrolling horizontally, else max(viewport, content)
58 new_w = viewport_w if not self.allow_horizontal_scroll else max(viewport_w, req_w)
59 new_h = max(viewport_h, req_h)
60
61 matrix_log("gui", "gui_render", "synchronize_to_viewport",
62 f"📏 [VIEWPORT] Sync -> View: {viewport_w}x{viewport_h} | "
63 f"Content Req: {req_w}x{req_h} | Target: {new_w}x{new_h}", "TRACE")
64
65 try:
66 # ⚡ STABILITY CHECK: Avoid redundant updates that trigger layout loops
67 curr_w = int(float(self.canvas.itemcget(self.window_id, "width")))
68 curr_h = int(float(self.canvas.itemcget(self.window_id, "height")))
69 if curr_w == int(new_w) and curr_h == int(new_h):
70 return {
71 "viewport": (viewport_w, viewport_h),
72 "content": (req_w, req_h),
73 "target": (new_w, new_h)
74 }
75
76 self.canvas.itemconfig(self.window_id, width=int(new_w), height=int(new_h))
77 self.canvas.configure(scrollregion=self.canvas.bbox("all"))
78 return {
79 "viewport": (viewport_w, viewport_h),
80 "content": (req_w, req_h),
81 "target": (new_w, new_h)
82 }
83 except tk.TclError:
84 return None
