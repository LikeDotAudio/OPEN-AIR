287 def send(self, address, value, meta=None):
288 """Handles Internal -> External OSC Sync (OSC Out)."""
289 with self._state_lock:
290 running = self._running
291
292 if not running or not self.run_bridge or not self.tx_client:
293 return
294
295 meta = meta or {}
296 if meta.get("message_type") == "LINK_FEEDBACK" and not meta.get("is_settled"):
297 return
298 if meta.get("origin_source") == "OSC":
299 return
300
301 matrix_log("comms", "osc", "send", f"TX: {address} <- {value}", "DEBUG")
302 self.tx_client.send_message(address, value)

305 def _on_protocol_event(self, message):
306 with self._state_lock:
307 if not self._running: return
308
309 source = message.get("source", "UNKNOWN").upper()
310 logical_source = message.get("logical_source", source).upper()
311 topic = str(message.get("topic", ""))
312 value = message.get("value")
313 meta = message.get("meta", {})
314
315 is_self_reflection = (source == "MQTT" and message.get("full_id") == app_constants.FULL_INSTANCE_ID)
316
317 # Skip non-OSC system/gui traffic
318 if source == "SYSTEM" or any(topic.startswith(x) for x in ["OPEN-AIR/System/", "OPEN-AIR/GUI/", "OPEN-AIR/oaGui/"]):
319 if "OSC" not in topic: return # Keep status updates
320
321 if "/Monitor/" in topic: return
322
323 # Map topic to OSC address
324 with self._state_lock:
325 osc_address = self.topic_to_osc.get(topic)
326
327 if not osc_address:
328 base_topic = topic.replace("OPEN-AIR/", "").lstrip("/")
329 osc_address = "/" + base_topic if base_topic.startswith("OSC/") else "/OSC/" + base_topic
330 osc_address = osc_address.replace("//", "/").strip("/")
331 if not osc_address.startswith("/"): osc_address = "/" + osc_address
332 if osc_address in ["/OSC", "/"]: osc_address = None
333
334 if osc_address:
335 real_val = value.get("value") if isinstance(value, dict) and "value" in value else value
336 origin_source = meta.get("origin_source", "UNKNOWN")
337
338 should_send = (origin_source != "OSC" and not is_self_reflection) or meta.get("is_settled")
339 is_valid_source = (logical_source in ["OSC", "GUI", "MQTT", "UNKNOWN"] or source in ["OSC", "OSC-TX", "MQTT"])
340
341 if should_send and self.run_bridge and is_valid_source:
342 if isinstance(real_val, (int, float, str, bool, list)):
343 self.send(osc_address, real_val, meta)
344
345 # Monitor Updates
346 if logical_source == "OSC":
347 self._notify_monitor(meta.get("direction", "RX"), meta.get("osc_address", topic),
348 value.get("value") if isinstance(value, dict) and "value" in value else value, topic)
349 elif source in ["OSC-TX", "MQTT"]:
350 self._notify_monitor(source.replace("-TX", ""), meta.get("osc_address", topic), value, topic)
351
352 return
