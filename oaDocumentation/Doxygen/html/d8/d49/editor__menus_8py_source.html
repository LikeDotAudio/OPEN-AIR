59 def _build_language_menu(editor):
60 """Builds the LANGUAGE menu for switching UI translation."""
61 matrix_log("ui", "gui_builder", "menu", "🎨🎨🎨 [RENDER] Constructing LANGUAGE menu", "DEBUG")
62 lang_menu = tk.Menu(editor.menubar, tearoff=0)
63 languages = [("English", "En"), ("French", "Fr"), ("Spanish", "Es"), ("German", "De")]
64 for label, code in languages:
65 lang_menu.add_command(label=label, command=lambda c=code: editor.change_language(c))
66 editor.menubar.add_cascade(label="LANGUAGE", menu=lang_menu)
