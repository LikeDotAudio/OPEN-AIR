71 # Pre-compile regex patterns for performance
72 # This matches 'import oaName_rs' or 'from oaName_rs import ...'
73 patterns = {}
74 for old, new in IMPORT_MAP.items():
75 patterns[old] = (

77 re.compile(rf"\bfrom {old} import\b")
78 )

80 for root, dirs, files in os.walk(root_dir):
81 # Skip the new Rust core directory to avoid circular confusion
82 if "oaRustCore" in root:
83 continue
84
85 for file in files:
86 if file.endswith(".py"):
87 file_path = os.path.join(root, file)
88
89 with open(file_path, encoding='utf-8') as f:
90 content = f.read()
91
92 original_content = content
93 for old, (imp_pattern, from_pattern) in patterns.items():
94 new_module = IMPORT_MAP[old]
95
96 # Replace 'import oaName_rs' -> 'from oaRustCore import oa_name_rs as oaName_rs'
97 # We use 'as' to keep your existing variable names working!
98 content = imp_pattern.sub(f"from oaRustCore import {new_module} as {old}", content)
99
100 # Replace 'from oaName_rs import X' -> 'from oaRustCore.oa_name_rs import X'
101 content = from_pattern.sub(f"from oaRustCore.{new_module} import", content)
102
103 if content != original_content:
104 with open(file_path, 'w', encoding='utf-8') as f:
105 f.write(content)
106 print(f"Refactored: {file_path}")
