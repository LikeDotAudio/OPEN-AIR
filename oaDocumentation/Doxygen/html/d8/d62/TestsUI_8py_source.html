248 def perform_stop_openair(self):
249 if not self.openair_process or self.openair_process.poll() is not None:
250 self.write_logwrite_log("ℹ️ [IDLE] OPEN-AIR is not currently running."); return
251 self.write_logwrite_log("🛑 [KILL] Terminating OPEN-AIR process tree...")
252 try:
253 os.killpg(os.getpgid(self.openair_process.pid), signal.SIGTERM)
254 self.openair_process.wait(timeout=5)
255 self.write_logwrite_log("✨ [TERMINATED] OPEN-AIR has been stopped.")
256 except subprocess.TimeoutExpired:
257 os.killpg(os.getpgid(self.openair_process.pid), signal.SIGKILL)
258 self.write_logwrite_log("⚠️ [FORCE] SIGKILL sent.")
259 except Exception as e: self.write_logwrite_log(f"💥 [ERROR] Error: {e}")
260 finally: self.openair_process = None
