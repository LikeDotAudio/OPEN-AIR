249 def perform_stop_openair(self):
250 if not self.openair_process or self.openair_process.poll() is not None:
251 self.write_logwrite_log("ℹ️ [IDLE] OPEN-AIR is not currently running."); return
252 self.write_logwrite_log("🛑 [KILL] Terminating OPEN-AIR process tree...")
253 try:
254 os.killpg(os.getpgid(self.openair_process.pid), signal.SIGTERM)
255 self.openair_process.wait(timeout=5)
256 self.write_logwrite_log("✨ [TERMINATED] OPEN-AIR has been stopped.")
257 except subprocess.TimeoutExpired:
258 os.killpg(os.getpgid(self.openair_process.pid), signal.SIGKILL)
259 self.write_logwrite_log("⚠️ [FORCE] SIGKILL sent.")
260 except Exception as e: self.write_logwrite_log(f"💥 [ERROR] Error: {e}")
261 finally: self.openair_process = None
