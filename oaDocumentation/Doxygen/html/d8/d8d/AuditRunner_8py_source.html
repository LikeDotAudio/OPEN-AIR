25def estimate_tokens(text):
26 """
27 Heuristically estimates the number of tokens in the given text.
28 Assumes ~4 characters per token for general text/code.
29 """
30 if not text:
31 return 0
32 # Common heuristic for LLM tokens
33 return max(1, len(text) // 4)

36def extract_prompt_from_toml(file_path):
37 """
38 Parses TOML to extract the multiline 'prompt' string.
39 """
40 prompt_lines = []
41 in_multiline = False
42
43 try:
44 with open(file_path, encoding='utf-8') as f:
45 for line in f:
46 stripped_line = line.strip()
47
48 if stripped_line.startswith('prompt = """'):
49 in_multiline = True
50 # Extract content after 'prompt = """' on the same line
51 prompt_part = stripped_line[len('prompt = """'):]
52 prompt_lines.append(prompt_part)
53 continue
54
55 if in_multiline:
56 if stripped_line.endswith('"""'):
57 # Extract content before '"""'
58 prompt_part = stripped_line[:-len('"""')]
59 prompt_lines.append(prompt_part)
60 in_multiline = False
61 break # Found the end of the prompt
62 else:
63 # Append the line as is, preserving internal newlines
64 prompt_lines.append(line.rstrip())
65
66 # Clean up and join
67 if prompt_lines:
68 return "\n".join(prompt_lines).strip()
69 return None
70
71 except FileNotFoundError:
72 print(f"Error: File not found at {file_path}")
73 return None
74 except Exception as e:
75 print(f"Error parsing TOML file {file_path}: {e}")
76 return None
