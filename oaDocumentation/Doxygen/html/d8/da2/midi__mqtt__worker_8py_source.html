34 def start(self):
35 if self._running: return
36
37 # Setup message handler before connecting
38 self.transport.set_message_handler(self._on_transport_message_on_transport_message)
39
40 # ⚡ UNIQUE IDENTITY: Prevent Client ID collisions in multi-partition environments
41 partition_id = os.environ.get("OPEN_AIR_PARTITION_ID", "SUP")
42 random_suffix = f"{random.getrandbits(16):04x}"
43 client_id = f"OPENAIR-{partition_id}-MIDIWORKER-{os.getpid()}-{random_suffix}"
44
45 connection_params = {
46 "destination_host": self.config.MQTT_BROKER_ADDRESS,
47 "destination_port": self.config.MQTT_BROKER_PORT,
48 "username": self.config.MQTT_USERNAME,
49 "password": self.config.MQTT_PASSWORD,
50 "client_id": client_id
51 }
52
53 if self.transport.connect(connection_params):
54 self._running = True
55 self.transport.subscribe(self.base_topic)
56 matrix_log("comms", "midi", "start", "📡 [MIDI-WORKER] Worker started with Core Transport.", "SUCCESS")
57 else:
58 logger.error("📡 [MIDI-WORKER] Failed to start Core Transport.")

60 def stop(self):
61 self._running = False

63 self.transport.disconnect()

65 def _on_transport_message(self, topic, payload):
66 """Unified callback from Core Transport."""
67 try:
68 # ⚡ EFFICIENT ROUTING: Forward to MidiManager
69 # The core transport already handled JSON decoding and echo prevention (src check)
70
71 meta = payload.get("meta", {}) if isinstance(payload, dict) else {}
72
73 event_message = {
74 "topic": topic,
75 "value": payload.get("value") if isinstance(payload, dict) else payload,
76 "meta": meta,
77 "source": "MIDI-MQTT",
78 "logical_source": meta.get("origin_source", "MQTT")
79 }
80 self.midi_manager._on_protocol_event(event_message)
81
82 except Exception as e:
83 logger.error(f"📡 [MIDI-WORKER] Error processing transport message: {e}")

85 def publish(self, topic, payload, meta=None):
86 """Legacy publish shim that delegates to core transport."""
87 if not self.transport: return
88
89 # Standardized payload structure
90 m = meta or {}

92 full_payload = {
93 "value": payload,
94 "meta": m
95 }
96 self.transport.publish(topic, full_payload, retain=self.config.MQTT_RETAIN_BEHAVIOR)
