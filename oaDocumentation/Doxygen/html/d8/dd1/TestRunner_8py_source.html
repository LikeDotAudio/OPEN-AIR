21 """

29 def startTest(self, test):

52 def _get_failure_cause(self, error):
53 import traceback
54 try:
55 exctype, value, tb = error
56 if tb:
57 tblist = traceback.extract_tb(tb)
58 # Filter out unittest internal frames for cleaner reporting
59 relevant_frames = [f for f in tblist if "unittest/case.py" not in f.filename]
60 if not relevant_frames and tblist:
61 relevant_frames = tblist
62
63 if relevant_frames:
64 filename, line, func, text = relevant_frames[-1]
65 # Using \n for breaks, let CSS white-space: pre-wrap or markdown handle it
66 return f"{value}\nAt {os.path.basename(filename)}:{line} in {func} -> {text}"
67 return str(value)
68 except:
69 return "Unknown Failure Cause"

98 def run(self, search_dirs, top_level_dir=None):
99 loader = unittest.TestLoader()
100 suites = []
101 for d in search_dirs:
102 if os.path.exists(d):
103 try:
104 suites.append(loader.discover(d, pattern="test_*.py", top_level_dir=top_level_dir))
105 except ImportError as e:
106 TEST_LOGGER.error(f"❌ ImportError in {d}: {e}")
107 if not self.record_callback:
108 print(f"❌ ImportError in {d}: {e}")
109
110 full_suite = unittest.TestSuite(suites)
111 result = CrashInterceptingResult(self)
112 full_suite.run(result)
113 return result
