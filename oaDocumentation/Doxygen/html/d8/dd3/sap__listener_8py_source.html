60 def _parse_sap_packet(self, data, addr):
61 # Look for the standard SDP start marker "v=0"
62 sdp_start = data.find(b"v=0")
63 if sdp_start == -1:
64 # Not a standard SDP payload or we missed it. Let's dump a debug if it's new.
65 try:
66 raw_text = data.decode('ascii', errors='ignore')
67 print(f"⚠️ [SAP] Non-SDP packet from {addr[0]}: {raw_text[:50]}...")
68 except:
69 pass
70 return
71
72 try:
73 sdp_payload = data[sdp_start:].decode('utf-8', errors='ignore')
74
75 session_name = "Unknown_SAP_Stream"
76 for line in sdp_payload.split('\n'):
77 line = line.strip()
78 if line.startswith('s='):
79 session_name = line[2:].strip()
80 break
81
82 payload = {
83 "source_ip": addr[0],
84 "session_name": session_name,
85 "sdp": sdp_payload,
86 "origin_source": "oaComSAP"
87 }
88
89 import json
90 payload_hash = hash(json.dumps(payload, sort_keys=True))
91 if self.known_streams.get(session_name) == payload_hash:
92 return # Skip republishing unchanged records
93 self.known_streams[session_name] = payload_hash
94
95 # Clean IP address for MQTT Topic path
96 safe_ip = addr[0].replace('.', '_')
97 topic = f"OPEN-AIR/SAP/Discovered/{safe_ip}"
98
99 print(f"📈 [SAP] Discovered/Updated: {session_name} from {addr[0]}")
100
101 if self.rx_callback:
102 self.rx_callback(addr[0], f"Stream: {session_name}", payload)
103
104 self.mqtt_publisher.publish(topic, payload)
105 except Exception as e:
106 print(f"⚠️ [SAP] Parsing error: {e}")
