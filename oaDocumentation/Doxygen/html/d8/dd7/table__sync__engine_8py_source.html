25 def update_full(self, payload, suppress_mqtt=False):
26 """Processes a full table refresh from a dictionary payload."""

29 if not isinstance(data, dict): return

34 if not data:

39 cols = list(next(iter(data.values())).keys())

42
43 for k, v in data.items():
44 iid = self.tree.insert("", tk.END, values=[v.get(c, "") for c in cols], tags=(k))
45 self.item_map[iid], self.device_key_map[k] = v, iid

52 def update_incremental(self, message):
53 """Processes a single row update or pulse event."""
54 topic, payload = message.topic, message.payload
55 try:
56 data = payload if isinstance(payload, (dict, list)) else orjson.loads(payload)
57 except Exception as e:

61 # 1. Handle Pulse (Radar Sync)
62 pa = data.get("angle") or data.get("position")
63 if isinstance(data, dict) and data.get("pulse") is True and pa is not None:
64 tid = self.device_key_map.get(str(int(pa)))
65 if not tid:
66 ac = next((c for c in self.tree["columns"] if c.lower() in ["angle", "deg", "position"]), None)
67 if ac:

74 return

76 # 2. Handle Data Update

79 # Extract the data key (dk) which is the part after the last "/data/"
