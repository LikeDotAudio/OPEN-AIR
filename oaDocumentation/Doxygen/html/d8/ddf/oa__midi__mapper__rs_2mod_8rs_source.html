17fn sanitize_id(port_name: Option<String>) -> PyResult<String> {
18 let name = match port_name {
19 Some(n) => n,
20 None => return Ok("unknown_device".to_string()),
21 };
22
23 // Split ALSA style colon if present
24 let name_part = name.split(':').next().unwrap_or(&name);
25
26 // Replace non-alphanumeric with underscore
27 let sanitized = RE_SANITIZE.replace_all(name_part, "_").to_lowercase();
28
29 // Collapse multiple underscores and trim
30 let collapsed = RE_UNDERSCORES.replace_all(&sanitized, "_");
31 let result = collapsed.trim_matches('_').to_string();
32
33 if result.is_empty() {
34 Ok("unknown".to_string())
35 } else {
36 Ok(result)
37 }
38}

41fn midi_to_topic(dev_id: String, message_type: String, channel: u8, note_or_cc: u8, value: u8) -> PyResult<(String, u8)> {
42 let base = format!("OPEN-AIR/MIDI/{}/ch{}", dev_id, channel);
43
44 match message_type.as_str() {
45 "control_change" => Ok((format!("{}/cc{}", base, note_or_cc), value)),
46 "note_on" => Ok((format!("{}/note{}", base, note_or_cc), value)),
47 "note_off" => Ok((format!("{}/note{}", base, note_or_cc), 0)),
48 _ => Ok((format!("{}/{}", base, message_type), 0)),
49 }
50}
