17def kill_all_profilers():
18 """Safety cleanup: stops any dangling profilers in the environment."""
19 sys.setprofile(None)
20 count = 0
21 for obj in gc.get_objects():
22 try:
23 if isinstance(obj, cProfile.Profile):
24 obj.disable()
25 count += 1
26 except Exception:
27 pass
28 if count > 0:
29 TEST_LOGGER.trace(f"FlameGraph: Watchdog killed {count} active profiler(s).")

31class MultiThreadProfiler:
32 """Orchestrates profiling across the main thread and all spawned child threads."""
33 def __init__(self):
34 self.profilers = []
35 self.lock = threading.Lock()

39 def install(self):
40 if self._is_installed:
41 return

45 # Ensure main profiler is enabled
46 self.main_profiler.enable()
47
48 if not hasattr(threading.Thread, "_original_run"):

50
51 original_run = threading.Thread._original_run
52 outer_self = self

54 def patched_run(self_thread):
55 # Clear any inherited profilers in this new thread
56 kill_all_profilers()
57
58 p = cProfile.Profile()
59 try:
60 p.enable()
61 with outer_self.lock:
62 outer_self.profilers.append(p)
63 try:
64 original_run(self_thread)
65 finally:
66 p.disable()
67 except ValueError as e:
68 TEST_LOGGER.warning(f"FlameGraph: Profiler enable failed: {e}")
69 # Fallback if another profiler is active (e.g. nested calls)
70 original_run(self_thread)
71
72 threading.Thread.run = patched_run
73 self._is_installed = True
74
75 def stop(self):
76 if not self._is_installed:
77 return
78 self.main_profiler.disable()
79 if hasattr(threading.Thread, "_original_run"):
