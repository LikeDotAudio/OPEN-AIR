116 # --- 1. Drop Shadow (The External Cast) ---
117 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, "👻🌀🔳 [LAYER] 1. Casting external drop shadow.", level="TRACE")

126 shadow_draw_context.ellipse((center_coordinate - screw_radius + offset_x, center_coordinate - screw_radius + offset_y,
127 center_coordinate + screw_radius + offset_x, center_coordinate + screw_radius + offset_y),
128 fill=(0, 0, 0, DEFAULT_SHADOW_ALPHA))
129 shadow_blur_radius = size_pixels * SHADOW_BLUR_RATIO_FILLISTER
130 else:
131 # Countersunk has almost no drop shadow, just a recess shade
132 shadow_blur_radius = size_pixels * SHADOW_BLUR_RATIO_COUNTERSUNK

137 # --- 2. The Head Geometry ---

165 lighting_rgba_image = ImageOps.colorize(lighting_mask_image, black="black", white="white").convert("RGBA")

186 # Coordinates of square vertices
187 rotation_angle_radians = math.radians(rotation_angle_degrees)
188 half_drive_size = drive_size_pixels / 2

190 # Points unrotated relative to center
191 square_points = [(-half_drive_size, -half_drive_size), (half_drive_size, -half_drive_size), (half_drive_size, half_drive_size), (-half_drive_size, half_drive_size)]
192 rotated_square_points = []
193 for local_x, local_y in square_points:
194 rotated_x = local_x * math.cos(rotation_angle_radians) - local_y * math.sin(rotation_angle_radians) + center_coordinate
195 rotated_y = local_x * math.sin(rotation_angle_radians) + local_y * math.cos(rotation_angle_radians) + center_coordinate
196 rotated_square_points.append((rotated_x, rotated_y))

199 drive_draw_context.polygon(rotated_square_points, fill=VOID_COLOR_RGBA)

201 # Internal Shadows simulation
202 point0, point1, point2, point3 = rotated_square_points
203
204 # "North" Wall: Illuminated
205 drive_draw_context.line((point0, point1), fill=NORTH_WALL_ILLUMINATION, width=2)
206 # "West" Wall: Illuminated

208 # "East" Wall: Shadow

210 # "South" Wall: Deep Shadow
211 drive_draw_context.line((point2, point3), fill=SOUTH_WALL_DEEP_SHADOW, width=2)
212
213 screw_image = Image.alpha_composite(screw_image, drive_layer_image)

215 # --- 4. Damage & Wear (Cam-out & Scratches) ---
216 if damage_intensity > 0:
217 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"🗡️🎨✨ [LAYER] 4. Applying screwdriver slippage wear (Int: {damage_intensity})", level="TRACE")
218 scratch_layer_image = Image.new('RGBA', (canvas_dimension, canvas_dimension), (0,0,0,0))
219 scratch_draw_context = ImageDraw.Draw(scratch_layer_image)
220
221 # Scratches radiating from the drive (Slippage)
222 num_slippage_scratches = int(SCRATCH_COUNT_BASE * damage_intensity) + 1
223 for _ in range(num_slippage_scratches):
224 scratch_angle_degrees = random.uniform(0, S_ANGLE_MAX)
225 start_distance = drive_size_pixels * START_DISTANCE_RATIO
226 end_distance = screw_radius * END_DISTANCE_RATIO
227
228 start_x = center_coordinate + math.cos(math.radians(scratch_angle_degrees)) * start_distance
229 start_y = center_coordinate + math.sin(math.radians(scratch_angle_degrees)) * start_distance
230 end_x = center_coordinate + math.cos(math.radians(scratch_angle_degrees)) * end_distance
231 end_y = center_coordinate + math.sin(math.radians(scratch_angle_degrees)) * end_distance
232
233 # Metal groove highlight
234 scratch_draw_context.line((start_x, start_y, end_x, end_y), fill=SCRATCH_SILVER_EXPOSURE, width=1)
235
236 screw_image = Image.alpha_composite(screw_image, scratch_layer_image)
237
238 # --- 5. Rust (Accumulation) ---
239 if rust_intensity > 0:
240 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"🟠🎨✨ [LAYER] 5. Accumulating iron oxide rust (Int: {rust_intensity})", level="TRACE")
241
242 # Noise texture for rust
243 noise_dimension = int(drive_size_pixels)
244 rust_noise_texture = Image.effect_noise((noise_dimension, noise_dimension), sigma=RUST_NOISE_SIGMA).resize((int(drive_size_pixels * RUST_RESIZE_RATIO), int(drive_size_pixels * RUST_RESIZE_RATIO)))
245 rust_noise_texture = rust_noise_texture.convert("RGBA")
246
247 # Colorize rust noise (Orange/Brown)
248 rust_alpha_level = int(MAX_ALPHA_VALUE * rust_intensity)
249 rust_color_layer = Image.new("RGBA", rust_noise_texture.size, RUST_BASE_COLOR_RGB + (rust_alpha_level,))
250 rust_composite_texture = ImageChops.multiply(rust_noise_texture, rust_color_layer)
251
252 # Paste over drive center
253 rust_offset_coordinate = int(center_coordinate - drive_size_pixels * RUST_OFFSET_RATIO)
254 screw_image.paste(rust_composite_texture, (rust_offset_coordinate, rust_offset_coordinate), rust_composite_texture)
255
256 # --- 6. Save to Cache ---
257 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, "🎨🆗💾 [SUCCESS] Procedural screw generation complete. Saving to cache.", level="SUCCESS")
258 AssetCacheManager.save_to_cache("screw", size_pixels, size_pixels, configuration_data, screw_image)
259
260 return screw_image
