76 # Generates a mapping from generic marker IDs (Marker_1, Marker_2, etc.) to specific device IDs.
77 # This function assumes a 'Device-###' format for device IDs and creates a mapping for
78 # a batch of markers based on a starting device ID.
79 # Inputs:
80 # start_id (str): The starting device ID (e.g., "Device-001").
81 # Outputs:
82 # dict: A dictionary mapping marker IDs (e.g., "Marker_1") to device IDs (e.g., "Device-001").
83 def _generate_device_map(self, start_id: str) -> dict:
84 """
85 Calculates the next 5 device IDs and maps Marker_1..Marker_6 to them.
86 Assumes ID format is 'Device-###'.
87 """
88 device_map = {}
89 # Extract the number from the starting device ID (e.g., 025 from Device-025)
90 match = re.search(r"Device-(\d+)", start_id)
91 if not match:
92 logger.error(f"❌ Error: Invalid starting Device ID format: {start_id}"
93 )
94 return {}
95
96 start_num = int(match.group(1))
97
98 for i in range(NUMBER_OF_MARKERS):
99 current_num = start_num + i
100 # Format the number back to three digits (e.g., 025)
101 device_id = f"Device-{current_num:03d}"
102 marker_key = f"Marker_{i + 1}"
103 device_map[marker_key] = device_id
104
105 return device_map
