46 @classmethod
47 def get_asset_hash(cls, key_prefix, width, height, config):
48 """Generates a unique hash for a specific asset configuration."""
49 # Convert config to a stable JSON string for hashing
50 config_json = orjson.dumps(config, option=orjson.OPT_SORT_KEYS)
51 hash_input = f"{key_prefix}_{width}x{height}_{config_json.decode()}"
52 # Using SHA256 for better collision resistance and to satisfy security audits.
53 return hashlib.sha256(hash_input.encode()).hexdigest()

55 @classmethod
56 def load_from_cache(cls, key_prefix, width, height, config):
57 """Loads an image from memory or disk if it exists and is healthy."""
58 asset_hash = cls.get_asset_hash(key_prefix, width, height, config)
59
60 # 1. Check Memory Cache First (FASTEST)
61 if asset_hash in _MEMORY_ASSET_CACHE:
62 return _MEMORY_ASSET_CACHE[asset_hash]
63
64 # 2. Check Disk Cache
65 cls._ensure_cache_dir()
66 cache_path = cls._CACHE_DIR / f"{asset_hash}.png"
67
68 if cache_path.exists():
69 try:
70 img = Image.open(cache_path)
71 # ⚡ MANDATORY INTEGRITY CHECK:
72 # Force PIL to actually read the pixel data.
73 img.load()
74
75 # Cache in memory for next time
76 _MEMORY_ASSET_CACHE[asset_hash] = img
77 return img
78 except Exception as e:
79 # If the image is corrupted (truncated, empty, etc), delete it!
80 logger.exception("❌ Corrupted cache file detected and removed: {} ({})", cache_path.name, str(e))
81 try:
82 os.remove(cache_path)
83 except:
84 pass
85 return None
