42def get_manager(context=None, state_cache_manager=None, mqtt_connection_manager=None, subscriber_router=None, run_bridge=True):
43 """
44 Returns the singleton OSCManager instance.
45 If not already initialized, it creates it with the provided managers.
46 Dependencies should be passed externally. If MQTT connection is not provided,
47 OSCManager will manage its own internal connection.
48 """
49 global _instance
50
51 # ⚡ ROBUST SINGLETON: Check if the manager is already initialized in another
52 # copy of this module (happens when run as __main__ and imported as a package)
53 if _instance is None:
54 try:
55 # Attempt to import from a potentially different instance if available
56 import oaComProtocols.oaComOSC.Entry as osc_entry
57 if osc_entry is not sys.modules[__name__] and osc_entry._instance:
58 _instance = osc_entry._instance
59 except (ImportError, AttributeError):
60 pass # Continue with fresh initialization
61
62 if _instance is None:
63 from oaComProtocols.oaComOSC.Managers.osc_manager import OSCManager
64
65 # Provide mocks if not supplied by the orchestrator
66 context = context if context else MockContext()
67 state_cache = state_cache_manager if state_cache_manager else MockStateCache()
68 # MQTT connection manager will be created internally if not provided
69 mqtt_conn = mqtt_connection_manager if mqtt_connection_manager else None
70 # subscriber_router is not directly used by OSCManager's init in this snippet
71
72 _instance = OSCManager(
73 context=context,
74 state_cache_manager=state_cache,
75 mqtt_connection_manager=mqtt_conn, # Will be None if not provided externally
76 run_bridge=run_bridge
77 )
78 matrix_log("comms", "osc", "get_manager", "OSCManager initialized.", "DEBUG")
79 else:
80 # Update existing instance if new dependencies are provided
81 if context: _instance.context = context
82 if state_cache_manager: _instance.state_cache_manager = state_cache_manager
83 if mqtt_connection_manager: _instance.mqtt_connection_manager = mqtt_connection_manager

85 return _instance
