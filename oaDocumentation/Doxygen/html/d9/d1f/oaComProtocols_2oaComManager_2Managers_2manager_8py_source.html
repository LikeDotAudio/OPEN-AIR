53 """

57 protocol_base_path = project_root / "oaComProtocols"

70 if entry_file_path.is_file():
71 module_full_name = f"oaComProtocols.{module_name_short}.Entry"

81 "module_name": module_full_name,
82 "start": entry_module.start,
83 "stop": entry_module.stop,

85 }

125 def start_all(self, **kwargs):
126 """
127 Starts all registered protocol modules. Each module's start function is called
128 in a separate thread to ensure they run concurrently.
129 Common dependencies (excluding MQTT) are passed.
130 """
131 with self._lock:
132 if self.running_threads:
133 matrix_log("manager", "com_protocol", "start_all", "⚠️ All protocols already running.", "WARNING")
134 return
135
136 matrix_log("manager", "com_protocol", "start_all", "🚀 Starting all registered protocols...", "INFO")
137
138 # Discover and register protocols if not already done
139 if not self.protocol_modules:
140 self.discover_and_register_protocols()
141
142 # Initialize common dependencies once before starting modules
143 # MQTT connection is managed internally by individual modules.
144 if not self.initialize_common_dependencies():
145 matrix_log("manager", "com_protocol", "start_all", "❌ Failed to initialize common dependencies. Aborting start.", "ERROR")
146 return
147
148 for name, info in self.protocol_modules.items():
149 try:
150 # Prepare arguments for the start function.
151 # This needs to be adaptive to each module's start() signature.
152 start_args = {}
153
154 # Pass common dependencies if the start function expects them.
155 # Exclude MQTT-related arguments as they are self-contained.
156
157 if "subscriber_router" in info["start"].__code__.co_varnames:
158 start_args["subscriber_router"] = self.subscriber_router
159 if "protocol_router" in info["start"].__code__.co_varnames:
160 start_args["protocol_router"] = self.protocol_router
161 if "run_bridge" in info["start"].__code__.co_varnames:
162 # ⚡ PARTITION PROPAGATION: Pass the run_bridge value from kwargs (from Entry.py)
163 start_args["run_bridge"] = kwargs.get("run_bridge", True)
164
165 # Handle specific dependencies like mqtt_publisher for DNSSD/MDNS/SAP
166 # These modules created their own StandaloneMqttPublisher internally.
167 # If they are refactored to accept an MQTT client, that would be handled here.
168 # For now, assuming they create their own.
169
170 # Add other common kwargs if identified
171 # start_args["some_other_param"] = kwargs.get("some_other_param")
172
173 thread = threading.Thread(target=info["start"], kwargs=start_args, name=f"{name}-Thread", daemon=True)
174 thread.start()
175 self.running_threads[name] = thread

182 def stop_all(self):
183 """
184 Stops all running protocol modules by calling their respective stop functions.
185 """
186 with self._lock:
187 if not self.running_threads:
188 matrix_log("manager", "com_protocol", "stop_all", "⚠️ No protocols were running.", "WARNING")
189 return
190
191 matrix_log("manager", "com_protocol", "stop_all", "🛑 Stopping all registered protocols...", "INFO")
192 for name, thread in list(self.running_threads.items()): # Iterate over a copy of items
193 try:
194 module_info = self.protocol_modules.get(name)
195 if module_info and callable(module_info["stop"]):
196 module_info["stop"]() # Call the module's stop function
197 matrix_log("manager", "com_protocol", "stop_all", f"✅ Stopped {name}.", "SUCCESS")
198 else:
199 matrix_log("manager", "com_protocol", "stop_all", f"⚠️ {name} has no callable stop function.", "WARNING")
200
201 # Clean up thread reference
202 if name in self.running_threads:
203 del self.running_threads[name]
204
205 except Exception as e:
206 matrix_log("manager", "com_protocol", "stop_all", f"❌ Error stopping {name}: {e}", "ERROR")
207
208 matrix_log("manager", "com_protocol", "stop_all", "All protocols stopped (or attempted).", "INFO")
209 self.running_threads.clear() # Ensure it's empty
