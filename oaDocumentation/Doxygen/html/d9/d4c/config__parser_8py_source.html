82 def get_requested_dimensions(self) -> tuple[int, int]:
83 """Calculates the total required width and height for the canvas."""
84 scale_text_buffer = 15
85
86 tick_h = self.tick_size if (self.show_ticks or self.tick_both_sides) else 0
87 if self.is_vertical:
88 label_thick = (self.font_size * 3) if (self.scale_position != "none" and self.show_scale_labels) else 0
89 else:
90 label_thick = (self.font_size + 4) if (self.scale_position != "none" and self.show_scale_labels) else 0
91
92 side_a = tick_h + label_thick
93 if self.peak_display: side_a += 6
94 side_b = tick_h if self.tick_both_sides else 0
95
96 if not self.is_vertical:
97 req_w = self.width + (scale_text_buffer * 2)
98 req_h = self.height + side_a + side_b
99 else:
100 req_w = self.width + side_a + side_b
101 req_h = self.height + (scale_text_buffer * 2)
102
103 return int(req_w), int(req_h)
