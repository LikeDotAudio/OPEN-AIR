22 def register_for_bg_sync(self, widget, canvas, config_data, context):
23 """
24 Registers a widget for automatic background synchronization with its parent.
25 Eliminates the need for local sync_bg() functions in widget creators.
26 """
27 builder_instance = context.builder_instance if context else None
28 parent = widget.master
29
30 def perform_sync(event=None):
31 if not widget.winfo_exists(): return
32
33 # ⚡ ROBUSTNESS: Prevent X11 BadValue (0x0) errors by avoiding configuration
34 # of widgets that have not yet been realized or have zero dimensions.
35 try:
36 w, h = widget.winfo_width(), widget.winfo_height()
37 if w <= 1 or h <= 1: return
38 except tk.TclError: return
39
40 try:
41 p_bg = parent.cget("bg")
42 if widget.cget("bg") != p_bg:
43 widget.configure(bg=p_bg)
44 if canvas and canvas.winfo_exists() and canvas.cget("bg") != p_bg:
45 canvas.configure(bg=p_bg)
46
47 # Use stored slice method if available
48 if hasattr(widget, '_perform_background_slice'):
49 widget._perform_background_slice()
50 else:
51 # Re-apply transparency slicing if not already registered
52 self._apply_transparency(widget, canvas, config_data, builder_instance)
53 except Exception as e:
54 matrix_log("UI", "GUI_MANAGER", inspect.currentframe().f_code.co_name, f"perform_sync error: {e}", level="DEBUG")
55 pass
56
57 # Bind to parent's configuration changes to keep in sync
58 parent.bind("<Configure>", perform_sync, add="+")
59 # Also sync immediately on Map
60 widget.bind("<Map>", perform_sync, add="+")
