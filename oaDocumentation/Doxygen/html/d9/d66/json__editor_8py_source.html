102 def _recurse(item, current_depth):

316 def _update_gutter(self):
317 self.gutter.config(state="normal")
318 self.gutter.delete("1.0", "end")
319
320 content = self.text_area.get("1.0", "end-1c")
321 lines = content.split('\n')
322 self.fold_regions.clear()
323
324 folded_ranges = self.text_area.tag_ranges("folded")
325
326 for i, line in enumerate(lines):
327 original_line_idx = i + 1
328 is_hidden = False
329 for r in range(0, len(folded_ranges), 2):
330 if self.text_area.compare(f"{original_line_idx}.0", ">=", folded_ranges[r]) and \
331 self.text_area.compare(f"{original_line_idx}.0", "<", folded_ranges[r+1]):
332 is_hidden = True; break
333
334 if is_hidden: continue
335
336 leading_spaces = len(line) - len(line.lstrip())
337 depth = leading_spaces // 2
338 color_idx = depth % len(self.DEPTH_COLORSDEPTH_COLORS)
339
340 indent_str = " " * depth
341 self.gutter.insert("end", indent_str)
342
343 if '{' in line or '[' in line:
344 glyph = "▶" if original_line_idx in self.folded_lines else "▼"
345 self.gutter.insert("end", glyph, ("fold_marker", f"depth_{color_idx}", f"fold_{original_line_idx}"))
346
347 nest = 0
348 for j in range(i, len(lines)):
349 nest += lines[j].count('{') + lines[j].count('[')
350 nest -= lines[j].count('}') + lines[j].count(']')
351 if nest == 0:
352 # ⚡ STORE DEPTH: Needed for level-specific folding
353 self.fold_regions[original_line_idx] = (f"{original_line_idx}.end", f"{j+1}.end", depth)
354 break
