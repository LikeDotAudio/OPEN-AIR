15 """

17 If message_callback is provided, shows a message emoji to view details.
18 """
19 h_frame = tk.Frame(parent, bg="#2d2d2d", pady=2)
20 h_frame.pack(fill="x", pady=(8, 2))
21
22 is_expanded = tk.BooleanVar(value=is_expanded_val)
23 fg_col = "#33A1FD" if not is_virtual else "#666666"
24 toggle_char = "▼" if is_expanded_val else "▶"
25
26 title_lbl = tk.Label(h_frame, text=f"{toggle_char} {key.upper()}", bg="#2d2d2d", fg=fg_col,
27 font=("Arial", 8, "bold"), cursor="hand2", anchor="w")
28 title_lbl.pack(side="left", fill="x", expand=True)
29
30 if message_callback:
31 msg_btn = tk.Label(h_frame, text="✉️", bg="#2d2d2d", fg="#FF9900", font=("Arial", 10), cursor="hand2")
32 msg_btn.pack(side="right", padx=5)
33 msg_btn.bind("<Button-1>", lambda e: message_callback())
34
35 if is_virtual:
36 tk.Button(h_frame, text="+", bg="#37373d", fg="#aaaaaa", relief="flat", font=("Arial", 7, "bold"),
37 command=on_add_callback).pack(side="right", padx=5)
38
39 def toggle_wrapper(e):
40 new_state = not is_expanded.get()

42 char = "▼" if new_state else "▶"
43 title_lbl.config(text=f"{char} {key.upper()}")
44 on_toggle_callback(new_state)
