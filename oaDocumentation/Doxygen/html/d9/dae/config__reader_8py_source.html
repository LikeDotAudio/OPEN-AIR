60 """Helper to get and parse config values with fallbacks."""

130 "CORE": 10,
131 "UI": 20

136 self.OSC_RX_PORTOSC_RX_PORT += offset
137 self.OSC_TX_PORTOSC_TX_PORT += offset

139 # ⚡ LOOPBACK SYNC: In UI partition, TX Target should point to Core's RX Port

161 def _parse_debug_matrix(self, config):
162 # ⚡ GRANULAR CONTROLS: Parse all DEBUG_* sections into the matrix
163 debug_sections = [
164 "DEBUG_MATRIX", "DEBUG_CORE_ORCHESTRATION", "DEBUG_STATE_DATA",
165 "DEBUG_TRANSLATOR", "DEBUG_COMMS_PROTOCOLS", "DEBUG_GUI",
166 "DEBUG_RUST_FFI", "DEBUG_ROUTER", "DEBUG_FILE_IO"
167 ]
168
169 for section_name in debug_sections:
170 if section_name in config:
171 section = config[section_name]
172 for key in section:
173 # ⚡ FIX: Skip known string-based keys to avoid ValueError
174 if key.lower() in ["mute_functions", "force_functions", "oid_map_source"]:
175 continue
176 # Store all keys in uppercase for consistent matrix lookup
177 try:
178 self.DEBUG_MATRIX[key.upper()] = section.getboolean(key, False)
179 except ValueError:
180 logger.warning(f"Config: Key '{key}' in '{section_name}' is not boolean. Skipping.")
181
182 # Special case for comma-separated strings (usually in DEBUG_MATRIX)
183 if "DEBUG_MATRIX" in config:
184 self.MUTE_FUNCTIONSMUTE_FUNCTIONS = config["DEBUG_MATRIX"].get("mute_functions", "")
