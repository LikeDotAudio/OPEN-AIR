185def _draw_body(canvas, center_x, center_y, radius, shape, color, gradient_level, rotation_angle=0, outline_thickness=0, fill_color="", teeth=8):
186 steps = gradient_level + 1
187 for index in range(steps):
188 current_radius = radius - (index * 2)
189 if current_radius <= 0: break
190 current_thickness = outline_thickness if index == 0 else 1
191 current_fill = fill_color if (index == 0 or steps == 1) else ""
192 if current_thickness == 0 and index == 0 and gradient_level == 0 and not current_fill:
193 continue
194
195 if shape == "circle":
196 if gradient_level > 0 or (index == 0 and (current_thickness > 0 or current_fill)):
197 canvas.create_oval(center_x-current_radius, center_y-current_radius, center_x+current_radius, center_y+current_radius, outline=color, width=current_thickness, fill=current_fill)
198 elif shape == "octagon":
199 OCTAGON_SIDES = 8
200 points = _get_poly_points(center_x, center_y, current_radius, sides=OCTAGON_SIDES, start_angle=rotation_angle)
201 canvas.create_polygon(points, outline=color, fill=current_fill, width=current_thickness)
202 elif shape == "gear":
203 GEAR_NOTCH_DEPTH = 0.15
204 points = _get_gear_points(center_x, center_y, current_radius, teeth=teeth, notch_depth=GEAR_NOTCH_DEPTH, start_angle=rotation_angle)
205 canvas.create_polygon(points, outline=color, fill=current_fill, width=current_thickness)

229def _draw_ticks(canvas, center_x, center_y, radius, arc_width, tick_length, style, color, min_val, max_val):
230 TICK_START_ANGLE = 240
231 TICK_TOTAL_EXTENT = 300
232 TICK_STEP_DEG = 30
233 end_angle = TICK_START_ANGLE - TICK_TOTAL_EXTENT
234
235 TICKS_PER_SPAN = 10.0
236 value_step = (max_val - min_val) / TICKS_PER_SPAN
237 current_angle, current_value = TICK_START_ANGLE, min_val

242 ts_dist, te_dist = TICK_BASE_DIST, TICK_BASE_DIST + tick_length

244 end_x, end_y = center_x + te_dist * math.cos(radians), center_y - te_dist * math.sin(radians)
245
246 if style == "dots":
247 DOT_RADIUS = 1
248 canvas.create_oval(end_x - DOT_RADIUS, end_y - DOT_RADIUS, end_x + DOT_RADIUS, end_y + DOT_RADIUS, fill=color, outline=color)

288def _get_poly_points(center_x, center_y, radius, sides=8, start_angle=0):
289 points = []
290 angle_step = 360 / sides
291 for index in range(sides):
292 degrees = index * angle_step + start_angle
293 radians = math.radians(degrees)
294 points.extend([center_x + radius * math.cos(radians), center_y - radius * math.sin(radians)])
295 return points

297def _get_gear_points(center_x, center_y, radius, teeth=8, notch_depth=0.15, start_angle=0):
298 """
299 Generates points for a gear shape with rounded (trapezoidal) teeth.
300 Each tooth consists of 4 segments to soften the points.
301 """
302 points = []
303 POINTS_PER_TOOTH = 4
304 num_segments = teeth * POINTS_PER_TOOTH # 4 points per tooth cycle
305 inner_radius = radius * (1 - notch_depth)
306 angle_step = 360 / num_segments
307
308 for index in range(num_segments):
309 degrees = index * angle_step + start_angle
310 radians = math.radians(degrees)
311
312 # Cycle through 4 states: Top-Left, Top-Right, Bottom-Right, Bottom-Left
313 tooth_state = index % 4
314 if tooth_state in [1, 2]: # "Top" of the tooth
315 current_radius = radius
316 else: # "Bottom" of the notch
317 current_radius = inner_radius
318
319 points.extend([center_x + current_radius * math.cos(radians), center_y - current_radius * math.sin(radians)])
320 return points
