78 def load_headers(self):
79 if not self.csv_filepath or not os.path.exists(self.csv_filepath):
80 return messagebox.showerror("Error", "Select valid CSV.")
81 try:
82 df = pd.read_csv(self.csv_filepath, nrows=1, keep_default_na=False)
83 self.headers = list(df.columns)
84 self.header_ui = HeaderConfigUI(self.h_frame, self.headers, self.preview)
85 except Exception as e:
86 logger.exception(f"Header loading failed: {e}")
87 messagebox.showerror("Error", str(e))
88
89 def generate_data(self):
90 try:
91 df = pd.read_csv(self.csv_filepath, keep_default_na=False)
92 h_map = self.header_ui.get_config_map()
93 sort_cols = [h for h, configuration in h_map.items() if configuration["role"] in ["Hierarchical Key", "Value as Key", "Key Name and Value"]]
94 df.sort_values(by=sort_cols, inplace=True, kind="stable")
95 result = CSVConverterEngine.build_hierarchy(df, h_map, "root", self.headers)
96 return {self.root_en.get(): result}
97 except Exception as e:
98 logger.exception(f"Generation failed: {e}")
99 messagebox.showerror("Error", f"Generation failed: {e}")
100 return None

106 def convert(self):
107 path = self.json_en.get()
108 if not path: return messagebox.showerror("Error", "Specify output path.")
109 data = self.generate_data()

111 try:
112 with open(path, "wb") as f: f.write(orjson.dumps(data))
113 messagebox.showinfo("Success", f"Saved to {path}")
114 except Exception as e:
115 logger.exception(f"Conversion or file saving failed: {e}")
116 messagebox.showerror("Error", str(e))
