199 def _add_entry(self, message):
200 if not self.tree.winfo_exists(): return
201
202 # Machine Time (UTP)
203 utp = f"{message['timestamp']:.6f}"
204 source = message["source"] # Transport (e.g. MQTT)
205 logical_source = message.get("logical_source", source) # Identity (e.g. MIDI)
206 # ⚡ V3.1.5 HARDENING: Ensure 'guid' exists before access to prevent Tkinter KeyError
207 guid = message.get("logical_guid", message.get("guid", "UNKNOWN"))
208
209 strategy = message.get("strategy", "BROADCAST")
210 topic = message["topic"]
211 value = message["value"]
212
213 # Tags are now pre-calculated by the router (no brains in GUI)
214 tags = message.get("ui_tags", [])
215
216 # ⚡ DISPLAY ENRICHMENT: Add emoji for visual confirmation in the tree ONLY
217 display_source = logical_source
218 if "SPLINK" in tags:
219 display_source = f"🔗 {logical_source}"
220
221 # ⚡ STACK BEHAVIOR: Insert at TOP (index 0)
222 item_id = self.tree.insert("", 0, values=(utp, display_source, guid, strategy, topic, value), tags=tuple(tags))
223
224 if len(self.tree.get_children()) > 100:
225 self.tree.delete(self.tree.get_children()[-1])
226
227 # If autoscroll is on, make sure the latest (top) is visible
228 if self.autoscroll_var.get():
229 self.tree.see(item_id)

231 def on_select_packet(self, event):
232 """Populates the investigation pane and handles SPLINK selection logic."""
233 selected = self.tree.selection()
234 if not selected: return
235
236 item = self.tree.item(selected[0])
237 utp = item["values"][0]
238 topic = item["values"][4]
239 tags = item["tags"]
240
241 # ⚡ AUTO-DUAL: If this is a splinked message, try to find and show BOTH halves
242 if "SPLINK" in tags:
243 trigger_ts, result_ts = self.router.get_splink_relationship(utp)
244 if trigger_ts and result_ts:
245 self._src_utp = f"{trigger_ts:.6f}"
246 self._dest_utp = f"{result_ts:.6f}"
247 # Update Entries
248 item_src = self._find_item_by_ts(self._src_utp)
249 item_dest = self._find_item_by_ts(self._dest_utp)
250 if item_src: self.src_var.set(item_src["values"][4])
251 if item_dest: self.dest_var.set(item_dest["values"][4])
252 self.splink_btn.configure(fg="#00ff00")
253 self._refresh_investigation()
254 return
255
256 # ⚡ STANDARD ALTERNATING LOGIC (for non-splinked messages)
257 if not self.src_var.get():
258 self.src_var.set(topic)
259 self._src_utp = utp
260 self.splink_btn.configure(fg="#888888")
261 elif not self.dest_var.get():
262 if topic != self.src_var.get():
263 self.dest_var.set(topic)
264 self._dest_utp = utp
265 self.splink_btn.configure(fg="#00ff00")
266 else:
267 self.src_var.set("")
268 self._src_utp = None
269 else:
270 # Shift
271 self.src_var.set(self.dest_var.get())
272 self._src_utp = getattr(self, "_dest_utp", None)
273 self.dest_var.set(topic)
274 self._dest_utp = utp
