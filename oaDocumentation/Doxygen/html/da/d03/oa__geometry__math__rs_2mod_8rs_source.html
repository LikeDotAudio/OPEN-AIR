12fn normalize_value(value: f64, min_val: f64, max_val: f64) -> f64 {
13 if (max_val - min_val).abs() < f64::EPSILON {
14 return 0.0;
15 }
16 (value - min_val) / (max_val - min_val)
17}

21 let norm = normalize_value(value, min_val, max_val);
22 if reverse {
23 (1.0 - norm) * pixel_length

29#[pyfunction]
30fn rotate_point(px: f64, py: f64, cx: f64, cy: f64, angle_deg: f64) -> (f64, f64) {
31 let rad = angle_deg.to_radians();
32 let cos_a = rad.cos();
33 let sin_a = rad.sin();
34
35 let nx = cos_a * (px - cx) - sin_a * (py - cy) + cx;
36 let ny = sin_a * (px - cx) + cos_a * (py - cy) + cy;
37
38 (nx, ny)
39}

42fn get_position(angle_deg: f64, distance: f64, center_x: f64, center_y: f64) -> (f64, f64) {
43 let rad = angle_deg.to_radians();
44 let x = center_x + distance * rad.cos();
45 let y = center_y + distance * rad.sin();
46 (x, y)
