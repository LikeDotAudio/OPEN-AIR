138 def _midi_listen_loop(self, port_or_name, _one_shot=False):
139 """High-priority loop for reading from a physical MIDI port."""
140 if isinstance(port_or_name, str):
141 port = self.ports.open_input(port_or_name)
142 else:
143 port = port_or_name # Support passing mock port for tests
144
145 if not port: return
146
147 matrix_log("comms", "midi", "_midi_listen_loop", f"▶️ [MIDI-LISTEN] Started listening on port: {getattr(port, 'name', 'unknown')}", "DEBUG")
148
149 while self._running:
150 try:
151 # Support tests that use iter_pending
152 if hasattr(port, 'iter_pending'):
153 messages = list(port.iter_pending())
154 else:
155 message = port.receive(timeout=0.005)
156 messages = [message] if message else []
157
158 for message in messages:
159 # 1. Translate MIDI to System Topic
160 topic, value = self.mapper.midi_to_topic(message, getattr(port, 'name', 'unknown'))
161 meta = {
162 "midi_type": message.type,
163 "guid": f"{topic.split('/')[2] if topic and '/' in topic else 'unknown'}/{getattr(message, 'channel', 0)}",
164 "midi_raw": str(message),
165 "origin_source": "MIDI" # Required for tests
166 }
167
168 if topic:
169 # 2. Update Hardware Lock to prevent echo fighting
170 self.lock_manager.lock(topic)
171
172 # ⚡ LOCAL FIRST: Notify internal monitors (Dashboard) immediately
173 self._notify_monitor("RX", {
174 "value": getattr(message, 'velocity', getattr(message, 'value', 0)),
175 "velocity": getattr(message, 'velocity', 0),
176 "channel": getattr(message, 'channel', 0),
177 "note": getattr(message, 'note', getattr(message, 'control', 0)),
178 "type": message.type,
179 "port": getattr(port, 'name', 'unknown'),
180 "raw": str(message)
181 })
182
183 # 3. Inject into ProtocolRouter
184 if self.state_cache_manager:
185 self.state_cache_manager.handle_external_update(topic, value, source="MIDI", metadata=meta)
186
187 # ⚡ DIRECT MQTT: Broadcast directly to broker if active (Standalone)
188 if self.mqtt_worker:
189 self.mqtt_worker.publish(topic, value, meta)
190
191 # 4. Cleanup lock based on event type
192 if message.type in ['note_on', 'note_off', 'control_change']:
193 if message.type == 'note_on' and getattr(message, 'velocity', 0) > 0:
194 pass # Keep locked while held? (Optional logic)
195 elif message.type == 'note_off':
196 self.lock_manager.unlock(topic)
197
198 if message.type == 'control_change':
199 self.lock_manager.delayed_unlock(topic)
200
201 if _one_shot: break
202 if not messages:
203 time.sleep(0.001) # Yield to CPU
204 except Exception as e:
205 matrix_log("comms", "midi", "_midi_listen_loop", f"🛑 [MIDI-LISTEN] Error on {getattr(port, 'name', 'unknown')}: {e}", "ERROR")
206 break
207
208 # Don't close if it was passed in (tests)
209 if isinstance(port_or_name, str):
210 self.ports.close_input(port_or_name)

214 def _on_protocol_event(self, message):
215 """
216 Triggered when a protocol event is received (from ProtocolRouter/MQTT).
217 Matches topics and determines if a MIDI message should be transmitted.
218 """
219 topic = message.get("topic")
220 value = message.get("value")
221 meta = message.get("meta", message.get("metadata", {}))
222 source = message.get("source", "UNKNOWN")
223 direction = "TX"
224
225 # 1. Loop Prevention: Ignore if WE were the source
226 if source == "MIDI" or source == "MIDI-TX" or meta.get("origin_source") == "MIDI":
227 return
228
229 # 2. Check if topic belongs to MIDI namespace
230 is_midi_topic = topic.startswith("OPEN-AIR/MIDI/")
231
232 # 3. Handle specific MIDI topics (Reflected from MQTT or link feedback)
233 if is_midi_topic:
234 # Reconstruct the monitor notification for GUI visualization
235 try:
236 real_val = value.get("value") if isinstance(value, dict) else value
237 import re
238 note_match = re.search(r"note(\d+)", topic)
239 note = int(note_match.group(1)) if note_match else 0
240
241 # Try to extract channel from topic as well
242 channel_match = re.search(r"ch(\d+)", topic)
243 channel = (int(channel_match.group(1)) - 1) if channel_match else 0
244
245 m_type = "note_on" if real_val > 0 else "note_off"
246 self._notify_monitor(direction, {
247 "value": real_val,
248 "topic": topic,
249 "note": note,
250 "channel": channel,
251 "velocity": real_val if isinstance(real_val, (int, float)) and real_val <= 127 else 127,
252 "type": m_type,
253 "raw": f"{m_type} note={note} channel={channel} velocity={real_val}"
254 })
255 except Exception: pass
256
257 if not self.run_bridge: return # Observers only monitor, they don't transmit
258
259 # 4. Translate back to MIDI and transmit
260 midi_message = self.mapper.topic_to_midi(topic, value)
261 if midi_message:
262 # Determine which port to send to (extracted from topic)
263 # topic: OPEN-AIR/MIDI/<port_name>/...
264 parts = topic.split('/')
265 if len(parts) > 2:
266 target_port = parts[2]
267 self.publish(target_port, midi_message)

269 def publish(self, *args, **kwargs):
270 """
271 Sends a MIDI message. Supports multiple signatures:
272 1. publish(port_name, midi_message) - Direct hardware send
273 2. publish(topic, value, meta) - Core router dispatch
274 """
275 if not self.run_bridge: return
276
277 # Signature 1: Direct hardware send (port_name, mido_message)
278 if len(args) == 2 and hasattr(args[1], 'type'):
279 port_name, midi_message = args
280 port = self.ports.open_output(port_name)
281 if port:
282 try:
283 port.send(midi_message)
284 except Exception as e:
285 matrix_log("comms", "midi", "publish", f"🛑 [MIDI-TX] Error sending to {port_name}: {e}", "ERROR")
286 return
287
288 # Signature 2: Core router dispatch (topic, value, meta)
289 if len(args) >= 2:
290 topic = args[0]
291 value = args[1]
292 meta = args[2] if len(args) > 2 else (kwargs.get('meta') or kwargs.get('metadata') or {})
293
294 # Anti-feedback check
295 if meta.get("origin_source") == "MIDI" or meta.get("source") == "MIDI":
296 return
297
298 midi_message = self.mapper.topic_to_midi(topic, value)
299 if midi_message:
300 # Hardware locks prevent us from moving a fader the user is currently touching
301 if self.lock_manager.is_locked(topic):
302 matrix_log("comms", "midi", "publish", f"🎹 [MIDI-TX] Dropping update for {topic} (Hardware Locked)", "DEBUG")
303 return
304
305 # Broadcast to ALL active outputs (simple Hub-and-Spoke model)
306 for outport in self.ports.outports:
307 try:
308 outport.send(midi_message)
309 except Exception as e:
310 logger.error(f"Failed to send MIDI to {getattr(outport, 'name', 'unknown')}: {e}")
