157 def refreshMixer(self) -> None:
158 try:
159 sinks = self.mixer.get_available_devices()
160 sources = self.mixer.get_available_sources()
161 apps = self.mixer.get_connected_software()
162
163 # Clear and populate
164 for list_id, items, type in [("#sink-list", sinks, 'device'),
165 ("#source-list", sources, 'source'),
166 ("#app-list", apps, 'app')]:
167 cont = self.query_one(list_id, ScrollableContainer)
168 cont.remove_children()
169 for item in items:
170 label = item['description'] if 'description' in item else item['name']
171 target_id = item['name'] if type != 'app' else item['pid']
172 cont.mount(UniversalVolumeControl(target_id, label, item['volume'], type, item.get('is_default', False)))
173 except Exception as e:
174 self.notify(f"Refresh Error: {e}", severity="error")

176 @work
177 async def start_pcm_visualizer(self) -> None:
178 bin_path = project_root / "oaAudioMixer/Core/oaAudioMixer_rs/target/release/pcm_visualizer"
179 monitor = self.query_one("#pcm-monitor", PCMBitMonitor)
180 if not bin_path.exists(): return
181 try:
182 self.visualizer_process = await asyncio.create_subprocess_exec(str(bin_path), stdout=asyncio.subprocess.PIPE)
183 current_frame = []
184 while True:
185 line_raw = await self.visualizer_process.stdout.readline()
186 if not line_raw: break
187 line = line_raw.decode('utf-8', errors='replace').strip()
188 if line == "---FRAME---":
189 if current_frame: monitor.update_bits("\n".join(current_frame)); current_frame = []
190 else:
191 if len(current_frame) < 16: current_frame.append(line)
192 except: pass
