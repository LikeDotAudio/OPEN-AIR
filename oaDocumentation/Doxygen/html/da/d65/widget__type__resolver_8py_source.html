50 def resolve_type(cls, config, orientation="vertical"):
51 """
52 Detects the implementation type based on aliases and context.
53 """
54 widget_type = config.get("type") or config.get("widget_type", "")
55
56 if widget_type == "_SmartMeter":
57 cosmetics = config.get("cosmetics", {})
58 viz = cosmetics.get("style_flags", {}).get("visualization", "bar").lower()
59 return "_NeedleVUMeter" if viz == "needle" else "_BarGraph"
60
61 if widget_type == "_SmartFader":
62 return "_CustomHorizontalFader" if orientation == "horizontal" else "_CustomFader"
63
64 resolved = cls.TYPE_MAP.get(widget_type, widget_type)
65 return resolved
