33class SnmpStatePersister:
34 """
35 Handles the background persistence of SNMP state (OID map) to a file.
36 This class is designed to be managed by a parent class (e.g., SNMPManager).
37 """
38
39 def __init__(self, state_provider, thread_lock: threading.RLock, notify_monitor_callback, run_bridge: bool, base_oid: str, oid_map_converter):
40 """
41 Initializes the state persister.
42
43 Args:
44 state_provider: Manager providing access to the reflected MQTT state.
45 thread_lock: The RLock used by the calling manager for thread-safe access.
46 notify_monitor_callback: Callback function to notify about SNMP activity.
47 run_bridge: Flag indicating if the SNMP bridge is active.
48 base_oid: The base OID for SNMP data.
49 oid_map_converter: The converter instance providing the OID map.
50 """
51 self.state_provider = state_provider
52 self._state_lock = thread_lock # Use the provided lock for external state access
53 self._notify_monitor = notify_monitor_callback
54 self.run_bridge = run_bridge
55 self.base_oid = base_oid
56 self.oid_map_converter = oid_map_converter # Reference to the converter
57
58 self._running = True # Internal flag to control the loop
59 self._thread = None
60
61 self.state_file = str(SNMP_STATE_FILE)
62 self._last_topic_count = 0
63 self._last_persisted_values = {} # ⚡ DELTA TRACKING: Only notify monitor on change

65 def start(self):
66 """Starts the background thread for state persistence."""
67 if self._thread and self._thread.is_alive():
68 snmp_logger.warning("SnmpStatePersister: Thread already running.")
69 return
70
71 self._running = True
72 self._thread = threading.Thread(target=self._persistence_loop, daemon=True, name="SNMP-StatePersistenceLoop")
73 self._thread.start()
74 matrix_log("comms", "snmp", "start",
75 "SnmpStatePersister: Started background persistence thread.", "INFO")
76
77 def stop(self):
78 """Signals the background thread to stop and waits for it to terminate."""
79 self._running = False
80 if self._thread and self._thread.is_alive():
81 matrix_log("comms", "snmp", "stop",
82 "SnmpStatePersister: Stopping background persistence thread...", "INFO")
83 self._thread.join(timeout=THREAD_JOIN_TIMEOUT) # Wait for thread to finish
84 if self._thread.is_alive():
85 snmp_logger.warning("SnmpStatePersister: Thread did not terminate gracefully.")

88 def reset_delta_tracking(self):
89 """⚡ MONITOR SYNC: Clears the last seen values to force a full re-notification of current state."""
90 with self._state_lock:
91 self._last_persisted_values.clear()
92 matrix_log("comms", "snmp", "reset_delta_tracking", "📡 [SNMP] Delta tracking reset. Full monitor refresh scheduled.", "DEBUG")

94 def _persistence_loop(self):
95 """The main loop for periodically saving SNMP state to a file."""
96 while self._running:
97 try:
98 if not self.state_provider:
99 # Non-matrix log for critical misconfiguration
100 snmp_logger.warning("SnmpStatePersister: State provider is not available. Skipping persistence.")
101 time.sleep(STATE_SYNC_INTERVAL)
102 continue
103
104 # Safely get OID map data for processing from the converter
105 oid_map_data = []
106 state_snapshot = {}
107
108 # Accessing state and converter requires the lock
109 with self._state_lock: # Use the lock provided by the manager
110 # ⚡ REFLECTION: Get the MQTT state from the provider instead of the global cache
111 state_snapshot = getattr(self.state_provider, "get_mqtt_state", lambda: {})()
112 if not state_snapshot: # Skip if state is empty
113 time.sleep(STATE_SYNC_INTERVAL)
114 continue
115
116 # 2. ⚡ REFRESH: Rebuild the OID map from this specific snapshot
117 self.oid_map_converter.build_oid_map(state_snapshot=state_snapshot)
118
119 # 3. Extract the items for iteration outside the lock
120 oid_map_data = list(self.oid_map_converter.oid_map.items())
121
122 sorted_items = sorted(oid_map_data, key=lambda x: x[1]['topic'])
123 lines = []
124 current_values = {} # Build fresh map for this loop
125
126 for oid, data in sorted_items:
127 topic = data['topic']
128 # Use the snapshot we took inside the lock for consistency
129 payload = state_snapshot.get(topic, {})
130 # Get payload for filtering if needed
131
132 # ⚡ ANTI-FEEDBACK SPEC: The Golden Rule for Transports
133 # Payloads from ProtocolRouter already have these normalized fields
134 message_type = payload.get("message_type")
135 origin_source = payload.get("origin_source")
136 is_settled = payload.get("is_settled", False)
137
138 # 1. If it's LINK_FEEDBACK, we only push to SNMP if it's SETTLED (confirmed state)
139 if message_type == "LINK_FEEDBACK" and not is_settled:
140 continue
141
142 # 2. If the origin_source is SNMP, don't send it back to SNMP
143 if origin_source == "SNMP":
144 continue
145
146 val_str = data['value']
147 lines.append(f"{oid}:{val_str}")
148 current_values[oid] = val_str
149
150 # ⚡ DELTA NOTIFICATION: Only notify monitor if the value has changed
151 # This prevents the Activity MQTT topic from being flooded every second.
152 if self._last_persisted_values.get(oid) != val_str:
153 self._notify_monitor("TX_DUMP", oid, val_str, topic, data)
154
155 # Update tracking for next loop
156 self._last_persisted_values = current_values
157
158 # Only perform file operations if the bridge is active and there's data
159 if self.run_bridge and lines:
160 current_count = 0
161 with self._state_lock: # Access converter's map for count
162 current_count = len(self.oid_map_converter.oid_map)
163
164 # Check if OID map has grown and sync MIB if necessary
165 if current_count > 0 and current_count != self._last_topic_count:
166 matrix_log("comms", "snmp", "_persistence_loop",
167 f"SnmpStatePersister: Tree Expansion ({self._last_topic_count} -> {current_count}). MIB sync might be needed.", "INFO")
168 with self._state_lock:
169 self._last_topic_count = current_count
170
171 # Sort lines for consistent file writing
172 def oid_key(oid_line):
173 oid_str = oid_line.split(':')[0]
174 return [int(x) for x in oid_str.strip('.').split('.')]
175 lines.sort(key=oid_key)
176
177 # Write to a temporary file and then replace
178 os.makedirs(os.path.dirname(self.state_file), exist_ok=True)
179 temp_path = self.state_file + ".tmp"
180 with open(temp_path, "w", encoding="utf-8") as f:
181 f.write("\n".join(lines) + "\n")
182
183 if os.path.exists(temp_path):
184 os.replace(temp_path, self.state_file)
185 os.chmod(self.state_file, 0o644) # Set appropriate permissions
186 matrix_log("comms", "snmp", "_persistence_loop",
187 f"SnmpStatePersister: State persisted to {self.state_file}", "DEBUG")
188
189 except Exception as e:
190 snmp_logger.error(f"SnmpStatePersister: Persistence loop error: {e}")
191
192 time.sleep(STATE_SYNC_INTERVAL) # Periodically check for updates
