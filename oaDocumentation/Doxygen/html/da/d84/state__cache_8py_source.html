295 def _parse_mqtt_payload(self, message: MqttMessage):
296 """Standardizes MQTT payload extraction into source, value, and metadata."""
297 raw = message.get_json_payload()
298 source, value, metadata = "MQTT", raw, {}
299
300 if isinstance(raw, dict):
301 if "value" in raw or "value" in raw:
302 source = str(raw.get("source", "MQTT")).upper()
303 value = raw.get("value") if "value" in raw else raw.get("value")
304 metadata = raw.copy()
305 for field in ["message_type", "message_guid", "origin_source",
306 "is_settled", "full_id", "boot"]:
307 if field in raw: metadata[field] = raw[field]
308 else:
309 value = raw
310 source = "MQTT"
311 metadata = {}

331 def handle_incoming_mqtt(self, client, userdata, message: MqttMessage) -> None:
332 """
333 Primary MQTT ingestion hook.
334
335 Normalizes topic paths (stripping Cmd/Tx prefixes) and updates the
336 cache based on traffic controller policy.
337
338 Args:
339 message (MqttMessage): Incoming network package.
340 """
341 topic = message.topic
342 base = app_constants.MQTT_BASE_TOPIC
343 normalized = False
344 if f"{base}/Cmd/" in topic:
345 topic = topic.replace(f"{base}/Cmd/", f"{base}/")
346 normalized = True
347 elif f"{base}/Tx/" in topic:
348 topic = topic.replace(f"{base}/Tx/", f"{base}/")
349 normalized = True
350
351 if normalized:
352 message = MqttMessage(topic=topic, payload=message.payload,
353 qos=message.qos, retain=message.retain)
354
355 if self.subscriber_router:
356 self.subscriber_router._on_message(client, userdata, message)
357
358 # ⚡ EXCLUSION: Skip state ingestion for high-bandwidth ST2138 traffic.
359 if topic.startswith("st2138/"):
360 return
361
362 try:
363 source, value, metadata, raw_payload = self._parse_mqtt_payload(message)
364
365 # ⚡ V3.1.25 GLOBAL PURGE COMMAND
366 if topic.endswith("/System/Control/ClearCache"):
367 if str(value).lower() in ["true", "1"]:
368 if app_constants.ROUTER_INGEST_LOGS:
369 matrix_log("core", "data", "handle_incoming_mqtt", "📡🗑️ [CACHE] Received global clear cache command over MQTT. Wiping memory.", "WARNING")
370 self.clear_all_state()
371 return
372
373 # ⚡ V3.1.16 REFLECTION DETECTION
374 # Identify if this message was authored by the local instance.
375 message_src_id = metadata.get("src") or metadata.get("full_id")
376 is_reflection = (message_src_id == app_constants.FULL_INSTANCE_ID)
377
378 if is_reflection:
379 metadata["is_reflection"] = True
380 if app_constants.ROUTER_INGEST_LOGS:
381 matrix_log("core", "data", "handle_incoming_mqtt", f"🛡️ [ECHO] Reflection detected for {topic}. Ingesting for visibility.", "TRACE")
382
383 from oaComBroker.Core.protocol_router.manager import ProtocolRouter
384 ProtocolRouter.get_instance().ingest("MQTT", topic, value, metadata)
385 self.observers.notify(topic, raw_payload)
386
387 # --- CACHE POLICY ---
388 # Self-reflections are NOT committed to the local cache because we
389 # already have the definitive state in memory (or will soon).
390 if is_reflection:
391 return
392
393 should_process, new_payload = cache_traffic_controller.process_traffic(
394 message, self.rust_cache)
395 if should_process:
396 if self.set_value(topic, new_payload):
397 self.search_engine.add_topic(topic)
398 self.save_engine.schedule_save(topic, new_payload)
399 self._updates_since_last_log += 1
400
401 except Exception as e:
402 import traceback
403 error_message = f"🔥 Error handling MQTT for {topic}: {e}\n" \
404 f"{traceback.format_exc()}"
