207 def _update_log(self, topic, payload):
208 """Performs the actual GUI update for the log tree."""
209 # Filter out visibility topics
210 if "visibility" in topic:
211 return
212
213 # Parse Topic: OPEN-AIR/Device Type/YAK/Model/YAK/Action/COMMAND
214 parts = topic.split('/')
215
216 device_type = parts[1] if len(parts) > 1 else "-"
217 # Index 2 is 'YAK' -> Skip
218 model = parts[3] if len(parts) > 3 else "-"
219 yak = parts[4] if len(parts) > 4 else "-"
220 action = parts[5] if len(parts) > 5 else "-"
221 command = "/".join(parts[6:]) if len(parts) > 6 else "-"
222
223 # ⚡ EXTRACT VALUE
224 val_display = "-"
225 tags = ()
226 try:
227 data = orjson.loads(payload)
228 if isinstance(data, dict):
229 if "value" in data:
230 val_display = str(data["value"])
231 tags = ("green_row")
232 elif "message" in data:
233 val_display = data["message"]
234 tags = ("orange_row")
235 elif "type" in data:
236 val_display = f"[{data['type']}]"
237 elif "message" in payload:
238 tags = ("orange_row")
239 except Exception as e:
240 matrix_log("core", "system", inspect.currentframe().f_code.co_name if "inspect" in globals() else "unknown", f"Payload not JSON or error parsing for topic {topic}: {e}", "DEBUG")
241 if "message" in payload:
242 tags = ("orange_row")
243
244 # Insert at the top
245 self.log_tree.insert("", 0, values=(device_type, model, yak, action, command, val_display, payload), tags=tags)

247 # Optional: Limit buffer size

315 def _populate_dissector(self, parent, data):
316 """Recursively populates the dissector tree."""
317 if isinstance(data, dict):
