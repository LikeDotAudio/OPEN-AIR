73 def _update_status_light(self, message: MqttMessage):
74 """Callback function to update the status light's color based on MQTT messages."""
75 try:
76 topic = message.topic
77 payload = message.payload
78 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"📥📶🔄 [MQTT] Incoming status light update on: {topic}", level="TRACE")
79
80 if isinstance(payload, bytes):
81 data = orjson.loads(payload)
82 elif isinstance(payload, str):
83 data = orjson.loads(payload)
84 else:
85 data = payload
86
87 # Support both 'color' and 'value' keys for status
88 color_val = data.get("color", data.get("value", "red"))
89 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"🔋🔄✨ [STATE] Status light color command: {color_val}", level="DEBUG")
90
91 # Determine final hex color
92 if color_val in ["green", True, 1, "1", "online"]:
93 fill_color = "#00ff00"
94 elif color_val in ["yellow", "warning"]:
95 fill_color = "#ffff00"
96 else:
97 fill_color = "#ff0000"
98

102 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"✨🔄🎨 [SYNC] Updating status light dot to color: {fill_color}", level="TRACE")
103 self.status_canvas.itemconfig(self.status_light_id, fill=fill_color)
