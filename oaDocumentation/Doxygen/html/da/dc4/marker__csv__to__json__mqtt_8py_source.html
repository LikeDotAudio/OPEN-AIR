34# Recursively publishes all key-value pairs of a nested dictionary to MQTT.
35# This function traverses a dictionary structure. For each non-dictionary value,
36# it constructs a full MQTT topic path by consmpte2138ting keys and publishes the value.
37# Inputs:
38# mqtt_util (MqttControllerUtility): The MQTT utility for publishing messages.
39# base_topic (str): The current base topic path for the recursion.
40# data (dict or any): The data (dictionary or value) to be published.
41# Outputs:
42# None.
43def _publish_recursive(mqtt_util, base_topic, data):
44 """
45 A simple recursive function to publish all parts of a nested dictionary.
46 """
47 if isinstance(data, dict):
48 for key, value in data.items():
49 new_topic = f"{base_topic}/{key}"
50 _publish_recursive(mqtt_util, new_topic, value)
51 else:
52 # When we hit a final value, publish it.
53 mqtt_util.publish_message(topic=base_topic, subtopic="", value=str(data), retain=True)
