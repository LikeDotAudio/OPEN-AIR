47 @classmethod
48 def from_config(cls, config, center_x, center_y):
49 style_ovr = config.cosmetics.get("style_overrides", {})
50 bezel_shape = style_ovr.get("bezel_shape", "").lower()
51 bezel_width = int(style_ovr.get("bezel_width", 12))
52
53 base_r = (config.size - config.scale_padding) / 2
54 ha = config.meter_viewable_angle / 2.0
55 sang, eang = config.meter_center_angle + ha, config.meter_center_angle - ha
56
57 return cls(
58 config=config, cx=center_x, cy=center_y,
59 base_r=base_r,
60 arc_r=base_r + (config.arc_radius_offset or 0),
61 tick_r=base_r + (config.tick_radius_offset or 0),
62 lab_r=base_r + (config.label_radius_offset or 0),
63 sang=sang, eang=eang, ext=sang - eang,
64 tl=config.tick_length_override if config.tick_length_override is not None else SCALE_TICK_LENGTH,
65 stl=config.sub_tick_length_override if config.sub_tick_length_override is not None else SCALE_SUB_TICK_LENGTH,
66 nsf=config.needle_length_factor_override if config.needle_length_factor_override is not None else config.needle_scale,
67 style_ovr=style_ovr, bezel_shape=bezel_shape, bezel_width=bezel_width

161 @staticmethod
162 def _get_pivots(ctx, val1, val2) -> list[tuple[float, float, float, bool]]:
163 """Determines pivot points based on meter mode and bezel shape."""
164 cx1 = ctx.cx + ctx.config.pivot_offset_x
165 cy1 = ctx.cy - ctx.config.pivot_offset_y
166 cx2 = ctx.cx + ctx.config.pivot_offset_x_2
167 cy2 = ctx.cy - ctx.config.pivot_offset_y_2
168
169 pivots = [(cx1, cy1, val1, ctx.config.counter_clockwise)]
170 is_stereo = ctx.bezel_shape in ["stereo_diamond", "intersecting_overlay"] or ctx.config.meter_mode == "stereo"
171
172 if is_stereo:
173 ccw2 = not ctx.config.counter_clockwise if ctx.bezel_shape == "stereo_diamond" else ctx.config.counter_clockwise
174 pivots.append((cx2, cy2, val2, ccw2))
175 return pivots

202 @staticmethod
203 def _draw_static_chassis(canvas, ctx):
204 """Draws the bezel/chassis and glass effects."""
205 config = ctx.config
206 cw, ch = int(canvas.cget("width")), int(canvas.cget("height"))
207 R, g_y, sk = BezelGeometry.get_scaling_params(cw, ch, ctx.bezel_shape, ctx.bezel_width)
208
209 # Draw background fill behind bezel if not transparent
210 if sk != "super_gem" and not (hasattr(canvas, 'panel_bg_image') and canvas.panel_bg_image) and not config.is_transparent:
211 from .constants import GEM_BASE_HEIGHT, GEM_BEZEL_EXPANSION, GEM_PEAK_HEIGHT
212 if sk == "gem": by = ctx.cy - ((GEM_BASE_HEIGHT * R * GEM_BEZEL_EXPANSION) + g_y)
213 elif sk == "super_gem": by = ctx.cy - (-(GEM_PEAK_HEIGHT * R * GEM_BEZEL_EXPANSION) + g_y)
214 elif sk == "octagon": by = ctx.cy - ((-0.923 * R * 1.4) + g_y)
215 else: by = ctx.cy - g_y
216
217 bg = canvas.cget("bg")
218 canvas.create_rectangle(0, by+1, cw, ch, fill=bg, outline=bg, tags="vu_static")
219
220 if config.cosmetics:
221 MeterModifier.draw_glass_layer(canvas, ctx.cx, ctx.cy, cw, ch, config.cosmetics)
222 MeterModifier.draw_foreground_overlay(canvas, ctx.cx, ctx.cy, cw, ch, config.cosmetics)
