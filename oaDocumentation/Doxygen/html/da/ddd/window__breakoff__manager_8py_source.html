23 def _setup_breakoff_monitoring(self):
24 """Initializes state and binds mapping events to trigger detection."""
25 if not hasattr(self, "state_mirror_engine") or not self.state_mirror_engine:
26 return
27 if not self.state_mirror_engine.root:
28 return
29
30 # ⚡ OPTIMIZATION: Disable monitoring in interactive editor preview

36 self.toplevel_window = None

47 def _check_breakoff_state(self, event=None):
48 """Determines if the current toplevel differs from the application root."""
49 if not self.winfo_exists():
50 return
51
52 if not hasattr(self, "main_root") or not self.main_root:
53 return
54
55 current_toplevel = self.winfo_toplevel()
56
57 if current_toplevel is not self.main_root:
58 self._handle_broken_off(current_toplevel)
59 else:
60 self._handle_docked()

62 def _handle_broken_off(self, toplevel):
63 """Transitions to broken-off state and binds window lifecycle events."""
64 if not self.is_broken_off:
65 self.is_broken_off = True
66 self.toplevel_window = toplevel
67
68 # Wire up geometry and destruction tracking
69 self.toplevel_window.bind("<Configure>", self._on_breakoff_configure_on_breakoff_configure)
70 self.toplevel_window.bind("<Destroy>", self._on_breakoff_destroy_on_breakoff_destroy)
71
72 self._publish_breakoff_state()
73
74 def _handle_docked(self):
75 """Transitions back to docked state and cleans up window bindings."""
76 if self.is_broken_off:
77 self.is_broken_off = False
78 if self.toplevel_window:
79 try:
80 self.toplevel_window.unbind("<Configure>")
81 self.toplevel_window.unbind("<Destroy>")
82 except tk.TclError:
83 pass # Window already dead
84 self.toplevel_window = None
85 self._publish_breakoff_state()

87 def _on_breakoff_configure(self, event):
88 """Reacts to physical window movement or resizing."""
89 if event.widget == self.toplevel_window:
90 self._publish_breakoff_state()
91
92 def _on_breakoff_destroy(self, event):
93 """Reacts to the Toplevel window being closed."""
94 if self.is_broken_off and event.widget == self.toplevel_window:
95 self.is_broken_off = False
96 self.toplevel_window = None
97 self._publish_breakoff_state()

99 def _publish_breakoff_state(self):
100 """Orchestrates MQTT payload creation and command publishing."""
101 if not is_connected() or not self.state_mirror_engine:
102 return
103
104 payload = self._create_breakoff_payload()
105 self.state_mirror_engine.publish_command(
106 self.breakoff_topic, orjson.dumps(payload).decode()
107 )

109 def _create_breakoff_payload(self):
110 """Calculates current window geometry for the MQTT payload."""
111 geom = {"w": 0, "h": 0, "x": 0, "y": 0}
112
113 if self.is_broken_off and self.toplevel_window:
114 try:
115 geom["w"] = self.toplevel_window.winfo_width()
116 geom["h"] = self.toplevel_window.winfo_height()
117 geom["x"] = self.toplevel_window.winfo_x()
118 geom["y"] = self.toplevel_window.winfo_y()
119 except tk.TclError:
120 self.is_broken_off = False # Fallback if window vanished
121
122 return {
123 "is_broken_off": self.is_broken_off,
124 "width": geom["w"],
125 "height": geom["h"],
126 "x": geom["x"],
127 "y": geom["y"],
128 "timestamp": time.time(),
129 "tab_name": getattr(self, "tab_name", "Unknown"),
130 }
