89 async def _receiver_task(self, client):

103 async def _queue_task(self, client):
104 """Processes outbound publish and subscribe queues."""
105 try:
106 while not self.stop_event.is_set():
107 try:
108 await asyncio.wait_for(self.kick_event.wait(), timeout=WORKER_KICK_TIMEOUT)
109 except TimeoutError: pass
110
111 self.kick_event.clear()
112 if self.stop_event.is_set(): break
113
114 # 1. Subscriptions
115 while True:
116 job = self.queue_manager.get_subscribe_request()
117 if job is None: break
118
119 try:
120 await client.subscribe(job["topic"], qos=job["qos"])
121 except Exception as e:
122 MQTT_LOGGER.error(f"aiomqtt: Subscribe Error: {e}")
123 finally:
124 self.queue_manager.task_done("subscribe")
125 self.queue_manager.remove_pending_subscription(job["topic"])
126
127 # 2. Publications
128 while True:
129 item = self.queue_manager.get_publish_message()
130 if item is None: break
131
132 topic, payload, qos, retain = self._parse_publish_item(item)
133 try:
134 await client.publish(topic, payload=payload, qos=qos, retain=retain)
135 except Exception as e:
136 MQTT_LOGGER.error(f"aiomqtt: Publish Error: {e}")
137 finally:
138 self.queue_manager.task_done("publish")
139
140 except asyncio.CancelledError: pass

142 def _parse_publish_item(self, item):
143 """Standardizes publish items and encodes payloads for aiomqtt."""
144 if isinstance(item, tuple):
145 topic, payload, qos, retain = item
146 elif isinstance(item, MqttMessage):
147 topic, payload, qos, retain = item.topic, item.payload, item.qos, item.retain
148 else:
149 topic = item.get("topic")
150 payload = item.get("payload")
151 qos = item.get("qos", 0)
152 retain = item.get("retain", False)
153
154 # ⚡ PROTOCOL ALIGNMENT: aiomqtt expects str, bytes, or None
155 if isinstance(payload, (dict, list)):
156 import orjson
157 # 🛡️ ROBUSTNESS: Use default=str to handle pathlib.Path and other non-JSON types
158 payload = orjson.dumps(payload, default=str)
159 elif isinstance(payload, bytes):
160 pass
161 elif payload is not None and not isinstance(payload, str):
162 payload = str(payload)
163
164 return topic, payload, qos, retain
