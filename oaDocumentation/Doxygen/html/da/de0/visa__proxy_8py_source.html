88 def _command_processor_worker(self):
89 """Background worker loop that executes queued SCPI commands.
90
91 Returns:
92 - None.
93
94 Side effects and thread-safety:
95 - Continuously polls the command queue until shutdown is signaled.
96 - Executes blocking VISA I/O operations (write/query).
97 - Handles and logs exceptions during command execution to prevent
98 thread death.
99 """
100 while not self.shutdown_flag.is_set():
101 # ⚡ ZERO EXCEPTION: Polling check instead of catching Empty
102 if self.command_queue.empty():
103 time.sleep(0.1)
104 continue
105
106 command_info = self.command_queue.get()
107
108 if command_info is None: # The Poison Pill
109 break
110
111 try:
112 command = command_info["command"]
113 query = command_info["query"]
114 correlation_id = command_info["correlation_id"]
115
116 if query:
117 self.query_safe(command, correlation_id)
118 else:
119 self.write_safe(command)
120 except Exception as e:
121 # ⚡ THREAD SAFETY: Catching all exceptions to prevent worker death
122 self._publish_proxy_error(
123 message=f"Exception during background command execution: {str(e)}",
124 command=command_info.get("command", "Unknown")
125 )
126 finally:
127 self.command_queue.task_done()
