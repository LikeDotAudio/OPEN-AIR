45 def draw_static_elements(self):
46 """Draws the bottom layer (The Color)"""
47 # We draw the bright neon color first. It sits at the very bottom.
48 if self.shape == "round":
49 self.canvas.create_oval(self.x, self.y, self.x + self.width, self.y + self.height,
50 fill=self.bg_color, outline="")

55 def update_physics(self):
56 # Smooth movement logic
57 if self.current_open < self.target_open:
58 self.current_open = min(self.target_open, self.current_open + self.open_speed)
59 elif self.current_open > self.target_open:
60 self.current_open = max(self.target_open, self.current_open - self.close_speed)
61
62 self.update_visuals()
63
64 def set_pressure(self, value):
65 self.target_open = max(0.0, min(1.0, float(value)))

67 def update_visuals(self):
68 # Clear moving parts
69 for item in self.shutter_ids:
70 self.canvas.delete(item)
71 self.shutter_ids = []
72
73 # 1. Calculate Shutter Positions (The "Barn Doors")
74 # Even for the round switch, the mechanism is fundamentally two sliding plates.
75 gap_size = self.width * self.current_open
76 center_x = self.x + self.width / 2
77
78 # Left and Right Shutter coordinates
79 l_x2 = center_x - (gap_size / 2)
80 r_x1 = center_x + (gap_size / 2)
81
82 # Draw Shutters (ALWAYS BLACK) to simulate the void
83 # Note: We make them slightly wider than the button to ensure they cover everything
84 s1 = self.canvas.create_rectangle(self.x - 5, self.y, l_x2, self.y + self.height,
85 fill="black", outline="")
86 s2 = self.canvas.create_rectangle(r_x1, self.y, self.x + self.width + 5, self.y + self.height,
87 fill="black", outline="")
88 self.shutter_ids.extend([s1, s2])
89
90 # 2. Draw the Mask/Housing (The Top Layer)
91 # This covers the ugly corners of the shutters and defines the shape.
92
93 if self.shape == "round":
94 # THE FIX: A Pure Black "Donut"
95 # We draw a very thick outline around the circle.
96 # This creates the black housing and hides the rectangular shutter corners.
97 mask_thickness = WINK_MASK_THICKNESS
98 bezel = self.canvas.create_oval(
99 self.x - mask_thickness/2,
100 self.y - mask_thickness/2,
101 self.x + self.width + mask_thickness/2,
102 self.y + self.height + mask_thickness/2,
103 outline=self.housing_color, # BLACK
104 width=mask_thickness
105 )
106
107 # Optional: A thin subtle grey ring just to show where the black button
108 # ends and the black background begins (if needed).
109 # I added a faint outline here for visibility.
110 trim = self.canvas.create_oval(self.x, self.y, self.x+self.width, self.y+self.height,
111 outline="#333333", width=2)
112
113 self.shutter_ids.extend([bezel, trim])
114
115 elif self.shape == "rect":
116 # For rectangle, we just need top/bottom bars if we want to frame it,
117 # but usually, the shutters form the whole face.
118 # Let's add a thick black frame to represent the housing thickness.
119 frame_thickness = WINK_FRAME_THICKNESS
120 frame = self.canvas.create_rectangle(
121 self.x - frame_thickness, self.y - frame_thickness,
122 self.x + self.width + frame_thickness, self.y + self.height + frame_thickness,
123 outline=self.housing_color, width=frame_thickness
124 )
125 self.shutter_ids.append(frame)
