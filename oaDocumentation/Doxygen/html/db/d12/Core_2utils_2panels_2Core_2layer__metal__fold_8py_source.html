14 def generate_metal_fold(width, height, config):
15 """
16 Generates 3D panel folds (creases) and edge shadows.
17 Now supports 'Segmented Panel' logic: each horizontal fold creates a new visual panel.
18 """
19 layer = Image.new('RGBA', (width, height), (0,0,0,0))
20 draw = ImageDraw.Draw(layer)
21 thickness = int(config.get("width_px", 20))
22
23 creases = config.get("creases", [])
24
25 # 1. Separate and sort creases to define segments
26 h_creases = sorted([float(c.get("position_pct", 0.5)) for c in creases if c.get("orientation", "horizontal") == "horizontal"])
27 # Add boundaries
28 h_bounds = [0.0] + h_creases + [1.0]
29
30 # 2. Draw Panel Effects for each segment
31 for i in range(len(h_bounds) - 1):
32 y_start = int(height * h_bounds[i])
33 y_end = int(height * h_bounds[i+1])
34 h_segment = y_end - y_start
35
36 if h_segment <= 0: continue
37
38 # --- Draw Segment Edges (Highlights and Shadows) ---
39 # This is what makes it look like a 'New Panel'
40 for j in range(thickness - 1, -1, -1):
41 # Quadratic falloff for smooth transitions
42 shadow_alpha = int(160 * math.pow(1.0 - j/thickness, 2))
43 highlight_alpha = int(60 * math.pow(1.0 - j/thickness, 2))
44
45 # Bottom Shadow of THIS segment (Ambient Occlusion)
46 draw.line((0, y_end - 1 - j, width - 1, y_end - 1 - j), fill=(0, 0, 0, shadow_alpha))
47
48 # Top Highlight of THIS segment (Simulated Overhead Light hitting the edge)
49 draw.line((0, y_start + j, width - 1, y_start + j), fill=(255, 255, 255, highlight_alpha))
50
51 # Left Highlight (Shared across full height, but we can segment it)
52 draw.line((j, y_start, j, y_end), fill=(255, 255, 255, highlight_alpha))
53
54 # Right Shadow (Shared across full height)
55 draw.line((width - 1 - j, y_start, width - 1 - j, y_end), fill=(0, 0, 0, shadow_alpha))
56
57 # --- Internal Crease Joint (The 'Gap') ---
58 if i < len(h_bounds) - 2: # Don't draw at the very bottom of the whole image
59 y = y_end
60 # Draw a deeper physical gap
61 # 1. Dark background for the gap
62 draw.line((0, y, width - 1, y), fill=(0, 0, 0, 255), width=2)
63 # 2. Subtle rim highlight on the top of the NEXT panel
64 draw.line((0, y + 1, width - 1, y + 1), fill=(255, 255, 255, 40), width=1)
65
66 # 3. Vertical Creases (Columns - keeping original logic for columns)
67 v_creases = [c for c in creases if c.get("orientation") == "vertical"]
68 for crease in v_creases:
69 pos_pct = float(crease.get("position_pct", 0.5))
70 if pos_pct < 0 or pos_pct > 1.0: continue
71 x = int(width * pos_pct)
72 # Highlight (left)
73 draw.line((x-1, 0, x-1, height - 1), fill=(255, 255, 255, 60), width=1)
74 # Shadow (right)
75 draw.line((x+1, 0, x+1, height - 1), fill=(0, 0, 0, 100), width=2)
76
77 return layer
