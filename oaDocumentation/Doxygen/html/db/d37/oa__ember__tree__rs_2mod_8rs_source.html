76fn parse_tag(input: &[u8]) -> IResult<&[u8], (u32, bool, u8)> {
77 let (input, first_byte) = u8(input)?;
78 let class = (first_byte & 0xC0) >> 6;
79 let is_constructed = (first_byte & 0x20) != 0;
80 let mut tag = (first_byte & 0x1F) as u32;
81
82 if tag == 0x1F {
83 // High-tag-number form
84 let mut input = input;
85 tag = 0;
86 loop {
87 let (next_input, byte) = u8(input)?;
88 input = next_input;
89 tag = (tag << 7) | (byte & 0x7F) as u32;
90 if (byte & 0x80) == 0 {
91 break;
92 }
93 }
94 Ok((input, (tag, is_constructed, class)))
95 } else {
96 Ok((input, (tag, is_constructed, class)))
97 }
98}

100fn parse_length(input: &[u8]) -> IResult<&[u8], usize> {
101 let (input, first_byte) = u8(input)?;
102 if first_byte == 0x80 {
103 // Indefinite length - not fully supported here for simplicity
104 return Err(nom::Err::Error(Error::new(input, nom::error::ErrorKind::Tag)));
105 }
106
107 if (first_byte & 0x80) == 0 {
108 // Short form
109 Ok((input, first_byte as usize))
110 } else {
111 // Long form
112 let n_bytes = (first_byte & 0x7F) as usize;
113 let (mut input, length_bytes) = take(n_bytes)(input)?;
114 let mut length = 0usize;
115 for &b in length_bytes {
116 length = (length << 8) | (b as usize);
117 }
118 Ok((input, length))
119 }
120}

122fn parse_ber_tlv(input: &[u8]) -> IResult<&[u8], BerTlv> {
123 let (input, (tag, is_constructed, class)) = parse_tag(input)?;
124 let (input, length) = parse_length(input)?;
125 let (input, value_bytes) = take(length)(input)?;
126
127 if is_constructed {
128 let mut children = Vec::new();
129 let mut remaining = value_bytes;
130 while !remaining.is_empty() {
131 let (next_remaining, child) = parse_ber_tlv(remaining)?;
132 remaining = next_remaining;
133 children.push(child);
134 }
135 Ok((input, BerTlv {
136 tag,
137 is_constructed,
138 class,
139 value: BerValue::Constructed(children),
140 }))
141 } else {
142 Ok((input, BerTlv {
143 tag,
144 is_constructed,
145 class,
146 value: BerValue::Primitive(value_bytes),
147 }))
148 }
149}
