112 # --- 1. Drop Shadow (The External Cast) ---
113 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, "👻🌀🔳 [LAYER] 1. Casting external drop shadow.", level="TRACE")

122 shadow_draw_context.ellipse((center_coordinate - screw_radius + offset_x, center_coordinate - screw_radius + offset_y,
123 center_coordinate + screw_radius + offset_x, center_coordinate + screw_radius + offset_y),
124 fill=(0, 0, 0, DEFAULT_SHADOW_ALPHA))

133 # --- 2. The Head Geometry ---
134 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"🎨🖌️🔘 [LAYER] 2. Lathe-turning screw head: {base_color_hex}", level="TRACE")

148 if screw_head_type == "fillister":
149 # Domed top: Radial gradient highlight shifted top-left

153 lighting_draw_context.ellipse((center_coordinate - screw_radius + specular_offset, center_coordinate - screw_radius + specular_offset,

156 else:

158 lighting_draw_context.ellipse((center_coordinate - screw_radius, center_coordinate - screw_radius, center_coordinate + screw_radius, center_coordinate + screw_radius), fill=COUNTERSUNK_LIGHT_LEVEL)
159
160 # Apply Lighting to Color using Multiply/Overlay
161 lighting_rgba_image = ImageOps.colorize(lighting_mask_image, black="black", white="white").convert("RGBA")
162 lighting_rgba_image.putalpha(Image.eval(lighting_mask_image, lambda alpha: int(alpha * LIGHTING_STRENGTH_RATIO))) # Adjust strength

175 # --- 3. The Robertson Void (Square Drive) ---
176 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, "🔳📐🕳️ [LAYER] 3. Punching Robertson square drive void.", level="TRACE")

179 drive_layer_image = Image.new('RGBA', (canvas_dimension, canvas_dimension), (0,0,0,0))
180 drive_draw_context = ImageDraw.Draw(drive_layer_image)

182 # Coordinates of square vertices
183 rotation_angle_radians = math.radians(rotation_angle_degrees)
184 half_drive_size = drive_size_pixels / 2

186 # Points unrotated relative to center
187 square_points = [(-half_drive_size, -half_drive_size), (half_drive_size, -half_drive_size), (half_drive_size, half_drive_size), (-half_drive_size, half_drive_size)]
188 rotated_square_points = []
189 for local_x, local_y in square_points:
190 rotated_x = local_x * math.cos(rotation_angle_radians) - local_y * math.sin(rotation_angle_radians) + center_coordinate
191 rotated_y = local_x * math.sin(rotation_angle_radians) + local_y * math.cos(rotation_angle_radians) + center_coordinate
192 rotated_square_points.append((rotated_x, rotated_y))

194 # Draw the Void (Dark Hole)
195 drive_draw_context.polygon(rotated_square_points, fill=VOID_COLOR_RGBA)
196
197 # Internal Shadows simulation
198 point0, point1, point2, point3 = rotated_square_points
199
200 # "North" Wall: Illuminated
201 drive_draw_context.line((point0, point1), fill=NORTH_WALL_ILLUMINATION, width=2)
202 # "West" Wall: Illuminated
203 drive_draw_context.line((point3, point0), fill=WEST_WALL_ILLUMINATION, width=2)
204 # "East" Wall: Shadow
205 drive_draw_context.line((point1, point2), fill=EAST_WALL_SHADOW, width=2)
206 # "South" Wall: Deep Shadow
207 drive_draw_context.line((point2, point3), fill=SOUTH_WALL_DEEP_SHADOW, width=2)

209 screw_image = Image.alpha_composite(screw_image, drive_layer_image)

211 # --- 4. Damage & Wear (Cam-out & Scratches) ---
212 if damage_intensity > 0:
213 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"🗡️🎨✨ [LAYER] 4. Applying screwdriver slippage wear (Int: {damage_intensity})", level="TRACE")
214 scratch_layer_image = Image.new('RGBA', (canvas_dimension, canvas_dimension), (0,0,0,0))
215 scratch_draw_context = ImageDraw.Draw(scratch_layer_image)
216
217 # Scratches radiating from the drive (Slippage)
218 num_slippage_scratches = int(SCRATCH_COUNT_BASE * damage_intensity) + 1
219 for _ in range(num_slippage_scratches):
220 scratch_angle_degrees = random.uniform(0, S_ANGLE_MAX)
221 start_distance = drive_size_pixels * START_DISTANCE_RATIO
222 end_distance = screw_radius * END_DISTANCE_RATIO
223
224 start_x = center_coordinate + math.cos(math.radians(scratch_angle_degrees)) * start_distance
225 start_y = center_coordinate + math.sin(math.radians(scratch_angle_degrees)) * start_distance
226 end_x = center_coordinate + math.cos(math.radians(scratch_angle_degrees)) * end_distance
227 end_y = center_coordinate + math.sin(math.radians(scratch_angle_degrees)) * end_distance
228
229 # Metal groove highlight
230 scratch_draw_context.line((start_x, start_y, end_x, end_y), fill=SCRATCH_SILVER_EXPOSURE, width=1)
231
232 screw_image = Image.alpha_composite(screw_image, scratch_layer_image)
233
234 # --- 5. Rust (Accumulation) ---
235 if rust_intensity > 0:
236 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"🟠🎨✨ [LAYER] 5. Accumulating iron oxide rust (Int: {rust_intensity})", level="TRACE")
237
238 # Noise texture for rust
239 noise_dimension = int(drive_size_pixels)
240 rust_noise_texture = Image.effect_noise((noise_dimension, noise_dimension), sigma=RUST_NOISE_SIGMA).resize((int(drive_size_pixels * RUST_RESIZE_RATIO), int(drive_size_pixels * RUST_RESIZE_RATIO)))
241 rust_noise_texture = rust_noise_texture.convert("RGBA")
242
243 # Colorize rust noise (Orange/Brown)
244 rust_alpha_level = int(MAX_ALPHA_VALUE * rust_intensity)
245 rust_color_layer = Image.new("RGBA", rust_noise_texture.size, RUST_BASE_COLOR_RGB + (rust_alpha_level,))
246 rust_composite_texture = ImageChops.multiply(rust_noise_texture, rust_color_layer)
247
248 # Paste over drive center
249 rust_offset_coordinate = int(center_coordinate - drive_size_pixels * RUST_OFFSET_RATIO)
250 screw_image.paste(rust_composite_texture, (rust_offset_coordinate, rust_offset_coordinate), rust_composite_texture)
251
252 # --- 6. Save to Cache ---
253 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, "🎨🆗💾 [SUCCESS] Procedural screw generation complete. Saving to cache.", level="SUCCESS")
254 CacheImageStore.save_to_cache("screw", size_pixels, size_pixels, configuration_data, screw_image)
