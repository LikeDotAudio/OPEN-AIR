13 @staticmethod
14 def _mdp_get_fader_at(x, y, frame):
15 # Access the plotter's canvas
16 canvas = frame.winfo_children()[0].canvas.get_tk_widget()
17 item_id = canvas.find_closest(x, y, halo=5)
18 if not item_id: return None
19 tags = canvas.gettags(item_id[0])
20 for tag in tags:
21 if tag.startswith("mdp_ltp_"): return frame.faders[0]
22 return None

32 @staticmethod
33 def _mdp_on_click(event, frame):
34 f = MDPInteractionMixin._mdp_get_fader_at(event.x, event.y, frame)
35 if f:
36 frame.active_fader = f; f.lift(); f.dragging = True
37 f.start_x, f.start_y = event.x, event.y
38 try: f.start_val, f.start_rot = float(f.linear_var.get()), float(f.rotation_var.get())
39 except: f.start_val, f.start_rot = f.val_min, f.rot_min

41 @staticmethod
42 def _mdp_on_drag(event, frame):
43 f = frame.active_fader
44 if f and f.dragging:
45 dx, dy = event.x - f.start_x, event.y - f.start_y
46 ldx, ldy = MDPMath.to_local_space(dx, dy, f.angle)
47
48 # Linear (Local Y)
49 dv = -(ldy / f.track_len) * (f.val_max - f.val_min)
50 f.linear_var.set(max(f.val_min, min(f.val_max, f.start_val + dv)))
51
52 # Rotary (Local X)
53 f.rotation_var.set(max(f.rot_min, min(f.rot_max, f.start_rot + ldx)))
54
55 @staticmethod
56 def _mdp_on_mid_click(event, frame):
57 f = MDPInteractionMixin._mdp_get_fader_at(event.x, event.y, frame)
58 if f: frame.active_fader = f; f.lift(); f.dragging = True; f.start_x, f.start_y = event.x, event.y; f.start_pos = (f.x, f.y)

60 @staticmethod
61 def _mdp_on_mid_drag(event, frame):
62 f = frame.active_fader
63 if f and f.dragging: f.x, f.y = f.start_pos[0] + (event.x - f.start_x), f.start_pos[1] + (event.y - f.start_y); f.render()

65 @staticmethod
66 def _mdp_on_release(event, frame):
67 if frame.active_fader: frame.active_fader.dragging = False; frame.active_fader = None

69 @staticmethod
70 def _mdp_on_scroll(event, frame):
71 f = MDPInteractionMixin._mdp_get_fader_at(event.x, event.y, frame)
72 if f:
73 delta = 1 if (event.num == 4 or (hasattr(event, "delta") and event.delta > 0)) else -1
74 if event.state & 0x0008: f.angle += delta * 3; f.render()
75 else:
76 curr = float(f.rotation_var.get())
77 f.rotation_var.set(max(f.rot_min, min(f.rot_max, curr + delta * 3)))
