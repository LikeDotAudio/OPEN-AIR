186 function filterLogs(containerId) {{
187 const container = document.getElementById(containerId);
188 const searchTerm = container.querySelector('.log-search').value.toLowerCase();
189 const levelFilter = container.querySelector('.filter-level').value.toLowerCase();
190 const systemFilter = container.querySelector('.filter-system').value.toLowerCase();
191 const elementFilter = container.querySelector('.filter-element') ? container.querySelector('.filter-element').value.toLowerCase() : "";
192 const moduleFilter = container.querySelector('.filter-module') ? container.querySelector('.filter-module').value.toLowerCase() : "";
193
194 const logEntries = container.querySelectorAll('.log-entry');
195 logEntries.forEach(entry => {{
196 const lines = entry.querySelectorAll('.log-line, .log-line-raw');
197 lines.forEach(line => {{
198 let show = true;
199 const text = line.textContent.toLowerCase();
200
201 if (searchTerm && !text.includes(searchTerm)) show = false;
202
203 if (show && levelFilter) {{
204 const levelCol = line.querySelector('.log-type');
205 if (levelCol && !levelCol.textContent.toLowerCase().includes(levelFilter)) show = false;
206 else if (!levelCol) show = false;
207 }}
208
209 if (show && systemFilter) {{
210 const systemCol = line.querySelector('.log-system');
211 if (systemCol && !systemCol.textContent.toLowerCase().includes(systemFilter)) show = false;
212 else if (!systemCol) show = false;
213 }}
214
215 if (show && elementFilter) {{
216 const elementCol = line.querySelector('.log-element');
217 if (elementCol && !elementCol.textContent.toLowerCase().includes(elementFilter)) show = false;
218 else if (!elementCol) show = false;
219 }}
220
221 if (show && moduleFilter) {{
222 const moduleCol = line.querySelector('.log-module');
223 if (moduleCol && !moduleCol.textContent.toLowerCase().includes(moduleFilter)) show = false;
224 else if (!moduleCol) show = false;
225 }}
226
227 line.style.display = show ? 'grid' : 'none'; // Use 'grid' for aligned lines
228 }});
229 }});
230 }}
