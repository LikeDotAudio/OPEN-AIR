63 def calculate_rotated_point(self, px, py, cx, cy, angle_deg):
64 rad = math.radians(angle_deg)
65 cos_a, sin_a = math.cos(rad), math.sin(rad)
66 new_x = cos_a * (px - cx) - sin_a * (py - cy) + cx
67 new_y = sin_a * (px - cx) + cos_a * (py - cy) + cy
68 return new_x, new_y

70 def render_visuals(self):
71 self.canvas.delete(self.tag_root)
72 center_x, center_y, angle, track_length = self.x, self.y, self.angle, self.track_len

74 # Hitbox
75 hitbox_width = HITBOX_WIDTH
76 hitbox_points = [
77 self.calculate_rotated_point(center_x - hitbox_width/2, center_y - track_length/2 - HITBOX_PADDING, center_x, center_y, angle),
78 self.calculate_rotated_point(center_x + hitbox_width/2, center_y - track_length/2 - HITBOX_PADDING, center_x, center_y, angle),

85 # Track
86 track_start = self.calculate_rotated_point(center_x, center_y - track_length/2, center_x, center_y, angle)

91 # Ticks
92 for i in range(TICK_COUNT):
93 local_y = (center_y + track_length/2) - (track_length * (i/TICK_DIVISOR))
94 tick_length = TICK_MAJOR_LENGTH if i % TICK_MAJOR_INTERVAL == 0 else TICK_MINOR_LENGTH
95 tick_p1, tick_p2 = self.calculate_rotated_point(center_x-TICK_OFFSET, local_y, center_x, center_y, angle), self.calculate_rotated_point(center_x-TICK_OFFSET-tick_length, local_y, center_x, center_y, angle)
96 self.canvas.create_line(tick_p1, tick_p2, fill="#666666", tags=self.tag_root)
97 tick_p3, tick_p4 = self.calculate_rotated_point(center_x+TICK_OFFSET, local_y, center_x, center_y, angle), self.calculate_rotated_point(center_x+TICK_OFFSET+tick_length, local_y, center_x, center_y, angle)
98 self.canvas.create_line(tick_p3, tick_p4, fill="#666666", tags=self.tag_root)
99
100 # Cap
101 norm = (self.val_current - self.val_min) / (self.val_max - self.val_min)
102 local_cap_y = (center_y + track_length/2) - (norm * track_length)
103 cap_center_x, cap_center_y = self.calculate_rotated_point(center_x, local_cap_y, center_x, center_y, angle)
104 radius = CAP_RADIUS
105
106 # Glow Effect
107 outline_col = self.cap_outline_hover if self.hovered else self.cap_outline_normal
108 outline_w = 3 if self.hovered else 2
109
110 self.canvas.create_oval(cap_center_x-radius, cap_center_y-radius, cap_center_x+radius, cap_center_y+radius, fill=self.cap_color, outline=outline_col, width=outline_w, tags=(self.tag_root, "cap"))
111
112 prad = math.radians(90 - self.rot_current - angle)
113 pointer_x, pointer_y = cap_center_x + (radius-POINTER_OFFSET)*math.cos(prad), cap_center_y - (radius-POINTER_OFFSET)*math.sin(prad)
114 self.canvas.create_line(cap_center_x, cap_center_y, pointer_x, pointer_y, fill=self.highlight_color, width=3, capstyle=tk.ROUND, tags=self.tag_root)
115 self.canvas.create_oval(cap_center_x-3, cap_center_y-3, cap_center_x+3, cap_center_y+3, fill=self.highlight_color, outline="", tags=self.tag_root)
116
117 self.canvas.create_text(cap_center_x, cap_center_y-INDICATOR_OFFSET, text=f"{self.val_current:.1f}", fill="white", font=("Arial", 8), tags=self.tag_root)

178 def get_fader_at(self, x, y):
179 item_id = self.canvas.find_closest(x, y, halo=5)
180 if not item_id: return None
181 tags = self.canvas.gettags(item_id[0])
182 for tag in tags:
183 if tag.startswith("fader_"):
184 try:
185 return self.faders[int(tag.split("_")[1])]
186 except:
187 pass

206 def on_drag(self, event):
207 fader = self.active_fader
208 if fader and fader.dragging:
209 dx, dy = event.x - fader.start_x, event.y - fader.start_y
210 rad = math.radians(fader.angle)
211 ldx = dx * math.cos(-rad) - dy * math.sin(-rad)
212 ldy = dx * math.sin(-rad) + dy * math.cos(-rad)
213
214 # Dual-Axis Control
215 # Vertical (ldy) -> Linear
216 # Horizontal (ldx) -> Rotary
217
218 # Linear
219 dv = -(ldy / fader.track_len) * (fader.val_max - fader.val_min)
220 fader.val_current = max(fader.val_min, min(fader.val_max, fader.start_val + dv))
221
222 # Rotary (Sensitivity: 1 pixel = 1 degree roughly? Scale it)
223 rot_sens = 1.0
224 fader.rot_current = max(fader.rot_min, min(fader.rot_max, fader.start_rot + (ldx * rot_sens)))
225
226 fader.render()
227 self.update_table(fader.widget_id, fader.val_current, fader.rot_current, fader.angle, fader.x, fader.y)

251 def on_scroll(self, event):
252 fader = self.get_fader_at(event.x, event.y)
253 if fader:
254 delta = 0
255 if event.num == 4: delta = 1
256 elif event.num == 5: delta = -1
257 elif hasattr(event, "delta"): delta = 1 if event.delta > 0 else -1
258
259 if event.state & 0x0008: # Alt + Scroll -> Rotate Widget
260 fader.angle += delta * 3
261 fader.render()
262 else: # Knob Value
263 fader.rot_current = max(fader.rot_min, min(fader.rot_max, fader.rot_current + delta * 3))
264 fader.render()
265 self.update_table(fader.widget_id, fader.val_current, fader.rot_current, fader.angle, fader.x, fader.y)
