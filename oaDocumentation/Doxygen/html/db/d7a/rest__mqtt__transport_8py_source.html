67 def connect(self, connection_params: dict[str, Any]) -> bool:
68 host = connection_params.get("destination_host", "localhost")
69 port = connection_params.get("destination_port", 1883)
70 username = connection_params.get("username")

73 # ⚡ UNIQUE IDENTITY: Prevent Client ID collisions in multi-partition environments
74 partition_id = os.environ.get("OPEN_AIR_PARTITION_ID", "SUP")
75 random_suffix = f"{random.getrandbits(16):04x}"
76 default_client_id = f"OPENAIR-{partition_id}-REST-{os.getpid()}-{random_suffix}"
77 client_id = connection_params.get("client_id", default_client_id)
78
79 matrix_log("comms", "rest_mqtt", "connect", f"📡📥 [REST-MQTT] Connecting to {host}:{port} as {client_id}.", "INFO")
80
81 # Support both Paho v1.x and v2.x
82 if hasattr(mqtt, 'CallbackVersion'):
83 self.client = mqtt.Client(callback_api_version=mqtt.CallbackVersion.VERSION2, client_id=client_id)
84 else:
85 self.client = mqtt.Client(client_id=client_id)
86
87 self.client.on_connect = self._on_connect
88 self.client.on_message = self._on_message
89 self.client.on_disconnect = self._on_disconnect

92 self.client.tls_set(tls_version=ssl.PROTOCOL_TLS)
93 if username and password:

96 try:
97 self.client.connect_async(host, port, 60)

100 import time
