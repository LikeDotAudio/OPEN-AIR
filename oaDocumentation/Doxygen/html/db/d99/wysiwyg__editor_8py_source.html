116 def new_workspace(self):
117 """Forces the user to Save As, then creates an empty canvas."""
118 matrix_log("ui", "gui_builder", "new", "🖱️🖱️🖱️ [ACTION] WysiwygEditor: NEW WORKSPACE triggered.", "INFO")
119
120 # 1. Force Save As (User MUST pick a file name)
121 if FileWriter.save_as(on_save_callback=self.on_save):
122 # 2. Reset State to empty structure
123 state_manager.initialize({}, state_manager.get_file_path())
124
125 # 3. Update UI
126 new_path = state_manager.get_file_path()
127 if not self.is_standalone:
128 self.window.title(f"WYSIWYG Editor - {new_path.name if new_path else 'Unsaved'}")
129 self.status_bar.set_status(f"New Workspace: {new_path.name if new_path else 'Unknown'}", "#00ffcc")
130
131 if hasattr(self, 'layout_view'):
132 self.layout_view._manual_rebuild()
133 return True
134
135 # User cancelled Save As or it failed
136 self.status_bar.set_status("NEW WORKSPACE CANCELLED - Save required.", "#ff8800")
137 return False
