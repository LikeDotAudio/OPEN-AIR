115 def new_workspace(self):
116 """Forces the user to Save As, then creates an empty canvas."""
117 matrix_log("ui", "gui_builder", "new", "🖱️🖱️🖱️ [ACTION] WysiwygEditor: NEW WORKSPACE triggered.", "INFO")
118
119 # 1. Force Save As (User MUST pick a file name)
120 if FileWriter.save_as(on_save_callback=self.on_save):
121 # 2. Reset State to empty structure
122 state_manager.initialize({}, state_manager.get_file_path())
123
124 # 3. Update UI
125 new_path = state_manager.get_file_path()
126 if not self.is_standalone:
127 self.window.title(f"WYSIWYG Editor - {new_path.name if new_path else 'Unsaved'}")
128 self.status_bar.set_status(f"New Workspace: {new_path.name if new_path else 'Unknown'}", "#00ffcc")
129
130 if hasattr(self, 'layout_view'):
131 self.layout_view._manual_rebuild()
132 return True
133
134 # User cancelled Save As or it failed
135 self.status_bar.set_status("NEW WORKSPACE CANCELLED - Save required.", "#ff8800")
136 return False
