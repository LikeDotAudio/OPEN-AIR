51def _set_debug_state(project_root, target_state: bool, console_print_func=None):
52 """
53 Scans the codebase and forces all debug flags to the specified state.
54 Utilizes Rust-native acceleration if available.
55 """
56 def log(message):
57 if console_print_func: console_print_func(message)
58 else: logger.info(message)
59
60 target_state_str = str(target_state)
61 action = "ENABLING" if target_state else "DISABLING"
62 log(f"🛠️ [CONFIG] Forcing all debug gates to {target_state_str.upper()}...")
63
64 if RUST_ENABLED:
65 try:
66 # Call the Rust implementation
67 modified = toggle_debug_flags_rs_native(str(project_root), target_state)
68 if modified:
69 log(f"🚀 [DEPLOY] {action} complete (Native Rust accelerated).")
70 return True
71 log("🛌 [SLEEPING] No state change needed (Native Rust accelerated).")
72 return False
73 except Exception as e:
74 logger.error(f"⚠️ [CONFIG] Rust execution failed: {e}. Falling back to Python...")

76 # --- Python Fallback Implementation ---
77 # Pattern captures: LOCAL_DEBUG, BUILDER_DEBUG, or generic DEBUG assignments
78 pattern = re.compile(
79 r'^(\s*(?:LOCAL_|BUILDER_)?[A-Z_]*DEBUG\s*=\s*)(True|False)(.*)$',
80 re.MULTILINE
81 )

83 py_files = []
84 for root, dirs, files in os.walk(project_root):
85 # Respect project boundaries and ignore hidden/temporary directories
86 dirs[:] = [d for d in dirs if not d.startswith('.') and
87 d not in ['venv', 'node_modules', '__pycache__', 'oaDataLogs']]
88 for file in files:
89 if file.endswith('.py'):
90 py_files.append(Path(root) / file)
91
92 files_modified = 0
93 flags_changed = 0
94
95 for file_path in py_files:
96 try:
97 with open(file_path, encoding='utf-8') as f:
98 content = f.read()
99
100 if not pattern.search(content):
101 continue
102
103 replacements_made = 0
104 def _debug_mode_wrapper(match):
105 nonlocal replacements_made
106 current_value = match.group(2)
107 # Only replace if the state is different to avoid disk churn
108 if current_value != target_state_str:
109 replacements_made += 1
110 return f"{match.group(1)}{target_state_str}{match.group(3)}"
111 return match.group(0)
112
113 new_content = pattern.sub(_debug_mode_wrapper, content)
114
115 if replacements_made > 0:

117 f.write(new_content)
118 files_modified += 1
119 flags_changed += replacements_made
120 except Exception as e:
121 logger.error(f"⚠️ [CONFIG] Failed to process {file_path.name}: {e}")

123 if files_modified > 0:

125 f"Changed {flags_changed} flags across {files_modified} files.")
126 return True

128 logger.info(f"🛌 [SLEEPING] No state change needed. "
129 f"All flags already {target_state_str.upper()}.")

140def toggle_debug_flags(project_root, console_print_func=None):
141 """
142 Toggles all debug flags based on the state of the first flag encountered.
143
144 This implements a 'Flip-Flop' logic: if the first file scanned has
145 debug enabled, the entire project is set to disabled, and vice-versa.
146 """
147 pattern = re.compile(r'^\s*(?:LOCAL_|BUILDER_)?[A-Z_]*DEBUG\s*=\s*(True|False)',
148 re.MULTILINE)
149
150 current_state_found = None
151 for root, dirs, files in os.walk(project_root):
152 dirs[:] = [d for d in dirs if not d.startswith('.') and
153 d not in ['venv', 'node_modules', '__pycache__']]
154 for file in files:
155 if file.endswith('.py'):
156 try:
157 with open(Path(root) / file, encoding='utf-8') as f:
158 content = f.read()
159 match = pattern.search(content)
160 if match:
161 current_state_found = (match.group(1) == "True")
162 break
163 except Exception:
164 continue
165 if current_state_found is not None:
166 break
167
168 if current_state_found is None:
169 return force_debug_on(project_root, console_print_func)
170
171 return _set_debug_state(project_root, not current_state_found, console_print_func)
