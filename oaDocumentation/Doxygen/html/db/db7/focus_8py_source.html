184 def _normalize_path(self, path):
185 """Standardizes the path string based on the current state's root keys."""
186 if path is None: return None

188 full_state = state_manager.get_state()
189 if not full_state:
190 return path
191
192 # If path is already valid, return as is

206 def _apply_array_redirection(self, path):
207 """Redirects focus from array elements to their blueprint for consistent editing."""
208 parts = str(path).split(".")
209 for i in range(len(parts)):
210 sub_path = ".".join(parts[:i+1])
211 value = state_manager.get_value_at_path(sub_path)
212
213 if isinstance(value, dict) and value.get("type") == "OcaArray":
214 # Redirect if path is deep within array fields
215 if len(parts) > i + 3 and parts[i+1] == "fields" and parts[i+3] == "fields":
216 return f"{sub_path}.blueprint.{'.'.join(parts[i+3:])}"
217 # Redirect if path points to the direct field list
218 elif len(parts) > i + 1 and parts[i+1] == "fields":
219 return f"{sub_path}.blueprint"
220 return path
