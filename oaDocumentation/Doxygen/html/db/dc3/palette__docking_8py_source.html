21 def undock_panel(self, panel_name):

24 return

31 def _focus_existing_detached(self, panel_name):
32 """Brings an already detached window to the front."""
33 if panel_name in self.detached_windows and self.detached_windows[panel_name].winfo_exists():

39 def _create_detached_window(self, panel_name, panel_widget):

58 def _handle_detached_close(self, panel_name, top_level, panel_widget):
59 """Cleans up state when a detached window is closed by the user."""
60 matrix_log("ui", "palette_manager", "on_toplevel_close", f"📱🎨🛌 [MOBILE] PaletteManager: Detached window for {panel_name} closed.", "INFO")
61 panel_widget.pack_forget()
