122 cols = self.data_manager.discover_columns()

143 def _toggle_all(self, state):
144 stack = [c for c in self.tree.get_children("")]
145 while stack:
146 item = stack.pop()
147 self.tree.item(item, open=state)
148 stack.extend(self.tree.get_children(item))
