94 def _on_geometry_change(self, widget, event):
95 meta = self._tracked_widgets.get(widget)

104 def _perform_geometry_publish(self, widget):
105 meta = self._tracked_widgets.get(widget)
106 if not meta: return
107 meta["geo_timer"] = None
108
109 try:
110 toplevel = widget.winfo_toplevel()
111 w, h, x, y = toplevel.winfo_width(), toplevel.winfo_height(), toplevel.winfo_x(), toplevel.winfo_y()
112
113 if not is_connected(): return
114
115 payload = {
116 "width": w, "height": h, "x": x, "y": y,
117 "timestamp": time.time(),
118 "tab_name": meta["tab_name"],
119 }
120 meta["engine"].publish_command(meta["geo_topic"], orjson.dumps(payload).decode())
121 matrix_log("UI", "GUI_MANAGER", inspect.currentframe().f_code.co_name, f"📏 UITrackingService: Published geometry for {meta['tab_name']}", level="DEBUG")
122 except Exception:
123 pass
