12fn format_clock_id(raw: &[u8]) -> String {
13 if raw.len() >= 10 {
14 let hex_part = format!("{:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}:{:02x}",

40fn parse_packet(py: Python, payload: &[u8], src_ip: String, dst_ip: String, udp_port: u16) -> PyResult<Py<PyAny>> {
41 let dict = PyDict::new_bound(py);
42
43 let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap().as_secs_f64();
44 dict.set_item("timestamp", now)?;
45 dict.set_item("source_ip", src_ip)?;
46 dict.set_item("dest_ip", dst_ip)?;
47 dict.set_item("udp_port", udp_port)?;
48
49 if payload.len() >= 34 {
50 let m_id = payload[0] & 0x0F;
51 let m_type = get_message_type(m_id);
52 let domain = payload[4];
53 let seq_id = u16::from_be_bytes([payload[30], payload[31]]);
54 let port_id = &payload[20..30];
55
56 let m_type_str = if m_type == "Unknown" {
57 format!("Unknown ({})", m_id)
58 } else {
59 m_type.to_string()
60 };
61
62 dict.set_item("message_type", m_type_str)?;
63 dict.set_item("domain", domain)?;
64 dict.set_item("sequence_id", seq_id)?;
65 dict.set_item("clock_identity", format_clock_id(port_id))?;
66 } else {
67 dict.set_item("message_type", "Unknown")?;
68 dict.set_item("domain", 0)?;
69 dict.set_item("sequence_id", 0)?;
70 dict.set_item("clock_identity", "Unknown")?;
71 }
72
73 Ok(dict.into_any().unbind())
74}
