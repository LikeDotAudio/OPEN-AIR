81def build_source(source_id, device_id, sdp_data):
82 """

84

89
90 Returns:
91 dict: The NMOS Source resource payload.
92 """
93 name = sdp_data.get("name", f"SAP Source {source_id[:4]}")

95

97 "id": source_id,
98 "version": now_ts(),
99 "label": name,
100 "description": "SAP discovered source",
101 "tags": {},
102 "device_id": device_id,
103 "format": "urn:x-nmos:format:audio",
104 "grain_rate": {"numerator": 25, "denominator": 1}, # Assuming a common grain rate
105 "clock_name": "clk0", # Default clock
106 "channels": [{"label": f"Ch{i}"} for i in range(ch)],
107 "parents": [],
108 "caps": {}
