102 def _traverse_and_build(self, node, current_path):
103 """
104 Recursively explores the JSON tree searching for "Table" markers.
105
106 Parameters:
107 - node: The current dictionary or list in the JSON structure.
108 - current_path: A list of strings representing the breadcrumb path.
109
110 Returns:
111 - None.
112 """
113 if not isinstance(node, dict):
114 return
115
116 for key, value in node.items():
117 new_path = current_path + [key]
118
119 # In our schema, a "Table" key indicates a collection of records.
120 if key == "Table" and isinstance(value, dict):
121 self._write_table_to_csv(value, new_path)
122 else:
123 self._traverse_and_build(value, new_path)

125 def _write_table_to_csv(self, table_node, table_path):
126 """
127 Extracts data from a table node and writes it to a specific CSV file.
128
129 Parameters:
130 - table_node: The dictionary containing the "data" map.
131 - table_path: The hierarchical path used to name the output file.
132
133 Returns:
134 - None.
135
136 Side Effects & Thread-Safety:
137 - Writes a new CSV file to the `csv_dir`.
138 """
139 data_dict = table_node.get("data")
140
141 if not isinstance(data_dict, dict) or not data_dict:
142 if LOCAL_DEBUG:
143 logger.debug(f"Warning: Skipping path {'/'.join(table_path)} "

145 return

147 data_list = list(data_dict.values())
148
149 # Dynamically determine headers from the first data row.
150 # Assumes all rows in the same table share the same schema.
151 headers = list(data_list[0].keys())
152
153 # Sanitize the full topic path to create a safe filesystem filename.
154 topic_string = "/".join(table_path)
155 sanitized_filename = (re.sub(r"[^a-zA-Z0-9_-]", "_", topic_string)
