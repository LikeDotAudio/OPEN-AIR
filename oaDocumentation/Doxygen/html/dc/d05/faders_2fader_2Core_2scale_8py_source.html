70 @staticmethod
71 def _calculate_smart_interval(frame, value_range):
72 """Calculate a human-friendly tick interval."""
73 if hasattr(frame, "tick_interval") and frame.tick_interval is not None:
74 return float(frame.tick_interval)
75 if value_range <= 0:
76 return SMART_INTERVAL_BASE
77
78 raw_interval = value_range / 10.0
79 exponent = math.floor(math.log10(raw_interval))
80 fractional_part = raw_interval / (10**exponent)
81
82 if fractional_part < 1.5:
83 snap_value = 1
84 elif fractional_part < 3.5:
85 snap_value = 2
86 elif fractional_part < 7.5:
87 snap_value = 5
88 else:
89 snap_value = 10
90 return snap_value * (10**exponent)

123 """Calculate safe offset for labels to avoid overlap with the fader cap."""
124 margin = LARGE_MARGIN
125 if width < MIN_CANVAS_WIDTH_FOR_MARGIN:
126 margin = MEDIUM_MARGIN

130 offset = max(tick_length_half, (cap_width / 2)) + margin
131
132 # Prevent overflowing the canvas boundaries
133 if label_position in ["right", "both"] and (center_x + offset > width - CANVAS_BOUNDARY_OFFSET):
134 offset = max(width - center_x - LABEL_ADJUSTMENT_OFFSET, slot_width / 2 + MIN_LABEL_DISTANCE)
135 if label_position in ["left", "both"] and (center_x - offset < CANVAS_BOUNDARY_OFFSET):
136 offset = max(center_x - LABEL_ADJUSTMENT_OFFSET, slot_width / 2 + MIN_LABEL_DISTANCE)

140 def _calculate_tick_y_coordinate(value, frame, available_height, vertical_padding):
141 """Convert a value to a vertical Y coordinate on the canvas."""
142 value_range = frame.max_val - frame.min_val
143 linear_normalization = max(0.0, min(1.0, (value - frame.min_val) / value_range if value_range != 0 else 0))
144 display_normalization = max(1e-7, linear_normalization) ** (1.0 / frame.log_exponent) if frame.log_exponent != 1.0 else linear_normalization
145 return available_height * (1 - display_normalization) + vertical_padding
