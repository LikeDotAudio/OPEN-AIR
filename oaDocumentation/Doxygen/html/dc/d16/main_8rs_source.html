34fn main() -> io::Result<()> {
35 let args = Args::parse();
36
37 // The standardized OPEN-AIR log pattern uses a leading float timestamp (UTP).
38 // We only capture the timestamp here as it is the primary sort key.
39 let log_pattern = Regex::new(r"^(?P<timestamp>\d+\.\d+)\s+\|\s+").unwrap();
40
41 let mut all_log_lines: Vec<LogLine> = Vec::new();
42
43 println!("📡📥📥 [INBOUND] Scanning directory: {:?}", args.dir);
44
45 for entry in WalkDir::new(&args.dir).max_depth(1) {
46 let entry = entry?;
47 // Only process .log files to avoid ingesting binary artifacts or reports.
48 if entry.path().extension().and_then(|s| s.to_str()) == Some("log") {
49 let file = File::open(entry.path())?;
50 let reader = BufReader::new(file);
51
52 for line in reader.lines() {
53 let line = line?;
54 if line.is_empty() {
55 continue;
56 }
57
58 if let Some(caps) = log_pattern.captures(&line) {
59 if let Ok(timestamp) = caps["timestamp"].parse::<f64>() {
60 all_log_lines.push(LogLine {
61 timestamp: timestamp,
62 content: line,
63 });
64 }
65 }
66 }
67 }
68 }
69
70 println!("📡 [PROCESS] Sorting {} log lines...", all_log_lines.len());
71 // Stable sort by timestamp reconstructs the system-wide event timeline.
72 all_log_lines.sort_by(|a, b| a.timestamp.partial_cmp(&b.timestamp).unwrap_or(std::cmp::Ordering::Equal));
73
74 println!("📡📤📤 [OUTBOUND] Writing to: {:?}", args.output);
75 let out_file = File::create(&args.output)?;
76 let mut writer = BufWriter::new(out_file);
77
78 for log_line in all_log_lines {
79 writer.write_all(log_line.content.as_bytes())?;
80 writer.write_all(b"\n")?;
81 }
82
83 writer.flush()?;
84 println!("📡 Successfully realigned logs.");
85
86 Ok(())
87}
