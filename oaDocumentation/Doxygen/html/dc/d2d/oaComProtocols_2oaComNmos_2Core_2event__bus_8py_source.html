13 """
14 A simple, thread-safe local event bus for the NMOS module.

17 self._subscribers: dict[str, list[Callable]] = {}

20 def subscribe(self, event_type: str, callback: Callable):
21 """Subscribes a callback to an event type."""
22 with self._lock:
23 if event_type not in self._subscribers:
24 self._subscribers[event_type] = []
25 if callback not in self._subscribers[event_type]:
26 self._subscribers[event_type].append(callback)
27
28 def unsubscribe(self, event_type: str, callback: Callable):
29 """Unsubscribes a callback from an event type."""
30 with self._lock:
31 if event_type in self._subscribers:
32 if callback in self._subscribers[event_type]:
33 self._subscribers[event_type].remove(callback)
34
35 def publish(self, event_type: str, *args, **kwargs):
36 """Publishes an event to all subscribers of that type."""
37 with self._lock:
38 callbacks = self._subscribers.get(event_type, []).copy()
39
40 for callback in callbacks:
41 try:
42 callback(*args, **kwargs)
43 except Exception as e:
44 print(f"[NmosEventBus] Error in callback for {event_type}: {e}")
