99 def _strip_constraints(self, data):
100 """Recursively removes fixed geometry from the ROOT ONLY to allow fluid resizing."""
101 if isinstance(data, dict):
102 # ⚡ ROOT PROTECTION: Only strip dimensions if this is a structural wrapper (no 'type')
103 # If it has a 'type', it's the main widget (e.g. OcaBin) and we should respect its intent.
104 if "type" not in data:
105 if "geometry" in data and isinstance(data["geometry"], dict):
106 data["geometry"].pop("width", None)
107 data["geometry"].pop("height", None)
108 data.pop("width", None)
109 data.pop("height", None)
110
111 # Recursively check children
112 for v in data.values():
113 self._strip_constraints(v)
114 else:
115 # If it has a type, it's a widget. We stop stripping here
116 # to ensure child containers inside the JSON keep their sizes.
117 pass
118 elif isinstance(data, list):
119 for item in data:
120 self._strip_constraints(item)
