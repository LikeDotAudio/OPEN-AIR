25 def _stop_all_managers(self):
26 """Internal helper to iterate and stop all shared manager instances."""
27 for name, instance in self.shared_instances.items():
28 if instance:
29 try:
30 matrix_log("UI", "GUI_MANAGER", inspect.currentframe().f_code.co_name, f"🛑 Stopping manager: {name}", level="DEBUG")
31 if hasattr(instance, "stop"): instance.stop()
32 elif hasattr(instance, "shutdown"): instance.shutdown()
33 elif hasattr(instance, "disconnect"): instance.disconnect()
34 except Exception as e:
35 logger.warning(f"⚠️ Error shutting down {name}: {e}")
