70fn parse_message(input: &[u8]) -> IResult<&[u8], OcaMessage> {
71 let (input, size) = be_u32(input)?;
72 let (input, handle) = be_u32(input)?;
73 let (input, target_ono) = be_u32(input)?;
74 let (input, method_id) = be_u32(input)?;
75
76 // Header is 16 bytes: size (4), handle (4), target_ono (4), method_id (4)
77 let param_size = if size > 16 { size - 16 } else { 0 };
78 let (input, parameters) = take(param_size)(input)?;
79
80 Ok((input, OcaMessage {
81 size,
82 handle,
83 target_ono,
84 method_id,
85 parameters,

89fn parse_pdu(input: &[u8]) -> IResult<&[u8], OcaPdu> {
90 let (input, version) = be_u16(input)?;
91 let (input, size) = be_u32(input)?;
92 let (input, message_count) = be_u16(input)?;
93
94 let (input, messages) = count(parse_message, message_count as usize)(input)?;

96 Ok((input, OcaPdu {
97 version,
98 size,
99 message_count,
