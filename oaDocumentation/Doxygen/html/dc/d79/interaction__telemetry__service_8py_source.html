25 """

28 if not state_mirror_engine or tab_name == "InteractivePreview":
29 return
30
31 visibility_topic = get_topic(
32 state_mirror_engine.base_topic,
33 base_mqtt_topic_from_path,
34 "visibility/visible",
35 )
36
37 geometry_topic = get_topic(
38 state_mirror_engine.base_topic,
39 base_mqtt_topic_from_path,
40 "visibility/geometry",
41 )
42
43 metadata = {
44 "tab_name": tab_name,
45 "engine": state_mirror_engine,
46 "vis_topic": visibility_topic,
47 "geo_topic": geometry_topic,
48 "geo_timer": None

54 widget,

56 on_hidden=lambda e: self._on_hidden(widget, e),
57 on_destroy=lambda e: self._on_destroy(widget, e),
58 on_configure=lambda e: self._on_geometry_change(widget, e)

90 def _on_geometry_change(self, widget, event):
91 if event.widget != widget: return

93 meta = self._tracked_widgets.get(widget)
94 if not meta: return

96 if meta["geo_timer"]:

99 meta["geo_timer"] = widget.after(500, lambda: self._perform_geometry_publish(widget))

101 def _perform_geometry_publish(self, widget):
102 meta = self._tracked_widgets.get(widget)
103 if not meta: return
104 meta["geo_timer"] = None
105
106 try:
107 width, height, pos_x, pos_y = widget.winfo_width(), widget.winfo_height(), widget.winfo_x(), widget.winfo_y()
108 TelemetryPublisher.publish_geometry(meta["engine"], meta["geo_topic"], meta["tab_name"], (width, height, pos_x, pos_y))
109
110 if hasattr(widget, '_log_telemetry_tx'):
111 widget._log_telemetry_tx(f"GEO: {width}x{height}")
112
113 matrix_log("ui", "gui_telemetry", "_perform_geometry_publish",
114 f"📡📏 [TELEMETRY] Transmitting {meta['tab_name']} geometry: {width}x{height} at {pos_x},{pos_y}", "TRACE")
115 except Exception:
