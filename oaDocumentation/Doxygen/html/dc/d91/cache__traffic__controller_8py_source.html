22# Processes incoming MQTT messages, decodes their payloads, and determines if an update to the cache is required.
23# This function acts as a gatekeeper, checking for redundancy by comparing the new payload
24# Inputs:
25# message (MqttMessage): The MQTT message object.
26# current_cache (Dict): The current state of the application cache.
27# Outputs:
28# Tuple[bool, Optional[Dict]]: A tuple indicating whether an update is needed (True/False)
29# and the new payload (if an update is needed, None otherwise).
30def process_traffic(
31 message: MqttMessage, current_cache: Any
32) -> tuple[bool, dict | None]:
33 """
34 Decodes payload and determines if update is needed.
35 Purity check is performed on the 'value' only, but METADATA is preserved.
36 """
37 topic = message.topic
38
39 # --- HEARTBEAT FILTER (V3.0.0 Refinement) ---
40 # Exclude high-frequency/volatile data from the persistent state cache.
41 # This prevents 'Heartbeat' or 'Status' messages from thrashing disk I/O.
42 volatile_patterns = ["/Heartbeat/", "/Status/Monitor/", "/Firehose/", "/Monitor/Telemetry/"]
43 if any(pattern in topic for pattern in volatile_patterns):
44 return False, None
45
46 try:
47 # 1. Decode original payload
48 raw_payload = message.get_json_payload()
49 if not isinstance(raw_payload, dict):
50 full_payload = {"value": raw_payload}
51 else:
52 full_payload = raw_payload
53
54 # 2. Extraction for Comparison (Purity Check)
55 if "value" in full_payload:
56 compare_val = {"value": full_payload["value"]}
57 # Carry timestamp if present for high-speed Rust comparison
58 if "timestamp" in full_payload: compare_val["timestamp"] = full_payload["timestamp"]
59 elif "pos" in full_payload:
60 compare_val = {"value": full_payload["pos"]}
61 if "timestamp" in full_payload: compare_val["timestamp"] = full_payload["timestamp"]
62 else:
63 # Complex state: strip noise for comparison only
64 keys_to_exclude = {"GUID", "type", "AES70", "source", "src"} # Keep 'timestamp' for comparison
65 compare_val = {k: v for k, v in full_payload.items() if k not in keys_to_exclude}
66
67 # 3. Decision: Only update if the VALUE changed
68 if state_comparator.should_update(topic, compare_val, current_cache):
69 # ⚡ RETURN FULL PAYLOAD: We need GUID/TS for the Investigation Engine
70 return True, full_payload
71
72 return False, None
73
74 except Exception:
75 return False, None
