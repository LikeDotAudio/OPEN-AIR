51 def _on_drag(event, mode):
52 if not widget.winfo_exists(): return
53 dx, dy = event.x_root - widget.winfo_rootx(), event.y_root - widget.winfo_rooty()
54 if mode == "padx":
55 pv = max(0, int(dx // 5))
56 _show_resize_tooltip(event, event.x_root, event.y_root, pv, None, label1="PAD-X", label2=None)
57 # Visualize the padding being added to the right
58 current_w, current_h = widget.winfo_width(), widget.winfo_height()
59 _show_ghost_box(current_w + pv, current_h) # Show widget + padding width
60 elif mode == "pady":
61 pv = max(0, int(dy // 5))
62 _show_resize_tooltip(event, event.x_root, event.y_root, pv, None, label1="PAD-Y", label2=None)
63 # Visualize the padding being added below
64 current_w, current_h = widget.winfo_width(), widget.winfo_height()
65 _show_ghost_box(current_w, current_h + pv) # Show widget + padding height
66 elif mode == "breath":
67 bpv = max(0, int(dx // 5)) # Breath Padding Value
68 _show_resize_tooltip(event, event.x_root, event.y_root, bpv, None, label1="BREATH", label2=None)
69 current_w, current_h = widget.winfo_width(), widget.winfo_height()
70 _show_ghost_box(current_w + bpv * 2, current_h + bpv * 2) # Visualize breath padding around the widget
71 else:
72 nw, nh = max(20, dx), max(20, dy)
73 if mode == "horiz": nh = widget.winfo_height()
74 if mode == "vert": nw = widget.winfo_width()
75 _show_resize_tooltip(event, event.x_root, event.y_root, nw, nh)
76 _show_ghost_box(nw, nh)

91 elif mode == "pady":

99 if mode in ["diag", "horiz"]: updates.append((int(nw), f"{path}.width"))
100 if mode in ["diag", "vert"]: updates.append((int(nh), f"{path}.height"))
101 else:
102 if mode in ["diag", "horiz"]: updates.append((int(nw), f"{path}.geometry.width"))
103 if mode in ["diag", "vert"]: updates.append((int(nh), f"{path}.geometry.height"))
104 state_manager.batch_update(updates, source=layout)
