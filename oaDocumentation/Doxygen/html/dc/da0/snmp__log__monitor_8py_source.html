29class SnmpLogMonitor:
30 """
31 Monitors the SNMP SET log file for incoming commands and translates them
32 into system events via the direct MQTT client.
33 """
34
35 def __init__(self, thread_lock: threading.RLock, notify_monitor_callback, running_flag_getter, mqtt_client=None):
36 """
37 Initializes the log monitor.
38 """
39 self._state_lock = thread_lock
40 self._notify_monitor = notify_monitor_callback
41 self._running_flag_getter = running_flag_getter
42 self._mqtt_client = mqtt_client
43
44 self._thread = None
45 self.log_file = str(SNMP_SET_LOG)
46
47 def start(self):
48 """Starts the background thread for log monitoring."""
49 if self._thread and self._thread.is_alive():
50 snmp_logger.warning("SnmpLogMonitor: Thread already running.")
51 return
52
53 self._thread = threading.Thread(target=self._log_monitoring_loop, daemon=True, name="SNMP-LogMonitorLoop")
54 self._thread.start()
55 matrix_log("comms", "snmp", "start",
56 "SnmpLogMonitor: Started background log monitoring thread.", "INFO")
57
58 def stop(self):
59 """Signals the background thread to stop and waits for it to terminate."""
60 if self._thread and self._thread.is_alive():
61 matrix_log("comms", "snmp", "stop",
62 "SnmpLogMonitor: Stopping background log monitoring thread...", "INFO")
63 self._thread.join(timeout=THREAD_JOIN_TIMEOUT)
64 if self._thread.is_alive():
65 snmp_logger.warning("SnmpLogMonitor: Thread did not terminate gracefully.")
66 matrix_log("comms", "snmp", "stop", "SnmpLogMonitor: Stopped.", "INFO")
67
68 def _log_monitoring_loop(self):
69 """The main loop for monitoring the SNMP SET log file."""
70 while self._running_flag_getter():
71 try:
72 if os.path.isfile(self.log_file) and os.path.getsize(self.log_file) > 0:
73 with open(self.log_file, "r+", encoding="utf-8") as f:
74 lines = f.readlines()
75 for line in lines:
76 parts = line.split()
77 if len(parts) >= MIN_LOG_PARTS and parts[0] == "-s":
78 oid, value = parts[1], parts[2]
79
80 matrix_log("comms", "snmp", "_log_monitoring_loop",
81 f"SnmpLogMonitor: RX SET: {oid} -> {value}", "DEBUG")
82
83 meta = {
84 "message_type": "SPLICE_ACTION",
85 "origin_source": "oaComSNMP"
86 }
87
88 if self._mqtt_client:
89 topic = f"OPEN-AIR/SNMP/gui_out/{oid}"
90 self._mqtt_client.publish(topic, {"value": value, "origin_source": "oaComSNMP"})
91
92 self._notify_monitor("RX_SET", oid, value, None, meta)
93 f.seek(0)
94 f.truncate()
95 except FileNotFoundError:
96 pass
97 except Exception as e:
98 snmp_logger.error(f"SnmpLogMonitor: Log monitoring loop error: {e}")
99
100 time.sleep(LOG_POLLING_INTERVAL)
