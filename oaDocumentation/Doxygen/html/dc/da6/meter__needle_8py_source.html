64 # 3. Animation Logic
65 def render_cb(full_redraw=False):
66 try:
67 now = time.time() * 1000; v1, v2 = frame.anim_current_value, (frame.anim_current_value_2 if config.meter_mode == "stereo" else None)
68 if v1 >= config.red_zone_start or (v2 is not None and v2 >= config.red_zone_start):
69 frame.anim_peak_on, frame.anim_peak_expiry = True, now + config.peak_hold_ms

72 # Wrap rendering in try-except to catch potential TclError (e.g., tagOrId not found)
73 MeterRenderingEngine.render(canvas, config, v1, v2, frame.anim_peak_on, ox, oy, full_redraw=full_redraw)
74 except tk.TclError as e:
75 # Log the error, but don't let it crash the application or tests
76 builder_logger.warning(f"TclError during meter rendering for '{config.label}': {e}. This might be due to missing tags or canvas items.")
77 except Exception as e:

79
80
81 frame.render = lambda: render_cb(full_redraw=True)
82 animator = MeterAnimator(frame, config, canvas, render_cb)
83 canvas.tag_bind("peak_dot", "<Button-1>", animator.reset_peak)

87 linker.setup_links(animator); frame.vu_value_var, frame.vu_value_var_2 = linker.vu_value_var, linker.vu_value_var_2
