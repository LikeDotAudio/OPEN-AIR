296 def _resolve_insertion_path(self):
297 """Helper to find the best path for programmatic insertion (Button Clicks)."""
298 path = self.last_focused_path
299
300 if not path:
301 # Default to root
302 full_state = state_manager.get_state()
303 if full_state:
304 path = list(full_state.keys())[0]
305 else:
306 return "" # Pure root
307
308 # Resolve container sub-path if needed
309 val = state_manager.get_value_at_path(path)
310 if isinstance(val, dict):
311 w_type = val.get("type", "")
312 if "Block" in w_type and "fields" not in path:
313 return f"{path}.fields"
314 if "Table" in w_type and "rows" not in path:
315 return f"{path}.rows.0"
316
317 return path
