17// AppConfig defines the strict structural and numerical bounds for
18// the core system parameters.
19#[derive(Debug, Serialize, Deserialize, Validate)]

41 // validate_config performs a two-stage check:
42 // 1. Extraction: Pulls raw Python types and converts them to native Rust types.
43 // 2. Bound Checking: Executes macro-based validation (e.g., range and length).
44 fn validate_config(&self, config_dict: &Bound<'_, PyDict>) -> PyResult<bool> {
45 let mqtt_port: u16 = config_dict.get_item("mqtt_port")?.and_then(|v| v.extract().ok()).unwrap_or(MQTT_DEFAULT_PORT);
46 let mqtt_broker: String = config_dict.get_item("mqtt_broker")?.and_then(|v| v.extract().ok()).unwrap_or_default();
47 let partition_id: String = config_dict.get_item("partition_id")?.and_then(|v| v.extract().ok()).unwrap_or_default();
48
49 let configuration = AppConfig {
50 partition_id,
51 mqtt_port,
52 mqtt_broker,
53 };
54
55 // Failure results in a PyValueError, preventing the system from
56 // proceeding with dangerous or illogical settings.
57 match configuration.validate() {
58 Ok(_) => Ok(true),
59 Err(e) => Err(pyo3::exceptions::PyValueError::new_err(format!("Config validation failed: {}", e))),
60 }
61 }
62}
