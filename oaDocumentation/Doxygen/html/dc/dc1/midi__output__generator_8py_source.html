121 def _send_midi(self, m_type, note, velocity):
122 if not self.midi_manager: return
123
124 channels = [i for i, v in enumerate(self.selected_channels) if v.get()]
125 if not channels: return
126
127 color = get_midi_color(channels[0])
128 if m_type == "note_on": self.keyboard.note_on(note, color)
129 else: self.keyboard.note_off(note)
130
131 port = self.selected_output_port.get()
132 for ch in channels:
133 topic = f"OPEN-AIR/MIDI/gui_out/ch{ch+1}/note{note}"
134 pld = {
135 "value": velocity, "channel": ch, "note": note, "velocity": velocity,
136 "type": m_type, "raw": f"{m_type} channel={ch} note={note} velocity={velocity}"
137 }
138 meta = {
139 "origin_source": "MIDI-TX", "target_port": port, "midi_type": m_type
140 }
141 import mido
142 midi_message = mido.Message(m_type, channel=ch, note=note, velocity=velocity)
143 try:
144 # ⚡ REFACTORED: MidiManager.publish expects (port_name, mido_message)
145 self.midi_manager.publish(port, midi_message)
146 except Exception as e:
147 logger.error(f"Failed to publish MIDI message: {e}")
