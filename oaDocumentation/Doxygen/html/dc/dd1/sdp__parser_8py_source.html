5def parse_sdp(sdp):
6 """
7 Parses an SDP string to extract relevant information for NMOS integration.
8
9 Args:
10 sdp (str): The SDP content as a string.
11
12 Returns:
13 dict: A dictionary containing parsed SDP attributes like name, IP, port,
14 bit depth, and sample rate.
15 """
16 data = {}
17
18 for line in sdp.splitlines():
19 if line.startswith("s="):
20 data["name"] = line[2:].strip()
21 elif line.startswith("o="):
22 parts = line.split()
23 if len(parts) >= 6:
24 data["src_ip"] = parts[5]
25 elif line.startswith("c=IN IP4"):
26 data["ip"] = line.split()[2].split("/")[0]
27 elif line.startswith("m=audio"):
28 try:
29 data["port"] = int(line.split()[1])
30 except (IndexError, ValueError):
31 pass # Ignore if port is not parseable
32 elif "rtpmap" in line and "L" in line:
33 try:
34 # Example format: a=rtpmap:96 L24/48000/2
35 # We are interested in bit depth, sample rate, and channels
36 parts = line.split("L")[1].split("/")
37 if len(parts) >= 2:
38 data["bit"] = int(parts[0])
39 data["rate"] = int(parts[1])
40 if len(parts) > 2:
41 data["ch"] = int(parts[2])
42 except (IndexError, ValueError):
43 pass # Ignore if rtpmap values are not parseable
44
45 return data
