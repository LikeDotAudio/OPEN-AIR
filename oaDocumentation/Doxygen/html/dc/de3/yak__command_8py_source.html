67 def process_fleet(self):
68 """
69 Reads the fleet data and loads the command tabs for each found device.
70 """
71 if not self.fleet_path.exists():
72 logger.error(f"❌ Fleet file not found: {self.fleet_path}")
73 return
74
75 try:
76 with open(self.fleet_path, "rb") as f:
77 fleet_data = orjson.loads(f.read())
78
79 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name, f"🚀 YakFleetCommandBuilder: Processing fleet data from {self.fleet_path}...", level="DEBUG")
80
81 # Iterate categories (e.g., Spectrum, DMM)
82 for category, cat_data in fleet_data.items():
83 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name, f"📂 Processing Category: {category}", level="DEBUG")
84 yak_data = cat_data.get("YAK", {})
85
86 # Iterate Models (e.g., 34401A, N9340B)
87 for model, model_data in yak_data.items():
88 if model == "Unknown": continue
89
90 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name, f"🔍 Searching for YAK tabs for Model: {model}...", level="DEBUG")
91
92 # Search for the specific device directory
93 # Structure: oaGui/Assets/left_50/top_100/<Category>/10_YAK/<Any_Number>_<Model>
94 # We use a glob to be flexible with the Category folder name and the Model prefix
95 search_pattern = f"oaGui/Assets/left_50/top_100/*/10_YAK/*_{model}"
96 found_dirs = list(GLOBAL_PROJECT_ROOT.glob(search_pattern))
97
98 if not found_dirs:
99 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name, f"⚠️ No YAK directory found for model {model}", level="DEBUG")
100 continue
101
102 for device_dir in found_dirs:
103 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name, f"✅ Found device directory: {device_dir}", level="SUCCESS")
104 self._load_tabs_for_device(device_dir, model)
105
106 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name, "🏁 YakFleetCommandBuilder: Fleet processing complete.", level="DEBUG")

135 def _process_staggered_queue(self, hidden_window, queue, model):
136 """Processes the next JSON file in the queue ONLY after the previous one completes."""
137 if not queue:
138 # All tabs loaded for this device. Schedule cleanup.
139 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name, f"✅ Serial load complete for {model}.", level="SUCCESS")
140 hidden_window.after(5000, lambda: self._cleanup_window(hidden_window, model))
141 return
142
143 json_path = queue.pop(0)
144 try:
145 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name, f"🐂 Serial Loading: {json_path.name}", level="DEBUG")
146
147 # Define completion callback to trigger next item in queue
148 def on_tab_complete():
149 # Schedule next load with a small gap for breathing room
150 hidden_window.after(100, lambda: self._process_staggered_queue(hidden_window, queue, model))
151
152 UniversalGuiLoader(
153 hidden_window,
154 str(json_path),
155 config={
156 "app_instance": self.app_instance,
157 "state_mirror_engine": getattr(self.app_instance, 'state_mirror_engine', None),
158 "subscriber_router": getattr(self.app_instance, 'subscriber_router', None),
159 "on_complete": on_tab_complete # PASS THE CALLBACK
160 }
161 )
162 except Exception:
163 logger.exception("❌ Failed to load tab {json_path.name}")
164 # Continue queue on failure
165 hidden_window.after(100, lambda: self._process_staggered_queue(hidden_window, queue, model))
