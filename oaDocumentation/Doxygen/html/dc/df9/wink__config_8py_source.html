10def extract_wink_config(config_data):
11 """Extracts and normalizes configuration for the Wink Button."""
12 config = config_data
13
14 # Shape and Size
15 shape_type = config.get("shape_type", "rect").lower()
16 width = config.get("width", 60)
17 height = config.get("height", 60) if shape_type != "round" else width
18 radius = config.get("radius", 5)
19
20 # Colors
21 bg_color = config.get("color", "#39FF14")
22 shutter_color = config.get("shutter_color", "black")
23
24 theme_colors = THEMES.get(DEFAULT_THEME, THEMES["dark"])
25 default_bg = theme_colors.get("bg", "#2b2b2b")
26 bezel_color = config.get("bezel_color", default_bg)
27
28 # Text
29 text_closed = config.get("text_closed", "")
30 if isinstance(text_closed, str) and " " in text_closed:
31 text_closed = text_closed.replace(" ", "\n")
32 text_closed_color = config.get("text_closed_color", "white" if shutter_color == "black" else "black")
33
34 text_inside = config.get("text_inside", "")
35 if text_inside == "":
36 text_inside = None
37 if isinstance(text_inside, str) and " " in text_inside:
38 text_inside = text_inside.replace(" ", "\n")
39 text_inside_color = config.get("text_inside_color", "black")
40
41 font_size_cfg = config.get("font_size")
42 font_size_closed_cfg = config.get("font_size_closed")
43
44 # Border/Lens
45 border_thickness = config.get("border_thickness", 2)
46 default_border_color = "#333333" if shutter_color.lower() in ["black", "#000000"] else "black"
47 border_color = config.get("border_color", default_border_color)
48
49 # Glass Lens specific (New)
50 use_glass_lens = config.get("use_glass_lens", True) # Default to true as per request? Or false for compatibility?
51 # User said: "intead of a border to the buttons, i would like it to resembel a glass lens."
52 # So if use_glass_lens is true, we might suppress the border or integrate it.
53
54 # Physics/Animation
55 open_duration = config.get("open_speed", 150)
56 close_duration = config.get("close_speed", 300)
57
58 if 0 < open_duration < 1.0: open_duration = 16 / open_duration
59 if 0 < close_duration < 1.0: close_duration = 16 / close_duration
60
61 open_inc = 16 / open_duration if open_duration > 0 else 1.0
62 close_inc = 16 / close_duration if close_duration > 0 else 1.0
63
64 blink_interval = config.get("blink_interval", 0)
65
66 # Logic
67 is_latching = config.get("latching", False)
68 is_locked_init = config.get("LOCKED", False)
69 value_default = config.get("value_default", False)
70
71 label_pos = config.get("label_position", "top").lower()
72
73 return {
74 "shape_type": shape_type,
75 "width": width,
76 "height": height,
77 "radius": radius,
78 "bg_color": bg_color,
79 "shutter_color": shutter_color,
80 "bezel_color": bezel_color,
81 "text_closed": text_closed,
82 "text_closed_color": text_closed_color,
83 "text_inside": text_inside,
84 "text_inside_color": text_inside_color,
85 "font_size": font_size_cfg,
86 "font_size_closed": font_size_closed_cfg,
87 "border_thickness": border_thickness,
88 "border_color": border_color,
89 "use_glass_lens": use_glass_lens,
90 "open_inc": open_inc,
91 "close_inc": close_inc,
92 "blink_interval": blink_interval,
93 "is_latching": is_latching,
94 "is_locked_init": is_locked_init,
95 "value_default": value_default,
96 "label_position": label_pos,
97 "text_align": config.get("text_align", "center").lower()
98 }
