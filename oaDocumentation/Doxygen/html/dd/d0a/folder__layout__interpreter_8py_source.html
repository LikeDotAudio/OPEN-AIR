42 def _scan_for_gui_files(self, path: pathlib.Path) -> bool:
43 """Recursively checks if a folder contains a '.json' or '.py' file."""
44 path_str = str(path)
45 if path_str in self._scan_cache:
46 return self._scan_cache[path_str]
47
48 result = False
49 try:
50 with os.scandir(path_str) as it:
51 for entry in it:
52 if entry.is_file():
53 name = entry.name
54 if (name.endswith(".json") or name.endswith(".py")) and \
55 name != "layout.json" and not name.startswith("__"):
56 result = True
57 break
58 elif entry.is_dir() and not entry.name.startswith("__"):
59 if self._scan_for_gui_files(pathlib.Path(entry.path)):
60 result = True
61 break
62 except (FileNotFoundError, PermissionError):
63 pass
64
65 self._scan_cache[path_str] = result
66 return result

87 def _parse_layout_json(self, layout_file: pathlib.Path, source_path: pathlib.Path) -> dict:
88 try:
89 if layout_file.stat().st_size == 0:
90 return {"type": "error", "data": {"error_message": "Empty layout.json"}}

97 def parse_layout_data(self, layout_data: dict, source_path: pathlib.Path) -> dict:
98 """Centralized interpreting logic for layout configuration dictionaries."""
99 layout_type = layout_data.get("type", "unknown")
100 behavior = layout_data.get("behavior", {})
101
102 parsed_data = {
103 "overflow_ew": behavior.get("overflow_ew", "none"),
104 "overflow_ns": behavior.get("overflow_ns", "none"),
105 "fluid_ew": behavior.get("fluid_ew", False),
106 "fluid_ns": behavior.get("fluid_ns", False)
107 }
108
109 if layout_type in ["horizontal_split", "vertical_split"]:
110 self._gather_split_data(layout_data, layout_type, source_path, parsed_data)
111 elif layout_type == "notebook":
112 self._gather_notebook_data(layout_data, source_path, parsed_data)
113 elif layout_type in ["monitors", "recursive_build"]:
114 self._gather_recursive_data(layout_data, source_path, parsed_data)
115 else:
116 return self._parse_directory_listing(source_path)
117
118 matrix_log("gui", "gui_manager", "FolderLayoutInterpreter", f"Interpreted: '{source_path}' | Type: '{layout_type}'", "DEBUG")
119 return {"type": layout_type, "data": parsed_data}

121 def _gather_split_data(self, layout_data, layout_type, source_path, parsed_data):
122 parsed_data["orientation"] = tk.HORIZONTAL if layout_type == "horizontal_split" else tk.VERTICAL
123 raw_panels = layout_data.get("panels", [])
124 percentages = layout_data.get("percentages", [])
125 resolved = []
126 for i, item in enumerate(raw_panels):
127 path_str = item if isinstance(item, str) else item.get("path")
128 if not path_str: continue
129 weight = percentages[i] if i < len(percentages) else (item.get("weight", 1) if isinstance(item, dict) else 1)
130 resolved.append({"path": source_path / path_str, "weight": weight})
131 parsed_data["panels"] = resolved
132 parsed_data["panel_percentages"] = [p["weight"] for p in resolved]
133
134 def _gather_notebook_data(self, layout_data, source_path, parsed_data):
135 tabs = layout_data.get("tabs", [])
136 resolved = []
137 for t in tabs:
138 if isinstance(t, dict) and "path" in t and "display_name" in t:
139 resolved.append({"path": source_path / t["path"], "display_name": t["display_name"]})
140 parsed_data["tabs"] = resolved
141
142 def _gather_recursive_data(self, layout_data, source_path, parsed_data):
143 parsed_data["gui_files"] = [source_path / f for f in layout_data.get("gui_files", [])]
144 children = []
145 for item in layout_data.get("child_containers", []):
146 children.append(source_path / item if isinstance(item, str) else item)
147 parsed_data["child_containers"] = children

149 def _parse_directory_listing(self, path: pathlib.Path) -> dict:
150 """Infers layout type from directory contents using a rule-based sequence."""
151 if path.is_file(): return self._parse_file_as_layout(path)
152 if not path.is_dir(): return {"type": "error", "data": {"error_message": f"Invalid path: {path}"}}
153
154 try:
155 sub_dirs = sorted([d for d in path.iterdir() if d.is_dir() and not d.name.startswith('.') and not d.name.startswith('__')])
156 gui_files = sorted([f for f in path.iterdir() if f.is_file() and f.suffix in [".json", ".py"] and f.name != "layout.json" and not f.name.startswith("__")])
157 except FileNotFoundError:
158 return {"type": "error", "data": {"error_message": "Directory not found."}}
159
160 # Rule Sequence
161 for detector in self._detectors:
162 result = detector.detect(path, sub_dirs, gui_files)
163 if result:
164 matrix_log("gui", "gui_manager", "FolderLayoutInterpreter", f"Interpreted: '{path}' | Type: '{result['type']}'", "DEBUG")
165 return result
166
167 matrix_log("gui", "gui_manager", "FolderLayoutInterpreter", f"Listing: '{path}' | Sub-dirs: {len(sub_dirs)} | Files: {len(gui_files)}", "DEBUG")
168 return {
169 "type": "directory_listing",
170 "data": {
171 "sub_dirs": [{"path": d} for d in sub_dirs],
172 "gui_files": gui_files,
173 },
