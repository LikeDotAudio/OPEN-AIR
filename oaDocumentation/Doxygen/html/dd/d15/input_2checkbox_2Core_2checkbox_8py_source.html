37 def _assemble_ui(self, parent_widget, config_data, context, **kwargs):
38 """Assembles the Checkbox UI elements."""
39 current_function_name = inspect.currentframe().f_code.co_name
40 config = config_data
41 label = get_text(config.get('label_active')) or get_text(config.get('label'), "")
42
43 matrix_log("UI", "GUI_ELEMENTS", inspect.currentframe().f_code.co_name, f"🔬 Entering _assemble_ui for '{label}'.", level="DEBUG")
44
45 try:
46 # Use tk.Canvas for transparency support
47 canvas = tk.Canvas(
48 parent_widget,
49 bd=0,
50 highlightthickness=0,
51 relief="flat",
52 height=30,
53 width=150
54 )
55
56 # We use a BooleanVar to track the state of the checkbox.
57 initial_value = bool(config.get("value", False))
58 state_var = kwargs.get("variable") or tk.BooleanVar(master=parent_widget, value=initial_value)
59 canvas.variable = state_var
60
61 def get_label_text():
62 current_state = state_var.get()
63 if current_state:
64 return get_text(config.get("label_active"), get_text(config.get("label"), ""))
65 else:
66 return get_text(config.get("label_inactive"), get_text(config.get("label"), ""))
67
68 def redraw_checkbox(*args):
69 if not canvas.winfo_exists(): return
70 canvas.delete("vu_element")
71 canvas.delete("industrial_text")
72 w = canvas.winfo_width()
73 h = canvas.winfo_height()
74 if w <= 1: return
75
76 current_state = state_var.get()
77 box_size = 16
78 bx, by = 10, h/2 - box_size/2
79
80 # Draw Box
81 canvas.create_rectangle(
82 bx, by, bx + box_size, by + box_size,
83 outline="white", width=1, tags="vu_element"
84 )
85
86 if current_state:
87 # Draw Checkmark
88 canvas.create_line(
89 bx+3, by+box_size/2, bx+box_size/2, by+box_size-3,
90 fill="#00ff00", width=2, tags="vu_element"
91 )
92 canvas.create_line(
93 bx+box_size/2, by+box_size-3, bx+box_size-3, by+3,
94 fill="#00ff00", width=2, tags="vu_element"
95 )
96
97 # Draw Label
98 canvas.create_text(
99 bx + box_size + 10, h/2, text=get_label_text(),
100 fill="white", font=("Helvetica", 9), anchor="w",
101 tags="industrial_text"
102 )
103
104 def toggle_state(event):
105 state_var.set(not state_var.get())
106 redraw_checkbox()
107
108 canvas.bind("<Button-1>", toggle_state)
109 canvas.bind("<Configure>", redraw_checkbox, add="+")
110 state_var.trace_add("write", lambda *a: redraw_checkbox())
111
112 redraw_checkbox()
113 return canvas, canvas
114
115 except Exception as e:
116 logger.exception(f"❌ Error in _assemble_ui for '{label}': {e}")
117 return None, None
