121 @staticmethod
122 def _get_shadow_pt(px, py, cx, cy, length):
123 """Calculates shadow projection with distance-based offset."""
124 dist = math.sqrt((px - cx)**2 + (py - cy)**2)
125 factor = min(1.0, max(0.0, dist / length)) if length != 0 else 0
126 sx = px + (ShadowDrawer.MAX_SHADOW_X * factor)
127 sy = py + (ShadowDrawer.MAX_SHADOW_Y * factor)
128 return sx, sy
