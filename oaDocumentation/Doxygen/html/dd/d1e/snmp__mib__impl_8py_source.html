109 def _check_for_disk_updates(self):
110 """Periodically checks if current.mib was updated by the Core process."""
111 if SNMP_CURRENT_MIB.exists():
112 try:
113 mtime = os.path.getmtime(SNMP_CURRENT_MIB)
114 if mtime > self._last_mtime:
115 self.load_mib_from_disk()
116 self._last_mtime = mtime
117 self.status_var.set("Status: Auto-Updated")
118 except: pass
119
120 if not getattr(self, '_shutdown', False):
121 self.after(5000, self._check_for_disk_updates_check_for_disk_updates)

123 def load_mib_from_disk(self, path=None):
124 target = path or SNMP_CURRENT_MIB
125 if target.exists():
126 try:
127 with open(target) as f:
128 content = f.read()
129 self.text_area.delete("1.0", tk.END)
130 self.text_area.insert("1.0", content)
131 self._last_mtime = os.path.getmtime(target)
132 except Exception as e:
133 logger.error(f"❌ SNMP UI: Failed to read MIB from disk: {e}")
