38 def generate_master_script(self):
39 """Generates a single master script to handle the entire OPEN-AIR OID tree."""
40 if self._verbose_logging_enabled(): snmp_logger.debug(f"📜 SNMP: Generating Master Bridge script at {self.master_script_path}...")
41
42 # ⚡ SYSTEM ACCESS: Use absolute paths for snmpd visibility
43 flat_file = os.path.abspath(str(SNMP_STATE_FILE))
44 log_file = os.path.abspath(str(SNMP_SET_LOG))
45 debug_log = os.path.join(os.path.dirname(flat_file), "bridge_debug.log")
46
47 lines = [
48 "#!/bin/bash",
49 "# OPEN-AIR Master SNMP Bridge",
50 f"FLAT_FILE=\"{flat_file}\"",
51 f"LOG_FILE=\"{log_file}\"",
52 f"DEBUG_LOG=\"{debug_log}\"",
53 "",
54 "# Log all requests for debugging",
55 "echo \"[$(date)] REQ: $1 $2\" >> $DEBUG_LOG",
56 "",
57 "# Handle SET (-s)",
58 "if [ \"$1\" = \"-s\" ]; then",
59 " echo \"-s $2 $3 $4\" >> $LOG_FILE",
60 " exit 0",
61 "fi",
62 "",
63 # Unified GET/GETNEXT Handler using AWK
64 "awk -F':' -v cmd=\"$1\" -v target=\"$2\" '",
65 " function norm(oid) {",
66 " gsub(/^[.\"]+|[.\"]+$/, \"\", oid);",
67 " split(oid, p, \".\");",
68 " out = \"\";",
69 " for (i=1; i<=length(p); i++) {",
70 " out = out sprintf(\"%010d.\", p[i]);",
71 " }",
72 " return out;",
73 " }",
74 " BEGIN { ",
75 " t = norm(target); ",
76 " found = 0; ",
77 " }",
78 " {",
79 " c = norm($1);",
80 " if (cmd == \"-g\") {",
81 " if (c == t) {",
82 " # Print OID without leading dot",
83 " print ($1 ~ /^\\./ ? substr($1, 2) : $1);",
84 " print \"string\";",
85 " print substr($0, index($0, \":\") + 1);",
86 " found = 1;",
87 " exit 0;",
88 " }",
89 " } else if (cmd == \"-n\") {",
90 " if (c > t) {",
91 " # Print OID without leading dot",
92 " print ($1 ~ /^\\./ ? substr($1, 2) : $1);",
93 " print \"string\";",
94 " print substr($0, index($0, \":\") + 1);",
95 " found = 1;",
96 " exit 0;",
97 " }",
98 " }",
99 " }",
100 " END { if (!found) exit 1; }",
101 " ' \"$FLAT_FILE\"",
102 "",
103 "if [ $? -ne 0 ]; then",
104 " echo \"[$(date)] NOT FOUND: $2\" >> $DEBUG_LOG",
105 "fi"
106 ]
107
108 try:
109 with open(self.master_script_path, "w") as f:
110 f.write("\n".join(lines) + "\n")
111 # 🔐 SYSTEM ACCESS: Ensure script is executable by all users (snmp user)
112 os.chmod(self.master_script_path, 0o755)
113 return self.master_script_path
114 except Exception as e:
115 snmp_logger.error(f"❌ Failed to generate master script: {e}")
116 return None
