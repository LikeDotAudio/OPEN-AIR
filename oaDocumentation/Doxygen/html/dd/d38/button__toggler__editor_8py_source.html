96 def _get_nested(self, key_path):
97 parts = key_path.split(".")
98 val = self.current_config
99 for p in parts:
100 if isinstance(val, dict):
101 val = val.get(p)
102 else:
103 return None
104 return val

106 def _set_nested(self, key_path, value):
107 parts = key_path.split(".")
108 target = self.current_config
109 for p in parts[:-1]:
110 if p not in target or not isinstance(target[p], dict):
111 target[p] = {}
112 target = target[p]
113 target[parts[-1]] = value
