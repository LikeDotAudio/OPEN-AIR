135 def _on_status_received(self, topic, payload):
136 """Callback for MQTT status updates."""
137 if not isinstance(payload, dict):
138 try:
139 import json
140 payload = json.loads(payload)
141 except:
142 return
143
144 self._last_status = payload
145 # Use .after() to update UI from MQTT thread safely
146 self.after(0, lambda: self._update_ui_with_status(payload))
