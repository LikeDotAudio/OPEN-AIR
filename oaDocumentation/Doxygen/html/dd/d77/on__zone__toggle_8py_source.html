18# Handles the event when a zone toggle button is clicked in the Showtime tab.
19# This function updates the `selected_zone` in the `showtime_tab_instance`,
20# managing the selection state and clearing group selections when a new zone is toggled.
21# It then refreshes the displayed zone, group, and device buttons, and triggers a tune request.
22# Inputs:
23# showtime_tab_instance: An instance of the Showtime tab.
24# zone_name (str): The name of the zone that was toggled.
25# Outputs:
26# None.
27def on_zone_toggle(showtime_tab_instance, zone_name):
28 matrix_log("UI", "SHOWTIME", inspect.currentframe().f_code.co_name, f"🟢️️️🔵 Zone toggle clicked for: {zone_name}. Current selection: {showtime_tab_instance.selected_zone}.", level="DEBUG")
29 if showtime_tab_instance.selected_zone == zone_name:
30 showtime_tab_instance.selected_zone = None
31 showtime_tab_instance.selected_group = None
32 matrix_log("UI", "SHOWTIME", inspect.currentframe().f_code.co_name, "🟢️️️🟡 Deselected Zone. Clearing Group selection.", level="DEBUG")
33 else:
34 showtime_tab_instance.selected_zone = zone_name
35 showtime_tab_instance.selected_group = None
36 matrix_log("UI", "SHOWTIME", inspect.currentframe().f_code.co_name, f"🟢️️️🟢 Selected new Zone: {showtime_tab_instance.selected_zone}. Clearing Group selection.", level="DEBUG")
37
38 showtime_tab_instance._create_zone_buttons()
39 showtime_tab_instance._create_group_buttons()
40 showtime_tab_instance._create_device_buttons()
