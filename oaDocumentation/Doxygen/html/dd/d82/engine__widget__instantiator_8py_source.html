27 Instantiates a widget class into the parent frame.

29 config_dict = {
30 "theme_colors": self.theme_colors,
31 "state_mirror_engine": self.state_mirror_engine,
32 "subscriber_router": self.subscriber_router,

37 # Wrap pure Python modules in a LoaderOrchestrator
38 builder = LoaderOrchestrator(parent_widget, json_path=None, config=config_dict)
39 self.builders.append(builder)
40
41 # Manually attach based on parent's geometry manager
42 try:
43 if parent_widget.grid_slaves():
44 builder.grid(row=0, column=0, sticky="nsew")
45 else:
46 builder.pack(fill=tk.BOTH, expand=True)
47 except tk.TclError:
48 builder.pack(fill=tk.BOTH, expand=True)
49
50 builder.start()
51 config_dict["builder_instance"] = builder
52
53 # Instantiate the actual Python GUI
54 try:
55 matrix_log("ui", "gui_builder", "instantiate_widget",
56 f"🔨 Instantiating {widget_class.__name__} (Parent: {parent_widget})", "DEBUG")
57

59
60 if hasattr(instance, "pack"):
61 instance.pack(fill=tk.BOTH, expand=True)
62 elif hasattr(instance, "grid"):
63 instance.grid(row=0, column=0, sticky="nsew")
64 except Exception as e:
65 matrix_log("ui", "gui_builder", "instantiate_widget",
66 f"🛑 [ERROR] Failed to instantiate {widget_class.__name__}: {e}", "ERROR")
67
68 return builder
