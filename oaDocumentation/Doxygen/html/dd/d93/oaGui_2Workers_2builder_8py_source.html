193 def _perform_canvas_resize(self, width):
194 self._resize_timer = None
195 if width <= 1 or not self.canvas_window_idcanvas_window_id: return
196
197 # ⚡ GEOMETRY SYNC: Stretch the content frame to fill the viewport
198 # Width depends on allow_horizontal_scroll lock.
199 # Height ALWAYS stretches to at least viewport height to ensure background coverage.
200 req_w = self.scroll_framescroll_frame.winfo_reqwidth()
201 req_h = self.scroll_framescroll_frame.winfo_reqheight()
202 canvas_h = self.canvas.winfo_height()
203
204 new_w = width if not self.allow_horizontal_scroll else max(width, req_w)
205 new_h = max(canvas_h, req_h)
