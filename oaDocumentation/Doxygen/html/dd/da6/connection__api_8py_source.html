34def build_connection_active(sender_id: str) -> dict[str, Any] | None:
35 sender_resource = STATE["SENDERS"].get(sender_id)
36 if not sender_resource:
37 return None
38
39 stream_data = next((s for s in STATE["STREAMS"].values() if s.get("sender_id") == sender_id), None)
40 if not stream_data:
41 return None
42
43 sdp_content = stream_data.get("sdp")
44 if not sdp_content:
45 return None
46
47 parsed_sdp = parse_sdp(sdp_content)
48 destination_ip = parsed_sdp.get("ip")
49
50 return {
51 "activation": {
52 "activation_time": now_ts(),
53 "mode": None,
54 "requested_time": None
55 },
56 "master_enable": True,
57 "receiver_id": None,
58 "transport_params": [{
59 "destination_port": parsed_sdp.get("port", 5004),
60 "source_port": parsed_sdp.get("port", 5004),
61 "source_ip": parsed_sdp.get("src_ip", get_ip()),
62 "destination_ip": destination_ip,
63 "rtp_enabled": True
64 }]
65 }

126 async def connect(self, websocket: WebSocket):
127 await websocket.accept()
128 self.active_connections.append(websocket)

131 def disconnect(self, websocket: WebSocket):
132 if websocket in self.active_connections:
133 self.active_connections.remove(websocket)
134 matrix_log("comms", "is07_ws", "disconnect", f"📡 [IS07-WS] Client disconnected. Remaining: {len(self.active_connections)}", "INFO")

136 async def broadcast(self, message: str):
137 for connection in self.active_connections:
138 try:
139 await connection.send_text(message)
