28def smooth_data(data: list[float], window_size: int) -> list[float]:
29 """Applies a simple moving average smoothing to the data."""
30 if window_size <= 1 or len(data) < (window_size // 2):
31 return data
32 window = np.ones(window_size) / window_size
33 smoothed = np.convolve(data, window, mode='same')
34 return smoothed.tolist()

36def update_line_data(line: Any, x_data: deque, y_data: deque, new_x: float, new_y: float, smoothing: int = 0):
37 """Updates line data with optional smoothing."""
38 x_data.append(new_x)
39 y_data.append(new_y)
40 if smoothing > 1 and len(y_data) >= smoothing:
41 y_plot = smooth_data(list(y_data), smoothing)
42 line.set_data(list(x_data), y_plot)
43 else:
44 line.set_data(list(x_data), list(y_data))

81def perform_fast_blit(ax: Any, canvas: Any, lines: list[Any]):
