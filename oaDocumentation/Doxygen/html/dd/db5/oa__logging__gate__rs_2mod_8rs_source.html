5// Description: High-speed native logging filter. Intercepts Python logger
6// calls to determine if a message should reach a sink based on a global
7// matrix. This offloads string-formatting and log-IO from the Python side
8// for disabled trace levels.
9
10use pyo3::prelude::*;
11use dashmap::DashMap;
12use once_cell::sync::Lazy;
13use std::sync::atomic::{AtomicBool, Ordering};
14
15// AtomicBool is used for the master toggle to ensure thread-safe, zero-lock
16// evaluation during high-frequency log bursts (e.g., protocol firehose).
17static MASTER_LOG_ENABLED: AtomicBool = AtomicBool::new(true);
18
19// DashMap provides lock-free concurrent reads for system-specific gates,
20// allowing multiple threads to evaluate logging levels simultaneously
21// without contention on the Global Interpreter Lock (GIL).
22static LOGGING_MATRIX: Lazy<DashMap<String, bool>> = Lazy::new(DashMap::new);
23
24#[pyfunction]
25fn set_master_toggle(enabled: bool) {
26 MASTER_LOG_ENABLED.store(enabled, Ordering::SeqCst);
27}
28
29#[pyfunction]
30#[pyo3(signature = (system, element=None, enabled=true))]
31fn set_gate_state(system: String, element: Option<String>, enabled: bool) {
32 let key = match element {
33 Some(element_name) => format!("{}:{}", system, element_name),
34 None => system,
35 };
36 LOGGING_MATRIX.insert(key, enabled);
37}
38
39#[pyfunction]
40#[pyo3(signature = (system, element=None, func_name=None))]
41fn is_debug_allowed(system: String, element: Option<String>, func_name: Option<String>) -> bool {
42 // Master kill-switch evaluation is the fastest path. If the system is
43 // in PRODUCTION mode, all trace/debug logs are dropped here.
44 let _ = func_name; // Reserved for future function-level granular gating.
45 if !MASTER_LOG_ENABLED.load(Ordering::Relaxed) {
46 return false;
47 }
48
49 // Element-specific gating allows for "surgical" debugging of a single
50 // widget or fader without flooding the console with other system traffic.
51 if let Some(element_name) = &element {
52 let key = format!("{}:{}", system, element_name);
53 if let Some(enabled) = LOGGING_MATRIX.get(&key) {
54 return *enabled;
55 }
56 }
57
58 // System-wide fallback ensures that protocol-level logs (e.g., all MIDI)
59 // are controlled if no specific element is targeted.
60 if let Some(enabled) = LOGGING_MATRIX.get(&system) {
61 return *enabled;
62 }
63
64 // Default to enabled ensures no logs are lost if a module hasn't
65 // explicitly registered with the gate.
66 true
67}

70pub fn oalogginggate_rs(m: &Bound<'_, PyModule>) -> PyResult<()> {

72 m.add_function(wrap_pyfunction!(set_gate_state, m)?)?;

74 Ok(())
75}
