10def deep_merge(target, source):
11 """Deep merges source dict into target dict without erasing nested blocks."""
12 for k, v in source.items():
13 if (isinstance(v, dict) and k in target and
14 isinstance(target[k], dict)):
15 deep_merge(target[k], v)
16 else:
17 target[k] = v
18 return target
