10def deep_merge(target, source):
11 """Deep merges source dict into target dict without erasing nested blocks."""
12 for key, value in source.items():
13 if (isinstance(value, dict) and key in target and
14 isinstance(target[key], dict)):
15 deep_merge(target[key], value)
16 else:
17 target[key] = value
18 return target

20def coerce_pixel_size(value, fallback):
21 """Coerce a width/height value to int pixels.
22
23 Percent strings like "100%" mean "fill the parent" — they're a layout
24 directive, not a literal pixel count. The widget_schema_normalizer turns
25 them into Tk sticky bits at the geometry pillar, but some widget creators
26 (graph plotter, composite dials, button toggler) read width/height directly
27 from sub-config blocks that don't go through that normalizer. For those
28 sites, fall back to a sensible default size and let grid sticky/weight
29 handle the actual stretch.
30 """
31 if value is None:
32 return int(fallback) if fallback is not None else 0
33 if isinstance(value, bool):
34 return int(fallback) if fallback is not None else 0
35 if isinstance(value, (int, float)):
36 return int(value)
37 if isinstance(value, str):
38 s = value.strip()
39 if not s or s.endswith("%"):
40 return int(fallback) if fallback is not None else 0
41 try:
42 return int(float(s))
43 except (ValueError, TypeError):
44 return int(fallback) if fallback is not None else 0

72def calculate_sticky(geometry):
73 """Translates Semantic Layout Model (Align/Anchor/Stretch) into Tkinter sticky bits."""
74 sticky_parts = set()
75 stretch = str(geometry.get("stretch", "")).lower()
76 anchor = str(geometry.get("anchor", "")).lower()

79 sticky_parts.update(["e", "w"])
