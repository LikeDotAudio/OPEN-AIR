52class YakTranslator:
53 """
54 Orchestrates the translation of high-level YAK commands into SCPI strings.
55
56 This manager maintains an in-memory repository of command definitions
57 and handles the transformation of MQTT-triggered events into instrument-
58 specific messages.
59
60 Responsibilities:
61 - Command Repository Management: Loads and caches YAK JSON definitions.
62 - SCPI Orchestration: Renders templates into hardware-ready strings.
63 - Transaction Correlation: Tracks IDs to link async instrument replies.
64 - Interface Isolation: Maps UI-neutral YAK paths to Core-level SCPI.
65
66 Constraints:
67 - Operates within the UI Partition (Translation Layer).
68 - Requires 'YAKETY_YAK_REPO_PATH' to be accessible for initialization.
69 - Depends on 'oaComProtocols.oaComMQTT' for trigger ingress.
70 """

99 def _load_yak_repository(self):
100 """
101 Loads YAK command definitions from the filesystem into memory.
102
103 Navigates to the project-defined repository path and ingests the
104 JSON command schema. If missing, attempts to initialize a skeletal
105 defaults file.
106 """
107 repo_path = YAKETY_YAK_REPO_PATH
108
109 if not repo_path.parent.exists():
110 repo_path.parent.mkdir(parents=True, exist_ok=True)
111
112 if repo_path.is_file() and repo_path.stat().st_size > 0:
113 with open(repo_path, "rb") as f:
114 self.yak_repository = orjson.loads(f.read())
115
116 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name,
117 f"📡📥📥 [CONFIG] YAK repository loaded: {repo_path}",
118 level="DEBUG")
119 else:
120 # ⚡ RESILIENCE: Handle missing repository with graceful fallback.
121 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name,
122 "ℹ️ YAK repository missing. Creating default.",
123 level="INFO")
124 self.yak_repository = {}
125 try:
126 with open(repo_path, "wb") as f:
127 f.write(orjson.dumps({}))
128 except Exception as e:
129 logger.error(f"❌ Failed to create default YAK repository: {e}")

141 def _on_yak_trigger_message(self, message: MqttMessage):
142 """
143 Processes incoming MQTT messages to trigger SCPI command generation.
144
145 Parses the topic to identify the command path, retrieves the
146 corresponding declaration, renders the final SCPI string using
147 payload parameters, and dispatches it to the Proxy Tx_Inbox.
148
149 Args:
150 message: The incoming MqttMessage object containing topic and payload.
151
152 Side Effects:
153 - Publishes a translated SCPI payload to 'OPEN-AIR/Proxy/Tx_Inbox'.
154 - Updates the internal context store for correlation.
155 """
156 topic = message.topic
157 payload = message.payload
158
159 # Strip the prefix to resolve the relative YAK path hierarchy.
160 yak_command_path = topic.replace("OPEN-AIR/yak/commands/", "").split("/")
161
162 declaration = self._get_command_declaration(yak_command_path)
163 if not declaration:
164 logger.error(f"❌ Unknown YAK command path: {yak_command_path}")
165 return
166
167 # Handle various incoming formats; prefer pre-parsed dict if available.
168 payload_parsed = orjson.loads(payload) if isinstance(payload, (bytes, str)) else payload
169
170 scpi_template = declaration.get("scpi_template")
171 if not scpi_template:
172 logger.error(f"❌ Missing 'scpi_template' for {yak_command_path}")
173 return
174
175 final_scpi_command = self._render_scpi_command(scpi_template, payload_parsed)
176 if not final_scpi_command:
177 return
178
179 is_query = declaration.get("is_query", False)
180 # Generate a lightweight 16-bit correlation ID for async tracking.
181 correlation_id = f"{random.getrandbits(16):04X}"
182
183 # Persist command context to map instrument responses back to original path.
184 self.command_context_store[correlation_id] = {
185 "path_parts": yak_command_path,
186 "command_details": declaration.get("Outputs", {}),
187 }
188
189 proxy_payload = {
190 "command": final_scpi_command,
191 "query": is_query,
192 "correlation_id": correlation_id,
193 }
194
195 self.mqtt_util.get_client_instance().publish(
196 topic="OPEN-AIR/Proxy/Tx_Inbox",
197 payload=orjson.dumps(proxy_payload).decode(),
198 qos=0,
199 retain=False,
200 )
201
202 matrix_log("UI", "TRANSLATOR", inspect.currentframe().f_code.co_name,
203 f"📡📤📤 [TRANSLATE] SCPI: '{final_scpi_command}' "
204 f"(ID: {correlation_id})", level="DEBUG")

206 def _get_command_declaration(self, path_parts: list):
207 """
208 Navigates the internal repository tree to locate a command node.
209
210 Args:
211 path_parts: Ordered list of keys representing the command path.
212
213 Returns:
214 dict: The command declaration node if found, otherwise None.
215 """
216 node = self.yak_repository
217 for part in path_parts:
218 node = node.get(part)
219 if node is None:
220 return None
221 return node

223 def _render_scpi_command(self, scpi_template: str, params: dict):
224 """
225 Interpolates parameters into a hardware-specific SCPI template.
226
227 Args:
228 scpi_template: Raw SCPI string containing brace-wrapped tokens.
229 params: Key-value pairs for string interpolation.
230
231 Returns:
232 str: The rendered, ready-to-transmit SCPI command.
233 """
234 return scpi_template.format(**params)

236 def retrieve_command_context(self, correlation_id: str):
237 """
238 Retrieves and purges context associated with a specific transaction.
239
240 Used by the RX manager to bridge instrument responses back to the
241 originating YAK command path and UI state updates.
242
243 Args:
244 correlation_id: The unique transaction identifier.
245
246 Returns:
247 dict: The stored context dictionary, or None on ID mismatch.
248 """
249 if correlation_id in self.command_context_store:
250 return self.command_context_store.pop(correlation_id)

252 logger.error(f"❌ Correlation ID mismatch: {correlation_id}")
