14fn process_log_file(file_path: String, max_lines: usize) -> PyResult<String> {
15 let file = File::open(file_path)?;
16 let reader = BufReader::new(file);
17
18 // Collect lines in reverse to easily grab the latest N
19 let mut lines: Vec<String> = reader.lines().filter_map(Result::ok).collect();
20
21 if lines.len() > max_lines {
22 lines = lines.drain((lines.len() - max_lines)..).collect();
23 }
24
25 let mut html_output = String::with_capacity(lines.len() * 300);
26
27 for line in lines {
28 let parts: Vec<&str> = line.split('|').map(|s| s.trim()).collect();
29
30 if parts.len() < 5 {
31 html_output.push_str(&format!(
32 "<div class=\"log-line-raw\" style=\"color: #666; padding: 5px 15px;\">{}</div>",
33 encode_safe(&line)
34 ));
35 continue;
36 }
37
38 let timestamp = encode_safe(parts[0]);
39 let level = parts[1];
40 let system = parts[2];
41 let element = encode_safe(parts[3]);
42 let module = encode_safe(parts[4]);
43
44 let message = if parts.len() > 5 {
45 let message_joined = parts[5..].join(" | ");
46 encode_safe(&message_joined).to_string()
47 } else {
48 "".to_string()
49 };
50
51 let level_class = format!("log-level-{}", level.to_lowercase());
52 let system_class = format!("log-system-{}", system.to_lowercase());
53
54 html_output.push_str(&format!(
55 "<div class=\"log-line\"> \
56 <span class=\"log-col log-timestamp\">{}</span> \
57 <span class=\"log-col log-type {}\">{}</span> \
58 <span class=\"log-col log-system {}\">{}</span> \
59 <span class=\"log-col log-element\">{}</span> \
60 <span class=\"log-col log-module\" title=\"{}\">{}</span> \
61 <span class=\"log-col log-col-message log-message\">{}</span> \
62 </div>",
63 timestamp, level_class, level, system_class, system, element, module, module, message
64 ));
65 }
66
67 Ok(html_output)
68}
