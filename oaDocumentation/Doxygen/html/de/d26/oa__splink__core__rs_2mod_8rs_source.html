38 fn new(handler_configs: &Bound<'_, PyList>) -> PyResult<Self> {

40
41 for config in handler_configs.iter() {
42 let config_dict = config.downcast::<PyDict>()?;
43
44 let enabled: bool = config_dict.get_item("enabled")?
45 .and_then(|v| v.extract().ok())
46 .unwrap_or(false);

50 }
51

53 .and_then(|v| v.extract().ok())
54 .unwrap_or_default();
55
56 let params = config_dict.get_item("params")?
57 .and_then(|v| v.downcast_into::<PyDict>().ok())
58 .unwrap_or_else(|| PyDict::new_bound(config_dict.py()));
59
60 match h_type.as_str() {

65 let d_max = params.get_item("dest_max")?.and_then(|v| v.extract().ok()).unwrap_or(255.0);

67 let d_min_val = params.get_item("dest_min")?;
68 let d_max_val = params.get_item("dest_max")?;
69 let is_int = if let (Some(v1), Some(v2)) = (d_min_val, d_max_val) {
70 v1.is_instance_of::<PyInt>() && v2.is_instance_of::<PyInt>()

72 false

76 },

94 fn process(&self, mut value: Py<PyAny>, _splink: &Bound<'_, PyDict>, state: &Bound<'_, PyDict>, direction: String) -> PyResult<Option<Py<PyAny>>> {
95 let py = state.py();
96
97 for handler in &self.handlers {
98 match handler {
99 HandlerType::Scale { s_min, s_max, d_min, d_max, is_int } => {
100 if let Ok(val_float) = value.extract::<f64>(py) {
101 let (src_in_min, src_in_max, dest_out_min, dest_out_max) = if direction == "REVERSE" {
102 (*d_min, *d_max, *s_min, *s_max)
103 } else {
104 (*s_min, *s_max, *d_min, *d_max)
105 };
106
107 let in_min = src_in_min.min(src_in_max);
108 let in_max = src_in_min.max(src_in_max);
109 let clamped_val = val_float.max(in_min).min(in_max);
110
111 let input_span = src_in_max - src_in_min;
112 let output_span = dest_out_max - dest_out_min;
113
114 let scaled_value = if input_span == 0.0 {
115 dest_out_min
116 } else {
117 dest_out_min + (((clamped_val - src_in_min) / input_span) * output_span)
118 };
119
120 if *is_int {
121 value = (scaled_value.round() as i64).into_py(py);
122 } else {
123 value = scaled_value.into_py(py);
124 }
125 }
126 },
127 HandlerType::Invert { min, max } => {
128 if let Ok(val_bool) = value.extract::<bool>(py) {
129 value = (!val_bool).into_py(py);
130 } else if let Ok(val_float) = value.extract::<f64>(py) {
131 let inverted_val = (max + min) - val_float;
132 // Use Bound for downcasting to check for int
133 let value_bound = value.bind(py);
134 if value_bound.is_instance_of::<PyInt>() {
135 value = (inverted_val.round() as i64).into_py(py);
136 } else {
137 value = inverted_val.into_py(py);
138 }
139 }
140 },
141 HandlerType::Deadband { threshold_percent, max_value } => {
142 let last_val_opt = state.get_item("last_deadband_value")?;
143
144 if let Some(last_val_any) = last_val_opt {
145 if let (Ok(val_float), Ok(last_val_float)) = (value.extract::<f64>(py), last_val_any.extract::<f64>()) {
146 let change_percent = if *max_value == 0.0 {
147 if val_float == last_val_float { 0.0 } else { 100.0 }
148 } else {
149 ((val_float - last_val_float).abs() / max_value) * 100.0
150 };
151
152 if change_percent < *threshold_percent {
153 return Ok(None); // Drop
154 }
155 } else if value.bind(py).to_string() == last_val_any.to_string() {
156 return Ok(None); // Drop
157 }
158 }
159 state.set_item("last_deadband_value", &value)?;
160 },
161 }
162 }
163
164 Ok(Some(value))
165 }
166}
