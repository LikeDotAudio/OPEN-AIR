22 def register_for_bg_sync(self, widget, canvas, configuration, context):
23 """
24 Registers a widget for automatic background synchronization with its parent.
25 """
26 builder_instance = context.builder_instance if context else None
27 parent = widget.master
28 widget_name = getattr(widget, 'path', type(widget).__name__)
29
30 def perform_sync(event=None):
31 if not widget.winfo_exists(): return
32
33 try:
34 # ⚡ LIGHTWEIGHT CHECK: Ensure widget is physically realized
35 w, h = widget.winfo_width(), widget.winfo_height()
36 if w <= 1 or h <= 1: return
37 except tk.TclError: return
38
39 try:
40 # ⚡ COLOR SYNC: Match parent's background color
41 p_bg = parent.cget("bg")
42 if widget.cget("bg") != p_bg:
43 widget.configure(bg=p_bg)
44 if canvas and canvas.winfo_exists() and canvas.cget("bg") != p_bg:
45 canvas.configure(bg=p_bg)
46
47 # ⚡ SLICING: Use stored slice method if available
48 # Note: Transparent widgets are already registered with the builder's
49 # centralized registry via EngineVisualEffects.
50 if hasattr(widget, '_perform_background_slice'):
51 widget._perform_background_slice()
52
53 except Exception as e:
54 matrix_log("UI", "GUI_MANAGER", "perform_sync", f"perform_sync error for {widget_name}: {e}", level="TRACE")
55
56 # Sync immediately once the widget is mapped to the screen
57 widget.bind("<Map>", perform_sync, add="+")
