217 def _get_color(self, norm_val):
218 MAX_RGB_VAL = 255
219 COLOR_SPLIT_THRESHOLD = 0.5
220 if norm_val < COLOR_SPLIT_THRESHOLD:
221 red = max(0, min(255, int(MAX_RGB_VAL * (norm_val * 2))))
222 green, blue = MAX_RGB_VAL, 0
223 else:
224 red = MAX_RGB_VAL
225 green = max(0, min(255, int(MAX_RGB_VAL * (1.0 - (norm_val - COLOR_SPLIT_THRESHOLD) * 2))))
226 blue = 0
227 return f"#{red:02x}{green:02x}{blue:02x}"

230 def _get_rgb_mixed_color(self):
231 MIN_RGB_CHANNELS = 3
232 if len(self.child_values) < MIN_RGB_CHANNELS: return "#888888"
233
234 val_range = self.max_val - self.min_val
235 normalize = lambda v: (self._safe_get(v) - self.min_val) / val_range if val_range else 0
236
237 MAX_RGB_VAL = 255
238 red = max(0, min(255, int(normalize(self.child_values[0]) * MAX_RGB_VAL)))
239 green = max(0, min(255, int(normalize(self.child_values[1]) * MAX_RGB_VAL)))
240 blue = max(0, min(255, int(normalize(self.child_values[2]) * MAX_RGB_VAL)))
241 return f"#{red:02x}{green:02x}{blue:02x}"

243 def _calculate_smart_interval(self, val_range):
244 DIVISOR = 10
245 raw_interval = val_range / DIVISOR
246 exponent = math.floor(math.log10(raw_interval))
247 fractional_part = raw_interval / (10**exponent)
248
249 if fractional_part < 1.5: snap_val = 1
250 elif fractional_part < 3.5: snap_val = 2
251 elif fractional_part < 7.5: snap_val = 5
252 else: snap_val = 10
253 return snap_val * (10**exponent)
