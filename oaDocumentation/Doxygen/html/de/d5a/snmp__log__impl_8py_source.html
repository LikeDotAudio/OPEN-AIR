153 def on_snmp_traffic(self, direction, oid, value, topic, metadata=None):
154 # Handle both periodic dumps and real-time changes
155 if direction not in ["TX_DUMP", "RX", "RX_SET"]: return
156
157 # ? BATCHING: Buffer updates to avoid flooding the event loop
158 self._update_buffer.append((oid, value, topic, metadata))
159
160 if not self._update_scheduled:
161 self._update_scheduled = True
162 # Process buffer every 100ms for responsiveness without lag
163 self.after(100, self._process_update_buffer_process_update_buffer)
164
165 def _process_update_buffer(self):
166 if not self._update_buffer:
167 self._update_scheduled = False
168 return
169
170 # ? CHUNKING: Process at most 100 items at a time to keep UI responsive
171 chunk_size = 100
172 chunk = self._update_buffer[:chunk_size]
173 self._update_buffer = self._update_buffer[chunk_size:]
174
175 for oid, value, topic, metadata in chunk:
176 self._update_oid_state(oid, value, topic, metadata)
177
178 # Update Counter once per batch
179 self.counter_var.set(f"Total Objects: {len(self._last_values)}")
180
181 # If more items remain, schedule another pass quickly
182 if self._update_buffer:
183 self.after(20, self._process_update_buffer_process_update_buffer)
184 else:
185 self._update_scheduled = False

187 def _update_oid_state(self, oid, value, topic, metadata):
188 prev_val = self._last_values.get(oid, None)
189
190 has_changed = (prev_val is not None and str(prev_val) != str(value))
191 is_new = (prev_val is None)
192
193 if not is_new and not has_changed:
194 # ? OPTIMIZATION: Only update tag if it's not already 'stale'
195 if oid in self._oid_to_item:
196 if self._item_tags.get(oid) != "stale":
197 item_id = self._oid_to_item[oid]
198 self.tree.item(item_id, tags=("stale",))
199 self._item_tags[oid] = "stale"
200 return
201
202 # Update State
203 self._last_values[oid] = value
204 if metadata:
205 self._oid_metadata[oid] = metadata
206
207 # Update UI
208 if oid in self._oid_to_item:
209 item_id = self._oid_to_item[oid]
210
211 # Update values if changed
212 if has_changed:
213 self.tree.item(item_id, values=(oid, value, prev_val or "-", topic or "-"))
214 # ? MOVE TO TOP and highlight green on change
215 # Only move if not already at the top to save CPU
216 if self.tree.index(item_id) != 0:
217 self.tree.move(item_id, "", 0)
218
219 self.tree.item(item_id, tags=("changed",))
220 self._item_tags[oid] = "changed"
221 else:
222 # ? NEW DISCOVERY: First Cycle Highlight (Yellow BG, Red Text)
223 item_id = self.tree.insert("", 0, values=(oid, value, "-", topic or "-"), tags=("new_discovery",))
224 self._oid_to_item[oid] = item_id
225 self._item_tags[oid] = "new_discovery"
