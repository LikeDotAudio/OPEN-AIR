14def post_resource(registrar_url, resource_type, resource_data, timeout=2):
15 """
16 Posts a resource to the NMOS registry.
17 """
18 global _registrar_available
19 resource_url = f"{registrar_url}/resource"
20 payload = {"type": resource_type, "data": resource_data}
21
22 try:
23 response = requests.post(resource_url, json=payload, timeout=timeout)
24 if response.status_code not in (200, 201):
25 print(f"[RegistrationManager] ERROR posting {resource_type}: {response.status_code} - {response.text}")
26 return False
27 else:
28 if not _registrar_available:
29 print(f"[RegistrationManager] Connectivity restored to {registrar_url}")
30 _registrar_available = True
31 print(f"[RegistrationManager] Successfully posted {resource_type} (ID: {resource_data.get('id', 'N/A')})")
32 return True
33 except (requests.exceptions.ConnectionError, requests.exceptions.ConnectTimeout):
34 if _registrar_available:
35 print(f"⚠️ [RegistrationManager] Registrar unreachable at {registrar_url}. NMOS Registration is suspended.")
36 _registrar_available = False
37 return False
38 except Exception as e:
39 print(f"[RegistrationManager] Unexpected error posting {resource_type}: {e}")
40 return False

57 node_resource (dict): The NMOS Node resource payload.

60 flows_dict (dict): Dictionary of NMOS Flow resources.
