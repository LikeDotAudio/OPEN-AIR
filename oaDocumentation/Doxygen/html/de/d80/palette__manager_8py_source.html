67 def _on_panel_dropdown_select(self, event):
68 """Event handler for manual panel selection via dropdown."""
69 name = self.selected_panel_var.get()
70 panel_widget = self._find_widget_by_name(name)
71 if panel_widget:
72 self.visibility_service.toggle_visibility(name, panel_widget)
73 self.selected_panel_var.set("")

75 def _find_widget_by_name(self, name):
76 """Helper to retrieve a panel widget from the configuration."""
77 for p in self.tool_panels:

79 return p["widget"]
80 return None
