22def fetch_all_senders(registrar_url):
23 """
24 Fetches all senders from the NMOS registry's query API with pagination.
25
26 Args:
27 registrar_url (str): The base URL of the NMOS registration API.
28
29 Returns:
30 list: A list of sender resource objects.
31 """
32 all_senders = []
33 query_url = registrar_url.replace("registration", "query")
34
35 limit = 100
36 until = None
37
38 while True:
39 url = f"{query_url}/senders?paging.limit={limit}&paging.order=update"
40
41 if until:
42 url += f"&paging.until={until}"
43
44 print(f"[SenderCache] Fetching senders: GET {url}")
45
46 try:
47 r = requests.get(url, timeout=2)
48 if r.status_code != 200:
49 print(f"[SenderCache] ERROR fetching senders: {r.status_code} - {r.text}")
50 break
51
52 data = r.json()
53 print(f"[SenderCache] Received {len(data)} senders in this page.")
54
55 if not data:
56 break
57
58 all_senders.extend(data)
59
60 link_header = r.headers.get("Link")
61 if not link_header:
62 break
63
64 # Parse Link header for next page
65 next_link_parts = [part.strip() for part in link_header.split(',')]
66 next_url_info = None
67 for part in next_link_parts:
68 if 'rel="next"' in part:
69 next_url_info = part.split(';')[0].strip('<>')
70 break
71
72 if next_url_info and "paging.until=" in next_url_info:
73 until = next_url_info.split("paging.until=")[1]
74 else:
75 break # No next page found or invalid format
76
77 except requests.exceptions.RequestException as e:
78 print(f"[SenderCache] Network error during sender fetch: {e}")
79 break
80 except Exception as e:
81 print(f"[SenderCache] Unexpected error during sender fetch: {e}")
82 break
83
84 print(f"[SenderCache] Total senders fetched: {len(all_senders)}")
85 return all_senders

88def rebuild_sender_key_cache(registrar_url):
89 """
90 Fetches all senders, parses their manifest SDPs, and builds a cache
91 mapping SDP-derived keys to sender resources.
92
93 Args:
94 registrar_url (str): The base URL of the NMOS registration API.
95 """
96 global SENDER_CACHE, SENDER_KEY_CACHE, CACHE_TS
97
98 print("[SenderCache] Rebuilding sender key cache...")
99
100 SENDER_CACHE = fetch_all_senders(registrar_url)
101 SENDER_KEY_CACHE = {}
102
103 for sender_resource in SENDER_CACHE:
104 sender_id = sender_resource.get("id")
105 manifest_url = sender_resource.get("manifest_href")
106
107 if not sender_id or not manifest_url:
108 continue
109
110 try:
111 # Fetch manifest (SDP)
112 response = requests.get(manifest_url, timeout=2)
113 if response.status_code != 200:
114 print(f"[SenderCache] Failed to fetch manifest for sender {sender_id}: {response.status_code}")
115 continue
116
117 sdp_content = response.text
118 parsed_sdp_data = parse_sdp(sdp_content)
119 match_key = build_match_key(parsed_sdp_data)
120
121 if match_key:
122 SENDER_KEY_CACHE[match_key] = sender_resource
123 print(f"[SenderCache] Mapped sender '{sender_id}' to key '{match_key}'")
124 else:
125 print(f"[SenderCache] Could not generate match key for sender '{sender_id}' from manifest.")
126
127 except requests.exceptions.RequestException as e:
128 print(f"[SenderCache] Network error fetching manifest for sender {sender_id}: {e}")
129 except Exception as e:
130 print(f"[SenderCache] Error processing manifest for sender {sender_id}: {e}")
131
132 CACHE_TS = time.time()
133 print(f"[SenderCache] Sender key cache rebuilt with {len(SENDER_KEY_CACHE)} entries.")
134

136def get_existing_senders_map(registrar_url):
137 """
138 Retrieves the sender cache, rebuilding it if it's expired.
139
140 Args:
141 registrar_url (str): The base URL of the NMOS registration API.
142
143 Returns:
144 dict: The map of sender keys to sender resources.
145 """
146 global CACHE_TS
147
148 current_time = time.time()
149 if current_time - CACHE_TS > CACHE_TTL:
150 print("[SenderCache] Cache expired, rebuilding...")
151 rebuild_sender_key_cache(registrar_url)
152 else:
153 print("[SenderCache] Using existing cached sender map.")
154
155 return SENDER_KEY_CACHE

158def find_existing_sender(sdp_content, registrar_url):
159 """
160 Finds an existing NMOS sender resource that matches the provided SDP content.
161
162 Args:
163 sdp_content (str): The SDP content of the incoming stream.
164 registrar_url (str): The base URL of the NMOS registration API.
165
166 Returns:
167 dict or None: The sender resource if found, otherwise None.
168 """
169 parsed_sdp_data = parse_sdp(sdp_content)
170 match_key = build_match_key(parsed_sdp_data)
171
172 if not match_key:
173 print("[SenderCache] Could not generate match key for SDP.")
174 return None
175
176 print(f"[SenderCache] Attempting to match SDP with key: '{match_key}'")
177 sender_map = get_existing_senders_map(registrar_url)
178
179 sender_resource = sender_map.get(match_key)
180
181 if sender_resource:
182 print(f"[SenderCache] FOUND existing sender: {sender_resource.get('id')}")
183 else:
184 print("[SenderCache] No existing sender found matching the SDP.")
185
186 return sender_resource
