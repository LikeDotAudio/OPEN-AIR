19def smooth_data(data: list[float], window_size: int) -> list[float]:
20 """Applies a simple moving average smoothing to the data."""
21 if window_size <= 1 or len(data) < (window_size // 2):
22 return data
23 window = np.ones(window_size) / window_size
24 smoothed = np.convolve(data, window, mode='same')
25 return smoothed.tolist()
