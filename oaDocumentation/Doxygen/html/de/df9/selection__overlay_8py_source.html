126 def _find_drop_target(self, root_x, root_y):
127 """Recursively finds the best drop container and insertion coordinate."""
128 top = self.widget.winfo_toplevel()
129 target_widget = top.winfo_containing(root_x, root_y)
130 if not target_widget: return None
131
132 # Resolve the widget under the mouse to its _oca_path
133 curr = target_widget
134 path = None
135 while curr:
136 if hasattr(curr, '_oca_path'):
137 path = curr._oca_path
138 break
139 curr = curr.master
140
141 if not path: return None
142
143 val = state_manager.get_value_at_path(path)
144 if not isinstance(val, dict):
145 # Leaf -> Move to parent container
146 path = ".".join(path.split(".")[:-1])
147 val = state_manager.get_value_at_path(path)
148 # Update curr to represent the container
149 temp = curr
150 while temp:
151 if getattr(temp, '_oca_path', None) == path:
152 curr = temp; break
153 temp = temp.master
154
155 if not isinstance(val, dict): return None
156
157 # Check if we are over a container (Block/Bin)
158 w_type = val.get("type", "")
159 if w_type in ["OcaBlock", "OcaBin", "OcaArray"]:
160 # Container -> Find which child we are near for insertion, or append
161 # For now, simple "append to container" visual
162 ox = self.workspace.render_area.winfo_rootx()
163 oy = self.workspace.render_area.winfo_rooty()
164 wx1 = curr.winfo_rootx(); wy1 = curr.winfo_rooty()
165 ww = curr.winfo_width(); wh = curr.winfo_height()
166
167 # Draw line near the bottom to indicate "append"
168 return (curr, path, "append", (wx1-ox, wy1+wh-5-oy, wx1+ww-ox, wy1+wh-5-oy))
169 else:
170 # Over a non-container widget -> Find its container and insert relative
171 container_path = ".".join(path.split(".")[:-1])
172 container_val = state_manager.get_value_at_path(container_path)
173
174 if isinstance(container_val, dict) and container_val.get("type") in ["OcaBlock", "OcaBin", "OcaArray"]:
175 cx1 = curr.winfo_rootx(); cy1 = curr.winfo_rooty()
176 cw = curr.winfo_width(); ch = curr.winfo_height()
177 ox = self.workspace.render_area.winfo_rootx(); oy = self.workspace.render_area.winfo_rooty()
178
179 if root_y < cy1 + (ch // 2):
180 return (curr.master, container_path, "before", (cx1-ox, cy1-oy, cx1+cw-ox, cy1-oy))
181 else:
182 return (curr.master, container_path, "after", (cx1-ox, cy1+ch-oy, cx1+cw-ox, cy1+ch-oy))
183
184 return None
