38 def get_section_flags(self, section_name):
39 """
40 Returns a dictionary of all boolean flags in a specific section.
41 """
42 self._ensure_config_exists()
43 flags = {}
44 if section_name in self.config:
45 section = self.config[section_name]
46 for key in section:
47 # Skip the list-based keys in DEBUG_MATRIX
48 if section_name == "DEBUG_MATRIX" and key.lower() in ["mute_functions", "force_functions"]:
49 continue
50 try:
51 flags[key] = self.config.getboolean(section_name, key)
52 except ValueError:
53 pass
54 return flags

56 def set_config_flag(self, section, key, value):
57 """
58 Surgically updates a specific flag in any section while preserving comments.
59 """
60 try:
61 with open(self.config_path) as f:
62 lines = f.readlines()
63 except Exception as e:
64 logger.error(f"❌ Failed to read config.ini: {e}")
65 return False
66
67 new_lines = []
68 current_section = None
69 key_found = False
70 val_str = "True" if value is True else "False" if value is False else str(value)
71
72 for line in lines:
73 stripped = line.strip()
74
75 # Identify section
76 if stripped.startswith('[') and stripped.endswith(']'):
77 current_section = stripped[1:-1].strip()
78 new_lines.append(line)
79 continue
80
81 # If in the correct section, look for the key
82 if current_section == section and not stripped.startswith('#') and '=' in stripped:
83 line_key, line_val = stripped.split('=', 1)
84 if line_key.strip().lower() == key.lower():
85 # Preserve existing indentation and comments on the same line
86 indent = line[:line.find(line_key)]
87 comment = ""
88 if '#' in line_val:
89 comment = " #" + line_val.split('#', 1)[1].rstrip()
90
91 new_lines.append(f"{indent}{line_key.strip()} = {val_str}{comment}\n")
92 key_found = True
93 continue
94
95 new_lines.append(line)
96
97 # If the key didn't exist in the section, add it at the end of the section
98 if not key_found:
99 insert_pos = -1
100 found_section = False
101 for i, line in enumerate(new_lines):
102 if line.strip() == f"[{section}]":
103 found_section = True
104 continue
105 if found_section and line.strip().startswith('['):
106 insert_pos = i
107 break
108
109 if insert_pos == -1: # End of file or section was the last one
110 new_lines.append(f"{key} = {val_str}\n")
111 else:
112 new_lines.insert(insert_pos, f"{key} = {val_str}\n")
113
114 try:
115 with open(self.config_path, 'w') as f:
116 f.writelines(new_lines)
117 matrix_log("core", "system", inspect.currentframe().f_code.co_name if "inspect" in globals() else "unknown", f"💾 [CONFIG] Surgically updated [{section}] {key} to {value}", "SUCCESS")
118
119 # Sync the internal configparser for immediate read consistency
120 self.config.read(self.config_path)
121 return True
122 except Exception as e:
123 logger.error(f"❌ Failed to write config.ini: {e}")
124 return False
