7def detect_visual_layout_folds(scroll_frame):
8 """
9 Scans child widgets for components defining a physical layout fold/separator.
10 Returns a sorted list of fold metadata (position_pct, orientation).
11 """
12 folds = []
13 if not scroll_frame or not scroll_frame.winfo_exists():
14 return folds
15
16 scroll_root_y = scroll_frame.winfo_rooty()
17 view_height = scroll_frame.winfo_height()
18
19 if view_height <= 0:
20 return folds
21
22 for child in scroll_frame.winfo_children():
23 path = getattr(child, '_oca_path', '')
24 if any(s in path for s in ['Fold', 'fold', 'Separator']):
25 # Calculate midpoint relative to scrollable area
26 mid_y = child.winfo_rooty() + (child.winfo_height() / 2) - scroll_root_y
27 folds.append({
28 "position_pct": mid_y / view_height,
29 "orientation": "horizontal"
30 })
31
32 # Ensure deterministic order
33 folds.sort(key=lambda x: x["position_pct"])
34 return folds
