98 fn get_bezel_points<'py>(&self, py: Python<'py>, cx: f64, cy: f64, w: f64, h: f64, shape: &str, line_width: f64, shrink_px: f64) -> PyResult<(Bound<'py, PyList>, bool)> {
99 let (radius, global_y_shift, shape_key) = self.get_scaling_params(w, h, shape, line_width);
100
101 let m_w = self.get_multiplier_w(&shape_key);
102 let m_h = self.get_multiplier_h(&shape_key);
103 let mut adj_radius = radius - (shrink_px / m_w.max(m_h));
104 if adj_radius < 1.0 { adj_radius = 1.0; }
105
106 let (pts, is_smooth) = match shape_key.as_str() {
107 "gem" => self._get_gem(adj_radius, global_y_shift),
108 "super_gem" => self._get_super_gem(adj_radius, global_y_shift),
109 "parking_meter" => self._get_parking_meter(adj_radius, global_y_shift),
110 "octagon" => self._get_octagon(adj_radius, global_y_shift),
111 "triangle" => self._get_triangle(adj_radius, global_y_shift),
112 "pyramid" => self._get_pyramid(adj_radius, global_y_shift),
113 "cylinder" | "hotdog" => self._get_hotdog_cylinder(adj_radius, global_y_shift, &shape_key),
114 "hex" => self._get_hex(adj_radius, global_y_shift),
115 "squectangle" => self._get_squectangle(adj_radius, global_y_shift),
116 "squimonde" => self._get_squimonde(adj_radius, global_y_shift),
117 "squircle" => self._get_squircle(adj_radius, global_y_shift),
118 "trapezoid" | "badge" => self._get_trapezoid(adj_radius, global_y_shift),
119 "crest" => self._get_crest(adj_radius, global_y_shift),
120 "stereo_diamond" => self._get_stereo_diamond(adj_radius, global_y_shift),
121 "intersecting_overlay" => self._get_intersecting_overlay(adj_radius, global_y_shift),
122 _ => (Vec::new(), false),
123 };
124
125 let flat_pts = PyList::empty_bound(py);
126 for (px, py_val) in pts {
127 let _ = flat_pts.append(cx + px);
128 let _ = flat_pts.append(cy - py_val);
129 }
130
131 Ok((flat_pts, is_smooth))
132 }

134 fn calculate_shadow_geometry<'py>(&self, py: Python<'py>, cx: f64, cy: f64, config: &Bound<'py, PyDict>) -> PyResult<Bound<'py, PyDict>> {
135 let value: f64 = config.get_item("value")?.unwrap().extract()?;
136 let min_val: f64 = config.get_item("min_val")?.unwrap().extract()?;
137 let max_val: f64 = config.get_item("max_val")?.unwrap().extract()?;
138 let start_angle_deg: f64 = config.get_item("start_angle_deg")?.unwrap().extract()?;
139 let end_angle_deg: f64 = config.get_item("end_angle_deg")?.unwrap().extract()?;
140 let extent_deg: f64 = config.get_item("extent_deg")?.unwrap().extract()?;
141 let main_arc_radius: f64 = config.get_item("main_arc_radius")?.unwrap().extract()?;
142 let text_offset_from_arc: f64 = config.get_item("text_offset_from_arc")?.unwrap().extract()?;
143 let style: String = config.get_item("style")?.unwrap().extract()?;
144 let thick: f64 = config.get_item("thick")?.unwrap().extract()?;
145 let counter_clockwise: bool = config.get_item("counter_clockwise")?.unwrap().extract()?;
146 let pivot_size: f64 = config.get_item("pivot_size")?.unwrap().extract()?;
147 let needle_scale: f64 = config.get_item("needle_scale")?.map(|v| v.extract().unwrap_or(1.0)).unwrap_or(1.0);
148
149 let bounded_val = value.clamp(min_val, max_val);
150 let range_val = max_val - min_val;
151 let norm_val = if range_val != 0.0 { (bounded_val - min_val) / range_val } else { 0.0 };
152
153 let angle_deg = if counter_clockwise {
154 end_angle_deg + (norm_val * extent_deg)
155 } else {
156 start_angle_deg - (norm_val * extent_deg)
157 };
158
159 let angle_rad = angle_deg * PI / 180.0;
160 let length = (main_arc_radius + text_offset_from_arc - 2.0) * needle_scale;
161
162 let tip_x = cx + length * angle_rad.cos();
163 let tip_y = cy - length * angle_rad.sin();
164
165 let result = PyDict::new_bound(py);
166 let coords_list = PyList::empty_bound(py);
167
168 let get_shadow_pt = |px: f64, py_val: f64| -> (f64, f64) {
169 let d = ((px - cx).powi(2) + (py_val - cy).powi(2)).sqrt();
170 let factor = (d / length).clamp(0.0, 1.0);
171 (px + MAX_SHADOW_X * factor, py_val + MAX_SHADOW_Y * factor)
172 };
173
174 match style.as_str() {
175 "teardrop" | "spade" => {
176 let d1 = length * 0.75;
177 let d2 = length * 0.875;
178 let p1x = cx + d1 * angle_rad.cos();
179 let p1y = cy - d1 * angle_rad.sin();
180 let p2x = cx + d2 * angle_rad.cos();
181 let p2y = cy - d2 * angle_rad.sin();
182 let bulb_w = thick * 2.5;
183 let perp_angle = angle_rad + (PI / 2.0);
184 let bx = cx + (d1 - thick) * angle_rad.cos();
185 let by = cy - (d1 - thick) * angle_rad.sin();
186 let mid_dist = (d1 + d2) / 2.0;
187 let mx = cx + mid_dist * angle_rad.cos();
188 let my = cy - mid_dist * angle_rad.sin();
189 let s1x = mx + bulb_w * perp_angle.cos();
190 let s1y = my - bulb_w * perp_angle.sin();
191 let s2x = mx - bulb_w * perp_angle.cos();
192 let s2y = my + bulb_w * perp_angle.sin();
193
194 let sc = get_shadow_pt(cx, cy);
195 let sp1 = get_shadow_pt(p1x, p1y);
196 let sb = get_shadow_pt(bx, by);
197 let ss1 = get_shadow_pt(s1x, s1y);
198 let sp2 = get_shadow_pt(p2x, p2y);
199 let ss2 = get_shadow_pt(s2x, s2y);
200 let stip = get_shadow_pt(tip_x, tip_y);
201
202 result.set_item("line1", vec![sc.0, sc.1, sp1.0, sp1.1])?;
203 result.set_item("poly", vec![sb.0, sb.1, ss1.0, ss1.1, sp2.0, sp2.1, ss2.0, ss2.1])?;
204 result.set_item("line2", vec![sp2.0, sp2.1, stip.0, stip.1])?;
205 result.set_item("type", "teardrop")?;
206 },
207 "knife-edge" | "taper" => {
208 let perp_angle = angle_rad + (PI / 2.0);
209 let base_rad = if style == "taper" { pivot_size / 2.0 } else { thick * 1.5 };
210 let bx1 = cx + base_rad * perp_angle.cos();
211 let by1 = cy - base_rad * perp_angle.sin();
212 let bx2 = cx - base_rad * perp_angle.cos();
213 let by2 = cy + base_rad * perp_angle.sin();
214
215 let sb1 = get_shadow_pt(bx1, by1);
216 let sb2 = get_shadow_pt(bx2, by2);
217 let stip = get_shadow_pt(tip_x, tip_y);
218
219 let _ = coords_list.append(sb1.0); let _ = coords_list.append(sb1.1);
220 let _ = coords_list.append(stip.0); let _ = coords_list.append(stip.1);
221 let _ = coords_list.append(sb2.0); let _ = coords_list.append(sb2.1);
222 result.set_item("coords", coords_list)?;
223 result.set_item("type", "polygon")?;
224 },
225 "baton" => {
226 let perp_angle = angle_rad + (PI / 2.0);
227 let ox = (thick / 2.0) * perp_angle.cos();
228 let oy = (thick / 2.0) * perp_angle.sin();
229
230 let s1 = get_shadow_pt(cx + ox, cy - oy);
231 let s2 = get_shadow_pt(tip_x + ox, tip_y - oy);
232 let s3 = get_shadow_pt(tip_x - ox, tip_y + oy);
233 let s4 = get_shadow_pt(cx - ox, cy + oy);
234
235 let _ = coords_list.append(s1.0); let _ = coords_list.append(s1.1);
236 let _ = coords_list.append(s2.0); let _ = coords_list.append(s2.1);
237 let _ = coords_list.append(s3.0); let _ = coords_list.append(s3.1);
238 let _ = coords_list.append(s4.0); let _ = coords_list.append(s4.1);
239 result.set_item("coords", coords_list)?;
240 result.set_item("type", "polygon")?;
241 },
242 _ => { // line
243 let sc = get_shadow_pt(cx, cy);
244 let stip = get_shadow_pt(tip_x, tip_y);
245 let _ = coords_list.append(sc.0); let _ = coords_list.append(sc.1);
246 let _ = coords_list.append(stip.0); let _ = coords_list.append(stip.1);
247 result.set_item("coords", coords_list)?;
248 result.set_item("type", "line")?;
249 }
250 }
251
252 Ok(result)
253 }
254}

257 fn get_scaling_params(&self, w: f64, h: f64, shape: &str, line_width: f64) -> (f64, f64, String) {
258 let mut shape_key = shape.to_lowercase();
259 if !self.has_multiplier(&shape_key) {
260 if shape_key == "triangle" || shape_key == "pyramid" { shape_key = "triangle".to_string(); }
261 else if shape_key == "cylinder" || shape_key == "hotdog" { shape_key = if shape_key == "hotdog" { "hotdog".to_string() } else { "cylinder".to_string() }; }
262 else if shape_key == "trapezoid" || shape_key == "badge" { shape_key = "trapezoid".to_string(); }
263 else { shape_key = "default".to_string(); }
264 }
265
266 let m_w = self.get_multiplier_w(&shape_key);
267 let m_h = self.get_multiplier_h(&shape_key);
268 let y_shift_factor = self.get_y_shift(&shape_key);
269
270 let avail_w = (w / 2.0) - (line_width / 2.0) - SAFE_MARGIN;
271 let avail_h = h - (line_width / 2.0) - SAFE_MARGIN;
272
273 let radius = (avail_w / m_w).min(avail_h / m_h);
274 let global_y_shift = y_shift_factor * radius;
275
276 (radius, global_y_shift, shape_key)
277 }

338 fn _get_parking_meter(&self, radius: f64, global_y_shift: f64) -> (Vec<(f64, f64)>, bool) {
339 let pm_rad = radius * PARKING_METER_BEZEL_EXPANSION;
340 let w_val = TRIANGLE_BASE_WIDTH * pm_rad;
341 let h_val = TRIANGLE_PEAK_HEIGHT * pm_rad;
342 let arc_radius = (w_val.powi(2) + h_val.powi(2)).sqrt();
343 let ang_start = h_val.atan2(w_val);
344 let ang_end = h_val.atan2(-w_val);
345 let mut pts = vec![(0.0, 0.0 + global_y_shift)];
346 let steps = 20;
347 for i in 0..=steps {
348 let ang = ang_start + (ang_end - ang_start) * (i as f64 / steps as f64);
349 let px = arc_radius * ang.cos();
350 let py = arc_radius * ang.sin() + global_y_shift;
351 pts.push((px, py));
352 }
353 (pts, false)
354 }

388 fn _get_hotdog_cylinder(&self, radius: f64, global_y_shift: f64, shape_key: &str) -> (Vec<(f64, f64)>, bool) {
389 let (w_straight, r_cap, cap_center_y) = if shape_key == "hotdog" {
390 (HOTDOG_WIDTH_STRAIGHT * radius, HOTDOG_CAP_RADIUS * radius, HOTDOG_CAP_CENTER_Y * radius)
391 } else {
392 (CYLINDER_WIDTH_STRAIGHT * radius, CYLINDER_CAP_RADIUS * radius, CYLINDER_CAP_CENTER_Y * radius)
393 };
394
395 let mut pts = vec![(0.0, 0.0 + global_y_shift), (w_straight, 0.0 + global_y_shift)];
396 let steps = CYLINDER_STEPS;
397 for i in 0..=steps {
398 let ang = (-90.0 + (180.0 * i as f64 / steps as f64)) * PI / 180.0;
399 let px = w_straight + r_cap * ang.cos();
400 let py = cap_center_y + r_cap * ang.sin();
401 pts.push((px, py + global_y_shift));
402 }
403 for i in 0..=steps {
404 let ang = (90.0 + (180.0 * i as f64 / steps as f64)) * PI / 180.0;
405 let px = -w_straight + r_cap * ang.cos();
406 let py = cap_center_y + r_cap * ang.sin();
407 pts.push((px, py + global_y_shift));
408 }
409 pts.push((0.0, 0.0 + global_y_shift));
410 (pts, false)
411 }

427 fn _get_squectangle(&self, radius: f64, global_y_shift: f64) -> (Vec<(f64, f64)>, bool) {
428 let mut pts = Vec::new();
429 let n = SQUIRCLE_N;
430 let w_sq = SQUECTANGLE_WIDTH_FACTOR * radius;
431 let h_sq = SQUECTANGLE_HEIGHT_FACTOR * radius;
432 let steps = SQUIRCLE_STEPS;
433 for i in 0..=steps {
434 let t = -PI/2.0 + (2.0 * PI * i as f64 / steps as f64);
435 let c = t.cos();
