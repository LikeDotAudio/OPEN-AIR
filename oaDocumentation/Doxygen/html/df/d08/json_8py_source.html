86 def discover_columns(self, depth_limit=3):
87 """Iteratively scans data to find common keys for table view columns."""
88 keys_found = set()
89 stack = [(self.raw_data, 0)]
90
91 while stack:
92 d, depth = stack.pop()
93 if depth > depth_limit: continue
94
95 if isinstance(d, dict):
96 for k, v in d.items():
97 if isinstance(v, dict):
98 for sub_k in v.keys():
99 if not isinstance(v[sub_k], (dict, list)): keys_found.add(sub_k)
100 stack.append((v, depth + 1))
101 elif isinstance(d, list):
102 for item in d:
103 if isinstance(item, dict):
104 for sub_k in item.keys():
105 if not isinstance(item[sub_k], (dict, list)): keys_found.add(sub_k)
106 stack.append((item, depth + 1))
107
108 priority = ["name", "id", "Value", "Start_MHz", "Stop_MHz", "MHz", "channel"]
109 cols = [k for k in priority if k in keys_found]
110 cols += sorted([k for k in keys_found if k not in priority])
111 self.dynamic_columns = cols
112 return cols
