16fn convert_csv_unknown(py: Python<'_>, file_path: String) -> PyResult<(Vec<String>, Py<PyList>)> {
17 let standard_headers = vec!["ZONE", "GROUP", "DEVICE", "NAME", "FREQ_MHZ", "PEAK"];
18 let mut header_aliases = HashMap::new();
19 header_aliases.insert("zone", vec!["zone", "area", "location"]);
20 header_aliases.insert("group", vec!["group", "channel_group"]);
21 header_aliases.insert("device", vec!["device", "dev_type", "model"]);
22 header_aliases.insert("name", vec!["name", "alias", "description"]);
23 header_aliases.insert("freq_mhz", vec!["freq", "frequency", "frequency_mhz", "freq_mhz"]);
24 header_aliases.insert("peak", vec!["peak", "peak_level", "max_level", "dbm"]);
25
26 let mut reader = ReaderBuilder::new()

28 .from_path(file_path)
29 .map_err(|e| PyErr::new::<pyo3::exceptions::PyIOError, _>(e.to_string()))?;

32 .map_err(|e| PyErr::new::<pyo3::exceptions::PyIOError, _>(e.to_string()))?
33 .iter()
34 .map(|h| h.trim().to_lowercase())
35 .collect();

38 for std_header in &standard_headers {
39 let std_key = std_header.to_lowercase();
40 if let Some(aliases) = header_aliases.get(std_key.as_str()) {
41 for alias in aliases {
42 if let Some(position) = headers.iter().position(|h| h == alias) {
43 header_map.insert(*std_header, position);
44 break;
45 }
46 }
47 }
48 }

52

55 let row_dict = PyDict::new_bound(py);
56
57 for std_header in &standard_headers {
58 let mut value_obj = py.None();
59 if let Some(&index) = header_map.get(std_header) {
60 if let Some(raw_value) = record.get(index) {
61 let trimmed = raw_value.trim();

63 if let Some(caps) = freq_regex.captures(trimmed) {
64 let value: f64 = caps["value"].parse().unwrap_or(0.0);
65 let unit = caps.name("unit").map(|m| m.as_str().to_lowercase());
66 let mut mhz_val = value;
67 if let Some(u) = unit {
68 if u == "k" { mhz_val /= 1000.0; }
69 else if u == "g" { mhz_val *= 1000.0; }
70 }
71 value_obj = mhz_val.into_py(py);
72 } else {
73 if let Ok(value) = trimmed.parse::<f64>() {
74 value_obj = value.into_py(py);
75 } else {

77 }
78 }
79 } else {
80 value_obj = trimmed.into_py(py);
81 }
82 }
83 }
84 let _ = row_dict.set_item(std_header, value_obj);
85 }

87 }

90 Ok((std_headers_vec, parsed_records.unbind()))
91}
