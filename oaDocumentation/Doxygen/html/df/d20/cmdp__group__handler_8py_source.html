41 def _init_group_state(self, group_name, color, initial_visible, initial_mute):
42 """Initialize Tkinter variables and store them in the widget references."""
43 iv = tk.BooleanVar(value=initial_visible)
44 im = tk.BooleanVar(value=initial_mute)
45 cv = tk.StringVar(value=color)
46 nv = tk.StringVar(value=group_name)
47
48 self.w.group_vars[group_name] = iv
49 self.group_mute_vars[group_name] = im
50 self.w.group_color_vars[group_name] = cv
51 self.w.group_name_vars[group_name] = nv
52
53 return {"visible": iv, "mute": im, "color": cv, "name": nv}

55 def _register_group_sme(self, group_name, vars):
56 """Handle State Mirror Engine registration and MQTT topic subscriptions."""
57 if not self.w.mixin_ref.state_mirror_engine:
58 return
59
60 sme = self.w.mixin_ref.state_mirror_engine
61 bp = f"{self.w.path}/groups/{group_name}"
62 iv, im, cv, nv = vars["visible"], vars["mute"], vars["color"], vars["name"]
63
64 sme.register_widget(f"{bp}/visible", iv, self.w.base_mqtt_topic, {"type": "_CMDP_GrpVis"})
65 sme.register_widget(f"{bp}/mute", im, self.w.base_mqtt_topic, {"type": "_CMDP_GrpMute"})
66 sme.register_widget(f"{bp}/color", cv, self.w.base_mqtt_topic, {"type": "_CMDP_GrpCol"})
67 sme.register_widget(f"{bp}/name", nv, self.w.base_mqtt_topic, {"type": "_CMDP_GrpName"})
68
69 def _broadcast_change(p):
70 if not getattr(sme, "_silent_update", False): sme.broadcast_gui_change_to_mqtt(p)
71
72 iv.trace_add("write", lambda *a: _broadcast_change(f"{bp}/visible"))
73 im.trace_add("write", lambda *a: _broadcast_change(f"{bp}/mute"))
74 cv.trace_add("write", lambda *a: _broadcast_change(f"{bp}/color"))
75 nv.trace_add("write", lambda *a: _broadcast_change(f"{bp}/name"))
76
77 for p, v in [("visible", iv), ("mute", im), ("color", cv), ("name", nv)]:
78 t = sme.get_widget_topic(f"{bp}/{p}")
79 if self.w.mixin_ref.subscriber_router and t:
80 self.w.mixin_ref.subscriber_router.subscribe_to_topic(t, sme.sync_incoming_mqtt_to_gui)
81 sme.initialize_widget_state(f"{bp}/{p}")

155 def _apply_group_mute(self, group_name):
156 muted = self.group_mute_vars[group_name].get()
157 if group_name in self.group_mute_buttons:
158 self.group_mute_buttons[group_name].config(text="🔈" if muted else "🔊", bg="#444" if muted else "#f4902c", fg="white" if muted else "black")
159 for f in self.w.faders:

163 idx = self.w.faders.index(f)
