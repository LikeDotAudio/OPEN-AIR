18 def _on_drag(self, event):
19 height = float(self.canvas.winfo_height())
20 if height <= 1: return
21
22 # Invert Y coordinate because Y increases downwards in Tkinter
23 norm_y_inverted = (event.y - 25) / (height - 45)
24 norm_y_inverted = max(0.0, min(1.0, norm_y_inverted))
25 norm_pos = 1.0 - norm_y_inverted
26
27 log_norm = max(0.0000001, norm_pos) ** self.log_exponent if self.log_exponent != 1.0 else norm_pos
28 current_value = self.min_val + log_norm * (self.max_val - self.min_val)
29
30 self.variable.set(current_value)
31 if self.state_mirror_engine:

33 self.path,
34 extra_payload={"SETTLED": False, "LOCKED": True}

48 def _on_mousewheel(self, event):
49 current_val = self.variable.get()
50 val_range = self.max_val - self.min_val
51 step = val_range * 0.05
52
53 delta = 1 if (event.num == 4 or (hasattr(event, 'delta') and event.delta > 0)) else -1
54 if sys.platform == "linux" and event.num == 5: delta = -1
55
56 new_val = max(self.min_val, min(self.max_val, current_val + (delta * step)))
