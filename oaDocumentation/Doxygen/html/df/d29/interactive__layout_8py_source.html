159 def _sync_rulers(self, event=None):
160 builder = self.preview_engine.preview_builder
161 if not builder: return
162 x_scroll, y_scroll = builder.canvas.xview(), builder.canvas.yview()
163 sr = builder.canvas.cget("scrollregion")
164 if not sr: return
165 sr_p = [float(p) for p in sr.split()]
166 fw, fh = sr_p[2] - sr_p[0], sr_p[3] - sr_p[1]
167 self.h_ruler.set_offset(int(x_scroll[0] * fw))
168 self.v_ruler.set_offset(int(y_scroll[0] * fh))
169 self.h_ruler.set_center(fw // 2)
170 self.v_ruler.set_center(fh // 2)

237 def _on_component_dragging(self, x, y, name):
238 """Handles real-time feedback for components being dragged from the Grab Bag."""
239 if not self.winfo_exists(): return
240
241 # 1. ⚡ GHOST MODE: Temporarily switch to ghost tier if not already there
242 # This provides immediate structural feedback.
243 if self.render_tier_var.get() != "Ghost":
244 self._previous_tier = self.render_tier_var.get()
245 self.render_tier_var.set("Ghost")
246 self._on_render_tier_change()
247
248 # 2. Ensure ghost overlay is visible and on top
249 self.ghost_overlay.place(x=0, y=0, relwidth=1, relheight=1)
250 tk.Widget.tkraise(self.ghost_overlay)
251
252 # 3. Find target and update visuals
253 target = self.focus_mgr.find_drop_target_at(x, y)
254 if target:
255 tw, tp, tmode, tcoords = target
256 self.ghost_overlay.draw_insertion_line(*tcoords)
257 else:
258 self.ghost_overlay.clear_insertion()
259
260 def _on_component_dropped_global(self, x, y, name, schema):
261 """Clears drag feedback when a component is dropped."""
262 if not self.winfo_exists(): return
263 self.ghost_overlay.clear()
264 self.ghost_overlay.place_forget()
265
266 # ⚡ RESTORE MODE: Switch back to the previous rendering quality
267 if hasattr(self, '_previous_tier'):
268 self.render_tier_var.set(self._previous_tier)
269 del self._previous_tier
270 self._on_render_tier_change()
271 else:
272 # Force rebuild anyway to show the new component
273 self._manual_rebuild()
