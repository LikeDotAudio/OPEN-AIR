169 def _sync_rulers(self, event=None):
170 builder = self.preview_engine.preview_builder
171 if not builder: return
172 x_scroll, y_scroll = builder.canvas.xview(), builder.canvas.yview()
173 sr = builder.canvas.cget("scrollregion")
174 if not sr: return
175 sr_p = [float(p) for p in sr.split()]
176 fw, fh = sr_p[2] - sr_p[0], sr_p[3] - sr_p[1]
177 self.h_ruler.set_offset(int(x_scroll[0] * fw))
178 self.v_ruler.set_offset(int(y_scroll[0] * fh))
179 self.h_ruler.set_center(fw // 2)
180 self.v_ruler.set_center(fh // 2)

248 def _on_component_dragging(self, x, y, name):
249 """Handles real-time feedback for components being dragged from the Grab Bag."""
250 if not self.winfo_exists(): return
251
252 # 1. ⚡ GHOST MODE: Temporarily switch to ghost tier if not already there
253 # This provides immediate structural feedback.
254 if self.render_tier_var.get() != "Ghost":
255 self._previous_tier = self.render_tier_var.get()
256 self.render_tier_var.set("Ghost")
257 self._on_render_tier_change()
258
259 # 2. Ensure ghost overlay is visible and on top
260 self.ghost_overlay.place(x=0, y=0, relwidth=1, relheight=1)
261 tk.Widget.tkraise(self.ghost_overlay)
262
263 # 3. Find target and update visuals
264 target = self.focus_mgr.find_drop_target_at(x, y)
265 if target:
266 tw, tp, tmode, tcoords = target
267 self.ghost_overlay.draw_insertion_line(*tcoords)
268 else:
269 self.ghost_overlay.clear_insertion()
270
271 def _on_component_dropped_global(self, x, y, name, schema):
272 """Clears drag feedback when a component is dropped."""
273 if not self.winfo_exists(): return
274 self.ghost_overlay.clear()
275 self.ghost_overlay.place_forget()
276
277 # ⚡ RESTORE MODE: Switch back to the previous rendering quality
278 if hasattr(self, '_previous_tier'):
279 self.render_tier_var.set(self._previous_tier)
280 del self._previous_tier
281 self._on_render_tier_change()
282 else:
283 # Force rebuild anyway to show the new component
284 self._manual_rebuild()
