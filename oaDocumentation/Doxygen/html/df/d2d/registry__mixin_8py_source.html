22class ThreadSafeRegistry:
23 """Encapsulates thread-safe storage for widget registrations."""
24 def __init__(self):
25 if HAS_RUST:
26 self._registry = RustWidgetRegistry()
27 else:
28 self._lock = threading.RLock()
29 self.widgets = {}
30 self.topic_map = {}

32 def register(self, widget_id, info):
33 if HAS_RUST:
34 self._registry.register(widget_id, info)
35 else:
36 with self._lock:
37 self.widgets[widget_id] = info
38 self.topic_map[info["topic"]] = widget_id

40 def is_registered(self, widget_id):
41 if HAS_RUST:
