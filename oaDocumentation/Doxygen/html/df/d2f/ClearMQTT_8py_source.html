26def check_mqtt_port(host, port, timeout=1.0):
27 """Verifies the MQTT broker port is reachable without using exceptions."""
28 sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
29 sock.settimeout(timeout)
30 result = sock.connect_ex((host, port))
31 sock.close()
32 return result == 0

52 def sweep(self):
53 """Discovers and deletes all topics (including retained) under the configured base topic."""
54 matrix_log("core", "system", inspect.currentframe().f_code.co_name if "inspect" in globals() else "unknown", f"📡📤📤 [CLEAR_MQTT] Starting MQTT Deep Sweep on {self.host}:{self.port} (Root: {self.base_topic})...", "INFO")
55
56 # ⚡ PRECONDITION VALIDATION: Verify port is open before connect()
57 if not check_mqtt_port(self.host, self.port):
58 logger.error(f"❌ Connection failed: Port {self.port} on {self.host} is unreachable.")
59 return
60
61 self.client.on_message = self.on_message
62
63 # We assume host and port are validated.
64 # connect() will only be called if the port check passed.
65 self.client.connect(self.host, self.port, 60)
66
67 # Subscribe to catch existing retained messages
68 wildcard = f"{self.base_topic}/#"
69 # Subscribe to both root and wildcard to be thorough
70 self.client.subscribe([(self.base_topic, 0), (wildcard, 0)])
71
72 # 1. Discovery Phase
73 matrix_log("core", "system", inspect.currentframe().f_code.co_name if "inspect" in globals() else "unknown", f" └─ 🕵️ Discovery: Scanning for active/retained topics under {self.base_topic}...", "INFO")
74 self.client.loop_start()
75
76 # Wait for retained messages to arrive.
77 # 2 seconds is usually enough for a local broker to dump retained state.
78 time.sleep(2.0)
79
80 self.client.loop_stop()
81
82 if not self.topics:
83 matrix_log("core", "system", inspect.currentframe().f_code.co_name if "inspect" in globals() else "unknown", f"✨ No topics found under {self.base_topic}. Broker is already clean.", "INFO")
84 self.client.disconnect()
85 return

87 matrix_log("core", "system", inspect.currentframe().f_code.co_name if "inspect" in globals() else "unknown", f" └─ 📋 Found {len(self.topics)} topics to clear.", "INFO")
88
89 # 2. Deletion Phase
90 self.client.loop_start() # Restart loop to handle PUBACKs
91
92 count = 0
93 publish_handles = []
94
95 # Sort reverse to potentially delete children before parents
96 for topic in sorted(list(self.topics), reverse=True):
97 # To delete a retained topic, publish a zero-length payload with retain=True
98 matrix_log("core", "system", inspect.currentframe().f_code.co_name if "inspect" in globals() else "unknown", f" Deleting MQTT topic: {topic}", "INFO")
99 message_info = self.client.publish(topic, payload=None, qos=1, retain=True)
100 publish_handles.append(message_info)
101 count += 1
102 if count % 100 == 0:
103 matrix_log("core", "system", inspect.currentframe().f_code.co_name if "inspect" in globals() else "unknown", f" ├─ Sent clear command for {count} topics...", "INFO")
104
105 # Wait for all deletion messages to be acknowledged by the broker
106 matrix_log("core", "system", inspect.currentframe().f_code.co_name if "inspect" in globals() else "unknown", " └─ ⏳ Finalizing: Waiting for broker acknowledgments...", "INFO")
107 for handle in publish_handles:
108 # wait_for_publish can still raise if the loop is not running or other state issues.
109 # but we assume the loop is running and we are within timeout.
110 # Using a polling check instead of a blocking call with exception
111 start_wait = time.time()
112 while not handle.is_published() and (time.time() - start_wait) < 1.0:
113 time.sleep(0.01)
114
115 self.client.loop_stop()
116 matrix_log("core", "system", inspect.currentframe().f_code.co_name if "inspect" in globals() else "unknown", f"✨ Successfully wiped {count} topics (and cleared retained state) from {self.base_topic} tree.", "INFO")
117 self.client.disconnect()
