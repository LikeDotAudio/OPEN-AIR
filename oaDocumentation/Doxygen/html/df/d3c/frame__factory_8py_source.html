29 def calculate_dimensions(config):
30 # Dynamic Geometry Calculation
31 padding = LAYOUT_PADDING_DEFAULT
32 main_arc_radius = (config.size - padding) / 2
33
34 # Calculate the extremes of the arc to determine bounding box
35 half_angle = config.meter_viewable_angle / 2.0
36 a1 = math.radians(config.meter_center_angle + half_angle)
37 a2 = math.radians(config.meter_center_angle - half_angle)
38
39 # Points to check: pivot(0,0), arc start, arc end, and any cardinal directions in between
40 pts = [(0, 0),
41 (main_arc_radius * math.cos(a1), -main_arc_radius * math.sin(a1)),
42 (main_arc_radius * math.cos(a2), -main_arc_radius * math.sin(a2))]
43
44 # Check for cardinal directions within the arc sweep
45 for angle_deg in [0, 90, 180, 270]:
46 # Normalize angle to be relative to center
47 diff = (angle_deg - config.meter_center_angle + 180) % 360 - 180
48 if abs(diff) <= half_angle:
49 ar = math.radians(angle_deg)
50 pts.append((main_arc_radius * math.cos(ar), -main_arc_radius * math.sin(ar)))
51
52 min_x = min(p[0] for p in pts)
53 max_x = max(p[0] for p in pts)
54 min_y = min(p[1] for p in pts)
55 max_y = max(p[1] for p in pts)
56
57 # Add padding for labels
58 if config.scale_numbers:
59 min_x -= LAYOUT_LABEL_PADDING_X; max_x += LAYOUT_LABEL_PADDING_X
60 min_y -= 5; max_y += 15 # Kept some internal ones as they seem specific to text bounding box logic
61
62 bb_width = max_x - min_x
63 bb_height = max_y - min_y
64
65 requested_w = config.width
66 requested_h = config.height
67
68 total_width = max(int(bb_width + LAYOUT_CANVAS_MARGIN_W), requested_w)
69 total_height = max(int(bb_height + LAYOUT_CANVAS_MARGIN_H), requested_h)
70
71 # Calculate pivot offset from top-left of canvas, centering the arc
72 offset_x = -min_x + (LAYOUT_CANVAS_MARGIN_W / 2)
73 offset_y = -min_y + (LAYOUT_CANVAS_MARGIN_H / 5) # Heuristic from original code (10 -> 2 ratio)
74
75 # Original code used: offset_x = -min_x + 10; offset_y = -min_y + 2
76 offset_x = -min_x + LAYOUT_OFFSET_X
77 offset_y = -min_y + LAYOUT_OFFSET_Y
78
79 if requested_w > bb_width + LAYOUT_CANVAS_MARGIN_W:
80 offset_x += (requested_w - (bb_width + LAYOUT_CANVAS_MARGIN_W)) / 2
81 if requested_h > bb_height + LAYOUT_CANVAS_MARGIN_H:
82 offset_y += (requested_h - (bb_height + LAYOUT_CANVAS_MARGIN_H)) / 2
83
84 # Apply Pivot Crop (Layout)
85 if config.pivot_crop > 0:
86 crop_pixels = main_arc_radius * (config.pivot_crop / 100.0)
87 total_height = max(LAYOUT_PIVOT_CROP_MIN_H, int(total_height - crop_pixels))
88
89 return total_width, total_height, offset_x, offset_y
