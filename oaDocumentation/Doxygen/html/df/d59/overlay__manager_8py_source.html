62 def _recursive_clear(self, container):

84 def _recursive_apply(self, container, depth=0):
85 """Recursively injects designer controls into widgets with valid paths."""
86 if depth > 10:
87 return
88
89 for child in container.winfo_children():
90 if getattr(child, '_is_design_overlay', False):
91 continue
92
93 path = getattr(child, '_oca_path', None)
94 if path and "unknown" not in path:
95 self._inject_modular_controls(child, path)
96
97 if isinstance(child, (tk.Frame, tk.Canvas, tk.LabelFrame)):
98 self._recursive_apply(child, depth + 1)

100 def _inject_modular_controls(self, widget, path):
101 """Instantiates and syncs modular overlay classes for a specific widget."""
102 is_focused = (self.workspace.focused_path == path)
103 overlay_instances = []
104
105 for overlay_cls in self.overlay_registry:
106 instance = overlay_cls(self.workspace, widget, path, is_focused)
107 overlay_instances.append(instance)
108
109 self.active_overlays[path] = overlay_instances
110
111 def _sync_pos(event=None):
112 """Syncs overlay elements with the widget's physical bounds."""
113 if not widget.winfo_exists():
114 return
115 try:
116 bounds = (widget.winfo_x(), widget.winfo_y(), widget.winfo_width(), widget.winfo_height())
117 if bounds[2] <= 1 or bounds[3] <= 1:
118 return
119 for instance in overlay_instances:
120 instance.sync(*bounds)
121 except (tk.TclError, ValueError):
122 pass
123
124 # Bind sync to widget resize and trigger initial sync
125 widget.bind("<Configure>", _sync_pos, add="+")
126 widget.after(100, _sync_pos)
