33 fn start(&mut self, host: String, port: u16, callback: Py<PyAny>) -> PyResult<()> {
34 if self.running.load(Ordering::SeqCst) {
35 return Err(PyErr::new::<pyo3::exceptions::PyRuntimeError, _>("Server already running"));
36 }
37
38 let addr = format!("{}:{}", host, port);
39 let socket = UdpSocket::bind(&addr).map_err(|e| {
40 PyErr::new::<pyo3::exceptions::PyIOError, _>(format!("Failed to bind to {}: {}", addr, e))
41 })?;
42
43 // Set read timeout so we can check the running flag
44 socket.set_read_timeout(Some(std::time::Duration::from_millis(500))).map_err(|e| {
45 PyErr::new::<pyo3::exceptions::PyIOError, _>(format!("Failed to set socket timeout: {}", e))
46 })?;
47
48 let running = self.running.clone();
49 running.store(true, Ordering::SeqCst);
50
51 let handle = thread::spawn(move || {
52 let mut buf = [0u8; 65535];
53 while running.load(Ordering::SeqCst) {
54 match socket.recv_from(&mut buf) {
55 Ok((size, _addr)) => {
56 let packet = rosc::decoder::decode_udp(&buf[..size]);
57 match packet {
58 Ok((_, packet)) => {
59 handle_packet(packet, &callback);
60 }
61 Err(e) => {
62 eprintln!("📡⚙️❌ [OSC-RS] Decode error: {:?}", e);
63 }
64 }
65 }
66 Err(e) if e.kind() == std::io::ErrorKind::WouldBlock => {
67 // Timeout reached, loop and check running flag
68 }
69 Err(e) => {
70 eprintln!("📡⚙️❌ [OSC-RS] Socket error: {:?}", e);
71 break;
72 }
73 }
74 }
75 running.store(false, Ordering::SeqCst);
76 });
77
78 self.handle = Some(handle);
79 Ok(())
80 }

95fn handle_packet(packet: OscPacket, callback: &Py<PyAny>) {
96 match packet {
97 OscPacket::Message(message) => {
98 dispatch_message(message.addr, message.args, callback);
99 }
100 OscPacket::Bundle(bundle) => {
101 for packet in bundle.content {
102 handle_packet(packet, callback);
103 }
104 }
105 }

108fn dispatch_message(addr: String, args: Vec<OscType>, callback: &Py<PyAny>) {
109 Python::with_gil(|py| {
110 let py_args = pyo3::types::PyList::empty_bound(py);
111 for arg in args {
112 match arg {
113 OscType::Int(i) => { let _ = py_args.append(i); }
114 OscType::Float(f) => { let _ = py_args.append(f); }
115 OscType::String(s) => { let _ = py_args.append(s); }
116 OscType::Blob(b) => { let _ = py_args.append(pyo3::types::PyBytes::new_bound(py, &b)); }
117 OscType::Long(l) => { let _ = py_args.append(l); }
118 OscType::Double(d) => { let _ = py_args.append(d); }
119 OscType::Char(c) => { let _ = py_args.append(c.to_string()); }
120 OscType::Color(c) => {
121 let dict = PyDict::new_bound(py);
122 let _ = dict.set_item("r", c.red);
123 let _ = dict.set_item("g", c.green);
124 let _ = dict.set_item("b", c.blue);
125 let _ = dict.set_item("a", c.alpha);
126 let _ = py_args.append(dict);
127 }
128 OscType::Midi(m) => {
129 let dict = PyDict::new_bound(py);
130 let _ = dict.set_item("port", m.port);
131 let _ = dict.set_item("status", m.status);
132 let _ = dict.set_item("data1", m.data1);
133 let _ = dict.set_item("data2", m.data2);
134 let _ = py_args.append(dict);
135 }
136 OscType::Bool(b) => { let _ = py_args.append(b); }
137 OscType::Nil => { let _ = py_args.append(py.None()); }
138 OscType::Inf => { let _ = py_args.append("INFINITY"); }
139 OscType::Time(t) => {
140 let dict = PyDict::new_bound(py);
141 let _ = dict.set_item("seconds", t.seconds);
142 let _ = dict.set_item("fraction", t.fractional);
143 let _ = py_args.append(dict);
144 }
145 _ => { let _ = py_args.append("UNKNOWN_TYPE"); }
146 }
147 }
148
149 let _ = callback.call1(py, (addr, py_args));
150 });
151}

180 fn send_message(&self, address: String, value: Py<PyAny>, py: Python<'_>) -> PyResult<()> {
181 if let Some(socket) = &self.socket {
182 let mut args = Vec::new();
183
184 // Extract value based on type
185 if let Ok(i) = value.extract::<i32>(py) {
186 args.push(OscType::Int(i));
187 } else if let Ok(f) = value.extract::<f32>(py) {
188 args.push(OscType::Float(f));
189 } else if let Ok(s) = value.extract::<String>(py) {
190 args.push(OscType::String(s));
191 } else if let Ok(b) = value.extract::<bool>(py) {
192 args.push(OscType::Bool(b));
193 } else if value.is_none(py) {
194 args.push(OscType::Nil);
195 }
196
197 let packet = OscPacket::Message(rosc::OscMessage {
198 addr: address,
199 args,
200 });
201
202 let message_buf = rosc::encoder::encode(&packet).map_err(|e| {
203 PyErr::new::<pyo3::exceptions::PyRuntimeError, _>(format!("OSC Encode error: {:?}", e))
204 })?;
205
206 socket.send(&message_buf).map_err(|e| {
207 PyErr::new::<pyo3::exceptions::PyIOError, _>(format!("OSC Send error: {}", e))
208 })?;
209 }
210 Ok(())
211 }
