16# Regex pattern to parse your specific log format:
17LOG_PATTERN = re.compile(
18 r"^(?P<timestamp>\d+\.\d+)\s+\|\s+(?P<level>\w+)\s+\|\s+(?P<partition>\w+)\s+\|\s+(?P<process>\w+)\s+\|\s+(?P<function>[\w\.]+)\s+\|\s+(?P<message>.*)$"
19)

21def extract_type(message):
22 match = re.search(r'\[(.*?)\]', message)
23 return match.group(1) if match else "UNKNOWN"
24
25def extract_emojis(message):
26 """Returns a list of all emojis found in the message string."""
27 return [char for char in message if emoji.is_emoji(char)]
28
29def parse_log_file(decoded_contents):
30 data = []
31 lines = decoded_contents.split('\n')
32 for idx, line in enumerate(lines):
33 match = LOG_PATTERN.match(line.strip())
34 if match:
35 row = match.groupdict()
36 row['type'] = extract_type(row['message'])
37 row['emojis'] = extract_emojis(row['message'])
38 row['sequence'] = idx # Keep track of order for timeline
39 data.append(row)
40 return pd.DataFrame(data)

137def update_data(contents):
138 if contents is None:
139 return [], [], [], []
140
141 content_type, content_string = contents.split(',')
142 decoded = base64.b64decode(content_string).decode('utf-8', errors='replace')
143 df = parse_log_file(decoded)
144
145 if df.empty:
146 return [], [], [], []
147
148 proc_opts = [{'label': i, 'value': i} for i in sorted(df['process'].unique())]
149 type_opts = [{'label': i, 'value': i} for i in sorted(df['type'].unique())]
150 func_opts = [{'label': i, 'value': i} for i in sorted(df['function'].unique())]
151
152 return df.to_dict('records'), proc_opts, type_opts, func_opts
