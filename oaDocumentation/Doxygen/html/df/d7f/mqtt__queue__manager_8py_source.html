30 # Internal Queues & State

36 def kick_worker(self):
37 """Signals the async worker to process pending queue items."""
38 if self._async_worker and self._async_worker.loop and self._async_worker.kick_event:
39 self._async_worker.loop.call_soon_threadsafe(self._async_worker.kick_event.set)

41 def put_publish_message(self, topic, payload=None, qos=0, retain=False):
42 """Adds a message to the publish queue."""
43 # Filter out internal monitor topics to avoid loop or unnecessary traffic
44 if "/System/Monitor/" in str(topic): return
45 self._publish_queue.put((topic, payload, qos, retain))
46 self.kick_worker()
47
48 def put_subscribe_request(self, topic, qos=0):
49 """Adds a subscription request to the queue, ensuring uniqueness."""
50 with self._pending_lock:
51 if topic in self._pending_subscriptions: return
52 self._pending_subscriptions.add(topic)
53 self._subscribe_queue.put({"topic": topic, "qos": qos})
54 self.kick_worker()
55
56 def get_publish_message(self):
57 """Retrieves a message from the publish queue. Returns None if empty."""

59 return self._publish_queue.get_nowait()
60 except queue.Empty:
61 return None

63 def get_subscribe_request(self):

70 def task_done(self, queue_type):
71 """Marks a task as done for the specified queue."""
72 if queue_type == "publish":

77 def remove_pending_subscription(self, topic):
78 """Removes a topic from the set of pending subscriptions."""
