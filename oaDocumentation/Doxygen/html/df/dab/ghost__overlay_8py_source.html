35 def draw_resize_handles(self, x, y, width, height):
36 """Draws 8 standard resizing handles around the specified bounding box."""
37 self.delete("handle")
38 hs = 3 # Handle half-size
39 coords = [
40 (x, y), (x+width//2, y), (x+width, y),
41 (x, y+height//2), (x+width, y+height//2),
42 (x, y+height), (x+width//2, y+height), (x+width, y+height)
43 ]
44 for cx, cy in coords:
45 self.create_rectangle(
46 cx-hs, cy-hs, cx+hs, cy+hs,
47 fill="white", outline="#007acc", tags="handle"
48 )
49 self.tag_raise("handle")
