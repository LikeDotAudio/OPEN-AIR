7class SemanticLayoutResolver:
8 """
9 Translates high-level layout intent (Align, Anchor, Stretch) into Tkinter sticky strings.
10 """
11 ANCHOR_MAP = {
12 "top": "n", "bottom": "s", "left": "w", "right": "e",
13 "north": "n", "south": "s", "west": "w", "east": "e",
14 "nw": "nw", "ne": "ne", "sw": "sw", "se": "se"
15 }
16
17 @classmethod
18 def resolve_sticky(cls, geometry, config):
19 """
20 Calculates the Tkinter sticky string based on semantic layout rules.
21 """
22 sticky_parts = set()
23 stretch = geometry.get("stretch", "").lower()
24 anchor = geometry.get("anchor", "").lower()
25 align = geometry.get("align", "").lower()
26
27 # 1. Handle Stretching (Size Change)
28 if stretch in ["width", "horizontal", "ew"]:
29 sticky_parts.update(["e", "w"])
30 elif stretch in ["height", "vertical", "ns"]:
31 sticky_parts.update(["n", "s"])
32 elif stretch in ["both", "all", "fill", "nsew"]:
33 sticky_parts.update(["n", "s", "e", "w"])
34
35 # 2. Handle Anchoring (Pinned Position)
36 for part in cls.ANCHOR_MAP.get(anchor, ""):
37 sticky_parts.add(part)
38
39 # 3. Handle Alignment (Justification)
40 if "e" not in sticky_parts and "w" not in sticky_parts:
41 if align in ["left", "west"]: sticky_parts.add("w")
42 if align in ["right", "east"]: sticky_parts.add("e")
43 if "n" not in sticky_parts and "s" not in sticky_parts:
44 if align in ["top", "north"]: sticky_parts.add("n")
45 if align in ["bottom", "south"]: sticky_parts.add("s")
46
47 # 4. Fallback to Deprecated 'sticky'
48 if not (stretch or anchor or align) and "sticky" in geometry:
49 sticky_parts.update(list(geometry["sticky"].lower()))
50
51 # 5. Fixed Size Enforcement
52 has_fixed_width = "width" in geometry or "width" in config
53 if has_fixed_width and stretch not in ["width", "both", "horizontal", "fill", "nsew"]:
54 if "e" in sticky_parts and "w" in sticky_parts:
55 sticky_parts.discard("e")
56 sticky_parts.discard("w")
57
58 has_fixed_height = "height" in geometry or "height" in config
59 if has_fixed_height and stretch not in ["height", "both", "vertical", "fill", "nsew"]:
60 if "n" in sticky_parts and "s" in sticky_parts:
61 sticky_parts.discard("s")
62 if not align: sticky_parts.add("n")
63
64 return "".join(sorted(list(sticky_parts)))
